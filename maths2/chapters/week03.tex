\section{Mathematical \& Theoretical Foundations}

Linear algebra abstracts geometric Euclidean vectors into algebraic entities spanning function spaces, polynomial spaces, and high-dimensional feature spaces.

\subsection{Formal Definition of a Vector Space}

\begin{definitionbox}{Vector Space Axioms}
A \textbf{vector space} over $\R$ is a non-empty set $V$ equipped with two binary operations:
\begin{itemize}
    \item \textbf{Vector Addition} $+: V \times V \to V$, mapping $(\mathbf{u}, \mathbf{v}) \mapsto \mathbf{u} + \mathbf{v} \in V$
    \item \textbf{Scalar Multiplication} $\cdot: \R \times V \to V$, mapping $(c, \mathbf{v}) \mapsto c\mathbf{v} \in V$
\end{itemize}
such that the following 8 axioms hold for all $\mathbf{u}, \mathbf{v}, \mathbf{w} \in V$ and all scalars $c, d \in \R$:
\begin{enumerate}
    \item \textbf{Commutativity of Addition:} $\mathbf{u} + \mathbf{v} = \mathbf{v} + \mathbf{u}$.
    \item \textbf{Associativity of Addition:} $(\mathbf{u} + \mathbf{v}) + \mathbf{w} = \mathbf{u} + (\mathbf{v} + \mathbf{w})$.
    \item \textbf{Additive Identity:} There exists a unique zero vector $\mathbf{0} \in V$ such that $\mathbf{v} + \mathbf{0} = \mathbf{v}$.
    \item \textbf{Additive Inverse:} For every $\mathbf{v} \in V$, there exists a unique $-\mathbf{v} \in V$ such that $\mathbf{v} + (-\mathbf{v}) = \mathbf{0}$.
    \item \textbf{Scalar Multiplicative Identity:} $1 \cdot \mathbf{v} = \mathbf{v}$.
    \item \textbf{Scalar Associativity:} $c(d\mathbf{v}) = (cd)\mathbf{v}$.
    \item \textbf{Vector Distributivity:} $c(\mathbf{u} + \mathbf{v}) = c\mathbf{u} + c\mathbf{v}$.
    \item \textbf{Scalar Distributivity:} $(c + d)\mathbf{v} = c\mathbf{v} + d\mathbf{v}$.
\end{enumerate}
\textbf{Standard Examples:} Real coordinate space $\R^n$, matrix spaces $\R^{m \times n}$, polynomial spaces $\mathbb{P}_n$, and continuous function spaces $\mathcal{C}([a, b])$.
\end{definitionbox}

\subsection{Vector Subspaces and the Three-Point Criterion}

A subset of a vector space is not automatically a vector space itself, as arbitrary subsets may not contain the origin or may step outside their boundaries under addition.

\begin{definitionbox}{Vector Subspace}
A subset $W \subseteq V$ of a vector space $V$ is called a \textbf{subspace} of $V$ (written $W \le V$) if $W$ is itself a vector space under the addition and scalar multiplication operations defined on $V$.
\end{definitionbox}

\begin{theorembox}{The Subspace Test (3-Point Criterion)}
A non-empty subset $W \subseteq V$ is a subspace of $V$ if and only if it satisfies three conditions:
\begin{enumerate}
    \item \textbf{Contains the Zero Vector:} $\mathbf{0} \in W$.
    \item \textbf{Closed Under Vector Addition:} If $\mathbf{u}, \mathbf{v} \in W$, then $\mathbf{u} + \mathbf{v} \in W$.
    \item \textbf{Closed Under Scalar Multiplication:} If $\mathbf{v} \in W$ and $c \in \R$, then $c\mathbf{v} \in W$.
\end{enumerate}
\textbf{Equivalent Single-Condition Test:} $W \subseteq V$ is a subspace if and only if $W \neq \emptyset$ and for all $\mathbf{u}, \mathbf{v} \in W$ and $c, d \in \R$, the linear combination $c\mathbf{u} + d\mathbf{v} \in W$.
\end{theorembox}

\subsection{Linear Combinations and the Span}

\begin{definitionbox}{Linear Combination and Span}
Let $S = \{\mathbf{v}_1, \mathbf{v}_2, \dots, \mathbf{v}_k\}$ be a set of vectors in a vector space $V$.
\begin{itemize}
    \item A vector $\mathbf{w} \in V$ is a \textbf{linear combination} of $S$ if there exist scalars $c_1, c_2, \dots, c_k \in \R$ such that:
    \begin{equation}
    \mathbf{w} = c_1 \mathbf{v}_1 + c_2 \mathbf{v}_2 + \dots + c_k \mathbf{v}_k = \sum_{i=1}^k c_i \mathbf{v}_i
    \end{equation}
    \item The \textbf{span} of $S$, denoted $\vspan(S)$ or $\vspan\{\mathbf{v}_1, \dots, \mathbf{v}_k\}$, is the set of all possible linear combinations of vectors in $S$:
    \begin{equation}
    \vspan(S) = \left\{ \sum_{i=1}^k c_i \mathbf{v}_i \ \middle|\ c_1, \dots, c_k \in \R \right\}
    \end{equation}
\end{itemize}
\textbf{Theorem:} For any non-empty set $S \subseteq V$, $\vspan(S)$ is \textbf{always a valid subspace} of $V$. It is the minimal subspace of $V$ containing $S$.
\end{definitionbox}

\subsection{Linear Independence and Redundancy}

Linear independence establishes whether every vector in a set provides a unique, non-redundant spatial dimension.

\begin{definitionbox}{Linear Independence}
A set of vectors $S = \{\mathbf{v}_1, \mathbf{v}_2, \dots, \mathbf{v}_k\}$ in a vector space $V$ is \textbf{linearly independent} if the vector equation:
\begin{equation}
c_1 \mathbf{v}_1 + c_2 \mathbf{v}_2 + \dots + c_k \mathbf{v}_k = \mathbf{0}
\end{equation}
admits \textbf{only the trivial solution}:
\begin{equation}
c_1 = c_2 = \dots = c_k = 0
\end{equation}
If there exists a non-trivial solution (where at least one scalar $c_i \neq 0$), the set $S$ is \textbf{linearly dependent}.
\end{definitionbox}

\begin{theorembox}{Characterization of Linear Dependence}
A set of vectors $S = \{\mathbf{v}_1, \mathbf{v}_2, \dots, \mathbf{v}_k\}$ with $k \ge 2$ is linearly dependent if and only if \textbf{at least one vector $\mathbf{v}_j$ can be expressed as a linear combination of the remaining vectors}:
\begin{equation}
\mathbf{v}_j \in \vspan(S \setminus \{\mathbf{v}_j\})
\end{equation}
\textbf{Dimensional Bound in $\R^n$:} Any set of $k$ vectors in $\R^n$ with $k > n$ is \textbf{guaranteed to be linearly dependent}.
\end{theorembox}

\begin{theorembox}{Matrix Column Independence Test}
Let $A = [\mathbf{v}_1 \ \mathbf{v}_2 \ \dots \ \mathbf{v}_k] \in \R^{n \times k}$ be the matrix having vectors $\mathbf{v}_i$ as its columns.
\begin{itemize}
    \item The vectors $\{\mathbf{v}_1, \dots, \mathbf{v}_k\}$ are linearly independent if and only if the homogeneous system $A\mathbf{c} = \mathbf{0}$ has only the trivial solution $\mathbf{c} = \mathbf{0}$.
    \item Equivalently, the columns are independent if and only if $\rank(A) = k$ (every column contains a pivot in REF).
    \item For a square set of $n$ vectors in $\R^n$ ($k = n$), the vectors are linearly independent if and only if:
    \begin{equation}
    \det(A) \neq 0
    \end{equation}
\end{itemize}
\end{theorembox}

\section{Deep Intuition \& Conceptual Derivations}

\subsection{Proof: More than $n$ Vectors in $\R^n$ Are Always Linearly Dependent}
Let $S = \{\mathbf{v}_1, \mathbf{v}_2, \dots, \mathbf{v}_k\}$ be a set of $k$ vectors in $\R^n$, where $k > n$.
To test linear independence, consider the vector equation:
\begin{equation}
c_1 \mathbf{v}_1 + c_2 \mathbf{v}_2 + \dots + c_k \mathbf{v}_k = \mathbf{0}
\end{equation}
Writing this in matrix form:
\begin{equation}
A \mathbf{c} = \mathbf{0}
\end{equation}
where $A = [\mathbf{v}_1 \ \mathbf{v}_2 \ \dots \ \mathbf{v}_k] \in \R^{n \times k}$ is an $n \times k$ matrix, and $\mathbf{c} = [c_1, \dots, c_k]^T \in \R^k$.
This represents a homogeneous system of $n$ linear equations in $k$ unknowns.
Because each elementary row operation creates at most one pivot row:
\begin{equation}
\rank(A) \le \min(n, k) = n
\end{equation}
The number of free variables in the solution space is given by the Rank-Nullity Theorem:
\begin{equation}
\text{number of free variables} = k - \rank(A) \ge k - n
\end{equation}
Since $k > n$, we have:
\begin{equation}
k - \rank(A) \ge k - n > 0
\end{equation}
Thus, there exists at least one free variable, guaranteeing the existence of infinitely many non-trivial solutions $\mathbf{c} \neq \mathbf{0}$. 
Therefore, $S$ is strictly linearly dependent $\blacksquare$.

\subsection{Proof that $\vspan(S)$ is Always a Vector Subspace}
Let $S = \{\mathbf{v}_1, \dots, \mathbf{v}_k\} \subseteq V$. We verify the three subspace conditions for $W = \vspan(S)$:
\begin{enumerate}
    \item \textbf{Zero Vector:} Choosing all scalar coefficients $c_1 = c_2 = \dots = c_k = 0$:
    \begin{equation}
    \sum_{i=1}^k 0 \cdot \mathbf{v}_i = \mathbf{0} \in W
    \end{equation}
    \item \textbf{Closure under Addition:} Let $\mathbf{u}, \mathbf{w} \in W$. Then $\mathbf{u} = \sum_{i=1}^k a_i \mathbf{v}_i$ and $\mathbf{w} = \sum_{i=1}^k b_i \mathbf{v}_i$ for scalars $a_i, b_i \in \R$.
    Adding them:
    \begin{equation}
    \mathbf{u} + \mathbf{w} = \sum_{i=1}^k a_i \mathbf{v}_i + \sum_{i=1}^k b_i \mathbf{v}_i = \sum_{i=1}^k (a_i + b_i) \mathbf{v}_i
    \end{equation}
    Since $(a_i + b_i) \in \R$, $\mathbf{u} + \mathbf{w}$ is a valid linear combination of $S$, so $\mathbf{u} + \mathbf{w} \in W$.
    \item \textbf{Closure under Scalar Multiplication:} For any $c \in \R$:
    \begin{equation}
    c\mathbf{u} = c \sum_{i=1}^k a_i \mathbf{v}_i = \sum_{i=1}^k (c a_i) \mathbf{v}_i
    \end{equation}
    Since $(c a_i) \in \R$, $c\mathbf{u} \in W$.
\end{enumerate}
Since all 3 criteria hold, $\vspan(S) \le V \quad \blacksquare$.

\section{Explicit Gotchas \& Diagnostic Misconceptions}

\begin{trapbox}{Trap 1: The Zero Vector Omission Fallacy}
A subset cannot be a vector subspace unless it passes through the origin.
Consider the hyperplane $W = \{[x, y, z]^T \in \R^3 \mid x + y + z = 1\}$. 
Testing the zero vector: $0 + 0 + 0 = 0 \neq 1$.
Because $\mathbf{0} \notin W$, $W$ is \textbf{not a subspace}. It is an affine hyperplane.
\end{trapbox}

\begin{trapbox}{Trap 2: Sets Containing the Zero Vector Are Always Linearly Dependent}
Any set of vectors $S = \{\mathbf{0}, \mathbf{v}_1, \mathbf{v}_2, \dots, \mathbf{v}_k\}$ that includes the zero vector is \textbf{automatically linearly dependent}.
We can form the non-trivial linear combination:
\begin{equation}
1 \cdot \mathbf{0} + 0 \cdot \mathbf{v}_1 + 0 \cdot \mathbf{v}_2 + \dots + 0 \cdot \mathbf{v}_k = \mathbf{0}
\end{equation}
Because the coefficient of $\mathbf{0}$ is $1 \neq 0$, a non-trivial solution exists, violating linear independence.
\end{trapbox}

\begin{trapbox}{Trap 3: Pairwise Independence Does NOT Imply Mutual Independence}
Students often test linear independence by checking whether any single vector is a scalar multiple of another. While this works for sets of 2 vectors, it fails completely for 3 or more vectors!
Consider:
\begin{equation}
\mathbf{v}_1 = \begin{bmatrix} 1 \\ 0 \end{bmatrix}, \quad \mathbf{v}_2 = \begin{bmatrix} 0 \\ 1 \end{bmatrix}, \quad \mathbf{v}_3 = \begin{bmatrix} 1 \\ 1 \end{bmatrix}
\end{equation}
No vector is a scalar multiple of any other ($\mathbf{v}_1 \nparallel \mathbf{v}_2, \mathbf{v}_1 \nparallel \mathbf{v}_3, \mathbf{v}_2 \nparallel \mathbf{v}_3$). Yet the set is linearly dependent because $\mathbf{v}_3 = 1\mathbf{v}_1 + 1\mathbf{v}_2$. You must test mutual linear independence across the entire set.
\end{trapbox}

\begin{trapbox}{Trap 4: Multicollinearity and Matrix Invertibility in Machine Learning}
In multiple linear regression, if two features are linearly dependent (e.g. $x_2 = 2.5 x_1$, or $x_{\text{temp\_F}} = 1.8 x_{\text{temp\_C}} + 32$), the design matrix $X \in \R^{N \times d}$ has rank strictly less than $d$.
Consequently, the normal equation matrix $X^T X \in \R^{d \times d}$ is \textbf{singular} ($\det(X^T X) = 0$). 
The OLS estimator $\hat{\mathbf{w}} = (X^T X)^{-1} X^T \mathbf{y}$ cannot be computed, producing catastrophic numerical failure in libraries unless regularized (e.g. Ridge Regression $(X^T X + \lambda I)^{-1}$).
\end{trapbox}

\section{Python Scientific Computing Lab}

The following laboratory illustrates testing linear independence, computing matrix ranks, and diagnosing multicollinearity in regression design matrices using NumPy and SymPy.

\begin{lstlisting}[language=Python, caption=Testing Vector Independence and Subspaces in Python]
import numpy as np
import sympy as sp

# 1. Testing Linear Independence via Determinant and Rank
v1 = np.array([1.0, 2.0, 3.0])
v2 = np.array([0.0, 1.0, 4.0])
v3 = np.array([5.0, 6.0, 0.0])

# Form matrix with vectors as columns
A = np.column_stack([v1, v2, v3])
det_A = np.linalg.det(A)
rank_A = np.linalg.matrix_rank(A)

print("Matrix A (columns are vectors):\n", A)
print(f"Determinant of A: {det_A:.4f}")
print(f"Rank of A: {rank_A}")
if rank_A == A.shape[1]:
    print("Vectors are Linearly Independent!")
else:
    print("Vectors are Linearly Dependent.")

# 2. Extracting Free Variables and Nullspace Basis with SymPy
A_sym = sp.Matrix([
    [1, 2, 3],
    [2, 4, 6],
    [3, 6, 9]
])
print("\nRank of Dependent Matrix:", A_sym.rank()) # Rank 1
print("Nullspace Basis (Dependency relations):\n", A_sym.nullspace())

# 3. Data Science: Multicollinearity Diagnostic in Design Matrices
# Feature 1: Square footage
# Feature 2: Square meters (exact scalar multiple: x2 = 0.092903 * x1)
# Feature 3: Number of bedrooms
np.random.seed(42)
sq_ft = np.array([1200.0, 1500.0, 1800.0, 2200.0, 2600.0])
sq_meters = sq_ft * 0.092903 # Perfectly collinear feature!
bedrooms = np.array([2.0, 3.0, 3.0, 4.0, 4.0])

X_multicollinear = np.column_stack([np.ones(5), sq_ft, sq_meters, bedrooms])
rank_X = np.linalg.matrix_rank(X_multicollinear)
print(f"\nDesign Matrix Shape: {X_multicollinear.shape}")
print(f"Design Matrix Rank: {rank_X} (Expected 4, but got {rank_X} due to collinearity!)")

# Matrix inversion attempt (X^T X)^(-1)
XT_X = X_multicollinear.T @ X_multicollinear
cond_num = np.linalg.cond(XT_X)
print(f"Condition Number of X^T X: {cond_num:.2e}") # > 1e15 indicates catastrophic singularity
\end{lstlisting}

\subsection*{Part B: Diagnostic Misconception MCQs}

\begin{enumerate}
    \setcounter{enumi}{5}
    \item A student is examining a set of 5 vectors in $\R^4$: $S = \{\mathbf{v}_1, \mathbf{v}_2, \mathbf{v}_3, \mathbf{v}_4, \mathbf{v}_5\}$. The student tests the first four vectors and finds that none of them is a scalar multiple of another. Can the set $S$ be linearly independent?
    \begin{enumerate}
        \item Yes, provided that all five vectors are non-zero.
        \item No; by the fundamental dimensional theorem, any set of $k$ vectors in $\R^n$ with $k > n$ is guaranteed to be linearly dependent because the homogeneous system $A\mathbf{c} = \mathbf{0}$ has more unknowns than equations ($5 > 4$), forcing at least one free variable.
        \item Yes, if the determinant of the 5 vectors is non-zero.
        \item It depends on whether the vectors are orthogonal.
    \end{enumerate}

    \item Consider the subset of the coordinate plane $W = \left\{ \begin{bmatrix} x \\ y \end{bmatrix} \in \R^2 \ \middle|\ xy \ge 0 \right\}$ (the union of the first and third quadrants). Which subspace criterion does $W$ violate?
    \begin{enumerate}
        \item It does not contain the zero vector.
        \item It is not closed under scalar multiplication.
        \item It is not closed under vector addition (for example, $\mathbf{u} = \begin{bmatrix} 1 \\ 3 \end{bmatrix} \in W$ and $\mathbf{v} = \begin{bmatrix} -3 \\ -1 \end{bmatrix} \in W$, but $\mathbf{u} + \mathbf{v} = \begin{bmatrix} -2 \\ 2 \end{bmatrix} \notin W$ because $(-2)(2) = -4 < 0$).
        \item It violates all three subspace criteria.
    \end{enumerate}
\end{enumerate}
