\chapter{Multivariable Functions, Partial Derivatives \& Gradient}

Machine learning operates fundamentally in multi-dimensional space: objective loss functions depend simultaneously on millions of model parameters $L(\mathbf{w}) = L(w_1, w_2, \dots, w_n)$. Moving from single-variable calculus to \index{Multivariable Calculus}\textbf{Multivariable Calculus} allows us to quantify how a function varies when moving in arbitrary multi-dimensional directions. In this chapter, we master \index{Partial Derivatives}\textbf{Partial Derivatives}, geometric visualization via \index{Level Sets}\textbf{Level Sets}, the \index{Gradient Vector}\textbf{Gradient Vector} $\nabla f$, and \index{Directional Derivatives}\textbf{Directional Derivatives}.

\vspace{0.5cm}
\begin{tcolorbox}[colback=primarylight, colframe=primarydark, title={\textbf{Executive Roadmap \& Learning Objectives}}]
\begin{itemize}
    \item \textbf{Target Competencies:} Compute first-order partial derivatives of multi-variable functions; construct and interpret the gradient vector $\nabla f$; evaluate directional derivatives $D_{\mathbf{u}} f$ along arbitrary unit vectors; prove that the gradient vector points in the direction of steepest ascent and is perpendicular to level curves; compute analytical gradients for loss functions (MSE and cross-entropy).
    \item \textbf{Prerequisites:} Vector algebra and inner products (Weeks 1 \& 7), single-variable differential calculus (IITM BS Maths 1).
    \item \textbf{Mathematical Notation:} Multivariable function $f: \mathbb{R}^n \to \mathbb{R}$; partial derivative $\frac{\partial f}{\partial x_i}$ or $f_{x_i}$; gradient vector $\nabla f(\mathbf{x}) \in \mathbb{R}^n$; directional derivative $D_{\mathbf{u}} f(\mathbf{x})$ where $\|\mathbf{u}\| = 1$; level curve $f(\mathbf{x}) = c$.
\end{itemize}
\end{tcolorbox}

\section{Mathematical \& Theoretical Foundations}

\subsection{Multivariable Functions and Level Curves}

\begin{definitionbox}{Multivariable Scalar Function}
\index{Multivariable Function}
A \textbf{scalar-valued multivariable function} is a rule $f: D \subseteq \mathbb{R}^n \to \mathbb{R}$ that assigns a unique real scalar $z = f(\mathbf{x}) = f(x_1, x_2, \dots, x_n)$ to each feature vector $\mathbf{x} \in D$.
\begin{itemize}
    \item The \textbf{Graph} of $f$ is the surface in $\mathbb{R}^{n+1}$ defined by $\{(\mathbf{x}, f(\mathbf{x})) \mid \mathbf{x} \in D\}$.
    \item The \textbf{Level Set} (or \textbf{Level Curve} for $n=2$, \textbf{Level Surface} for $n=3$) of $f$ at value $c \in \mathbb{R}$ is the set of all domain points producing constant output $c$:
    \begin{equation}
    L_c(f) = \left\{ \mathbf{x} \in \mathbb{R}^n \;\middle|\; f(\mathbf{x}) = c \right\}
    \end{equation}
\end{itemize}
\end{definitionbox}

\subsection{Partial Derivatives}

\begin{definitionbox}{Partial Derivatives}
\index{Partial Derivatives}
The \textbf{first-order partial derivative} of $f(\mathbf{x})$ with respect to the coordinate variable $x_i$ measures the instantaneous rate of change of $f$ when moving parallel to the $i$-th coordinate axis while freezing all other $n-1$ variables as constants:
\begin{equation}
\frac{\partial f}{\partial x_i}(\mathbf{x}) = \lim_{h \to 0} \frac{f(x_1, \dots, x_i + h, \dots, x_n) - f(x_1, \dots, x_i, \dots, x_n)}{h}
\end{equation}
\end{definitionbox}

\subsection{The Gradient Vector and Directional Derivatives}

\begin{definitionbox}{The Gradient Vector ($\nabla f$)}
\index{Gradient Vector}
For a differentiable scalar function $f: \mathbb{R}^n \to \mathbb{R}$, the \textbf{Gradient Vector} $\nabla f(\mathbf{x})$ is the column vector composed of all $n$ first-order partial derivatives:
\begin{equation}
\nabla f(\mathbf{x}) = \begin{bmatrix} \frac{\partial f}{\partial x_1}(\mathbf{x}) \\ \frac{\partial f}{\partial x_2}(\mathbf{x}) \\ \vdots \\ \frac{\partial f}{\partial x_n}(\mathbf{x}) \end{bmatrix} \in \mathbb{R}^n
\end{equation}
\end{definitionbox}

\begin{definitionbox}{Directional Derivative}
\index{Directional Derivative}
Let $\mathbf{u} \in \mathbb{R}^n$ be a \textbf{unit vector} ($\|\mathbf{u}\| = 1$). The \textbf{directional derivative} of $f$ at point $\mathbf{x}$ in direction $\mathbf{u}$ is:
\begin{equation}
D_{\mathbf{u}} f(\mathbf{x}) = \lim_{h \to 0} \frac{f(\mathbf{x} + h\mathbf{u}) - f(\mathbf{x})}{h}
\end{equation}
For any differentiable function, the directional derivative is calculated as the standard Euclidean inner product of the gradient with the unit direction:
\begin{equation}
D_{\mathbf{u}} f(\mathbf{x}) = \nabla f(\mathbf{x}) \cdot \mathbf{u} = \|\nabla f(\mathbf{x})\| \cos(\theta)
\end{equation}
where $\theta$ is the angle between $\nabla f(\mathbf{x})$ and $\mathbf{u}$.
\end{definitionbox}

\begin{enumerate}
    \item It points in the direction of \textbf{steepest increase} of $f(\mathbf{x})$.
    \item It is strictly \textbf{orthogonal (perpendicular)} to the level curve passing through that point.
\end{enumerate}

\section{Deep Intuition \& Conceptual Derivations}

\subsection{Proof of Gradient Orthogonality to Level Curves}

\begin{theorembox}{Orthogonality of Gradient to Level Sets}
Let $f: \mathbb{R}^n \to \mathbb{R}$ be continuously differentiable. At any point $\mathbf{x}_0$ on the level surface $S = \{\mathbf{x} \in \mathbb{R}^n \mid f(\mathbf{x}) = c\}$ where $\nabla f(\mathbf{x}_0) \neq \mathbf{0}$, the gradient vector $\nabla f(\mathbf{x}_0)$ is \textbf{strictly orthogonal} to the tangent plane of $S$ at $\mathbf{x}_0$.
\end{theorembox}

\begin{proof}
Let $\mathbf{r}: (-\epsilon, \epsilon) \to \mathbb{R}^n$ be any smooth parameter curve lying entirely within the level surface $S$, passing through $\mathbf{x}_0$ at $t = 0$:
\begin{equation*}
\mathbf{r}(0) = \mathbf{x}_0 \quad \text{and} \quad f(\mathbf{r}(t)) = c \quad \forall t \in (-\epsilon, \epsilon)
\end{equation*}
The velocity vector $\mathbf{v} = \mathbf{r}'(0)$ is a tangent vector to the level surface at $\mathbf{x}_0$.
Differentiating the constant identity $f(\mathbf{r}(t)) = c$ with respect to $t$ using the multivariable chain rule:
\begin{equation*}
\frac{d}{dt} \left[ f(\mathbf{r}(t)) \right] = \frac{d}{dt}[c] = 0
\end{equation*}
By the multivariable chain rule:
\begin{equation*}
\sum_{i=1}^n \frac{\partial f}{\partial x_i}(\mathbf{r}(t)) \frac{dx_i}{dt}(t) = \nabla f(\mathbf{r}(t)) \cdot \mathbf{r}'(t) = 0
\end{equation*}
Evaluating at $t = 0$:
\begin{equation}
\nabla f(\mathbf{x}_0) \cdot \mathbf{r}'(0) = 0 \iff \nabla f(\mathbf{x}_0) \perp \mathbf{r}'(0)
\end{equation}
Because this orthogonality holds for every smooth curve passing through $\mathbf{x}_0$ on $S$, the gradient vector $\nabla f(\mathbf{x}_0)$ is perpendicular to every tangent vector, and therefore perpendicular to the entire tangent space of the level set.
\end{proof}

\subsection{Proof of Steepest Ascent and Descent Bounds}

From $D_{\mathbf{u}} f(\mathbf{x}) = \nabla f(\mathbf{x}) \cdot \mathbf{u} = \|\nabla f(\mathbf{x})\| \|\mathbf{u}\| \cos(\theta)$:
Since $\|\mathbf{u}\| = 1$ and $-1 \le \cos(\theta) \le 1$:
\begin{equation}
-\|\nabla f(\mathbf{x})\| \le D_{\mathbf{u}} f(\mathbf{x}) \le \|\nabla f(\mathbf{x})\|
\end{equation}
\begin{itemize}
    \item \textbf{Maximum Rate ($\cos\theta = 1 \implies \theta = 0$):} Occurs when $\mathbf{u} = \frac{\nabla f(\mathbf{x})}{\|\nabla f(\mathbf{x})\|}$. The maximal instantaneous rate of increase is precisely $\|\nabla f(\mathbf{x})\|$.
    \item \textbf{Minimum Rate ($\cos\theta = -1 \implies \theta = \pi$):} Occurs when $\mathbf{u} = -\frac{\nabla f(\mathbf{x})}{\|\nabla f(\mathbf{x})\|}$. The maximal instantaneous rate of decrease is $-\|\nabla f(\mathbf{x})\|$.
    \item \textbf{Zero Rate ($\cos\theta = 0 \implies \theta = \pi/2$):} Moving tangent to the level curve produces zero rate of change ($D_{\mathbf{u}} f = 0$).
\end{itemize}

\section{Explicit Gotchas \& Diagnostic Misconceptions}

\begin{trapbox}{Trap 1: Forgetting to Normalize the Direction Vector}
The formula $D_{\mathbf{v}} f = \nabla f \cdot \mathbf{v}$ is valid \textbf{only when $\mathbf{v}$ is a unit vector} ($\|\mathbf{v}\| = 1$). If given an unnormalized direction vector $\mathbf{v} = [3, 4]^T$, evaluating $\nabla f \cdot \mathbf{v}$ yields an answer that is inflated by $\|\mathbf{v}\| = 5$. Always normalize first: $\mathbf{u} = \frac{\mathbf{v}}{\|\mathbf{v}\|}$.
\end{trapbox}

\begin{trapbox}{Trap 2: Differentiating Inactive Variables into Zero in Product Terms}
When taking $\frac{\partial}{\partial x}(5 x y^2)$, $y^2$ is a multiplicative constant, NOT an additive constant! The derivative is $5 y^2 \cdot \frac{d}{dx}(x) = 5 y^2$. Confusing multiplicative constants with additive constants (e.g., falsely asserting $\frac{\partial}{\partial x}(5 x y^2) = 0$ because $y$ is constant) is one of the most common student calculation errors.
\end{trapbox}

\begin{trapbox}{Trap 3: Believing the Gradient Vector Lives in the Output Space}
If $f: \mathbb{R}^n \to \mathbb{R}$, students often expect the gradient to be a scalar because $f$ outputs a scalar. The gradient $\nabla f(\mathbf{x}) \in \mathbb{R}^n$ lives in the \textbf{domain feature space}. It specifies an $n$-dimensional direction vector along which to nudge the input parameters $\mathbf{x}$.
\end{trapbox}

\begin{trapbox}{Trap 4: Confusing Level Curves with Surface Cross-Sections}
A level curve $f(x, y) = c$ lives entirely in the 2D input plane $\mathbb{R}^2$. A vertical cross-section $z = f(x, y_0)$ is a 2D slice of the 3D surface in $(x, z)$-space. Level curves are the horizontal "isoline" footprints seen from directly above the terrain.
\end{trapbox}

\section{Python / Scientific Computing Lab}

The following Python script computes analytical gradients, verifies them against numerical finite differences, and validates gradient orthogonality to level curves.

\begin{lstlisting}[language=Python, caption={Python Lab: Analytical vs. Numerical Gradients and Directional Derivatives}]
import numpy as np

# 1. Objective Function: f(x, y) = 4*x^2*y - 3*y^2 + 5*x
def f(p):
    x, y = p[0], p[1]
    return 4.0 * x**2 * y - 3.0 * y**2 + 5.0 * x

# Analytical Gradient: [8*x*y + 5, 4*x^2 - 6*y]^T
def grad_f_analytical(p):
    x, y = p[0], p[1]
    return np.array([8.0 * x * y + 5.0, 4.0 * x**2 - 6.0 * y])

# Numerical Gradient via Central Finite Differences
def grad_f_numerical(p, h=1e-6):
    grad = np.zeros_like(p)
    for i in range(len(p)):
        p_plus = p.copy()
        p_minus = p.copy()
        p_plus[i] += h
        p_minus[i] -= h
        grad[i] = (f(p_plus) - f(p_minus)) / (2.0 * h)
    return grad

# Test at Point (1.0, 2.0)
p0 = np.array([1.0, 2.0])
grad_anal = grad_f_analytical(p0)
grad_num = grad_f_numerical(p0)
print(f"Analytical Gradient at (1, 2): {grad_anal}")
print(f"Numerical Gradient at (1, 2):  {grad_num}")
assert np.allclose(grad_anal, grad_num, atol=1e-5), "Gradient verification failed!"

# 2. Directional Derivative along v = [3, 4]^T
v = np.array([3.0, 4.0])
u = v / np.linalg.norm(v)  # Normalized unit vector
# Function from Exercise 3: g(x, y) = x^2 + 3*x*y + y^2 at (2, 1)
grad_g = np.array([2.0*(2.0) + 3.0*(1.0), 3.0*(2.0) + 2.0*(1.0)])  # [7, 8]^T
dir_deriv = np.dot(grad_g, u)
print(f"\nUnit direction u: {u}")
print(f"Gradient of g at (2, 1): {grad_g}")
print(f"Directional Derivative D_u g(2, 1): {dir_deriv:.4f}")

# 3. Orthogonality to Level Curve Tangent
# Level curve of g(x, y) = 11 passing through (2, 1)
# Tangent vector t must satisfy: grad_g . t = 0  ==>  t = [-grad_g[1], grad_g[0]]
t_tangent = np.array([-grad_g[1], grad_g[0]])
t_unit = t_tangent / np.linalg.norm(t_tangent)
dir_along_tangent = np.dot(grad_g, t_unit)
print(f"Rate of change strictly along contour tangent: {dir_along_tangent:.2e}")
assert np.isclose(dir_along_tangent, 0.0), "Orthogonality violated!"
\end{lstlisting}

    \item Determine the maximum instantaneous rate of increase.
    \item Determine the unit vector direction of steepest descent.

\item \textbf{Data Science Application (MSE Loss Gradient):} In linear regression with a single sample observation $(x_i, y_i)$, the squared loss as a function of weight $m$ and bias $b$ is:
\begin{equation*}
L(m, b) = (m x_i + b - y_i)^2
\end{equation*}
Compute the gradient vector $\nabla L(m, b) = \begin{bmatrix} \frac{\partial L}{\partial m} \\ \frac{\partial L}{\partial b} \end{bmatrix}$ using the chain rule.
\end{enumerate}
