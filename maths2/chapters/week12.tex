\chapter{Eigenvalues, Eigenvectors, Diagonalization \& SVD}

Every linear transformation contains privileged coordinate axes: special directions that experience pure scalar stretching or compression without any rotation. These privileged axes are the \index{Eigenvectors}\textbf{Eigenvectors}, and their scaling factors are the \index{Eigenvalues}\textbf{Eigenvalues}. Diagonalizing a matrix decouples complex dynamic interactions into independent 1-dimensional scalings. For rectangular or noisy real-world data matrices, \index{Singular Value Decomposition (SVD)}\textbf{Singular Value Decomposition (SVD)} extends this spectral insight universally, providing the mathematical foundation for \index{Principal Component Analysis (PCA)}\textbf{Principal Component Analysis (PCA)}, low-rank matrix compression, and latent representation learning.

\end{itemize}

\section{Mathematical \& Theoretical Foundations}

\subsection{The Eigenvalue Problem and Characteristic Polynomial}

\begin{definitionbox}{Eigenvalues and Eigenvectors}
\index{Eigenvalue} \index{Eigenvector}
Let $A \in \mathbb{R}^{n \times n}$ be a square matrix. A non-zero vector $\mathbf{v} \in \mathbb{R}^n$ ($\mathbf{v} \neq \mathbf{0}$) is an \textbf{eigenvector} of $A$ corresponding to \textbf{eigenvalue} $\lambda \in \mathbb{C}$ if and only if:
\begin{equation}
A \mathbf{v} = \lambda \mathbf{v} \iff (A - \lambda I) \mathbf{v} = \mathbf{0}
\end{equation}
The set of all solutions to $(A - \lambda I)\mathbf{v} = \mathbf{0}$ is the \textbf{eigenspace} $E_\lambda = \operatorname{Null}(A - \lambda I)$.
\end{definitionbox}

\begin{theorembox}{The Characteristic Equation and Trace-Determinant Invariants}
Non-trivial solutions ($\mathbf{v} \neq \mathbf{0}$) exist if and only if $(A - \lambda I)$ is singular:
\begin{equation}
p(\lambda) = \det(A - \lambda I) = 0
\end{equation}
The roots of the degree-$n$ polynomial $p(\lambda)$ are the eigenvalues $\lambda_1, \lambda_2, \dots, \lambda_n$. They satisfy two universal algebraic invariants:
\begin{enumerate}
    \item \textbf{Sum of Eigenvalues equals Trace:} $\sum_{i=1}^n \lambda_i = \trace(A)$
    \item \textbf{Product of Eigenvalues equals Determinant:} $\prod_{i=1}^n \lambda_i = \det(A)$
\end{enumerate}
\end{theorembox}

\subsection{Matrix Diagonalization}

\begin{definitionbox}{Diagonalization Theorem}
\index{Diagonalization}
An $n \times n$ matrix $A$ is \textbf{diagonalizable} if and only if it possesses $n$ linearly independent eigenvectors $\{\mathbf{v}_1, \dots, \mathbf{v}_n\}$.
Assembling the eigenvectors into column matrix $P = [\mathbf{v}_1 \; \dots \; \mathbf{v}_n]$ and the corresponding eigenvalues into diagonal matrix $D = \operatorname{diag}(\lambda_1, \dots, \lambda_n)$ yields:
\begin{equation}
A = P D P^{-1} \iff D = P^{-1} A P
\end{equation}
\textbf{Computational Superpower (Matrix Powers):} For any integer $k \ge 1$:
\begin{equation}
A^k = (P D P^{-1})(P D P^{-1}) \dots (P D P^{-1}) = P D^k P^{-1}
\end{equation}
where $D^k = \operatorname{diag}(\lambda_1^k, \dots, \lambda_n^k)$.
\end{definitionbox}

\begin{theorembox}{The Spectral Theorem for Real Symmetric Matrices}
\index{Spectral Theorem}
If $A \in \mathbb{R}^{n \times n}$ is real symmetric ($A^T = A$), then:
\begin{enumerate}
    \item All $n$ eigenvalues of $A$ are strictly \textbf{real numbers}.
    \item Eigenvectors corresponding to distinct eigenvalues are \textbf{mutually orthogonal}.
    \item $A$ is \textbf{orthogonally diagonalizable}: there exists an orthogonal matrix $Q \in \mathbb{R}^{n \times n}$ ($Q^T Q = I$) such that:
    \begin{equation}
    A = Q D Q^T = \sum_{i=1}^n \lambda_i \mathbf{q}_i \mathbf{q}_i^T
    \end{equation}
\end{enumerate}
\end{theorembox}

\subsection{Singular Value Decomposition (SVD)}

\begin{definitionbox}{Singular Value Decomposition ($A = U \Sigma V^T$)}
\index{Singular Value Decomposition}
Let $A \in \mathbb{R}^{m \times n}$ be an arbitrary rectangular matrix with $\rank(A) = r \le \min(m, n)$. There exist orthogonal matrices $U \in \mathbb{R}^{m \times m}$ and $V \in \mathbb{R}^{n \times n}$ such that:
\begin{equation}
A = U \Sigma V^T
\end{equation}
where:
\begin{itemize}
    \item $U = [\mathbf{u}_1 \; \dots \; \mathbf{u}_m] \in \mathbb{R}^{m \times m}$ are the \textbf{left singular vectors} (orthonormal eigenvectors of $A A^T$).
    \item $V = [\mathbf{v}_1 \; \dots \; \mathbf{v}_n] \in \mathbb{R}^{n \times n}$ are the \textbf{right singular vectors} (orthonormal eigenvectors of $A^T A$).
    \item $\Sigma \in \mathbb{R}^{m \times n}$ is a rectangular diagonal matrix containing the \textbf{singular values} sorted in descending order:
    \begin{equation}
    \sigma_1 \ge \sigma_2 \ge \dots \ge \sigma_r > \sigma_{r+1} = \dots = 0, \quad \sigma_i = \sqrt{\lambda_i(A^T A)}
    \end{equation}
\end{itemize}
\end{definitionbox}

\begin{theorembox}{Eckart-Young-Mirsky Low-Rank Approximation Theorem}
\index{Low-Rank Approximation}
The optimal rank-$k$ ($k < r$) approximation to $A$ under both the Frobenius norm and spectral norm is obtained by truncating the SVD to its top $k$ terms:
\begin{equation}
A_k = \sum_{i=1}^k \sigma_i \mathbf{u}_i \mathbf{v}_i^T = U_k \Sigma_k V_k^T
\end{equation}
The approximation error is precisely the sum of neglected singular values:
\begin{equation}
\|A - A_k\|_2 = \sigma_{k+1}, \quad \|A - A_k\|_F^2 = \sum_{j=k+1}^r \sigma_j^2
\end{equation}
\end{theorembox}

\section{Deep Intuition \& Conceptual Derivations}

\subsection{Orthogonality of Symmetric Matrix Eigenvectors}

Why are the eigenvectors of every real symmetric matrix strictly orthogonal?
\begin{proof}
Let $A \in \mathbb{R}^{n \times n}$ be real symmetric ($A^T = A$). Let $\mathbf{v}_1, \mathbf{v}_2$ be eigenvectors corresponding to distinct eigenvalues $\lambda_1 \neq \lambda_2$:
\begin{equation*}
A \mathbf{v}_1 = \lambda_1 \mathbf{v}_1 \quad \text{and} \quad A \mathbf{v}_2 = \lambda_2 \mathbf{v}_2
\end{equation*}
Consider the inner product $\langle A\mathbf{v}_1, \mathbf{v}_2 \rangle$. Using the definition of matrix transpose:
\begin{equation*}
\langle A\mathbf{v}_1, \mathbf{v}_2 \rangle = (A\mathbf{v}_1)^T \mathbf{v}_2 = \mathbf{v}_1^T A^T \mathbf{v}_2 = \mathbf{v}_1^T (A\mathbf{v}_2) = \langle \mathbf{v}_1, A\mathbf{v}_2 \rangle
\end{equation*}
Now evaluate both sides using the eigenvalue equations:
\begin{align*}
\langle A\mathbf{v}_1, \mathbf{v}_2 \rangle &= \langle \lambda_1 \mathbf{v}_1, \mathbf{v}_2 \rangle = \lambda_1 \langle \mathbf{v}_1, \mathbf{v}_2 \rangle \\
\langle \mathbf{v}_1, A\mathbf{v}_2 \rangle &= \langle \mathbf{v}_1, \lambda_2 \mathbf{v}_2 \rangle = \lambda_2 \langle \mathbf{v}_1, \mathbf{v}_2 \rangle
\end{align*}
Equating the two expressions:
\begin{equation}
\lambda_1 \langle \mathbf{v}_1, \mathbf{v}_2 \rangle = \lambda_2 \langle \mathbf{v}_1, \mathbf{v}_2 \rangle \implies (\lambda_1 - \lambda_2) \langle \mathbf{v}_1, \mathbf{v}_2 \rangle = 0
\end{equation}
Since $\lambda_1 \neq \lambda_2$ by assumption, $\lambda_1 - \lambda_2 \neq 0$. Dividing by $(\lambda_1 - \lambda_2)$ yields:
\begin{equation}
\langle \mathbf{v}_1, \mathbf{v}_2 \rangle = 0 \iff \mathbf{v}_1 \perp \mathbf{v}_2
\end{equation}
Thus, eigenvectors corresponding to distinct eigenvalues are strictly perpendicular.
\end{proof}

\subsection{Connection between SVD and Principal Component Analysis (PCA)}

Let $X \in \mathbb{R}^{N \times d}$ be a centered data matrix ($\frac{1}{N}\sum \mathbf{x}_i = \mathbf{0}$).
The sample covariance matrix is:
\begin{equation}
C = \frac{1}{N} X^T X \in \mathbb{R}^{d \times d}
\end{equation}
Computing the SVD of $X$:
\begin{equation}
X = U \Sigma V^T \implies X^T X = (V \Sigma^T U^T)(U \Sigma V^T) = V (\Sigma^T \Sigma) V^T
\end{equation}
Therefore:
\begin{equation}
C = \frac{1}{N} V \Sigma^2 V^T = V \left( \frac{\Sigma^2}{N} \right) V^T
\end{equation}
This establishes three profound mathematical identities:
\begin{enumerate}
    \item The \textbf{right singular vectors $V$} of $X$ are precisely the \textbf{principal component directions} (eigenvectors of $C$).
    \item The dataset variances along each principal axis are $\lambda_i(C) = \frac{\sigma_i^2}{N}$.
    \item The low-dimensional projected coordinates of the data are given by the \textbf{principal scores} $Z = X V = U \Sigma$.
\end{enumerate}

\section{Explicit Gotchas \& Diagnostic Misconceptions}

\begin{trapbox}{Trap 1: Assuming Every Square Matrix Can Be Diagonalized}
A matrix is diagonalizable if and only if the geometric multiplicity of each eigenvalue equals its algebraic multiplicity. If an eigenvalue $\lambda$ is repeated but yields fewer independent eigenvectors than its multiplicity, the matrix is \textbf{defective}. For example, $A = \begin{bmatrix} 3 & 1 \\ 0 & 3 \end{bmatrix}$ has $\lambda = 3$ (multiplicity 2), but $\nullity(A - 3I) = 1$. It possesses only 1 independent eigenvector and cannot be diagonalized!
\end{trapbox}

\begin{trapbox}{Trap 2: Believing Singular Values Can Be Negative}
Unlike eigenvalues (which can be positive, negative, or complex), singular values are \textbf{strictly non-negative real numbers} by definition ($\sigma_i = \sqrt{\lambda_i(A^T A)} \ge 0$). Because $A^T A$ is always Symmetric Positive Semi-Definite, its eigenvalues are guaranteed to be non-negative reals ($\lambda_i \ge 0$), making their square roots strictly real and non-negative.
\end{trapbox}

\begin{trapbox}{Trap 3: Scrambling the Order of Columns in $P$ and Eigenvalues in $D$}
In the diagonalization $A = P D P^{-1}$, the sequence of eigenvector columns in $P = [\mathbf{v}_1, \dots, \mathbf{v}_n]$ must \textbf{strictly match} the sequence of eigenvalues along the diagonal of $D = \operatorname{diag}(\lambda_1, \dots, \lambda_n)$. Putting $\lambda_2$ in the first diagonal entry while placing $\mathbf{v}_1$ in the first column destroys the equality $A P = P D$.
\end{trapbox}

\begin{trapbox}{Trap 4: Confusing Left Singular Vectors with Right Singular Vectors}
For a non-square matrix $A \in \mathbb{R}^{m \times n}$:
\begin{itemize}
    \item The left singular vectors $U$ reside in the codomain $\mathbb{R}^m$ (eigenvectors of $A A^T \in \mathbb{R}^{m \times m}$).
    \item The right singular vectors $V$ reside in the domain $\mathbb{R}^n$ (eigenvectors of $A^T A \in \mathbb{R}^{n \times n}$).
\end{itemize}
They have completely different dimensions when $m \neq n$ and must never be interchanged.
\end{trapbox}

\section{Python / Scientific Computing Lab}

The following scientific computing script demonstrates complete analytical and numerical eigen-decomposition, matrix power acceleration, full SVD, and PCA variance calculation.

\begin{lstlisting}[language=Python, caption={Python Lab: Eigen-decomposition, SVD, and PCA Variance Explained}]
import numpy as np

# 1. Eigen-decomposition and Rapid Matrix Power
A = np.array([[4.0, 1.0],
              [2.0, 3.0]])

eigenvalues, P = np.linalg.eig(A)
D = np.diag(eigenvalues)
P_inv = np.linalg.inv(P)

print("Matrix A:\n", A)
print(f"Eigenvalues: {eigenvalues}")
print("Eigenvector Matrix P:\n", P)

# Verification A = P @ D @ P_inv
A_reconstructed = P @ D @ P_inv
assert np.allclose(A, A_reconstructed), "Diagonalization failed!"

# Rapid 4th power: A^4 = P @ (D^4) @ P_inv
D4 = np.diag(eigenvalues**4)
A4_fast = P @ D4 @ P_inv
A4_direct = np.linalg.matrix_power(A.astype(int), 4)
print("\nA^4 via Diagonalization:\n", np.round(A4_fast, 2))
print("A^4 via Direct Multiplication:\n", A4_direct)
assert np.allclose(A4_fast, A4_direct)

# 2. Orthogonal Diagonalization of Symmetric Matrix
S = np.array([[3.0, 1.0],
              [1.0, 3.0]])
eig_S, Q = np.linalg.eigh(S)  # eigh is specialized for symmetric matrices
print("\nSymmetric Matrix S Eigenvalues:", eig_S)
print("Orthogonal Matrix Q (Q^T @ Q == I):\n", np.round(Q.T @ Q, 4))
assert np.allclose(Q.T @ Q, np.eye(2))

# 3. Singular Value Decomposition (SVD) on Rectangular Matrix
X = np.array([[3.0, 0.0],
              [0.0, -2.0]])
U, s, Vt = np.linalg.svd(X)
print("\nRectangular SVD:")
print("  Singular Values (strictly non-negative):", s)
print("  Left Singular Vectors U:\n", U)
print("  Right Singular Vectors V^T:\n", Vt)

# 4. PCA Variance Explained
# Given covariance eigenvalues: [80.0, 15.0, 5.0]
cov_eigenvals = np.array([80.0, 15.0, 5.0])
total_variance = np.sum(cov_eigenvals)
var_explained_2pc = np.sum(cov_eigenvals[:2]) / total_variance * 100.0
print(f"\nPCA Total Dataset Variance: {total_variance}")
print(f"Variance Explained by Top 2 Principal Components: {var_explained_2pc:.2f}%")
assert np.isclose(var_explained_2pc, 95.0)
\end{lstlisting}
