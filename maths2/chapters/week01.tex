\section{Mathematical \& Theoretical Foundations}

Linear algebra provides the mathematical language for multi-dimensional data representation, linear mappings, and structural transformations in data science and machine learning.

\subsection{Vectors in $\R^n$ and Inner Product Spaces}

\begin{definitionbox}{Vector in $\R^n$}
An $n$-dimensional real vector $\mathbf{v} \in \R^n$ is an ordered $n$-tuple of real numbers, conventionally represented as an $n \times 1$ column matrix:
\begin{equation}
\mathbf{v} = \begin{bmatrix} v_1 \\ v_2 \\ \vdots \\ v_n \end{bmatrix} \in \R^n
\end{equation}
The vector space $\R^n$ is equipped with two fundamental operations:
\begin{enumerate}
    \item \textbf{Vector Addition:} For $\mathbf{u}, \mathbf{v} \in \R^n$, $\mathbf{u} + \mathbf{v} = [u_1 + v_1, u_2 + v_2, \dots, u_n + v_n]^T$.
    \item \textbf{Scalar Multiplication:} For $c \in \R$, $c\mathbf{v} = [c v_1, c v_2, \dots, c v_n]^T$.
\end{enumerate}
\end{definitionbox}

\begin{definitionbox}{Dot Product, Norm, and Angle}
The standard \textbf{inner product} (or \textbf{dot product}) of two vectors $\mathbf{u}, \mathbf{v} \in \R^n$ is the scalar:
\begin{equation}
\mathbf{u} \cdot \mathbf{v} = \mathbf{u}^T \mathbf{v} = \sum_{i=1}^n u_i v_i = u_1 v_1 + u_2 v_2 + \dots + u_n v_n
\end{equation}
\begin{itemize}
    \item \textbf{Euclidean Norm ($L_2$ Norm):} The length or magnitude of vector $\mathbf{u}$ is:
    \begin{equation}
    \|\mathbf{u}\| = \sqrt{\mathbf{u} \cdot \mathbf{u}} = \sqrt{\sum_{i=1}^n u_i^2}
    \end{equation}
    \item \textbf{Cauchy-Schwarz Inequality:} For all $\mathbf{u}, \mathbf{v} \in \R^n$:
    \begin{equation}
    |\mathbf{u} \cdot \mathbf{v}| \le \|\mathbf{u}\| \|\mathbf{v}\|
    \end{equation}
    with equality if and only if $\mathbf{u}$ and $\mathbf{v}$ are linearly collinear ($\mathbf{u} = c \mathbf{v}$).
    \item \textbf{Angle and Cosine Similarity:} The angle $\theta \in [0, \pi]$ between non-zero vectors $\mathbf{u}$ and $\mathbf{v}$ satisfies:
    \begin{equation}
    \cos \theta = \frac{\mathbf{u} \cdot \mathbf{v}}{\|\mathbf{u}\| \|\mathbf{v}\|}
    \end{equation}
    \item \textbf{Orthogonality:} $\mathbf{u}$ and $\mathbf{v}$ are \textbf{orthogonal} ($\mathbf{u} \perp \mathbf{v}$) if and only if $\mathbf{u} \cdot \mathbf{v} = 0$.
\end{itemize}
\end{definitionbox}

\subsection{Matrix Algebra and Transposition}

A matrix $A \in \R^{m \times n}$ is a rectangular array with $m$ rows and $n$ columns.

\begin{definitionbox}{Matrix Multiplication}
Let $A \in \R^{m \times k}$ and $B \in \R^{k \times n}$. The product $C = AB \in \R^{m \times n}$ is defined entrywise by:
\begin{equation}
c_{ij} = \sum_{p=1}^k a_{ip} b_{pj} = \text{row}_i(A) \cdot \text{col}_j(B)
\end{equation}
\textbf{Crucial Algebraic Properties:}
\begin{itemize}
    \item \textbf{Associativity:} $(AB)C = A(BC)$
    \item \textbf{Distributivity:} $A(B + C) = AB + AC$
    \item \textbf{Non-Commutativity:} In general, $AB \neq BA$. Even if $A$ and $B$ are both square of size $n \times n$, $AB$ and $BA$ are typically unequal.
\end{itemize}
\end{definitionbox}

\begin{theorembox}{The Transpose and Its Reversal Property}
The \textbf{transpose} of $A \in \R^{m \times n}$, denoted $A^T \in \R^{n \times m}$, reflects entries across the main diagonal: $(A^T)_{ij} = a_{ji}$.
\begin{itemize}
    \item $(A + B)^T = A^T + B^T$
    \item $(cA)^T = c A^T$
    \item $(A^T)^T = A$
    \item \textbf{Socks-and-Shoes Reversal Property:}
    \begin{equation}
    (AB)^T = B^T A^T
    \end{equation}
    For $k$ matrices: $(A_1 A_2 \dots A_k)^T = A_k^T \dots A_2^T A_1^T$.
\end{itemize}
\end{theorembox}

\subsection{The Duality of Linear Systems: Row vs. Column Picture}

A system of $m$ linear equations in $n$ variables $x_1, \dots, x_n$ is represented in matrix notation as:
\begin{equation}
A \mathbf{x} = \mathbf{b} \iff \begin{bmatrix} a_{11} & a_{12} & \dots & a_{1n} \\ a_{21} & a_{22} & \dots & a_{2n} \\ \vdots & \vdots & \ddots & \vdots \\ a_{m1} & a_{m2} & \dots & a_{mn} \end{bmatrix} \begin{bmatrix} x_1 \\ x_2 \\ \vdots \\ x_n \end{bmatrix} = \begin{bmatrix} b_1 \\ b_2 \\ \vdots \\ b_m \end{bmatrix}
\end{equation}
Linear systems possess two complementary geometric interpretations:
\begin{enumerate}
    \item \textbf{The Row Picture:} Each equation represents an $(n-1)$-dimensional affine hyperplane in $\R^n$. The solution vector $\mathbf{x}^*$ is the geometric intersection point of all $m$ hyperplanes.
    \item \textbf{The Column Picture:} The matrix-vector product $A\mathbf{x}$ is viewed as a \textbf{linear combination of the column vectors} of $A$:
    \begin{equation}
    x_1 \begin{bmatrix} a_{11} \\ a_{21} \\ \vdots \\ a_{m1} \end{bmatrix} + x_2 \begin{bmatrix} a_{12} \\ a_{22} \\ \vdots \\ a_{m2} \end{bmatrix} + \dots + x_n \begin{bmatrix} a_{1n} \\ a_{2n} \\ \vdots \\ a_{mn} \end{bmatrix} = \mathbf{b}
    \end{equation}
    Solving $A\mathbf{x} = \mathbf{b}$ asks: \textit{"What combination of the column vectors of $A$ reconstructs the target vector $\mathbf{b}$?"} A solution exists if and only if $\mathbf{b}$ lies in the \textbf{span of the columns} of $A$ (the Column Space $\operatorname{Col}(A)$).
\end{enumerate}

\subsection{Determinants as Oriented Volume Distortion}

The \textbf{determinant} is a multilinear, alternating function mapping square matrices $A \in \R^{n \times n}$ to a scalar $\det(A) \in \R$.

\begin{definitionbox}{Laplace Cofactor Expansion}
For any $n \times n$ matrix $A$, the determinant can be expanded along any row $i$:
\begin{equation}
\det(A) = \sum_{j=1}^n (-1)^{i+j} a_{ij} \det(M_{ij}) = \sum_{j=1}^n a_{ij} C_{ij}
\end{equation}
where $M_{ij}$ is the $(n-1) \times (n-1)$ submatrix formed by deleting row $i$ and column $j$, and $C_{ij} = (-1)^{i+j} \det(M_{ij})$ is the $(i, j)$-th \textbf{cofactor}.
\begin{itemize}
    \item For $2 \times 2$: $\det \begin{bmatrix} a & b \\ c & d \end{bmatrix} = ad - bc$.
    \item For $3 \times 3$: $\det \begin{bmatrix} a & b & c \\ d & e & f \\ g & h & i \end{bmatrix} = a(ei - fh) - b(di - fg) + c(dh - eg)$.
\end{itemize}
\end{definitionbox}

\begin{theorembox}{Essential Determinant Properties}
\begin{itemize}
    \item $\det(I) = 1$
    \item $\det(A^T) = \det(A)$
    \item \textbf{Multiplicative Property:} $\det(AB) = \det(A) \det(B)$
    \item \textbf{Scalar Multiple:} $\det(c A) = c^n \det(A)$ for $A \in \R^{n \times n}$ and $c \in \R$.
    \item \textbf{Invertibility Theorem:} Matrix $A$ is invertible ($A^{-1}$ exists) if and only if $\det(A) \neq 0$. If $A$ is invertible, $\det(A^{-1}) = \frac{1}{\det(A)}$.
    \item If two rows or columns of $A$ are identical, or if a row/column is all zeros, $\det(A) = 0$.
\end{itemize}
\end{theorembox}

\section{Deep Intuition \& Conceptual Derivations}

\subsection{Proof of the Socks-and-Shoes Transpose Identity: $(AB)^T = B^T A^T$}
Let $A \in \R^{m \times k}$ and $B \in \R^{k \times n}$. The product $C = AB$ has dimensions $m \times n$.
The $(i, j)$-th entry of the transpose $C^T$ is by definition the $(j, i)$-th entry of $C$:
\begin{equation}
( (AB)^T )_{ij} = (AB)_{ji} = \sum_{p=1}^k a_{jp} b_{pi}
\end{equation}
Since real scalar multiplication commutes ($a_{jp} b_{pi} = b_{pi} a_{jp}$):
\begin{equation}
( (AB)^T )_{ij} = \sum_{p=1}^k b_{pi} a_{jp}
\end{equation}
Now, consider the product $D = B^T A^T$. 
$B^T \in \R^{n \times k}$ with entries $(B^T)_{ip} = b_{pi}$, and $A^T \in \R^{k \times m}$ with entries $(A^T)_{pj} = a_{jp}$.
Evaluating entry $(i, j)$ of $D$:
\begin{equation}
(B^T A^T)_{ij} = \sum_{p=1}^k (B^T)_{ip} (A^T)_{pj} = \sum_{p=1}^k b_{pi} a_{jp}
\end{equation}
Both expressions are identical for all $1 \le i \le n$ and $1 \le j \le m$. Thus:
\begin{equation}
(AB)^T = B^T A^T \quad \blacksquare
\end{equation}

\subsection{Geometric Derivation of the $2 \times 2$ Determinant as Signed Area}
Consider two column vectors $\mathbf{v}_1 = \begin{bmatrix} a \\ c \end{bmatrix}$ and $\mathbf{v}_2 = \begin{bmatrix} b \\ d \end{bmatrix}$ in $\R^2$. 
The parallelogram spanned by $\mathbf{v}_1$ and $\mathbf{v}_2$ is enclosed inside a bounding rectangle with base $(a + b)$ and height $(c + d)$.
The total area of the bounding rectangle is:
\begin{equation}
A_{\text{rect}} = (a + b)(c + d) = ac + ad + bc + bd
\end{equation}
The excess region outside the parallelogram consists of:
\begin{enumerate}
    \item Two identical right triangles of base $a$ and height $c$: $2 \times \frac{1}{2} ac = ac$.
    \item Two identical right triangles of base $b$ and height $d$: $2 \times \frac{1}{2} bd = bd$.
    \item Two identical corner rectangles of width $b$ and height $c$: $2 \times (bc) = 2bc$.
\end{enumerate}
Subtracting the excess areas from the bounding rectangle:
\begin{equation}
A_{\text{parallelogram}} = (ac + ad + bc + bd) - (ac + bd + 2bc) = ad - bc
\end{equation}
This establishes why $\det \begin{bmatrix} a & b \\ c & d \end{bmatrix} = ad - bc$ measures the geometric area of the spanned 2D surface $\blacksquare$.

\section{Explicit Gotchas \& Diagnostic Misconceptions}

\begin{trapbox}{Trap 1: Assuming Matrix Commutativity ($AB = BA$)}
Unlike elementary arithmetic, matrix multiplication is \textbf{strictly non-commutative}. For almost all matrices, $AB \neq BA$. In fact, if $A \in \R^{3 \times 2}$ and $B \in \R^{2 \times 3}$, $AB \in \R^{3 \times 3}$, while $BA \in \R^{2 \times 2}$---they do not even share the same dimensions! Even for square matrices, $AB$ and $BA$ represent different sequences of transformations.
\end{trapbox}

\begin{trapbox}{Trap 2: Transposition Without Order Reversal}
Students often incorrectly distribute transposes across products as $(AB)^T = A^T B^T$. 
If $A \in \R^{3 \times 4}$ and $B \in \R^{4 \times 2}$, then $AB \in \R^{3 \times 2}$, so $(AB)^T \in \R^{2 \times 3}$. 
Evaluating $A^T B^T$: $A^T \in \R^{4 \times 3}$ and $B^T \in \R^{2 \times 4}$. The inner dimensions (3 and 2) do not match, so $A^T B^T$ is \textbf{mathematically undefined}! The transpose must invert the order: $(AB)^T = B^T A^T$.
\end{trapbox}

\begin{trapbox}{Trap 3: Zero Determinant Means "No Solution"}
Students frequently believe that if $\det(A) = 0$, the system $A\mathbf{x} = \mathbf{b}$ has no solutions. 
This is false: $\det(A) = 0$ means the system does not have a \textit{unique} solution. It may have \textbf{infinitely many solutions} (if $\mathbf{b}$ lies within the column space $\operatorname{Col}(A)$) or \textbf{no solutions} (if $\mathbf{b}$ lies outside $\operatorname{Col}(A)$).
\end{trapbox}

\begin{trapbox}{Trap 4: Conflating Inner (Dot) Product with Outer Product}
For vectors $\mathbf{u}, \mathbf{v} \in \R^n$:
\begin{itemize}
    \item \textbf{Inner (Dot) Product:} $\mathbf{u}^T \mathbf{v}$ multiplies a $1 \times n$ row by an $n \times 1$ column, resulting in a \textbf{scalar} ($1 \times 1$).
    \item \textbf{Outer Product:} $\mathbf{u} \mathbf{v}^T$ multiplies an $n \times 1$ column by a $1 \times n$ row, resulting in an \textbf{$n \times n$ rank-1 matrix}.
\end{itemize}
Confusing the order of transposition turns scalars into matrices and vice versa.
\end{trapbox}

\section{Python Scientific Computing Lab}

The following Python laboratory demonstrates vector norms, cosine similarities, matrix operations, and determinant evaluations using NumPy.

\begin{lstlisting}[language=Python, caption=Vector and Matrix Operations in NumPy]
import numpy as np

# 1. Vector Dot Product, Norms, and Cosine Similarity
u = np.array([2.0, 3.0, -1.0])
v = np.array([1.0, -2.0, 4.0])

dot_prod = np.dot(u, v)
norm_u = np.linalg.norm(u)
norm_v = np.linalg.norm(v)
cos_sim = dot_prod / (norm_u * norm_v)

print(f"Dot Product u . v: {dot_prod:.4f}")
print(f"Norm ||u||: {norm_u:.4f}, Norm ||v||: {norm_v:.4f}")
print(f"Cosine Similarity: {cos_sim:.4f}")

# 2. Matrix Multiplication and Non-Commutativity
A = np.array([[1, 2], [3, 4]])
B = np.array([[0, -1], [2, 5]])

AB = A @ B
BA = B @ A
print("\nProduct AB:\n", AB)
print("Product BA:\n", BA)
print("Is AB == BA?", np.array_equal(AB, BA))

# 3. Transpose Identity: (AB)^T == B^T A^T
AB_T = (A @ B).T
BT_AT = B.T @ A.T
print("\n(AB)^T == B^T A^T:", np.allclose(AB_T, BT_AT))

# 4. Determinant and Matrix Invertibility
M = np.array([[3.0, 1.0, 2.0],
              [0.0, 4.0, 5.0],
              [1.0, 2.0, 1.0]])

det_M = np.linalg.det(M)
print(f"\nDeterminant of M: {det_M:.4f}")
if not np.isclose(det_M, 0.0):
    M_inv = np.linalg.inv(M)
    print("M is invertible. Inverse M^(-1):\n", np.round(M_inv, 4))
    print("M @ M^(-1) == I:\n", np.round(M @ M_inv, 4))

# 5. Data Science: Covariance Matrix Generalized Variance
Sigma = np.array([[4.0, 3.0],
                  [3.0, 9.0]])
gen_var = np.linalg.det(Sigma)
print(f"\nGeneralized Variance |Sigma|: {gen_var:.4f}")
\end{lstlisting}

\subsection*{Part A: Analytical \& Computational Problems}

\begin{enumerate}
    \item \textbf{Vector Algebra, Norms, and Orthogonality:}
    Given vectors $\mathbf{u} = \begin{bmatrix} 3 \\ -2 \\ 4 \end{bmatrix}$ and $\mathbf{v} = \begin{bmatrix} 2 \\ 5 \\ 1 \end{bmatrix}$:
    \begin{enumerate}
        \item Compute the dot product $\mathbf{u} \cdot \mathbf{v}$.
        \item Compute the Euclidean norms $\|\mathbf{u}\|$ and $\|\mathbf{v}\|$.
        \item Calculate the cosine of the angle between $\mathbf{u}$ and $\mathbf{v}$. Are $\mathbf{u}$ and $\mathbf{v}$ orthogonal?
        \item Find the outer product matrix $P = \mathbf{u} \mathbf{v}^T$.
    \end{enumerate}

    \item \textbf{Matrix Multiplication and Transpose Identities:}
    Let matrices $A = \begin{bmatrix} 2 & 1 \\ -1 & 3 \end{bmatrix}$ and $B = \begin{bmatrix} 4 & -2 \\ 1 & 0 \end{bmatrix}$.
    \begin{enumerate}
        \item Compute the matrix product $AB$.
        \item Compute the matrix product $BA$. State whether $A$ and $B$ commute.
        \item Directly compute $(AB)^T$ and compare it with the product $B^T A^T$.
    \end{enumerate}

    \item \textbf{Linear Systems: Row Picture vs. Column Picture:}
    Consider the system of linear equations:
    \begin{align*}
    x_1 + 2x_2 &= 5 \\
    3x_1 - x_2 &= 1
    \end{align*}
    \begin{enumerate}
        \item Formulate the system in matrix notation $A\mathbf{x} = \mathbf{b}$.
        \item Describe the Row Picture: find the equations and slopes of the two lines in $\R^2$ and their point of intersection.
        \item Describe the Column Picture: express $\mathbf{b} = \begin{bmatrix} 5 \\ 1 \end{bmatrix}$ as an explicit linear combination of the column vectors of $A$.
    \end{enumerate}

    \item \textbf{Determinants and Singularity Conditions:}
    Consider the parameterized matrix:
    \begin{equation}
    M_k = \begin{bmatrix} 2 & 1 & 3 \\ 0 & k & 4 \\ 1 & -1 & 2 \end{bmatrix}
    \end{equation}
    \begin{enumerate}
        \item Calculate the determinant $\det(M_k)$ as a function of the parameter $k$ using cofactor expansion.
        \item For what exact value of $k$ is the matrix $M_k$ singular (non-invertible)?
        \item If $k = 5$, what is the determinant of the inverse matrix $M_5^{-1}$?
    \end{enumerate}

    \item \textbf{Covariance Matrices and Mahalanobis Distance:}
    In a 2-feature machine learning dataset, features have mean vector $\boldsymbol{\mu} = \begin{bmatrix} 0 \\ 0 \end{bmatrix}$ and covariance matrix:
    \begin{equation}
    \Sigma = \begin{bmatrix} 5 & 2 \\ 2 & 2 \end{bmatrix}
    \end{equation}
    \begin{enumerate}
        \item Compute the generalized variance given by $\det(\Sigma)$.
        \item Find the analytical matrix inverse $\Sigma^{-1}$.
        \item For a test observation $\mathbf{x} = \begin{bmatrix} 1 \\ 2 \end{bmatrix}$, calculate the squared Mahalanobis distance $d_M^2 = (\mathbf{x} - \boldsymbol{\mu})^T \Sigma^{-1} (\mathbf{x} - \boldsymbol{\mu})$.
    \end{enumerate}
\end{enumerate}

\subsection*{Part B: Diagnostic Misconception MCQs}

\begin{enumerate}
    \setcounter{enumi}{5}
    \item A student is simplifying the transpose of a sequence of matrix multiplications involving three invertible matrices $A \in \R^{4 \times 3}, B \in \R^{3 \times 5}, C \in \R^{5 \times 2}$. The student claims that $(ABC)^T = A^T B^T C^T$. Which statement correctly evaluates this claim?
    \begin{enumerate}
        \item The claim is correct because transposition is a linear operator that distributes directly across all factors.
        \item The claim is false because transposition reverses the multiplication order: $(ABC)^T = C^T B^T A^T$. In fact, the expression $A^T B^T C^T$ is not even mathematically defined because the inner matrix dimensions do not match.
        \item The claim is false only if one of the matrices has a determinant equal to zero.
        \item The claim is correct provided that $A, B,$ and $C$ are symmetric matrices.
    \end{enumerate}

    \item If a square matrix $A \in \R^{3 \times 3}$ has $\det(A) = 0$, what does this geometrically imply about the linear transformation represented by $A$?
    \begin{enumerate}
        \item The transformation rotates all vectors by $90^\circ$ without altering their length.
        \item The transformation completely collapses 3-dimensional space into a lower-dimensional subspace (a plane, a line, or the origin), destroying volume and making the transformation irreversible.
        \item The transformation scales the volume of any 3D body by a factor of 2.
        \item The transformation has no effect on any vector in $\R^3$.
    \end{enumerate}
\end{enumerate}
