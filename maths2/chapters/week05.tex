\chapter{Null Space, Nullity \& Linear Mappings}

Matrices are not static numerical arrays; they represent dynamic geometric transformations mapping vectors between vector spaces. When a matrix transformation compresses higher-dimensional data, certain non-zero input directions inevitably collapse into the origin. The collection of all such collapsed vectors forms the \index{Null Space}\textbf{Null Space} (or Kernel), while its dimension—the \index{Nullity}\textbf{Nullity}—quantifies the exact loss of geometric degrees of freedom. In this chapter, we formalize \index{Linear Transformations}\textbf{Linear Transformations}, their structural invariants, and the mechanics of the null space.

\section{Mathematical \& Theoretical Foundations}

\subsection{The Null Space (Kernel) of a Matrix}

\begin{definitionbox}{Null Space and Nullity}
\index{Null Space} \index{Nullity}
Let $A \in \mathbb{R}^{m \times n}$ represent a linear transformation from $\mathbb{R}^n$ (domain) to $\mathbb{R}^m$ (codomain). The \textbf{Null Space} (or \textbf{Kernel}) of $A$, denoted $\operatorname{Null}(A)$ or $\ker(A)$, is the set of all input vectors $\mathbf{x} \in \mathbb{R}^n$ mapped to the zero vector $\mathbf{0} \in \mathbb{R}^m$:
\begin{equation}
\operatorname{Null}(A) = \left\{ \mathbf{x} \in \mathbb{R}^n \;\middle|\; A \mathbf{x} = \mathbf{0} \right\}
\end{equation}
The dimension of the null space is called the \textbf{Nullity} of $A$:
\begin{equation}
\nullity(A) = \dim(\operatorname{Null}(A))
\end{equation}
\end{definitionbox}

\begin{theorembox}{Subspace Property of the Null Space}
For any matrix $A \in \mathbb{R}^{m \times n}$, the null space $\operatorname{Null}(A)$ is a valid vector subspace of the \textbf{domain} $\mathbb{R}^n$.
\end{theorembox}

\begin{proof}
We test the three fundamental subspace criteria:
\begin{enumerate}
    \item \textbf{Contains Zero Vector:} $A \mathbf{0} = \mathbf{0}$, hence $\mathbf{0} \in \operatorname{Null}(A)$.
    \item \textbf{Closed under Addition:} Let $\mathbf{u}, \mathbf{v} \in \operatorname{Null}(A)$, so $A\mathbf{u} = \mathbf{0}$ and $A\mathbf{v} = \mathbf{0}$. By the distributive property of matrix multiplication:
    \begin{equation*}
    A(\mathbf{u} + \mathbf{v}) = A\mathbf{u} + A\mathbf{v} = \mathbf{0} + \mathbf{0} = \mathbf{0} \implies \mathbf{u} + \mathbf{v} \in \operatorname{Null}(A)
    \end{equation*}
    \item \textbf{Closed under Scalar Multiplication:} Let $\mathbf{u} \in \operatorname{Null}(A)$ and $c \in \mathbb{R}$. Then:
    \begin{equation*}
    A(c\mathbf{u}) = c(A\mathbf{u}) = c(\mathbf{0}) = \mathbf{0} \implies c\mathbf{u} \in \operatorname{Null}(A)
    \end{equation*}
\end{enumerate}
Thus, $\operatorname{Null}(A)$ is a subspace of $\mathbb{R}^n$.
\end{proof}

\subsection{Linear Transformations and Mappings}

\begin{definitionbox}{Axioms of a Linear Transformation}
\index{Linear Transformation}
Let $V$ and $W$ be vector spaces over $\mathbb{R}$. A mapping $T: V \to W$ is called a \textbf{linear transformation} (or linear operator) if it satisfies two conditions for all $\mathbf{u}, \mathbf{v} \in V$ and all scalars $c \in \mathbb{R}$:
\begin{enumerate}
    \item \textbf{Additivity:} $T(\mathbf{u} + \mathbf{v}) = T(\mathbf{u}) + T(\mathbf{v})$
    \item \textbf{Homogeneity of Degree 1:} $T(c\mathbf{u}) = c\, T(\mathbf{u})$
\end{enumerate}
Equivalently, $T$ preserves arbitrary linear combinations:
\begin{equation}
T(c_1 \mathbf{u} + c_2 \mathbf{v}) = c_1 T(\mathbf{u}) + c_2 T(\mathbf{v})
\end{equation}
\end{definitionbox}

\begin{theorembox}{Fundamental Properties of Linear Transformations}
If $T: V \to W$ is a linear transformation, then:
\begin{enumerate}
    \item $T(\mathbf{0}_V) = \mathbf{0}_W$ (the zero vector of the domain maps strictly to the zero vector of the codomain).
    \item $T(-\mathbf{v}) = -T(\mathbf{v})$ for all $\mathbf{v} \in V$.
    \item $T\left(\sum_{i=1}^k c_i \mathbf{v}_i\right) = \sum_{i=1}^k c_i T(\mathbf{v}_i)$.
\end{enumerate}
\end{theorembox}

\section{Deep Intuition \& Conceptual Derivations}

\subsection{Affine Solution Geometry: $A\mathbf{x} = \mathbf{b}$}

Why does the null space matter when solving inhomogeneous real-world equations $A\mathbf{x} = \mathbf{b}$?
\begin{theorembox}{General Solution Decomposition Theorem}
Let $A \in \mathbb{R}^{m \times n}$ and $\mathbf{b} \in \mathbb{R}^m$. Suppose $\mathbf{x}_p$ is a specific \textbf{particular solution} satisfying $A\mathbf{x}_p = \mathbf{b}$. Then the complete solution set $S$ is the affine space:
\begin{equation}
S = \left\{ \mathbf{x}_p + \mathbf{x}_h \;\middle|\; \mathbf{x}_h \in \operatorname{Null}(A) \right\} = \mathbf{x}_p + \operatorname{Null}(A)
\end{equation}
\end{theorembox}

\begin{proof}
Let $\mathbf{x}$ be any valid solution to $A\mathbf{x} = \mathbf{b}$. Consider the difference vector $\mathbf{x}_h = \mathbf{x} - \mathbf{x}_p$.
Evaluating $A\mathbf{x}_h$:
\begin{equation*}
A\mathbf{x}_h = A(\mathbf{x} - \mathbf{x}_p) = A\mathbf{x} - A\mathbf{x}_p = \mathbf{b} - \mathbf{b} = \mathbf{0}
\end{equation*}
This proves that $\mathbf{x}_h \in \operatorname{Null}(A)$. Therefore, every solution can be expressed as $\mathbf{x} = \mathbf{x}_p + \mathbf{x}_h$.
Conversely, if $\mathbf{x}_h \in \operatorname{Null}(A)$, then $A(\mathbf{x}_p + \mathbf{x}_h) = A\mathbf{x}_p + A\mathbf{x}_h = \mathbf{b} + \mathbf{0} = \mathbf{b}$.
Thus, the solution manifold is a translated copy of $\operatorname{Null}(A)$ shifted away from the origin by $\mathbf{x}_p$.
\end{proof}

\section{Explicit Gotchas \& Diagnostic Misconceptions}

\begin{trapbox}{Trap 1: Confusing Domain Ambient Dimension with Codomain Dimension}
For a non-square matrix $A \in \mathbb{R}^{m \times n}$, the null space resides in the \textbf{domain} $\mathbb{R}^n$, NOT the codomain $\mathbb{R}^m$. For instance, if $A$ is $3 \times 5$, vectors in $\operatorname{Null}(A)$ have length 5, not 3. In contrast, the column space $\operatorname{Col}(A)$ resides in the codomain $\mathbb{R}^3$.
\end{trapbox}

\begin{trapbox}{Trap 2: Believing $T(\mathbf{0}) = \mathbf{0}$ is Sufficient to Prove Linearity}
While every linear transformation must map zero to zero ($T(\mathbf{0}) = \mathbf{0}$), this condition is merely \textbf{necessary, not sufficient}. Functions such as $f(x) = x^2$, $f(x) = |x|$, or $T(x, y) = [x^3, y]^T$ all satisfy $T(\mathbf{0}) = \mathbf{0}$, yet all fail additivity and scalar homogeneity.
\end{trapbox}

\begin{trapbox}{Trap 3: Classifying Affine Mappings as Linear Transformations}
A mapping of the form $T(\mathbf{x}) = A\mathbf{x} + \mathbf{b}$ with constant offset $\mathbf{b} \neq \mathbf{0}$ (e.g., $f(x) = 3x + 5$) is an \textbf{affine transformation}, NOT a linear transformation. Evaluating at zero yields $T(\mathbf{0}) = \mathbf{b} \neq \mathbf{0}$, violating the foundational axiom of vector space linearity.
\end{trapbox}

\begin{trapbox}{Trap 4: Assuming a Trivial Null Space Means Invertibility for Rectangular Matrices}
If a tall rectangular matrix $A \in \mathbb{R}^{10 \times 4}$ has $\operatorname{Null}(A) = \{\mathbf{0}\}$, its columns are linearly independent and the mapping is injective (one-to-one). However, $A$ is \textbf{not invertible} because it is not square; it possesses no two-sided inverse. It maps $\mathbb{R}^4$ onto a 4D subspace of $\mathbb{R}^{10}$, leaving a 6D orthogonal complement unreachable.
\end{trapbox}

\section{Python / Scientific Computing Lab}

The following Python script computes the exact rational basis for the null space using SymPy, verifies the Rank-Nullity Theorem, and demonstrates how non-trivial null space vectors produce zero activations in linear layers.

\begin{lstlisting}[language=Python, caption={Python Lab: Null Space Extraction and Transformation Verification}]
import numpy as np
import sympy as sp
from scipy.linalg import null_space

# 1. Exact Symbolic Null Space Basis via SymPy
A_sym = sp.Matrix([[1, 2, -1],
                   [2, 4,  5]])

null_basis_sym = A_sym.nullspace()
print("Symbolic Null Space Basis:")
for v in null_basis_sym:
    sp.pprint(v)

# 2. Numerical Null Space via SciPy (Orthonormal Basis)
A_num = np.array([[1.0, 2.0, -1.0],
                  [2.0, 4.0,  5.0]])

ns = null_space(A_num)
print(f"\nNumerical Null Space Basis (Shape {ns.shape}):\n{ns}")
print("Verification A @ null_space == 0:")
print(np.allclose(A_num @ ns, 0))

# 3. Rank-Nullity Theorem Check
n_cols = A_num.shape[1]
rank_A = np.linalg.matrix_rank(A_num)
nullity_A = ns.shape[1]
print(f"\nDomain Dim (n): {n_cols}")
print(f"Rank(A): {rank_A}")
print(f"Nullity(A): {nullity_A}")
assert rank_A + nullity_A == n_cols, "Rank-Nullity Theorem Violated!"
print("Rank-Nullity Theorem confirmed: 2 + 1 = 3")

# 4. Neural Network Linear Projection Bottleneck Simulation
# Layer mapping R^50 -> R^10
np.random.seed(42)
W = np.random.randn(10, 50)
ns_W = null_space(W)
print(f"\nLinear bottleneck W (10x50): Nullity = {ns_W.shape[1]} dimensions of information lost.")
# Feeding a vector from the null space produces exactly zero output
x_null = ns_W[:, 0]
y_out = W @ x_null
print(f"Output norm for null space input: {np.linalg.norm(y_out):.2e}")
\end{lstlisting}
