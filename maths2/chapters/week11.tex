\chapter{Hessian Matrix, Multivariable Extrema \& Differentiability}

Finding a point where the gradient vanishes ($\nabla f(\mathbf{x}^*) = \mathbf{0}$) tells us only that the multivariable surface is momentarily flat; it does not reveal whether we stand at the bottom of a valley, the crest of a ridge, or a precarious saddle pass. To resolve the local topography, we must examine the second-order curvature of the surface. This curvature is encoded by the \index{Hessian Matrix}\textbf{Hessian Matrix}. In this chapter, we master \index{Clairaut's Theorem}\textbf{Clairaut's Theorem}, the \index{Second Derivative Test}\textbf{Second Derivative Test via Quadratic Forms}, convexity in machine learning loss functions, and the rigorous mathematical criteria for \index{Multivariable Differentiability}\textbf{Multivariable Differentiability}.

\vspace{0.5cm}

    \item \textbf{Prerequisites:} Multivariable partial derivatives and gradients (Week 9), critical points (Week 10), quadratic forms and positive definiteness (Week 7).
    \item \textbf{Mathematical Notation:} Hessian matrix $H(f) \in \mathbb{R}^{n \times n}$; Hessian determinant $D = \det(H) = f_{xx} f_{yy} - (f_{xy})^2$; quadratic form $\mathbf{h}^T H \mathbf{h}$; positive definite $H \succ 0$; positive semi-definite $H \succeq 0$; differentiability error limit.
\end{itemize}

\section{Mathematical \& Theoretical Foundations}

\subsection{Second-Order Partial Derivatives and Clairaut's Theorem}

\begin{definitionbox}{Higher-Order Partial Derivatives}
For a function of two variables $f(x, y)$, there are four second-order partial derivatives:
\begin{equation}
f_{xx} = \frac{\partial^2 f}{\partial x^2}, \quad f_{yy} = \frac{\partial^2 f}{\partial y^2}, \quad f_{xy} = \frac{\partial}{\partial y}\left(\frac{\partial f}{\partial x}\right), \quad f_{yx} = \frac{\partial}{\partial x}\left(\frac{\partial f}{\partial y}\right)
\end{equation}
\end{definitionbox}

\begin{theorembox}{Clairaut's Theorem on the Equality of Mixed Partials}
\index{Clairaut's Theorem}
Let $f: D \subseteq \mathbb{R}^n \to \mathbb{R}$. If the mixed second-order partial derivatives $\frac{\partial^2 f}{\partial x_i \partial x_j}$ and $\frac{\partial^2 f}{\partial x_j \partial x_i}$ are continuous on an open disk containing point $\mathbf{x}_0$, then they are identically equal:
\begin{equation}
\frac{\partial^2 f}{\partial x_i \partial x_j}(\mathbf{x}_0) = \frac{\partial^2 f}{\partial x_j \partial x_i}(\mathbf{x}_0) \quad \forall i, j
\end{equation}
Consequently, the Hessian matrix of any twice continuously differentiable function ($C^2$) is strictly \textbf{symmetric}: $H(f)^T = H(f)$.
\end{theorembox}

\subsection{The Hessian Matrix and Second-Order Taylor Expansion}

\begin{definitionbox}{The Hessian Matrix $H(f)$}
\index{Hessian Matrix}
For a scalar function $f: \mathbb{R}^n \to \mathbb{R}$, the \textbf{Hessian matrix} $H(f)(\mathbf{x})$ is the $n \times n$ matrix of all second partial derivatives:
\begin{equation}
H(f)(\mathbf{x}) = \begin{bmatrix}
\frac{\partial^2 f}{\partial x_1^2} & \frac{\partial^2 f}{\partial x_1 \partial x_2} & \dots & \frac{\partial^2 f}{\partial x_1 \partial x_n} \\
\frac{\partial^2 f}{\partial x_2 \partial x_1} & \frac{\partial^2 f}{\partial x_2^2} & \dots & \frac{\partial^2 f}{\partial x_2 \partial x_n} \\
\vdots & \vdots & \ddots & \vdots \\
\frac{\partial^2 f}{\partial x_n \partial x_1} & \frac{\partial^2 f}{\partial x_n \partial x_2} & \dots & \frac{\partial^2 f}{\partial x_n^2}
\end{bmatrix} \in \mathbb{R}^{n \times n}
\end{equation}
\end{definitionbox}

\begin{theorembox}{Multivariable Second-Order Taylor Formula}
\index{Taylor Expansion}
Let $f: \mathbb{R}^n \to \mathbb{R}$ be twice continuously differentiable. Near a critical point $\mathbf{x}^*$ where $\nabla f(\mathbf{x}^*) = \mathbf{0}$, the deviation of $f$ along perturbation vector $\mathbf{h} \in \mathbb{R}^n$ is governed by the Hessian quadratic form:
\begin{equation}
f(\mathbf{x}^* + \mathbf{h}) = f(\mathbf{x}^*) + \underbrace{\nabla f(\mathbf{x}^*)^T \mathbf{h}}_{= 0} + \frac{1}{2} \mathbf{h}^T H(f)(\mathbf{x}^*) \mathbf{h} + \mathcal{O}(\|\mathbf{h}\|^3)
\end{equation}
\end{theorembox}

\subsection{The Second Derivative Test for Multivariable Extrema}

\begin{theorembox}{General Eigenvalue Classification of Critical Points}
Let $\mathbf{x}^*$ be a critical point of $f: \mathbb{R}^n \to \mathbb{R}$ ($\nabla f(\mathbf{x}^*) = \mathbf{0}$). Let $\lambda_1, \lambda_2, \dots, \lambda_n$ be the real eigenvalues of the symmetric Hessian matrix $H(f)(\mathbf{x}^*)$:
\begin{enumerate}
    \item \textbf{Local Minimum:} All eigenvalues are strictly positive ($\lambda_i > 0$ for all $i \iff H \succ 0$, Positive Definite).
    \item \textbf{Local Maximum:} All eigenvalues are strictly negative ($\lambda_i < 0$ for all $i \iff H \prec 0$, Negative Definite).
    \item \textbf{Saddle Point:} At least one eigenvalue is positive and at least one is negative ($H$ is Indefinite).
    \item \textbf{Inconclusive:} At least one eigenvalue is zero and all other eigenvalues have the same sign ($H$ is only Semi-Definite).
\end{enumerate}
\end{theorembox}

\begin{theorembox}{The 2D Hessian Determinant Test ($n = 2$)}
For $f(x, y)$ with critical point $(x_0, y_0)$, let $D = \det(H(f)) = f_{xx} f_{yy} - (f_{xy})^2$:
\begin{enumerate}
    \item If $D > 0$ and $f_{xx} > 0$: $(x_0, y_0)$ is a \textbf{Local Minimum}.
    \item If $D > 0$ and $f_{xx} < 0$: $(x_0, y_0)$ is a \textbf{Local Maximum}.
    \item If $D < 0$: $(x_0, y_0)$ is a \textbf{Saddle Point}.
    \item If $D = 0$: The test is \textbf{Inconclusive} (higher-order derivatives determine behavior).
\end{enumerate}
\end{theorembox}

\section{Deep Intuition \& Conceptual Derivations}

\subsection{Derivation of the 2D Hessian Determinant Test}

\begin{proof}
Let $(x_0, y_0)$ be a critical point of $f(x, y)$ with $\nabla f(x_0, y_0) = \mathbf{0}$.
Let $\mathbf{h} = [h, k]^T$ be a small perturbation. By Taylor's theorem:
\begin{equation*}
f(x_0 + h, y_0 + k) - f(x_0, y_0) \approx \frac{1}{2} \begin{bmatrix} h & k \end{bmatrix} \begin{bmatrix} f_{xx} & f_{xy} \\ f_{xy} & f_{yy} \end{bmatrix} \begin{bmatrix} h \\ k \end{bmatrix} = \frac{1}{2} Q(h, k)
\end{equation*}
where the quadratic form is $Q(h, k) = f_{xx} h^2 + 2 f_{xy} h k + f_{yy} k^2$.
Assuming $f_{xx} \neq 0$, multiply and divide by $f_{xx}$ and complete the square in $h$:
\begin{align*}
f_{xx} Q(h, k) &= (f_{xx})^2 h^2 + 2 f_{xx} f_{xy} h k + f_{xx} f_{yy} k^2 \\
&= (f_{xx} h + f_{xy} k)^2 + \underbrace{(f_{xx} f_{yy} - f_{xy}^2)}_{D} k^2
\end{align*}
Dividing through by $f_{xx}$:
\begin{equation}
Q(h, k) = \frac{(f_{xx} h + f_{xy} k)^2 + D k^2}{f_{xx}}
\end{equation}
Now examine the sign of $Q(h, k)$ for all $(h, k) \neq (0, 0)$:
\begin{enumerate}
    \item \textbf{Case $D > 0$:} The numerator $(f_{xx} h + f_{xy} k)^2 + D k^2$ is strictly positive for all non-zero perturbations.
    Therefore, the sign of $Q(h, k)$ is determined completely by the denominator $f_{xx}$:
    \begin{itemize}
        \item If $f_{xx} > 0$, then $Q(h, k) > 0$ for all $(h, k) \neq (0, 0)$, meaning $f(x_0+h, y_0+k) > f(x_0, y_0)$. Thus, $(x_0, y_0)$ is a \textbf{strict local minimum}.
        \item If $f_{xx} < 0$, then $Q(h, k) < 0$ for all $(h, k) \neq (0, 0)$, so $(x_0, y_0)$ is a \textbf{strict local maximum}.
    \end{itemize}
    \item \textbf{Case $D < 0$:} For $k = 0$ ($h \neq 0$), $Q(h, 0) = f_{xx} h^2$, which has the sign of $f_{xx}$.
    However, setting $f_{xx} h + f_{xy} k = 0$ yields $Q(h, k) = \frac{D k^2}{f_{xx}}$, which has the opposite sign of $f_{xx}$ (since $D < 0$).
    Because $Q$ takes strictly positive values in some directions and strictly negative values in other directions, $(x_0, y_0)$ is a \textbf{saddle point}.
\end{enumerate}
\end{proof}

\subsection{Convexity and the Hessian in Machine Learning}

A loss function $L(\mathbf{w})$ is \textbf{convex} on $\mathbb{R}^d$ if and only if its Hessian matrix is positive semi-definite everywhere:
\begin{equation}
H(L)(\mathbf{w}) \succeq 0 \quad \forall \mathbf{w} \in \mathbb{R}^d
\end{equation}
For convex loss surfaces:
\begin{enumerate}
    \item Every local minimum is automatically a \textbf{global minimum}.
    \item There are \textbf{no saddle points} and no local maxima.
    \item Gradient Descent with appropriate step size is mathematically guaranteed to converge to the global optimum.
\end{enumerate}

\section{Explicit Gotchas \& Diagnostic Misconceptions}

\begin{trapbox}{Trap 1: Assuming Existence of Partial Derivatives Guarantees Differentiability}
In single-variable calculus, if $f'(x_0)$ exists, $f$ is automatically continuous and differentiable. In multivariable calculus, this is \textbf{completely false}! Consider the textbook counterexample:
\begin{equation*}
f(x, y) = \begin{cases} \frac{x y}{x^2 + y^2} & (x, y) \neq (0, 0) \\ 0 & (x, y) = (0, 0) \end{cases}
\end{equation*}
Both partial derivatives exist at the origin: $f_x(0, 0) = 0$ and $f_y(0, 0) = 0$. Yet along the line $y = m x$, $f(x, mx) = \frac{m}{1 + m^2}$, which varies with $m$. The function is not even continuous at $(0, 0)$, let alone differentiable! Full differentiability requires all partial derivatives to be \textbf{continuous}.
\end{trapbox}

\begin{trapbox}{Trap 2: Declaring a Local Minimum Based Solely on $f_{xx} > 0$}
A student often checks $f_{xx} > 0$ and prematurely concludes that a point is a local minimum. If $D = f_{xx} f_{yy} - (f_{xy})^2 < 0$, the point is a \textbf{saddle point}, regardless of the sign of $f_{xx}$! The cross-term $(f_{xy})^2$ represents twisting curvature; if the twist is too strong, it warps the bowl into a mountain pass.
\end{trapbox}

\begin{trapbox}{Trap 3: Declaring the Inconclusive Case ($D = 0$) to Be a Saddle Point}
When $D = 0$, the Hessian test provides zero information. It does NOT mean the point is a saddle point. For example:
\begin{itemize}
    \item $f(x, y) = x^4 + y^4$ has $D(0, 0) = 0$, but $(0, 0)$ is a strict \textbf{local minimum}.
    \item $f(x, y) = -x^4 - y^4$ has $D(0, 0) = 0$, but $(0, 0)$ is a strict \textbf{local maximum}.
    \item $f(x, y) = x^4 - y^4$ has $D(0, 0) = 0$, and $(0, 0)$ is a \textbf{saddle point}.
\end{itemize}
\end{trapbox}

\begin{trapbox}{Trap 4: Forgetting That Clairaut's Symmetry Requires Continuous Mixed Partials}
If the mixed partial derivatives are discontinuous, $f_{xy}$ does not necessarily equal $f_{yx}$. In non-smooth machine learning models (such as architectures with discontinuous activation thresholds), the Hessian can become asymmetric, violating standard optimization guarantees.
\end{trapbox}

\section{Python / Scientific Computing Lab}

The following scientific computing script evaluates analytical vs. numerical Hessian matrices, computes eigenvalues, classifies critical points, and verifies loss convexity.

\begin{lstlisting}[language=Python, caption={Python Lab: Hessian Construction, Eigenvalue Classification, and Loss Convexity}]
import numpy as np

# 1. Hessian Construction and Eigenvalue Classification
# Function from Exercise 1 & 2: f(x, y) = x^3 + y^3 - 3*x*y
def f(p):
    x, y = p[0], p[1]
    return x**3 + y**3 - 3.0 * x * y

def hessian_analytical(p):
    x, y = p[0], p[1]
    # f_xx = 6*x, f_yy = 6*y, f_xy = f_yx = -3
    return np.array([[6.0 * x, -3.0],
                     [-3.0,     6.0 * y]])

# Test Critical Points: (0, 0) and (1, 1)
crit_points = [np.array([0.0, 0.0]), np.array([1.0, 1.0])]

print("Classification of Critical Points for f(x, y) = x^3 + y^3 - 3xy:")
for cp in crit_points:
    H = hessian_analytical(cp)
    D = np.linalg.det(H)
    eigenvalues = np.linalg.eigvals(H)
    print(f"\nPoint ({cp[0]:.0f}, {cp[1]:.0f}):")
    print(f"  Hessian Matrix:\n{H}")
    print(f"  Determinant D: {D:.1f}")
    print(f"  Eigenvalues:   {eigenvalues}")
    if D < 0:
        print("  Classification: SADDLE POINT (Indefinite Hessian)")
    elif D > 0 and H[0, 0] > 0:
        print("  Classification: LOCAL MINIMUM (Positive Definite Hessian)")
    elif D > 0 and H[0, 0] < 0:
        print("  Classification: LOCAL MAXIMUM (Negative Definite Hessian)")
    else:
        print("  Classification: INCONCLUSIVE (D = 0)")

# 2. Convexity Verification of Ordinary Least Squares MSE Loss
# L(w) = (1/N) * ||Xw - y||^2  ==>  Hessian = (2/N) * X^T X
np.random.seed(42)
N, d = 100, 3
X = np.random.randn(N, d)
H_mse = (2.0 / N) * (X.T @ X)

eig_mse = np.linalg.eigvals(H_mse)
print(f"\nMSE Loss Hessian Eigenvalues (N={N}, d={d}):")
print(f"  Eigenvalues: {eig_mse}")
is_convex = np.all(eig_mse >= -1e-10)
print(f"  Is Loss Function Convex Everywhere? {is_convex}")
assert is_convex, "MSE Loss must be globally convex!"
\end{lstlisting}

    \item Construct the $3 \times 3$ Hessian matrix.
    \item Prove that the critical point is a global minimum.
\end{enumerate}

\item \textbf{Data Science Application (Convexity of Mean Squared Error):} The single-parameter linear regression loss function is $L(w) = \frac{1}{N} \sum_{i=1}^N (w x_i - y_i)^2$.
\begin{enumerate}
    \item Compute the second derivative $\frac{d^2 L}{dw^2}$.
    \item Show that $\frac{d^2 L}{dw^2} \ge 0$ for all $w$, proving that the loss surface is strictly convex and has a unique global minimum when $\sum x_i^2 > 0$.
\end{enumerate}
\end{enumerate}

\subsection*{Part B: Diagnostic Multiple-Choice Questions}
\begin{enumerate}
\setcounter{enumi}{5}
\item \textbf{Hessian Test Invariants and Saddle Points:} Let $f(x, y)$ be a smooth function. At critical point $(2, 3)$, the second partial derivatives are $f_{xx} = 4$, $f_{yy} = 1$, and $f_{xy} = 3$. How is $(2, 3)$ classified?
\begin{enumerate}
    \item Local Minimum because $f_{xx} = 4 > 0$.
    \item Local Maximum because $\trace(H) = 5 > 0$.
    \item Saddle Point because the Hessian determinant $D = f_{xx} f_{yy} - (f_{xy})^2 = -5 < 0$.
    \item Inconclusive because higher-order derivatives must be evaluated.
\end{enumerate}

\item \textbf{Multivariable Differentiability vs. Partial Derivatives:} Which of the following statements concerning multivariable differentiability is mathematically correct?
\begin{enumerate}
    \item If both partial derivatives $f_x$ and $f_y$ exist at $(x_0, y_0)$, then $f$ is guaranteed to be continuous and differentiable at $(x_0, y_0)$.
    \item Existence of all directional derivatives at $(x_0, y_0)$ is necessary and sufficient for differentiability.
    \item If the partial derivatives $f_x$ and $f_y$ exist and are continuous on an open neighborhood of $(x_0, y_0)$, then $f$ is guaranteed to be differentiable at $(x_0, y_0)$.
    \item Differentiability of a multivariable function requires the Hessian determinant to be strictly non-zero.
\end{enumerate}
\end{enumerate}
