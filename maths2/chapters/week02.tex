\section{Mathematical \& Theoretical Foundations}

Solving linear systems $A\mathbf{x} = \mathbf{b}$ is the computational workhorse of machine learning, appearing in linear regression, Kalman filtering, Gaussian processes, and network flow optimization.

\subsection{Cramer's Rule for Square Systems}

Let $A \in \R^{n \times n}$ be a square coefficient matrix and $\mathbf{b} \in \R^n$.

\begin{theorembox}{Cramer's Rule}
If $\det(A) \neq 0$, the system $A\mathbf{x} = \mathbf{b}$ has a \textbf{unique solution} given explicitly by:
\begin{equation}
x_i = \frac{\det(A_i(\mathbf{b}))}{\det(A)}, \quad \text{for } i = 1, 2, \dots, n
\end{equation}
where $A_i(\mathbf{b})$ denotes the matrix formed by replacing the $i$-th column of $A$ with the target vector $\mathbf{b}$:
\begin{equation}
A_i(\mathbf{b}) = \Big[ \mathbf{a}_1 \ \dots \ \mathbf{a}_{i-1} \ \mathbf{b} \ \mathbf{a}_{i+1} \ \dots \ \mathbf{a}_n \Big]
\end{equation}
\textbf{Computational Complexity:} Computing $n + 1$ determinants of size $n \times n$ via cofactor expansion requires $\mathcal{O}((n+1)!)$ operations. For $n = 20$, $(21)! \approx 5 \times 10^{19}$ operations, requiring thousands of years on a supercomputer! Cramer's rule is valuable for theoretical derivations and small ($2 \times 2$ or $3 \times 3$) closed forms, but unusable for big data.
\end{theorembox}

\subsection{Elementary Row Operations and Elementary Matrices}

To solve large systems efficiently, we transform the augmented matrix $[A \mid \mathbf{b}] \in \R^{m \times (n+1)}$ into an upper-triangular or staircase form without altering the solution set.

\begin{definitionbox}{Elementary Row Operations (EROs)}
The three valid row operations that preserve the solution space of $A\mathbf{x} = \mathbf{b}$ are:
\begin{enumerate}
    \item \textbf{Row Swap ($R_i \leftrightarrow R_j$):} Exchange row $i$ and row $j$.
    \item \textbf{Scalar Multiplication ($R_i \to c R_i$):} Multiply row $i$ by a non-zero scalar $c \neq 0$.
    \item \textbf{Row Addition ($R_i \to R_i + c R_j$):} Add a scalar multiple of row $j$ to row $i$.
\end{enumerate}
\textbf{Matrix Representation:} Each ERO corresponds to pre-multiplying $[A \mid \mathbf{b}]$ by an invertible \textbf{elementary matrix} $E \in \R^{m \times m}$ obtained by applying the same ERO to the identity matrix $I_m$.
\end{definitionbox}

\subsection{Row Echelon Form (REF) vs. Reduced Row Echelon Form (RREF)}

\begin{definitionbox}{Row Echelon Form (REF)}
A matrix is in \textbf{Row Echelon Form (REF)} if:
\begin{enumerate}
    \item All rows consisting entirely of zeros are placed at the bottom.
    \item In any non-zero row, the first non-zero entry (called the \textbf{leading coefficient} or \textbf{pivot}) appears strictly to the right of the pivot in the row above it.
    \item All entries directly below a pivot are zero.
\end{enumerate}
\end{definitionbox}

\begin{definitionbox}{Reduced Row Echelon Form (RREF)}
A matrix is in \textbf{Reduced Row Echelon Form (RREF)} if it satisfies REF, and additionally:
\begin{enumerate}
    \item Every pivot is equal to $1$.
    \item Each pivot is the \textbf{only non-zero entry in its entire column} (all entries both above and below the pivot are zero).
\end{enumerate}
\textbf{Uniqueness Theorem:} While a matrix has infinitely many different valid REFs depending on the sequence of row operations chosen, its \textbf{RREF is mathematically unique}.
\end{definitionbox}

\subsection{The Rouché-Capelli Solvability Classification}

Let $A \in \R^{m \times n}$ with augmented matrix $[A \mid \mathbf{b}]$. The \textbf{rank} of a matrix, $\rank(A)$, is the number of non-zero rows (or equivalently, the number of pivots) in its row echelon form.

\begin{theorembox}{The Rouché-Capelli Theorem}
A linear system $A\mathbf{x} = \mathbf{b}$ with $n$ variables is:
\begin{enumerate}
    \item \textbf{Inconsistent (No Solution):} If and only if:
    \begin{equation}
    \rank(A) < \rank([A \mid \mathbf{b}])
    \end{equation}
    Geometrically, this occurs when forward elimination produces a contradiction row:
    \begin{equation}
    \begin{bmatrix} 0 & 0 & \dots & 0 & \mid & c \end{bmatrix} \quad \text{with } c \neq 0 \implies 0 = c
    \end{equation}
    \item \textbf{Consistent with a Unique Solution:} If and only if:
    \begin{equation}
    \rank(A) = \rank([A \mid \mathbf{b}]) = n
    \end{equation}
    Every variable is a pivot variable; there are no free variables.
    \item \textbf{Consistent with Infinitely Many Solutions:} If and only if:
    \begin{equation}
    \rank(A) = \rank([A \mid \mathbf{b}]) = r < n
    \end{equation}
    There are $r$ pivot variables and $k = n - r > 0$ \textbf{free variables}. The solution set forms an affine subspace of dimension $n - r$.
\end{enumerate}
\end{theorembox}

\section{Deep Intuition \& Conceptual Derivations}

\subsection{Proof of Cramer's Rule via Matrix Column Multilinearity}
Let $A = [\mathbf{a}_1, \mathbf{a}_2, \dots, \mathbf{a}_n] \in \R^{n \times n}$ with $\det(A) \neq 0$, and let $\mathbf{x} = [x_1, x_2, \dots, x_n]^T$ satisfy $A\mathbf{x} = \mathbf{b}$.
Expressing $A\mathbf{x} = \mathbf{b}$ as a linear combination of column vectors:
\begin{equation}
\mathbf{b} = \sum_{j=1}^n x_j \mathbf{a}_j = x_1 \mathbf{a}_1 + x_2 \mathbf{a}_2 + \dots + x_n \mathbf{a}_n
\end{equation}
Now consider the matrix $A_i(\mathbf{b})$, in which the $i$-th column is replaced by $\mathbf{b}$:
\begin{equation}
A_i(\mathbf{b}) = \Big[ \mathbf{a}_1 \ \dots \ \mathbf{a}_{i-1} \ \left(\sum_{j=1}^n x_j \mathbf{a}_j\right) \ \mathbf{a}_{i+1} \ \dots \ \mathbf{a}_n \Big]
\end{equation}
By the multilinearity of the determinant with respect to column vectors, we expand the determinant across the sum:
\begin{equation}
\det(A_i(\mathbf{b})) = \sum_{j=1}^n x_j \det\Big[ \mathbf{a}_1 \ \dots \ \mathbf{a}_{i-1} \ \mathbf{a}_j \ \mathbf{a}_{i+1} \ \dots \ \mathbf{a}_n \Big]
\end{equation}
Notice the structure of the summand for $j \neq i$: the matrix contains the column $\mathbf{a}_j$ at position $j$ and also at position $i$. 
Because any matrix with two identical columns has determinant zero:
\begin{equation}
\det\Big[ \mathbf{a}_1 \ \dots \ \mathbf{a}_{i-1} \ \mathbf{a}_j \ \mathbf{a}_{i+1} \ \dots \ \mathbf{a}_n \Big] = 0 \quad \text{for all } j \neq i
\end{equation}
The only term that survives the summation is the single index $j = i$:
\begin{equation}
\det(A_i(\mathbf{b})) = x_i \det\Big[ \mathbf{a}_1 \ \dots \ \mathbf{a}_{i-1} \ \mathbf{a}_i \ \mathbf{a}_{i+1} \ \dots \ \mathbf{a}_n \Big] = x_i \det(A)
\end{equation}
Since $\det(A) \neq 0$, dividing both sides by $\det(A)$ yields Cramer's Rule:
\begin{equation}
x_i = \frac{\det(A_i(\mathbf{b}))}{\det(A)} \quad \blacksquare
\end{equation}

\section{Explicit Gotchas \& Diagnostic Misconceptions}

\begin{trapbox}{Trap 1: Applying Cramer's Rule to Singular or Rectangular Systems}
Cramer's Rule requires two strict mathematical preconditions:
\begin{enumerate}
    \item The coefficient matrix $A$ must be \textbf{square} ($m = n$).
    \item The matrix must be \textbf{non-singular} ($\det(A) \neq 0$).
\end{enumerate}
If $\det(A) = 0$, the denominator in $\frac{\det(A_i(\mathbf{b}))}{\det(A)}$ is zero, and Cramer's Rule fails completely. It cannot distinguish between a system with infinitely many solutions and a system with no solutions.
\end{trapbox}

\begin{trapbox}{Trap 2: Conflating Row Echelon Form (REF) with Reduced REF (RREF)}
Students often stop row operations prematurely, claiming a matrix is in RREF when it is only in REF.
In REF, entries \textit{above} a pivot may be non-zero, and pivots can be any non-zero real number (e.g. 5, -2). 
In RREF, every pivot \textbf{must be normalized to 1}, and \textbf{all entries above and below every pivot must be eliminated to 0}. Back-substitution is required for REF; RREF allows solutions to be read off directly.
\end{trapbox}

\begin{trapbox}{Trap 3: Performing Illegal Row Operations}
Elementary row operations must be strictly invertible. The following two blunders are common:
\begin{itemize}
    \item \textbf{Multiplying a row by zero ($R_i \to 0 \cdot R_i$):} This destroys all information in the equation and changes the solution space.
    \item \textbf{Adding a constant scalar to a row ($R_i \to R_i + c$):} You can only add a scalar multiple of \textit{another row} ($R_i \to R_i + c R_j$). Adding a constant scalar turns a linear equation into an affine constant, creating false contradictions.
\end{itemize}
\end{trapbox}

\begin{trapbox}{Trap 4: Missing Contradiction Pivots in the Augmented Column}
When reducing $[A \mid \mathbf{b}]$ to REF, always inspect the final column.
If forward elimination yields a row $[0 \ 0 \ \dots \ 0 \mid c]$ with $c \neq 0$, the pivot resides in the augmented column! This translates to the equation $0 x_1 + 0 x_2 + \dots + 0 x_n = c \implies 0 = c$, which is impossible. The system is \textbf{strictly inconsistent (no solution)}, regardless of how many variables or equations were given.
\end{trapbox}

\section{Python Scientific Computing Lab}

The following laboratory executes Cramer's Rule, manual Gaussian Elimination with back-substitution, SymPy RREF extraction, and Ordinary Least Squares normal equations.

\begin{lstlisting}[language=Python, caption=Cramer's Rule and Gaussian Elimination in Python]
import numpy as np
import sympy as sp

# 1. Cramer's Rule Implementation
def cramers_rule(A, b):
    det_A = np.linalg.det(A)
    if np.isclose(det_A, 0.0):
        raise ValueError("Matrix is singular! Cramer's rule fails.")
    
    n = A.shape[0]
    x = np.zeros(n)
    for i in range(n):
        A_i = A.copy()
        A_i[:, i] = b
        x[i] = np.linalg.det(A_i) / det_A
    return x

A_cramer = np.array([[2.0, 3.0], 
                     [5.0, -1.0]])
b_cramer = np.array([8.0, 3.0])
sol_cramer = cramers_rule(A_cramer, b_cramer)
print("Cramer's Rule Solution:", np.round(sol_cramer, 4))

# 2. Symbolic RREF and Rank using SymPy
# Augmented matrix for system:
# x1 + 2*x2 + x3 = 8
# 2*x1 + 3*x2 + 4*x3 = 20
# 4*x1 + 3*x2 + 2*x3 = 16
M_sym = sp.Matrix([
    [1, 2, 1, 8],
    [2, 3, 4, 20],
    [4, 3, 2, 16]
])

rref_matrix, pivot_cols = M_sym.rref()
print("\nUnique RREF Form:\n", rref_matrix)
print("Pivot Column Indices:", pivot_cols)
print(f"Matrix Rank: {len(pivot_cols)}")

# 3. Solving Underdetermined System with Free Variables
# x1 + 2*x2 - x3 = 4
# 2*x1 + 4*x2 + x3 = 11
M_under = sp.Matrix([
    [1, 2, -1, 4],
    [2, 4, 1, 11]
])
rref_under, pivots_under = M_under.rref()
print("\nUnderdetermined System RREF:\n", rref_under)
print("Pivots at columns:", pivots_under) # Columns 0 and 2 are pivots; Column 1 is free!

# 4. Data Science: OLS Normal Equations (X^T X) w = X^T y
X = np.array([[1.0, 1.0],
              [1.0, 2.0],
              [1.0, 3.0],
              [1.0, 4.0]])
y = np.array([2.2, 2.8, 3.6, 4.5])

XT_X = X.T @ X
XT_y = X.T @ y

# Solve for regression weights w = [intercept, slope]
w_ols = np.linalg.solve(XT_X, XT_y)
print("\nOLS Optimal Regression Weights w*:", np.round(w_ols, 4))
\end{lstlisting}

\subsection*{Part B: Diagnostic Misconception MCQs}

\begin{enumerate}
    \setcounter{enumi}{5}
    \item A student row-reduces an augmented matrix $[A \mid \mathbf{b}]$ representing a system of 4 linear equations in 3 variables, arriving at the following Row Echelon Form:
    \begin{equation}
    \left[\begin{array}{ccc|c}
    1 & 2 & -1 & 3 \\
    0 & 1 & 4 & 2 \\
    0 & 0 & 0 & 5 \\
    0 & 0 & 0 & 0
    \end{array}\right]
    \end{equation}
    The student observes that row 4 is all zeros, row 3 has no pivot in the $A$ portion, and concludes that $x_3$ is a free variable, yielding infinitely many solutions. Which statement diagnoses the student's reasoning?
    \begin{enumerate}
        \item The reasoning is correct; whenever a zero row occurs in the coefficient matrix, the system has infinitely many solutions.
        \item The reasoning is false because row 3 represents the equation $0x_1 + 0x_2 + 0x_3 = 5 \implies 0 = 5$, which is an impossible contradiction; the system is strictly inconsistent and has \textbf{no solutions}.
        \item The reasoning is false because row 4 must be moved to the top of the matrix before determining solvability.
        \item The reasoning is false because Cramer's Rule should have been used instead of Gaussian Elimination.
    \end{enumerate}

    \item Why is Gaussian Elimination ($\mathcal{O}(n^3)$) preferred over Cramer's Rule in machine learning algorithms that solve systems of linear equations?
    \begin{enumerate}
        \item Cramer's Rule only works when all entries in the matrix are positive numbers.
        \item Cramer's Rule requires computing determinants that scale factorially as $\mathcal{O}((n+1)!)$ via cofactor expansion, making it computationally intractable for datasets with more than a few features.
        \item Gaussian Elimination works only on singular matrices, which are common in machine learning.
        \item Cramer's Rule changes the eigenvalues of the matrix.
    \end{enumerate}
\end{enumerate}
