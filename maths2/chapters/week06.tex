\chapter{Matrix Representation, Kernel, Image \& Rank-Nullity}

Linear transformations are abstract geometric mappings that preserve vector space structure. Once ordered bases are chosen for both the domain and the codomain, every abstract linear transformation manifests concretely as a unique matrix. The fundamental conservation law linking the dimension of what collapses to zero (\index{Kernel}\textbf{Kernel} or Null Space) and what survives into the output space (\index{Image}\textbf{Image} or Column Space) is codified by the celebrated \index{Rank-Nullity Theorem}\textbf{Rank-Nullity Theorem}.

\vspace{0.5cm}
\begin{tcolorbox}[colback=primarylight, colframe=primarydark, title={\textbf{Executive Roadmap \& Learning Objectives}}]
\begin{itemize}
    \item \textbf{Target Competencies:} Construct matrix representations of linear operators with respect to standard and non-standard ordered bases; compute explicit bases for the kernel ($\ker(T)$) and image ($\operatorname{im}(T)$); prove and apply the Rank-Nullity Theorem; evaluate injectivity, surjectivity, and bijectivity of linear maps; diagnose compression loss in linear autoencoder bottlenecks.
    \item \textbf{Prerequisites:} Systems of linear equations and REF (Week 2), vector spaces and linear independence (Week 3), basis and dimension (Week 4), null space and linear mappings (Week 5).
    \item \textbf{Mathematical Notation:} Linear transformation $T: V \to W$; transformation matrix $[T]_{\mathcal{B}_V, \mathcal{B}_W}$; kernel $\ker(T)$; image/range $\operatorname{im}(T)$; domain dimension $\dim(V) = n$; rank $\rank(T) = \dim(\operatorname{im}(T))$; nullity $\nullity(T) = \dim(\ker(T))$.
\end{itemize}
\end{tcolorbox}

\section{Mathematical \& Theoretical Foundations}

\subsection{Matrix Representation with Respect to Ordered Bases}

\begin{definitionbox}{Transformation Matrix Relative to Bases}
\index{Matrix Representation}
Let $V$ and $W$ be finite-dimensional vector spaces over $\mathbb{R}$ with $\dim(V) = n$ and $\dim(W) = m$. Let $\mathcal{B}_V = \{\mathbf{v}_1, \mathbf{v}_2, \dots, \mathbf{v}_n\}$ be an ordered basis for $V$, and let $\mathcal{B}_W = \{\mathbf{w}_1, \mathbf{w}_2, \dots, \mathbf{w}_m\}$ be an ordered basis for $W$.
The \textbf{matrix representation} of a linear transformation $T: V \to W$ relative to $\mathcal{B}_V$ and $\mathcal{B}_W$, denoted $[T]_{\mathcal{B}_V, \mathcal{B}_W} \in \mathbb{R}^{m \times n}$, is constructed column-by-column:
\begin{equation}
[T]_{\mathcal{B}_V, \mathcal{B}_W} = \begin{bmatrix} [T(\mathbf{v}_1)]_{\mathcal{B}_W} & [T(\mathbf{v}_2)]_{\mathcal{B}_W} & \dots & [T(\mathbf{v}_n)]_{\mathcal{B}_W} \end{bmatrix}
\end{equation}
For any arbitrary vector $\mathbf{x} \in V$, the image coordinate vector is computed via standard matrix-vector multiplication:
\begin{equation}
[T(\mathbf{x})]_{\mathcal{B}_W} = [T]_{\mathcal{B}_V, \mathcal{B}_W} [\mathbf{x}]_{\mathcal{B}_V}
\end{equation}
\end{definitionbox}

\subsection{Kernel and Image Subspaces}

\begin{definitionbox}{Kernel and Image Subspaces}
\index{Kernel} \index{Image}
For any linear transformation $T: V \to W$:
\begin{itemize}
    \item The \textbf{Kernel} (or Null Space) of $T$ is the set of all vectors in domain $V$ that map to the zero vector of $W$:
    \begin{equation}
    \ker(T) = \left\{ \mathbf{v} \in V \;\middle|\; T(\mathbf{v}) = \mathbf{0}_W \right\} \subseteq V
    \end{equation}
    \item The \textbf{Image} (or Range) of $T$ is the set of all reachable output vectors in codomain $W$:
    \begin{equation}
    \operatorname{im}(T) = \left\{ \mathbf{w} \in W \;\middle|\; \exists \, \mathbf{v} \in V \text{ such that } T(\mathbf{v}) = \mathbf{w} \right\} \subseteq W
    \end{equation}
\end{itemize}
$\ker(T)$ is a subspace of $V$, and $\operatorname{im}(T)$ is a subspace of $W$.
\end{definitionbox}

\begin{theorembox}{Injectivity and Surjectivity Criteria}
Let $T: V \to W$ be a linear transformation:
\begin{enumerate}
    \item \textbf{Injective (One-to-One):} $T$ is injective if and only if $\ker(T) = \{\mathbf{0}_V\}$, or equivalently $\nullity(T) = 0$.
    \item \textbf{Surjective (Onto):} $T$ is surjective if and only if $\operatorname{im}(T) = W$, or equivalently $\rank(T) = \dim(W)$.
    \item \textbf{Bijective (Isomorphism):} If $\dim(V) = \dim(W)$, $T$ is injective $\iff$ $T$ is surjective $\iff$ $T$ is an isomorphism (invertible).
\end{enumerate}
\end{theorembox}

\section{Deep Intuition \& Conceptual Derivations}

\subsection{Rigorous Proof of the Rank-Nullity Theorem}

\begin{theorembox}{The Rank-Nullity Theorem (Dimension Theorem for Linear Mappings)}
Let $V$ and $W$ be vector spaces over $\mathbb{R}$, with $\dim(V) = n < \infty$. For any linear transformation $T: V \to W$:
\begin{equation}
\dim(\ker(T)) + \dim(\operatorname{im}(T)) = \dim(V)
\end{equation}
In matrix formulation, for any $A \in \mathbb{R}^{m \times n}$:
\begin{equation}
\nullity(A) + \rank(A) = n
\end{equation}
\end{theorembox}

\begin{proof}
Let $k = \dim(\ker(T))$. Since $\ker(T)$ is a subspace of $V$, $k \le n$.
Let $\mathcal{B}_{\ker} = \{\mathbf{u}_1, \mathbf{u}_2, \dots, \mathbf{u}_k\}$ be an ordered basis for $\ker(T)$.
By the Basis Extension Theorem, any linearly independent set can be extended to form a basis for $V$. Append $r = n - k$ vectors $\{\mathbf{v}_1, \mathbf{v}_2, \dots, \mathbf{v}_r\}$ so that:
\begin{equation*}
\mathcal{B}_V = \{\mathbf{u}_1, \dots, \mathbf{u}_k, \mathbf{v}_1, \dots, \mathbf{v}_r\}
\end{equation*}
forms a complete basis for $V$. To complete the proof, it suffices to prove that the set of image vectors:
\begin{equation*}
S = \{T(\mathbf{v}_1), T(\mathbf{v}_2), \dots, T(\mathbf{v}_r)\}
\end{equation*}
is a valid basis for $\operatorname{im}(T)$, which requires showing $S$ spans $\operatorname{im}(T)$ and is linearly independent.
\begin{enumerate}
    \item \textbf{$S$ Spans $\operatorname{im}(T)$:} Let $\mathbf{w} \in \operatorname{im}(T)$. There exists $\mathbf{x} \in V$ such that $T(\mathbf{x}) = \mathbf{w}$. Express $\mathbf{x}$ in basis $\mathcal{B}_V$:
    \begin{equation*}
    \mathbf{x} = \sum_{i=1}^k c_i \mathbf{u}_i + \sum_{j=1}^r d_j \mathbf{v}_j
    \end{equation*}
    Applying linear transformation $T$:
    \begin{equation*}
    T(\mathbf{x}) = \sum_{i=1}^k c_i T(\mathbf{u}_i) + \sum_{j=1}^r d_j T(\mathbf{v}_j)
    \end{equation*}
    Because each $\mathbf{u}_i \in \ker(T)$, $T(\mathbf{u}_i) = \mathbf{0}$. Therefore:
    \begin{equation*}
    \mathbf{w} = T(\mathbf{x}) = \mathbf{0} + \sum_{j=1}^r d_j T(\mathbf{v}_j) = \sum_{j=1}^r d_j T(\mathbf{v}_j)
    \end{equation*}
    Thus, $S$ spans $\operatorname{im}(T)$.
    \item \textbf{$S$ is Linearly Independent:} Suppose there exist scalars $d_1, \dots, d_r$ such that:
    \begin{equation*}
    d_1 T(\mathbf{v}_1) + d_2 T(\mathbf{v}_2) + \dots + d_r T(\mathbf{v}_r) = \mathbf{0}_W
    \end{equation*}
    By linearity of $T$:
    \begin{equation*}
    T\left( \sum_{j=1}^r d_j \mathbf{v}_j \right) = \mathbf{0}_W \implies \left( \sum_{j=1}^r d_j \mathbf{v}_j \right) \in \ker(T)
    \end{equation*}
    Since $\mathcal{B}_{\ker} = \{\mathbf{u}_1, \dots, \mathbf{u}_k\}$ is a basis for $\ker(T)$, this vector can be written as:
    \begin{equation*}
    \sum_{j=1}^r d_j \mathbf{v}_j = \sum_{i=1}^k c_i \mathbf{u}_i \implies \sum_{i=1}^k (-c_i) \mathbf{u}_i + \sum_{j=1}^r d_j \mathbf{v}_j = \mathbf{0}_V
    \end{equation*}
    Because $\mathcal{B}_V = \{\mathbf{u}_1, \dots, \mathbf{u}_k, \mathbf{v}_1, \dots, \mathbf{v}_r\}$ is a basis for $V$, all coefficients must be identically zero:
    \begin{equation*}
    c_1 = \dots = c_k = 0 \quad \text{and} \quad d_1 = d_2 = \dots = d_r = 0
    \end{equation*}
    Hence, $S$ is strictly linearly independent.
\end{enumerate}
Because $S$ spans $\operatorname{im}(T)$ and is linearly independent, it forms a basis for $\operatorname{im}(T)$.
Therefore, $\dim(\operatorname{im}(T)) = r = n - k$. Rearranging gives:
\begin{equation*}
\dim(\ker(T)) + \dim(\operatorname{im}(T)) = k + r = n = \dim(V)
\end{equation*}
This establishes the theorem.
\end{proof}

\section{Explicit Gotchas \& Diagnostic Misconceptions}

\begin{trapbox}{Trap 1: Swapping Basis Order in Matrix Construction}
When forming $[T]_{\mathcal{B}_V, \mathcal{B}_W}$, the columns must be ordered strictly according to the basis vectors of the domain $\mathcal{B}_V = (\mathbf{v}_1, \dots, \mathbf{v}_n)$, while each column contains coordinates relative to $\mathcal{B}_W$. Permuting the sequence of basis vectors changes the row and column order of the matrix, resulting in invalid coordinate transformations.
\end{trapbox}

\begin{trapbox}{Trap 2: Believing Domain Dimension Depends on Matrix Rank}
The domain dimension $n$ in $\rank(A) + \nullity(A) = n$ is a fixed structural constant determined entirely by the number of columns in $A$, not its rank. If rank increases, nullity decreases by the exact same amount. The sum always remains precisely $n$.
\end{trapbox}

\begin{trapbox}{Trap 3: Confusing Kernel Vectors with Output Values}
Vectors in the kernel $\ker(T)$ are \textbf{inputs} that produce zero outputs: $T(\mathbf{v}) = \mathbf{0}$. The kernel is not equal to $\{\mathbf{0}\}$ simply because its elements evaluate to zero. A matrix can have a rich, multi-dimensional kernel filled with non-zero vectors that all map to zero.
\end{trapbox}

\begin{trapbox}{Trap 4: Applying Rank-Nullity to the Codomain Dimension $m$}
Students frequently write $\rank(A) + \nullity(A) = m$ (the number of rows). This is fundamentally wrong. The columns of $A$ represent how the domain basis vectors are transformed; hence, the conservation law applies strictly to the \textbf{domain dimension $n$} (the number of columns).
\end{trapbox}

\section{Python / Scientific Computing Lab}

The following Python script computes transformation matrices relative to non-standard bases, extracts explicit bases for the image and kernel, and verifies coordinate transformations.

\begin{lstlisting}[language=Python, caption={Python Lab: Non-Standard Basis Representation and Rank-Nullity Verification}]
import numpy as np
import sympy as sp
from scipy.linalg import null_space

# 1. Non-Standard Basis Matrix Representation: [T]_B
# Transformation: T([x, y]^T) = [x + y, 2x - y]^T
# Standard matrix: A_E = [[1, 1], [2, -1]]
# Ordered basis: B = {v1 = [1, 1]^T, v2 = [1, 0]^T}
A_E = np.array([[1.0,  1.0],
                [2.0, -1.0]])
P = np.array([[1.0, 1.0],
              [1.0, 0.0]])  # Transition matrix from B to E

# Similarity Transformation: [T]_B = P^(-1) @ A_E @ P
P_inv = np.linalg.inv(P)
A_B = P_inv @ A_E @ P
print("Transformation Matrix in Basis B ([T]_B):\n", A_B)

# Verify with test vector: v = 3*v1 + 2*v2 = [5, 3]^T
coords_B = np.array([3.0, 2.0])
v_E = P @ coords_B

# Forward image in standard basis
Tv_E = A_E @ v_E
# Forward image via [T]_B
Tv_coords_B = A_B @ coords_B
print("\nReconstructed T(v) from Basis B:", P @ Tv_coords_B)
print("Direct Standard T(v):", Tv_E)
assert np.allclose(P @ Tv_coords_B, Tv_E), "Coordinate mapping mismatch!"

# 2. Kernel and Image Decomposition for 3x3 Singular Matrix
A = np.array([[1, 2, 3],
              [2, 4, 6],
              [1, 1, 1]], dtype=float)

# SymPy exact column space (Image) and null space (Kernel)
M = sp.Matrix(A)
img_basis = M.columnspace()
ker_basis = M.nullspace()

print(f"\nMatrix A (3x3): Rank = {len(img_basis)}, Nullity = {len(ker_basis)}")
print("Image (Column Space) Basis:")
for col in img_basis:
    sp.pprint(col.T)
print("Kernel (Null Space) Basis:")
for k in ker_basis:
    sp.pprint(k.T)

# Rank-Nullity Verification
assert len(img_basis) + len(ker_basis) == A.shape[1]
print(f"Rank-Nullity confirmed: {len(img_basis)} + {len(ker_basis)} = {A.shape[1]}")
\end{lstlisting}

    \item $B = P^{-1} A P$
    \item $B = P A P^{-1}$
    \item $B = A P$
