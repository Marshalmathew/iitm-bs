\chapter{Orthogonality, Gram-Schmidt Process \& Projections}

Orthogonality is the rigorous mathematical generalization of perpendicularity to arbitrary dimensions. Mutually orthogonal coordinate axes decouple multi-dimensional systems into completely independent, non-interfering 1-dimensional components. When noisy data vectors lie outside our model's representational capability, \index{Orthogonal Projections}\textbf{Orthogonal Projections} compute the mathematically optimal closest approximation. In this chapter, we master \index{Orthonormal Bases}\textbf{Orthonormal Bases}, the algorithmic \index{Gram-Schmidt Process}\textbf{Gram-Schmidt Process}, and the projection foundation of \index{Least Squares}\textbf{Ordinary Least Squares (OLS)}.

\vspace{0.5cm}

    \item \textbf{Prerequisites:} Inner products, norms, and Cauchy-Schwarz inequality (Week 7), basis and dimension (Week 4), systems of equations (Week 2).
    \item \textbf{Mathematical Notation:} Orthogonality $\mathbf{u} \perp \mathbf{v}$ ($\langle \mathbf{u}, \mathbf{v} \rangle = 0$); orthogonal complement $W^\perp$; orthogonal projection $\proj_W(\mathbf{y}) = \hat{\mathbf{y}}$; projection matrix $P = A(A^T A)^{-1} A^T$; orthonormal basis $\{\mathbf{q}_1, \dots, \mathbf{q}_k\}$; orthogonal matrix $Q$ ($Q^T Q = I$).
\end{itemize}

\section{Mathematical \& Theoretical Foundations}

\subsection{Orthogonality and the Generalized Pythagorean Theorem}

\begin{definitionbox}{Orthogonal Vectors and Orthonormal Sets}
\index{Orthogonality} \index{Orthonormal Basis}
Let $V$ be a real inner product space.
\begin{itemize}
    \item Two vectors $\mathbf{u}, \mathbf{v} \in V$ are \textbf{orthogonal} (denoted $\mathbf{u} \perp \mathbf{v}$) if and only if:
    \begin{equation}
    \langle \mathbf{u}, \mathbf{v} \rangle = 0
    \end{equation}
    \item A set of vectors $\{\mathbf{q}_1, \mathbf{q}_2, \dots, \mathbf{q}_k\}$ is an \textbf{orthonormal set} if every vector has unit length and all distinct pairs are orthogonal:
    \begin{equation}
    \langle \mathbf{q}_i, \mathbf{q}_j \rangle = \delta_{ij} = \begin{cases} 1 & \text{if } i = j \\ 0 & \text{if } i \neq j \end{cases}
    \end{equation}
\end{itemize}
\end{definitionbox}

\begin{theorembox}{Generalized Pythagorean Theorem}
Two vectors $\mathbf{u}, \mathbf{v} \in V$ are orthogonal if and only if:
\begin{equation}
\|\mathbf{u} + \mathbf{v}\|^2 = \|\mathbf{u}\|^2 + \|\mathbf{v}\|^2
\end{equation}
More generally, if $\{\mathbf{u}_1, \dots, \mathbf{u}_k\}$ is an orthogonal collection of vectors:
\begin{equation}
\left\| \sum_{i=1}^k \mathbf{u}_i \right\|^2 = \sum_{i=1}^k \|\mathbf{u}_i\|^2
\end{equation}
\end{theorembox}

\begin{theorembox}{Trivial Coordinate Extraction in an Orthonormal Basis}
If $\mathcal{B} = \{\mathbf{q}_1, \mathbf{q}_2, \dots, \mathbf{q}_n\}$ is an orthonormal basis for $V$, finding the coordinate coefficients of any vector $\mathbf{v} \in V$ requires \textbf{no matrix inversions or Gaussian elimination}:
\begin{equation}
\mathbf{v} = \sum_{i=1}^n \langle \mathbf{v}, \mathbf{q}_i \rangle \mathbf{q}_i = \langle \mathbf{v}, \mathbf{q}_1 \rangle \mathbf{q}_1 + \langle \mathbf{v}, \mathbf{q}_2 \rangle \mathbf{q}_2 + \dots + \langle \mathbf{v}, \mathbf{q}_n \rangle \mathbf{q}_n
\end{equation}
\end{theorembox}

\subsection{Orthogonal Projections and the Best Approximation Theorem}

\begin{theorembox}{The Best Approximation Theorem}
\index{Best Approximation Theorem}
Let $W$ be a finite-dimensional subspace of an inner product space $V$, and let $\mathbf{y} \in V$. The \textbf{orthogonal projection} of $\mathbf{y}$ onto $W$, denoted $\hat{\mathbf{y}} = \proj_W(\mathbf{y})$, is the unique vector in $W$ that minimizes Euclidean distance to $\mathbf{y}$:
\begin{equation}
\|\mathbf{y} - \hat{\mathbf{y}}\| < \|\mathbf{y} - \mathbf{w}\| \quad \forall \mathbf{w} \in W, \; \mathbf{w} \neq \hat{\mathbf{y}}
\end{equation}
The approximation error vector $\mathbf{e} = \mathbf{y} - \hat{\mathbf{y}}$ is strictly orthogonal to every vector in $W$ ($\mathbf{e} \in W^\perp$).
\end{theorembox}

\subsection{The Orthogonal Projection Matrix}

\begin{definitionbox}{Projection Matrix $P$}
\index{Projection Matrix}
Let the columns of matrix $A \in \mathbb{R}^{m \times k}$ ($k \le m$) form a linearly independent basis for subspace $W = \operatorname{Col}(A)$. The orthogonal projection of any vector $\mathbf{y} \in \mathbb{R}^m$ onto $W$ is given by matrix multiplication:
\begin{equation}
\hat{\mathbf{y}} = \proj_W(\mathbf{y}) = P \mathbf{y}
\end{equation}
where the projection matrix $P \in \mathbb{R}^{m \times m}$ is defined by:
\begin{equation}
P = A (A^T A)^{-1} A^T
\end{equation}
If the columns of $A$ are already orthonormal (so $A = Q$ and $Q^T Q = I_k$), the formula simplifies dramatically to:
\begin{equation}
P = Q Q^T
\end{equation}
\end{definitionbox}

\section{Deep Intuition \& Conceptual Derivations}

\subsection{Derivation of the Normal Equations via Error Orthogonality}

Why does the normal equation $(A^T A)\hat{\mathbf{x}} = A^T \mathbf{y}$ always produce the optimal least-squares solution?
\begin{proof}
Let $A \in \mathbb{R}^{m \times k}$ have linearly independent columns. We seek $\hat{\mathbf{x}} \in \mathbb{R}^k$ such that $\hat{\mathbf{y}} = A\hat{\mathbf{x}} \in \operatorname{Col}(A)$ is the orthogonal projection of $\mathbf{y} \in \mathbb{R}^m$ onto $\operatorname{Col}(A)$.
By the Best Approximation Theorem, the residual error vector $\mathbf{e} = \mathbf{y} - A\hat{\mathbf{x}}$ must be orthogonal to every vector in the column space $\operatorname{Col}(A)$.
Since the columns of $A$ ($\mathbf{a}_1, \dots, \mathbf{a}_k$) form a basis for $\operatorname{Col}(A)$, $\mathbf{e}$ must be orthogonal to each column $\mathbf{a}_j$:
\begin{equation*}
\mathbf{a}_j^T (\mathbf{y} - A\hat{\mathbf{x}}) = 0 \quad \forall j \in \{1, 2, \dots, k\}
\end{equation*}
Stacking these $k$ inner products into a single matrix equation:
\begin{equation*}
\begin{bmatrix} \mathbf{a}_1^T \\ \mathbf{a}_2^T \\ \vdots \\ \mathbf{a}_k^T \end{bmatrix} (\mathbf{y} - A\hat{\mathbf{x}}) = \mathbf{0} \iff A^T (\mathbf{y} - A\hat{\mathbf{x}}) = \mathbf{0}
\end{equation*}
Distributing $A^T$ across the parentheses yields the celebrated \textbf{Normal Equations}:
\begin{equation}
A^T \mathbf{y} - A^T A \hat{\mathbf{x}} = \mathbf{0} \implies (A^T A)\hat{\mathbf{x}} = A^T \mathbf{y}
\end{equation}
Since the columns of $A$ are linearly independent, the Gram matrix $A^T A \in \mathbb{R}^{k \times k}$ is strictly invertible. Multiplying by $(A^T A)^{-1}$:
\begin{equation}
\hat{\mathbf{x}} = (A^T A)^{-1} A^T \mathbf{y}
\end{equation}
Finally, multiplying by $A$ to obtain the projected vector $\hat{\mathbf{y}} = A\hat{\mathbf{x}}$ yields:
\begin{equation}
\hat{\mathbf{y}} = A (A^T A)^{-1} A^T \mathbf{y} = P \mathbf{y}
\end{equation}
\end{proof}

\subsection{Algebraic Properties of Projection Matrices}

\begin{theorembox}{Symmetry and Idempotence of $P$}
The orthogonal projection matrix $P = A(A^T A)^{-1} A^T$ satisfies:
\begin{enumerate}
    \item \textbf{Symmetry ($P^T = P$):}
    \begin{equation*}
    P^T = \left( A (A^T A)^{-1} A^T \right)^T = (A^T)^T \left( (A^T A)^{-1} \right)^T A^T = A (A^T A)^{-1} A^T = P
    \end{equation*}
    \item \textbf{Idempotence ($P^2 = P$):} Projecting a vector that has already been projected onto $W$ causes zero further change:
    \begin{align*}
    P^2 &= \left( A (A^T A)^{-1} A^T \right) \left( A (A^T A)^{-1} A^T \right) \\
    &= A (A^T A)^{-1} \underbrace{\left( A^T A (A^T A)^{-1} \right)}_{I_k} A^T = A (A^T A)^{-1} A^T = P
    \end{align*}
\end{enumerate}
\end{theorembox}

\section{Explicit Gotchas \& Diagnostic Misconceptions}

\begin{trapbox}{Trap 1: Forgetting to Normalize During the Gram-Schmidt Process}
The standard Gram-Schmidt subtraction $\mathbf{u}_k = \mathbf{v}_k - \sum \frac{\langle \mathbf{v}_k, \mathbf{u}_j \rangle}{\|\mathbf{u}_j\|^2} \mathbf{u}_j$ generates an \textbf{orthogonal} basis, but NOT an \textbf{orthonormal} basis. To obtain an orthonormal basis, every orthogonal vector must be explicitly normalized by its Euclidean length: $\mathbf{q}_k = \frac{\mathbf{u}_k}{\|\mathbf{u}_k\|}$.
\end{trapbox}

\begin{trapbox}{Trap 2: Applying $P = A(A^T A)^{-1} A^T$ When Columns are Collinear}
The projection formula $P = A(A^T A)^{-1} A^T$ strictly requires the columns of $A$ to be linearly independent. If two feature columns are collinear, $\det(A^T A) = 0$, rendering $(A^T A)^{-1}$ mathematically undefined. To project onto a rank-deficient subspace, one must first extract a minimal independent column basis or use the Moore-Penrose pseudoinverse $A^+$.
\end{trapbox}

\begin{trapbox}{Trap 3: Believing All Orthogonal Matrices Have $\det(Q) = 1$}
An orthogonal matrix satisfies $Q^T Q = I$, which implies $\det(Q^T Q) = \det(Q)^2 = 1 \implies \det(Q) = \pm 1$. Matrices with $\det(Q) = +1$ are \textbf{proper rotations} (Special Orthogonal Group $\operatorname{SO}(n)$). Matrices with $\det(Q) = -1$ are \textbf{reflections} or roto-reflections (e.g., $Q = \begin{bmatrix} 1 & 0 \\ 0 & -1 \end{bmatrix}$).
\end{trapbox}

\begin{trapbox}{Trap 4: Confusing Orthogonal Projections with Oblique Projections}
An operator $P$ is an \textbf{orthogonal} projection if and only if it is both idempotent ($P^2 = P$) AND symmetric ($P^T = P$). If $P^2 = P$ but $P^T \neq P$, the operator is an \textbf{oblique projection}: it projects onto $W$ along a direction that is not perpendicular to $W$, destroying minimum-distance optimality!
\end{trapbox}

\section{Python / Scientific Computing Lab}

The following scientific computing script demonstrates Gram-Schmidt orthogonalization from scratch, computes QR factorizations, and verifies the projection matrix properties.

\begin{lstlisting}[language=Python, caption={Python Lab: Gram-Schmidt, QR Decomposition, and Orthogonal Projections}]
import numpy as np

# 1. Classical Gram-Schmidt Orthogonalization Algorithm
def gram_schmidt(V):
    """V: (m x k) matrix with linearly independent columns."""
    Q = np.zeros_like(V, dtype=float)
    for j in range(V.shape[1]):
        q = V[:, j].astype(float)
        for i in range(j):
            # Subtract projection onto previous orthonormal vector
            q -= np.dot(V[:, j], Q[:, i]) * Q[:, i]
        norm_q = np.linalg.norm(q)
        assert norm_q > 1e-10, "Collinear vectors detected!"
        Q[:, j] = q / norm_q
    return Q

# Test vectors from Exercise 2
V2 = np.array([[3.0, 1.0],
               [4.0, 0.0]])
Q2 = gram_schmidt(V2)
print("Computed Orthonormal Basis Q (Gram-Schmidt):\n", Q2)
print("Orthogonality check Q^T @ Q:\n", np.round(Q2.T @ Q2, 4))

# 2. QR Decomposition via NumPy
A = np.array([[1.0, 1.0, 0.0],
              [1.0, 0.0, 1.0],
              [0.0, 1.0, 1.0]])
Q, R = np.linalg.qr(A)
print("\nNumPy QR - Q (Orthonormal):\n", Q)
print("NumPy QR - R (Upper Triangular):\n", np.round(R, 4))
assert np.allclose(Q @ R, A), "Reconstruction A = QR failed!"

# 3. Orthogonal Projection Matrix P = A (A^T A)^(-1) A^T
P = A @ np.linalg.inv(A.T @ A) @ A.T
print("\nProjection Matrix P Shape:", P.shape)
print("Symmetry Error ||P^T - P||:", np.linalg.norm(P.T - P))
print("Idempotence Error ||P^2 - P||:", np.linalg.norm(P @ P - P))
assert np.allclose(P.T, P), "P is not symmetric!"
assert np.allclose(P @ P, P), "P is not idempotent!"

# 4. Ordinary Least Squares via Normal Equations vs QR
y = np.array([4.0, 5.0, 7.0])
# Normal Equations: (A^T A) x_hat = A^T y
x_hat_normal = np.linalg.solve(A.T @ A, A.T @ y)
# Stable QR Solver: R x_hat = Q^T y
x_hat_qr = np.linalg.solve(R, Q.T @ y)
print(f"\nOLS Solution (Normal Equations): {x_hat_normal}")
print(f"OLS Solution (Stable QR):         {x_hat_qr}")
assert np.allclose(x_hat_normal, x_hat_qr)
\end{lstlisting}

\subsection*{Part A: Computational \& Analytical Problems}
\begin{enumerate}
\item \textbf{Projection onto a Vector:} Compute the orthogonal projection of vector $\mathbf{v} = \begin{bmatrix} 4 \\ 3 \end{bmatrix}$ onto the 1D line spanned by $\mathbf{u} = \begin{bmatrix} 1 \\ 2 \end{bmatrix}$. Verify that the residual error vector $\mathbf{e} = \mathbf{v} - \proj_{\mathbf{u}}(\mathbf{v})$ is perpendicular to $\mathbf{u}$.

\item \textbf{Gram-Schmidt Process in $\mathbb{R}^2$:} Use the Gram-Schmidt process to convert the basis $\mathcal{B} = \left\{ \mathbf{v}_1 = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \mathbf{v}_2 = \begin{bmatrix} 1 \\ 0 \end{bmatrix} \right\}$ into an orthonormal basis $\{\mathbf{q}_1, \mathbf{q}_2\}$.

\item \textbf{Gram-Schmidt Process in $\mathbb{R}^3$:} Apply the Gram-Schmidt process to construct an orthonormal basis from:
\begin{equation*}
\mathbf{v}_1 = \begin{bmatrix} 1 \\ 1 \\ 0 \end{bmatrix}, \quad \mathbf{v}_2 = \begin{bmatrix} 1 \\ 0 \\ 1 \end{bmatrix}, \quad \mathbf{v}_3 = \begin{bmatrix} 0 \\ 1 \\ 1 \end{bmatrix}
\end{equation*}

\item \textbf{Orthogonal Matrix Properties:} Let $Q = \begin{bmatrix} \cos \theta & -\sin \theta \\ \sin \theta & \cos \theta \end{bmatrix}$ be a 2D rotation matrix.
\begin{enumerate}
    \item Compute $Q^T Q$ and show that $Q^{-1} = Q^T$.
    \item Prove that $\|Q \mathbf{x}\| = \|\mathbf{x}\|$ for all $\mathbf{x} \in \mathbb{R}^2$ (isometry / length preservation).
    \item Prove that $\langle Q\mathbf{x}, Q\mathbf{y} \rangle = \langle \mathbf{x}, \mathbf{y} \rangle$ for all $\mathbf{x}, \mathbf{y} \in \mathbb{R}^2$ (angle preservation).
\end{enumerate}

\item \textbf{Data Science Application (Least Squares Projection Properties):} In linear regression, the fitted target vector is $\hat{\mathbf{y}} = P \mathbf{y}$, where $P = X(X^T X)^{-1} X^T$.
\begin{enumerate}
    \item Prove that $P$ is symmetric: $P^T = P$.
    \item Prove that $P$ is idempotent: $P^2 = P$.
    \item If $X \in \mathbb{R}^{100 \times 5}$ has full column rank 5, what are the eigenvalues of $P$, and what is the numerical value of $\trace(P)$?
\end{enumerate}
\end{enumerate}
