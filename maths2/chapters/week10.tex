\chapter{Tangent Planes, Steepest Descent \& Critical Points}

Optimization is the computational engine of modern artificial intelligence. Training a neural network or fitting a statistical model requires navigating an ultra-high-dimensional loss surface $L(\mathbf{w})$ to find parameter configurations where the error cannot be decreased. Locally, every smooth curved surface can be approximated by a flat \index{Tangent Plane}\textbf{Tangent Plane} (or Hyperplane). Moving along the direction of steepest loss reduction leads directly to the \index{Gradient Descent}\textbf{Gradient Descent} algorithm, terminating at stationary \index{Critical Points}\textbf{Critical Points} where the gradient vector vanishes ($\nabla f(\mathbf{x}^*) = \mathbf{0}$).

\vspace{0.5cm}

    \item \textbf{Prerequisites:} Multivariable partial derivatives, gradient vectors, and directional derivatives (Week 9), vector algebra and dot products (Week 1).
    \item \textbf{Mathematical Notation:} Surface $z = f(x, y)$; tangent plane at $(x_0, y_0, z_0)$; linear approximation $L(\mathbf{x})$; surface normal vector $\mathbf{n} = [f_x, f_y, -1]^T$; gradient update $\mathbf{w}^{(k+1)} = \mathbf{w}^{(k)} - \alpha \nabla L(\mathbf{w}^{(k)})$; learning rate $\alpha$; critical point condition $\nabla f(\mathbf{x}^*) = \mathbf{0}$.
\end{itemize}
\end{tcolorbox}

\section{Mathematical \& Theoretical Foundations}

\subsection{Tangent Planes and Hyperplanes}

\begin{definitionbox}{Tangent Plane to a Surface $z = f(x, y)$}
\index{Tangent Plane}
Let $f: \mathbb{R}^2 \to \mathbb{R}$ have continuous partial derivatives at $(x_0, y_0)$, and let $z_0 = f(x_0, y_0)$. The \textbf{tangent plane} to the surface $z = f(x, y)$ at the point $(x_0, y_0, z_0)$ is the plane containing all tangent lines to the surface at that point:
\begin{equation}
z - z_0 = f_x(x_0, y_0)(x - x_0) + f_y(x_0, y_0)(y - y_0)
\end{equation}
Equivalently, rewriting as a level surface $F(x, y, z) = f(x, y) - z = 0$, the \textbf{normal vector} to the tangent plane is:
\begin{equation}
\mathbf{n} = \begin{bmatrix} f_x(x_0, y_0) \\ f_y(x_0, y_0) \\ -1 \end{bmatrix} \implies \mathbf{n} \cdot \begin{bmatrix} x - x_0 \\ y - y_0 \\ z - z_0 \end{bmatrix} = 0
\end{equation}
\end{definitionbox}

\begin{definitionbox}{First-Order Taylor Linear Approximation}
\index{Linear Approximation}
For a general multivariable function $f: \mathbb{R}^n \to \mathbb{R}$, the \textbf{tangent hyperplane} (linear approximation) centered at $\mathbf{x}_0$ is:
\begin{equation}
L(\mathbf{x}) = f(\mathbf{x}_0) + \nabla f(\mathbf{x}_0)^T (\mathbf{x} - \mathbf{x}_0)
\end{equation}
The approximation error satisfies $\lim_{\mathbf{x} \to \mathbf{x}_0} \frac{|f(\mathbf{x}) - L(\mathbf{x})|}{\|\mathbf{x} - \mathbf{x}_0\|} = 0$.
\end{definitionbox}

\subsection{Steepest Descent and the Gradient Descent Algorithm}

\begin{definitionbox}{Gradient Descent Update Rule}
\index{Gradient Descent}
To minimize a differentiable objective loss function $L(\mathbf{w}): \mathbb{R}^d \to \mathbb{R}$, the \textbf{Gradient Descent} algorithm initializes at a starting weight vector $\mathbf{w}^{(0)}$ and iteratively updates:
\begin{equation}
\mathbf{w}^{(k+1)} = \mathbf{w}^{(k)} - \alpha \nabla L(\mathbf{w}^{(k)})
\end{equation}
where $\alpha > 0$ is the \textbf{learning rate} (step size parameter).
\end{definitionbox}

\subsection{Critical (Stationary) Points}

\begin{definitionbox}{Critical Points}
\index{Critical Points}
An interior point $\mathbf{x}^* \in \mathbb{R}^n$ is called a \textbf{critical point} (or stationary point) of $f$ if the gradient vanishes completely:
\begin{equation}
\nabla f(\mathbf{x}^*) = \mathbf{0} \iff \frac{\partial f}{\partial x_i}(\mathbf{x}^*) = 0 \quad \forall i \in \{1, 2, \dots, n\}
\end{equation}
\end{definitionbox}

\begin{theorembox}{Fermat's Theorem for Multivariable Functions}
If $f: \mathbb{R}^n \to \mathbb{R}$ attains a local minimum or local maximum at an interior point $\mathbf{x}^*$, and if all partial derivatives exist at $\mathbf{x}^*$, then:
\begin{equation}
\nabla f(\mathbf{x}^*) = \mathbf{0}
\end{equation}
Vanishing gradient is a \textbf{necessary condition} for a local extremum, but NOT a sufficient condition (due to saddle points).
\end{theorembox}

\section{Deep Intuition \& Conceptual Derivations}

\subsection{Optimal Learning Rate and Convergence of Gradient Descent}

Consider a canonical convex quadratic loss function:
\begin{equation}
L(\mathbf{w}) = \frac{1}{2} \mathbf{w}^T A \mathbf{w}
\end{equation}
where $A \in \mathbb{R}^{d \times d}$ is Symmetric Positive Definite with eigenvalues $\lambda_{\max} \ge \dots \ge \lambda_{\min} > 0$. The unique minimum is $\mathbf{w}^* = \mathbf{0}$.
The gradient is $\nabla L(\mathbf{w}) = A \mathbf{w}$. The Gradient Descent recurrence is:
\begin{equation}
\mathbf{w}^{(k+1)} = \mathbf{w}^{(k)} - \alpha A \mathbf{w}^{(k)} = (I - \alpha A) \mathbf{w}^{(k)}
\end{equation}
Applying this recurrence $k$ times:
\begin{equation}
\mathbf{w}^{(k)} = (I - \alpha A)^k \mathbf{w}^{(0)}
\end{equation}

\begin{theorembox}{Convergence Criterion for Gradient Descent}
The weight sequence converges to the global minimum $\mathbf{w}^* = \mathbf{0}$ for all initializations $\mathbf{w}^{(0)}$ if and only if all eigenvalues of $(I - \alpha A)$ have absolute value strictly less than 1:
\begin{equation}
|1 - \alpha \lambda_i| < 1 \quad \forall i \in \{1, \dots, d\}
\end{equation}
This yields the fundamental stability bounds on the learning rate $\alpha$:
\begin{equation}
-1 < 1 - \alpha \lambda_{\max} < 1 \iff 0 < \alpha < \frac{2}{\lambda_{\max}(A)}
\end{equation}
The optimal learning rate minimizing the spectral radius is:
\begin{equation}
\alpha^* = \frac{2}{\lambda_{\min}(A) + \lambda_{\max}(A)}
\end{equation}
If $\alpha \ge \frac{2}{\lambda_{\max}}$, Gradient Descent oscillates wildly and diverges (gradient explosion).
\end{theorembox}

\section{Explicit Gotchas \& Diagnostic Misconceptions}

\begin{trapbox}{Trap 1: Confusing Normal Vectors of Explicit Surfaces with Level Curves}
For a 2D level curve $f(x, y) = c$, the 2D normal vector is the gradient $\nabla f = [f_x, f_y]^T$.
However, for an explicit 3D surface $z = f(x, y)$, the 3D normal vector is $\mathbf{n} = [f_x, f_y, -1]^T$. Forgetting the $-1$ component in the $z$-direction is an extremely common error that collapses 3D plane equations into 2D line equations.
\end{trapbox}

\begin{trapbox}{Trap 2: Believing $\nabla f(\mathbf{x}^*) = \mathbf{0}$ Guarantees a Local Minimum or Maximum}
A vanishing gradient is only a \textbf{stationary condition}. At a saddle point (such as $f(x, y) = x^2 - y^2$ at $(0, 0)$), $\nabla f(0, 0) = \mathbf{0}$. Yet walking in the $x$-direction increases $f$, while walking in the $y$-direction decreases $f$. Without second-order derivative testing (the Hessian matrix), one cannot assert whether a critical point is a minimum, maximum, or saddle point.
\end{trapbox}

\begin{trapbox}{Trap 3: Believing Gradient Descent Always Decreases Loss in a Single Step}
If the learning rate $\alpha$ is chosen too large ($\alpha > 2/\lambda_{\max}$), the update $\mathbf{w}^{(1)} = \mathbf{w}^{(0)} - \alpha \nabla L$ overshoots the valley entirely, landing higher up the opposite wall ($L(\mathbf{w}^{(1)}) > L(\mathbf{w}^{(0)})$). Descent is guaranteed only for sufficiently small step sizes $\alpha$.
\end{trapbox}

\begin{trapbox}{Trap 4: Using Linear Approximations Far from the Expansion Point}
The linear approximation $L(\mathbf{x}) = f(\mathbf{x}_0) + \nabla f(\mathbf{x}_0)^T(\mathbf{x} - \mathbf{x}_0)$ is accurate only in an infinitesimal neighborhood of $\mathbf{x}_0$. For non-linear loss surfaces, quadratic and higher-order curvature terms dominate rapidly as $\|\mathbf{x} - \mathbf{x}_0\|$ grows, rendering first-order Taylor estimates highly inaccurate.
\end{trapbox}

\section{Python / Scientific Computing Lab}

The following scientific computing script verifies tangent plane approximations, simulates gradient descent trajectories on an anisotropic loss surface, and demonstrates divergence when the learning rate exceeds the theoretical ceiling.

\begin{lstlisting}[language=Python, caption={Python Lab: Tangent Planes, Gradient Descent, and Step Size Stability}]
import numpy as np

# 1. Tangent Plane Linear Approximation
# f(x, y) = sqrt(x^2 + y^2) at (3.0, 4.0)
def f(p):
    return np.sqrt(p[0]**2 + p[1]**2)

p0 = np.array([3.0, 4.0])
z0 = f(p0)  # 5.0
grad_p0 = p0 / z0  # [3/5, 4/5] = [0.6, 0.8]

# Linear approximation function: L(p) = z0 + grad . (p - p0)
def L_approx(p):
    return z0 + np.dot(grad_p0, p - p0)

test_point = np.array([3.06, 3.92])
exact_val = f(test_point)
approx_val = L_approx(test_point)
print(f"Exact f(3.06, 3.92):       {exact_val:.6f}")
print(f"Linear Approx L(3.06, 3.92): {approx_val:.6f}")
print(f"Absolute Error:             {abs(exact_val - approx_val):.2e}")

# 2. Gradient Descent Simulation: L(w) = w1^2 + 4*w2^2
# Hessian A = [[2, 0], [0, 8]] --> lambda_max = 8.0, lambda_min = 2.0
lambda_max = 8.0
alpha_max = 2.0 / lambda_max
print(f"\nTheoretical Maximum Stable Learning Rate: alpha < {alpha_max:.3f}")

def run_gd(alpha, steps=5):
    w = np.array([3.0, 2.0])
    trajectory = [w.copy()]
    for _ in range(steps):
        grad = np.array([2.0 * w[0], 8.0 * w[1]])
        w = w - alpha * grad
        trajectory.append(w.copy())
    return np.array(trajectory)

# Stable run with alpha = 0.1
traj_stable = run_gd(alpha=0.1, steps=3)
print("\nStable Gradient Descent Trajectory (alpha = 0.1):")
for step, pos in enumerate(traj_stable):
    loss = pos[0]**2 + 4.0*pos[1]**2
    print(f"  Step {step}: w = [{pos[0]:.4f}, {pos[1]:.4f}], Loss = {loss:.4f}")

# Divergent run with alpha = 0.28 (> 0.25)
traj_divergent = run_gd(alpha=0.28, steps=3)
print("\nDivergent Gradient Descent Trajectory (alpha = 0.28):")
for step, pos in enumerate(traj_divergent):
    loss = pos[0]**2 + 4.0*pos[1]**2
    print(f"  Step {step}: w = [{pos[0]:.4f}, {pos[1]:.4f}], Loss = {loss:.4f}")
\end{lstlisting}

    \item Starting from initial weight vector $\mathbf{w}^{(0)} = \begin{bmatrix} 3 \\ 2 \end{bmatrix}$ with learning rate $\alpha = 0.1$, compute the updated weight vector $\mathbf{w}^{(1)}$.
