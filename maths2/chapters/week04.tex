\chapter{Basis, Dimension \& Matrix Rank}

A \index{Basis}\textbf{Basis} is the minimal generating skeletal coordinate framework of a vector space: every vector can be synthesized in exactly one unique way as a linear combination of its elements. The cardinality of this minimal generating set defines the intrinsic, invariant \index{Dimension}\textbf{Dimension} of the space. In computational matrix theory, \index{Matrix Rank}\textbf{Matrix Rank} quantifies the true degrees of geometric freedom within the row and column spaces, distinguishing genuine signals from redundant linear multicollinearities.

\end{itemize}
\end{tcolorbox}

\section{Mathematical \& Theoretical Foundations}

\subsection{Axiomatic Definition of a Basis}

\begin{definitionbox}{Basis for a Vector Space}
\index{Basis}
Let $V$ be a vector space over $\mathbb{R}$. A subset $\mathcal{B} = \{\mathbf{b}_1, \mathbf{b}_2, \dots, \mathbf{b}_n\} \subseteq V$ is called an \textbf{ordered basis} for $V$ if and only if it satisfies two simultaneous structural conditions:
\begin{enumerate}
    \item \textbf{Linear Independence:} $\mathcal{B}$ contains no redundant vectors, meaning
    \begin{equation}
    c_1 \mathbf{b}_1 + c_2 \mathbf{b}_2 + \dots + c_n \mathbf{b}_n = \mathbf{0} \implies c_1 = c_2 = \dots = c_n = 0
    \end{equation}
    \item \textbf{Spanning Set:} $\mathcal{B}$ generates the entire ambient space $V$:
    \begin{equation}
    \vspan(\mathcal{B}) = \left\{ \sum_{i=1}^n c_i \mathbf{b}_i \;\middle|\; c_i \in \mathbb{R} \right\} = V
    \end{equation}
\end{enumerate}
\end{definitionbox}

\begin{theorembox}{Unique Representation Theorem}
\index{Unique Coordinates}
Let $\mathcal{B} = \{\mathbf{b}_1, \mathbf{b}_2, \dots, \mathbf{b}_n\}$ be a basis for $V$. For every vector $\mathbf{x} \in V$, there exists an \textbf{uniquely determined} sequence of scalar coordinates $(c_1, c_2, \dots, c_n) \in \mathbb{R}^n$ such that:
\begin{equation}
\mathbf{x} = c_1 \mathbf{b}_1 + c_2 \mathbf{b}_2 + \dots + c_n \mathbf{b}_n
\end{equation}
The vector $[\mathbf{x}]_\mathcal{B} = [c_1, c_2, \dots, c_n]^T \in \mathbb{R}^n$ is the \textbf{coordinate vector of $\mathbf{x}$ relative to $\mathcal{B}$}.
\end{theorembox}

\begin{proof}
Suppose there exist two representation expansions for the same vector $\mathbf{x} \in V$:
\begin{equation*}
\mathbf{x} = \sum_{i=1}^n c_i \mathbf{b}_i \quad \text{and} \quad \mathbf{x} = \sum_{i=1}^n d_i \mathbf{b}_i
\end{equation*}
Subtracting the second equation from the first yields:
\begin{equation*}
\mathbf{0} = \mathbf{x} - \mathbf{x} = \sum_{i=1}^n (c_i - d_i) \mathbf{b}_i
\end{equation*}
Since $\mathcal{B}$ is linearly independent by definition, the only linear combination that equals the zero vector is the trivial linear combination. Consequently:
\begin{equation*}
c_i - d_i = 0 \implies c_i = d_i \quad \forall i \in \{1, 2, \dots, n\}
\end{equation*}
Hence, the coordinate representation is strictly unique.
\end{proof}

\subsection{Dimension of Vector Spaces and Subspaces}

\begin{theorembox}{Dimension Invariance Theorem}
\index{Dimension}
If a vector space $V$ is spanned by a finite set of vectors, then every basis of $V$ contains the exact same number of vectors. This fundamental topological invariant is called the \textbf{dimension} of $V$, denoted $\dim(V)$.
\begin{itemize}
    \item Standard Euclidean space: $\dim(\mathbb{R}^n) = n$.
    \item Polynomial space of degree at most $n$: $\dim(\mathbb{P}_n) = n + 1$ (basis $\{1, t, t^2, \dots, t^n\}$).
    \item Matrix space: $\dim(\mathbb{R}^{m \times n}) = m \cdot n$.
    \item Trivial zero space: $\dim(\{\mathbf{0}\}) = 0$ (the empty set $\emptyset$ serves as its basis).
\end{itemize}
\end{theorembox}

\begin{theorembox}{Subspace Dimension Bounds \& Equality Criterion}
Let $W$ be a vector subspace of a finite-dimensional space $V$. Then:
\begin{enumerate}
    \item $\dim(W) \le \dim(V)$.
    \item If $\dim(W) = \dim(V)$, then $W = V$.
\end{enumerate}
\end{theorembox}

\subsection{Matrix Rank via Row and Column Spaces}

\begin{definitionbox}{Row Space, Column Space, and Matrix Rank}
\index{Matrix Rank}
For any matrix $A \in \mathbb{R}^{m \times n}$:
\begin{itemize}
    \item The \textbf{Row Space} $\vspan(\text{rows of } A) \subseteq \mathbb{R}^n$ has dimension equal to the row rank.
    \item The \textbf{Column Space} $\vspan(\text{columns of } A) \subseteq \mathbb{R}^m$ has dimension equal to the column rank.
    \item \textbf{Fundamental Theorem of Matrix Rank:} For any matrix, the row rank and column rank are identically equal:
    \begin{equation}
    \dim(\operatorname{Row}(A)) = \dim(\operatorname{Col}(A)) = \rank(A) \le \min(m, n)
    \end{equation}
    \item $\rank(A)$ is exactly the number of leading pivot entries in any Row Echelon Form (REF) of $A$.
\end{itemize}
\end{definitionbox}

\section{Deep Intuition \& Conceptual Derivations}

\subsection{The Equivalence of Row Rank and Column Rank}

A profound result in linear algebra is that the maximum number of linearly independent rows in any matrix $A \in \mathbb{R}^{m \times n}$ equals the maximum number of linearly independent columns, despite their residing in completely different ambient vector spaces ($\mathbb{R}^n$ versus $\mathbb{R}^m$).

\begin{proof}
Let $A \in \mathbb{R}^{m \times n}$ have row rank $r$. Let $\mathbf{r}_1^T, \mathbf{r}_2^T, \dots, \mathbf{r}_r^T \in \mathbb{R}^{1 \times n}$ be a basis for the row space $\operatorname{Row}(A)$.
Since every row of $A$ belongs to $\operatorname{Row}(A)$, each row $\mathbf{a}_i^T$ can be expressed as a linear combination:
\begin{equation*}
\mathbf{a}_i^T = c_{i1}\mathbf{r}_1^T + c_{i2}\mathbf{r}_2^T + \dots + c_{ir}\mathbf{r}_r^T
\end{equation*}
In matrix form, this yields the low-rank factorization:
\begin{equation}
A = C R
\end{equation}
where $C \in \mathbb{R}^{m \times r}$ contains the coefficient coordinates $c_{ij}$, and $R \in \mathbb{R}^{r \times n}$ contains the basis row vectors.
Now examine the columns of $A$. Multiplying $C$ by the columns of $R$, each column $\mathbf{a}_{*j}$ of $A$ is given by:
\begin{equation*}
\mathbf{a}_{*j} = C \mathbf{r}_{*j} = \sum_{k=1}^r R_{kj} \mathbf{c}_{*k}
\end{equation*}
This explicitly proves that every column of $A$ lies in the span of the $r$ columns of $C$. Therefore:
\begin{equation*}
\dim(\operatorname{Col}(A)) \le r = \dim(\operatorname{Row}(A)) \implies \text{Column Rank}(A) \le \text{Row Rank}(A)
\end{equation*}
Applying this identical argument to $A^T$ yields:
\begin{equation*}
\text{Row Rank}(A) = \text{Column Rank}(A^T) \le \text{Row Rank}(A^T) = \text{Column Rank}(A)
\end{equation*}
Sandwiching the two inequalities establishes exact equality: $\text{Row Rank}(A) = \text{Column Rank}(A) = \rank(A)$.
\end{proof}

\subsection{The Basis Extension and Contraction Principle}

In an $n$-dimensional vector space $V$:
\begin{enumerate}
    \item Any linearly independent subset of fewer than $n$ vectors can always be \textbf{extended} to form a full basis by appending vectors outside its current span.
    \item Any spanning set of more than $n$ vectors can always be \textbf{contracted} to form a basis by sequentially eliminating redundant vectors.
    \item If a subset contains \textbf{exactly} $n$ vectors, linear independence and spanning are completely equivalent: proving one automatically guarantees the other!
\end{enumerate}

\section{Explicit Gotchas \& Diagnostic Misconceptions}

\begin{trapbox}{Trap 1: Conflating a Spanning Set with a Minimal Basis}
Every basis spans the space, but an arbitrary spanning set is almost never a basis. Appending ten collinear vectors to a basis produces a valid spanning set that completely fails linear independence. A basis requires \textbf{both} generation ($\vspan(S) = V$) and minimality ($\text{redundancy} = 0$).
\end{trapbox}

\begin{trapbox}{Trap 2: Assuming Basis Vectors Must Be Perpendicular (Orthogonal)}
A widespread misconception is that basis vectors must be orthogonal (at $90^\circ$ angles) and normalized to unit length. While \textbf{orthonormal bases} offer tremendous computational advantages (such as trivial coordinate projections via inner products), any collection of linearly independent vectors that spans the space constitutes a perfectly valid basis, regardless of skew angles or unequal vector lengths.
\end{trapbox}

\begin{trapbox}{Trap 3: The Off-by-One Dimension Error in Polynomial and Matrix Spaces}
Students frequently assert that the space of polynomials of degree at most $n$, denoted $\mathbb{P}_n$, has dimension $n$. This misses the constant polynomial term $t^0 = 1$. The standard monomial basis is $\mathcal{B} = \{1, t, t^2, \dots, t^n\}$, which contains precisely $n + 1$ elements. Thus $\dim(\mathbb{P}_n) = n + 1$. Similarly, the space of $m \times n$ matrices has dimension $m \cdot n$, not $m + n$.
\end{trapbox}

\begin{trapbox}{Trap 4: Confusing Matrix Rank with Matrix Dimensions}
A matrix $X \in \mathbb{R}^{1000 \times 10}$ has 10,000 numerical entries and 1000 rows, but its rank can \textbf{never exceed 10}. Because $\rank(X) \le \min(m, n)$, the dimension of both the row space and column space is capped by the smaller dimension: $\min(1000, 10) = 10$. If two feature columns are linearly dependent, the rank drops further to $\le 8$.
\end{trapbox}

\section{Python / Scientific Computing Lab}

The following self-contained scientific computing script demonstrates basis verification, coordinate computation, matrix rank determination, and redundant column elimination using NumPy and SciPy.

\begin{lstlisting}[language=Python, caption={Python Lab: Basis Coordinates, Rank, and Subspace Basis Extraction}]
import numpy as np
from scipy.linalg import null_space

# 1. Coordinate Computation in an Oblique Basis
# Basis vectors for R^2: b1 = [2, 0.5]^T, b2 = [0.5, 1.5]^T
B = np.array([[2.0, 0.5],
              [0.5, 1.5]])
v = np.array([3.0, 3.5])

# Coordinates [v]_B satisfy: B @ c = v  ==>  c = inv(B) @ v
coords = np.linalg.solve(B, v)
print(f"Oblique Basis Coordinates [c1, c2]: {coords}")
assert np.allclose(B @ coords, v), "Verification failed!"

# 2. Matrix Rank & Pivot Identification
A = np.array([[1, 2, 1, 4],
              [2, 4, 3, 11],
              [3, 6, 4, 15]], dtype=float)

rank_A = np.linalg.matrix_rank(A)
print(f"\nMatrix Shape: {A.shape}, Computed Rank: {rank_A}")

# QR Decomposition with Column Pivoting to extract a minimal column basis
Q, R, P = np.linalg.qr(A, pivoting=True)
independent_cols = P[:rank_A]
print(f"Indices of Linearly Independent Columns (Basis for Col(A)): {independent_cols}")
column_basis = A[:, independent_cols]
print("Extracted Column Space Basis:\n", column_basis)

# 3. Diagnosing Multicollinearity in a Design Matrix
# Simulating 5 features where feature 2 is 2*f0 + 3*f1 (perfect multicollinearity)
np.random.seed(42)
X = np.random.randn(100, 4)
col_redundant = 2.0 * X[:, 0] + 3.0 * X[:, 1]
X_collinear = np.column_stack([X[:, :2], col_redundant, X[:, 2:]])

rank_X = np.linalg.matrix_rank(X_collinear)
print(f"\nData Matrix (100 samples, 5 features) Rank: {rank_X}")
if rank_X < X_collinear.shape[1]:
    print(f"WARNING: Matrix is rank deficient by {X_collinear.shape[1] - rank_X} dimensions (Multicollinearity present!).")
\end{lstlisting}

A = \begin{bmatrix} 1 & 2 & 1 & 4 \\ 2 & 4 & 3 & 11 \\ 3 & 6 & 4 & 15 \end{bmatrix}
