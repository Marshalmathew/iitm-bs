\section*{Week 1: Vectors, Matrices \& Determinants}

\subsection*{Detailed Solutions}

\textbf{1. Vector Algebra, Norms, and Orthogonality}
Given vectors $\mathbf{u} = \begin{bmatrix} 3 \\ -2 \\ 4 \end{bmatrix}$ and $\mathbf{v} = \begin{bmatrix} 2 \\ 5 \\ 1 \end{bmatrix}$:
\begin{enumerate}
    \item \textbf{Dot Product $\mathbf{u} \cdot \mathbf{v}$:}
    \begin{equation*}
    \mathbf{u} \cdot \mathbf{v} = (3)(2) + (-2)(5) + (4)(1) = 6 - 10 + 4 = \mathbf{0}
    \end{equation*}
    \item \textbf{Euclidean Norms $\|\mathbf{u}\|$ and $\|\mathbf{v}\|$:}
    \begin{align*}
    \|\mathbf{u}\| &= \sqrt{3^2 + (-2)^2 + 4^2} = \sqrt{9 + 4 + 16} = \mathbf{\sqrt{29}} \approx 5.3852 \\
    \|\mathbf{v}\| &= \sqrt{2^2 + 5^2 + 1^2} = \sqrt{4 + 25 + 1} = \mathbf{\sqrt{30}} \approx 5.4772
    \end{align*}
    \item \textbf{Angle and Orthogonality:}
    \begin{equation*}
    \cos \theta = \frac{\mathbf{u} \cdot \mathbf{v}}{\|\mathbf{u}\| \|\mathbf{v}\|} = \frac{0}{\sqrt{29}\sqrt{30}} = 0 \implies \theta = \frac{\pi}{2} = 90^\circ
    \end{equation*}
    Because their dot product is zero, $\mathbf{u}$ and $\mathbf{v}$ are \textbf{strictly orthogonal} ($\mathbf{u} \perp \mathbf{v}$).
    \item \textbf{Outer Product Matrix $P = \mathbf{u} \mathbf{v}^T$:}
    \begin{equation*}
    P = \mathbf{u} \mathbf{v}^T = \begin{bmatrix} 3 \\ -2 \\ 4 \end{bmatrix} \begin{bmatrix} 2 & 5 & 1 \end{bmatrix} = \begin{bmatrix} 3(2) & 3(5) & 3(1) \\ -2(2) & -2(5) & -2(1) \\ 4(2) & 4(5) & 4(1) \end{bmatrix} = \mathbf{\begin{bmatrix} 6 & 15 & 3 \\ -4 & -10 & -2 \\ 8 & 20 & 4 \end{bmatrix}}
    \end{equation*}
    Notice that $P$ is a $3 \times 3$ rank-1 matrix whose columns are scalar multiples of $\mathbf{u}$.
\end{enumerate}

\textbf{2. Matrix Multiplication and Transpose Identities}
Given $A = \begin{bmatrix} 2 & 1 \\ -1 & 3 \end{bmatrix}$ and $B = \begin{bmatrix} 4 & -2 \\ 1 & 0 \end{bmatrix}$:
\begin{enumerate}
    \item \textbf{Matrix Product $AB$:}
    \begin{equation*}
    AB = \begin{bmatrix} 2(4) + 1(1) & 2(-2) + 1(0) \\ -1(4) + 3(1) & -1(-2) + 3(0) \end{bmatrix} = \mathbf{\begin{bmatrix} 9 & -4 \\ -1 & 2 \end{bmatrix}}
    \end{equation*}
    \item \textbf{Matrix Product $BA$:}
    \begin{equation*}
    BA = \begin{bmatrix} 4(2) + (-2)(-1) & 4(1) + (-2)(3) \\ 1(2) + 0(-1) & 1(1) + 0(3) \end{bmatrix} = \mathbf{\begin{bmatrix} 10 & -2 \\ 2 & 1 \end{bmatrix}}
    \end{equation*}
    Because $\begin{bmatrix} 9 & -4 \\ -1 & 2 \end{bmatrix} \neq \begin{bmatrix} 10 & -2 \\ 2 & 1 \end{bmatrix}$, $AB \neq BA$. Matrices $A$ and $B$ \textbf{do not commute}.
    \item \textbf{Direct Transpose vs. Reversal Property:}
    Transposing $AB$ directly:
    \begin{equation*}
    (AB)^T = \mathbf{\begin{bmatrix} 9 & -1 \\ -4 & 2 \end{bmatrix}}
    \end{equation*}
    Evaluating $B^T A^T$:
    \begin{equation*}
    B^T = \begin{bmatrix} 4 & 1 \\ -2 & 0 \end{bmatrix}, \quad A^T = \begin{bmatrix} 2 & -1 \\ 1 & 3 \end{bmatrix}
    \end{equation*}
    \begin{equation*}
    B^T A^T = \begin{bmatrix} 4(2) + 1(1) & 4(-1) + 1(3) \\ -2(2) + 0(1) & -2(-1) + 0(3) \end{bmatrix} = \mathbf{\begin{bmatrix} 9 & -1 \\ -4 & 2 \end{bmatrix}}
    \end{equation*}
    This confirms the socks-and-shoes reversal identity: $(AB)^T = B^T A^T$.
\end{enumerate}

\textbf{3. Linear Systems: Row Picture vs. Column Picture}
System: $x_1 + 2x_2 = 5$ and $3x_1 - x_2 = 1$.
\begin{enumerate}
    \item \textbf{Matrix Form:}
    \begin{equation*}
    A\mathbf{x} = \mathbf{b} \iff \mathbf{\begin{bmatrix} 1 & 2 \\ 3 & -1 \end{bmatrix} \begin{bmatrix} x_1 \\ x_2 \end{bmatrix} = \begin{bmatrix} 5 \\ 1 \end{bmatrix}}
    \end{equation*}
    \item \textbf{Row Picture:}
    Equation 1: $x_2 = -\frac{1}{2}x_1 + \frac{5}{2}$ (Line $L_1$ with slope $m_1 = -0.5$). \\
    Equation 2: $x_2 = 3x_1 - 1$ (Line $L_2$ with slope $m_2 = 3$). \\
    Setting the two lines equal to find their intersection:
    \begin{equation*}
    3x_1 - 1 = -\frac{1}{2}x_1 + \frac{5}{2} \implies \frac{7}{2}x_1 = \frac{7}{2} \implies x_1 = 1 \implies x_2 = 3(1) - 1 = 2
    \end{equation*}
    The Row Picture interprets the solution as the unique point of intersection: $\mathbf{(x_1, x_2) = (1, 2)}$.
    \item \textbf{Column Picture:}
    The system is expressed as a linear combination of the column vectors of $A$:
    \begin{equation*}
    x_1 \begin{bmatrix} 1 \\ 3 \end{bmatrix} + x_2 \begin{bmatrix} 2 \\ -1 \end{bmatrix} = \begin{bmatrix} 5 \\ 1 \end{bmatrix}
    \end{equation*}
    Substituting the coordinates $x_1 = 1$ and $x_2 = 2$:
    \begin{equation*}
    \mathbf{1 \begin{bmatrix} 1 \\ 3 \end{bmatrix} + 2 \begin{bmatrix} 2 \\ -1 \end{bmatrix} = \begin{bmatrix} 1 + 4 \\ 3 - 2 \end{bmatrix} = \begin{bmatrix} 5 \\ 1 \end{bmatrix}}
    \end{equation*}
    The Column Picture demonstrates that 1 unit of column 1 added to 2 units of column 2 exactly constructs the target vector $\mathbf{b}$.
\end{enumerate}

\textbf{4. Determinants and Singularity Conditions}
Matrix $M_k = \begin{bmatrix} 2 & 1 & 3 \\ 0 & k & 4 \\ 1 & -1 & 2 \end{bmatrix}$.
\begin{enumerate}
    \item \textbf{Determinant via Laplace Expansion along Column 1:}
    \begin{align*}
    \det(M_k) &= 2 \begin{vmatrix} k & 4 \\ -1 & 2 \end{vmatrix} - 0 \begin{vmatrix} 1 & 3 \\ -1 & 2 \end{vmatrix} + 1 \begin{vmatrix} 1 & 3 \\ k & 4 \end{vmatrix} \\
    &= 2(2k - (-4)) + (4 - 3k) = 2(2k + 4) + 4 - 3k \\
    &= 4k + 8 + 4 - 3k = \mathbf{k + 12}
    \end{align*}
    \item \textbf{Singularity Condition:}
    A matrix is singular if and only if its determinant is zero:
    \begin{equation*}
    \det(M_k) = 0 \iff k + 12 = 0 \iff \mathbf{k = -12}
    \end{equation*}
    \item \textbf{Inverse Matrix Determinant for $k = 5$:}
    For $k = 5$, $\det(M_5) = 5 + 12 = 17 \neq 0$.
    By the determinant multiplicative property $\det(A^{-1}) = \frac{1}{\det(A)}$:
    \begin{equation*}
    \det(M_5^{-1}) = \frac{1}{\det(M_5)} = \mathbf{\frac{1}{17}}
    \end{equation*}
\end{enumerate}

\textbf{5. Covariance Matrices and Mahalanobis Distance}
Given $\boldsymbol{\mu} = \begin{bmatrix} 0 \\ 0 \end{bmatrix}$, $\Sigma = \begin{bmatrix} 5 & 2 \\ 2 & 2 \end{bmatrix}$, and $\mathbf{x} = \begin{bmatrix} 1 \\ 2 \end{bmatrix}$:
\begin{enumerate}
    \item \textbf{Generalized Variance $\det(\Sigma)$:}
    \begin{equation*}
    \det(\Sigma) = (5)(2) - (2)(2) = 10 - 4 = \mathbf{6}
    \end{equation*}
    \item \textbf{Analytical Matrix Inverse $\Sigma^{-1}$:}
    Using the $2 \times 2$ formula $\begin{bmatrix} a & b \\ c & d \end{bmatrix}^{-1} = \frac{1}{ad - bc}\begin{bmatrix} d & -b \\ -c & a \end{bmatrix}$:
    \begin{equation*}
    \Sigma^{-1} = \frac{1}{6} \begin{bmatrix} 2 & -2 \\ -2 & 5 \end{bmatrix} = \mathbf{\begin{bmatrix} 1/3 & -1/3 \\ -1/3 & 5/6 \end{bmatrix}}
    \end{equation*}
    \item \textbf{Squared Mahalanobis Distance $d_M^2$:}
    Since $\boldsymbol{\mu} = \mathbf{0}$, $\mathbf{x} - \boldsymbol{\mu} = \begin{bmatrix} 1 \\ 2 \end{bmatrix}$.
    \begin{align*}
    d_M^2 &= \begin{bmatrix} 1 & 2 \end{bmatrix} \begin{bmatrix} 1/3 & -1/3 \\ -1/3 & 5/6 \end{bmatrix} \begin{bmatrix} 1 \\ 2 \end{bmatrix} \\
    &= \begin{bmatrix} 1(1/3) + 2(-1/3) & 1(-1/3) + 2(5/6) \end{bmatrix} \begin{bmatrix} 1 \\ 2 \end{bmatrix} \\
    &= \begin{bmatrix} -1/3 & 4/3 \end{bmatrix} \begin{bmatrix} 1 \\ 2 \end{bmatrix} = \left(-\frac{1}{3}\right)(1) + \left(\frac{4}{3}\right)(2) = -\frac{1}{3} + \frac{8}{3} = \mathbf{\frac{7}{3} \approx 2.3333}
    \end{align*}
\end{enumerate}

\subsection*{Diagnostic MCQ Explanations}

\textbf{6. Transpose of Matrix Product Sequence}
\begin{itemize}
    \item \textbf{Correct Answer: (b)}
    \item \textbf{Why (b) is correct:} Transposing a matrix product reverses the order of every term: $(ABC)^T = C^T B^T A^T$. If $A \in \R^{4 \times 3}, B \in \R^{3 \times 5}, C \in \R^{5 \times 2}$, their transposes have dimensions $A^T \in \R^{3 \times 4}, B^T \in \R^{5 \times 3}, C^T \in \R^{2 \times 5}$. Evaluating $A^T B^T$ attempts to multiply a $3 \times 4$ matrix by a $5 \times 3$ matrix; because the inner dimensions (4 and 5) do not match, the expression $A^T B^T C^T$ is completely undefined. Conversely, $C^T B^T A^T$ multiplies $(2 \times 5)(5 \times 3)(3 \times 4) = 2 \times 4$, which matches $(ABC)^T \in \R^{2 \times 4}$.
    \item \textbf{Why (a) is false:} Transposition does not distribute without order reversal.
    \item \textbf{Why (c) is false:} The matrices are non-square ($4 \times 3$, etc.), so determinants do not exist for them anyway.
    \item \textbf{Why (d) is false:} Even if matrices are square and symmetric, $A^T B^T = AB \neq (AB)^T$ unless $A$ and $B$ commute.
\end{itemize}

\textbf{7. Geometric Meaning of $\det(A) = 0$}
\begin{itemize}
    \item \textbf{Correct Answer: (b)}
    \item \textbf{Why (b) is correct:} The determinant represents the factor by which the linear transformation scales the volume of any region in space. If $\det(A) = 0$, the transformation compresses non-zero volumes in 3D space down to zero volume, flattening the entire space into a 2D plane, a 1D line, or a 0D point. This loss of dimensionality means multiple distinct input vectors are mapped to the same output vector, making it impossible to invert the transformation.
    \item \textbf{Why (a) is false:} Rotations preserve volumes and have $\det(R) = +1$.
    \item \textbf{Why (c) is false:} Scaling volume by 2 corresponds to $\det(A) = 2$.
    \item \textbf{Why (d) is false:} The identity transformation has $\det(I) = 1$.
\end{itemize}

\section*{Week 2: Cramer's Rule, Row Echelon Form \& Gaussian Elimination}

\subsection*{Detailed Solutions}

\textbf{1. Cramer's Rule on a $2 \times 2$ Linear System}
Given the linear system $3x_1 + 4x_2 = 10$ and $x_1 - 2x_2 = -3$:
\begin{enumerate}
    \item \textbf{Determinant of Coefficient Matrix $A$:}
    \begin{equation*}
    \det(A) = \begin{vmatrix} 3 & 4 \\ 1 & -2 \end{vmatrix} = 3(-2) - 4(1) = -6 - 4 = \mathbf{-10}
    \end{equation*}
    Since $\det(A) = -10 \neq 0$, the system has a unique solution.
    \item \textbf{Modified Matrices and Determinants:}
    Replace column 1 with $\mathbf{b} = \begin{bmatrix} 10 \\ -3 \end{bmatrix}$:
    \begin{equation*}
    \det(A_1(\mathbf{b})) = \begin{vmatrix} 10 & 4 \\ -3 & -2 \end{vmatrix} = 10(-2) - 4(-3) = -20 + 12 = \mathbf{-8}
    \end{equation*}
    Replace column 2 with $\mathbf{b}$:
    \begin{equation*}
    \det(A_2(\mathbf{b})) = \begin{vmatrix} 3 & 10 \\ 1 & -3 \end{vmatrix} = 3(-3) - 10(1) = -9 - 10 = \mathbf{-19}
    \end{equation*}
    \item \textbf{Solution Coordinates via Cramer's Rule:}
    \begin{align*}
    x_1 &= \frac{\det(A_1(\mathbf{b}))}{\det(A)} = \frac{-8}{-10} = \mathbf{\frac{4}{5} = 0.8} \\
    x_2 &= \frac{\det(A_2(\mathbf{b}))}{\det(A)} = \frac{-19}{-10} = \mathbf{\frac{19}{10} = 1.9}
    \end{align*}
    The unique solution vector is $\mathbf{x} = \begin{bmatrix} 0.8 \\ 1.9 \end{bmatrix}$.
\end{enumerate}

\textbf{2. Classifying Echelon Forms and Pivot Structures}
\begin{itemize}
    \item \textbf{Matrix $A = \begin{bmatrix} 1 & 3 & 0 & 2 \\ 0 & 1 & 0 & 4 \\ 0 & 0 & 0 & 0 \end{bmatrix}$:}
    \textbf{Row Echelon Form (REF)}. Pivots are at $(1, 1)$ and $(2, 2)$, the zero row is at the bottom, and all entries below pivots are zero. It is \textbf{not RREF} because the entry above the second pivot is $3 \neq 0$.
    \item \textbf{Matrix $B = \begin{bmatrix} 1 & 0 & 5 \\ 0 & 1 & -2 \\ 0 & 0 & 1 \end{bmatrix}$:}
    \textbf{Row Echelon Form (REF)}. Pivots are at $(1, 1), (2, 2), (3, 3)$. It is \textbf{not RREF} because entries above the pivot in column 3 are $5 \neq 0$ and $-2 \neq 0$.
    \item \textbf{Matrix $C = \begin{bmatrix} 0 & 1 & 2 \\ 1 & 0 & 3 \end{bmatrix}$:}
    \textbf{Neither REF nor RREF}. The leading non-zero entry of row 1 is in column 2, while the leading entry of row 2 is in column 1. The pivot of row 2 lies to the left of the pivot of row 1, violating the strict staircase property. Performing row swap $R_1 \leftrightarrow R_2$ would place it in RREF.
    \item \textbf{Matrix $D = \begin{bmatrix} 1 & 0 & 3 \\ 0 & 1 & -1 \\ 0 & 0 & 0 \end{bmatrix}$:}
    \textbf{Reduced Row Echelon Form (RREF)}. Pivots are both 1, entries above and below pivots are strictly 0, and the zero row is at the bottom. Column 3 has non-zero entries, but it is a non-pivot column representing a free variable.
\end{itemize}

\textbf{3. Gaussian Elimination with Back-Substitution ($3 \times 3$)}
Linear system:
\begin{align*}
x_1 + 2x_2 + x_3 &= 8 \\
2x_1 + 3x_2 + 4x_3 &= 20 \\
4x_1 + 3x_2 + 2x_3 &= 16
\end{align*}
\begin{enumerate}
    \item \textbf{Augmented Matrix Setup:}
    \begin{equation*}
    [A \mid \mathbf{b}] = \left[\begin{array}{ccc|c}
    1 & 2 & 1 & 8 \\
    2 & 3 & 4 & 20 \\
    4 & 3 & 2 & 16
    \end{array}\right]
    \end{equation*}
    \item \textbf{Forward Elimination to REF:}
    Eliminate column 1 entries below row 1 via $R_2 \to R_2 - 2R_1$ and $R_3 \to R_3 - 4R_1$:
    \begin{equation*}
    \left[\begin{array}{ccc|c}
    1 & 2 & 1 & 8 \\
    0 & 3 - 4 & 4 - 2 & 20 - 16 \\
    0 & 3 - 8 & 2 - 4 & 16 - 32
    \end{array}\right]
    = \left[\begin{array}{ccc|c}
    1 & 2 & 1 & 8 \\
    0 & -1 & 2 & 4 \\
    0 & -5 & -2 & -16
    \end{array}\right]
    \end{equation*}
    Eliminate column 2 entry below row 2 via $R_3 \to R_3 - 5R_2$:
    \begin{equation*}
    \left[\begin{array}{ccc|c}
    1 & 2 & 1 & 8 \\
    0 & -1 & 2 & 4 \\
    0 & -5 - 5(-1) & -2 - 5(2) & -16 - 5(4)
    \end{array}\right]
    = \left[\begin{array}{ccc|c}
    1 & 2 & 1 & 8 \\
    0 & -1 & 2 & 4 \\
    0 & 0 & -12 & -36
    \end{array}\right]
    \end{equation*}
    The augmented matrix is now in Row Echelon Form.
    \item \textbf{Back-Substitution:}
    \begin{itemize}
        \item From Row 3: $-12 x_3 = -36 \implies \mathbf{x_3 = 3}$.
        \item From Row 2: $-x_2 + 2(3) = 4 \implies -x_2 + 6 = 4 \implies \mathbf{x_2 = 2}$.
        \item From Row 1: $x_1 + 2(2) + 1(3) = 8 \implies x_1 + 4 + 3 = 8 \implies \mathbf{x_1 = 1}$.
    \end{itemize}
    Unique Solution Vector: $\mathbf{x} = \mathbf{\begin{bmatrix} 1 \\ 2 \\ 3 \end{bmatrix}}$.
\end{enumerate}

\textbf{4. Free Variables, Rank, and Parametric Solution Families}
Underdetermined system $x_1 + 2x_2 - x_3 = 4$ and $2x_1 + 4x_2 + x_3 = 11$:
\begin{enumerate}
    \item \textbf{Row Reduction to RREF:}
    \begin{equation*}
    \left[\begin{array}{ccc|c}
    1 & 2 & -1 & 4 \\
    2 & 4 & 1 & 11
    \end{array}\right]
    \xrightarrow{R_2 \to R_2 - 2R_1}
    \left[\begin{array}{ccc|c}
    1 & 2 & -1 & 4 \\
    0 & 0 & 3 & 3
    \end{array}\right]
    \end{equation*}
    Normalize pivot in row 2 via $R_2 \to \frac{1}{3} R_2$:
    \begin{equation*}
    \left[\begin{array}{ccc|c}
    1 & 2 & -1 & 4 \\
    0 & 0 & 1 & 1
    \end{array}\right]
    \end{equation*}
    Eliminate entry above pivot in column 3 via $R_1 \to R_1 + R_2$:
    \begin{equation*}
    \left[\begin{array}{ccc|c}
    1 & 2 & 0 & 5 \\
    0 & 0 & 1 & 1
    \end{array}\right] \quad \text{\textbf{(RREF)}}
    \end{equation*}
    \item \textbf{Rank and Variable Identification:}
    \begin{itemize}
        \item $\rank(A) = 2$ and $\rank([A \mid \mathbf{b}]) = 2$. Since ranks are equal, the system is consistent.
        \item Number of variables $n = 3$. Since $\rank < 3$, there are infinitely many solutions.
        \item \textbf{Pivot Columns:} Columns 1 and 3 $\implies x_1$ and $x_3$ are \textbf{pivot variables}.
        \item \textbf{Free Columns:} Column 2 has no pivot $\implies x_2$ is a \textbf{free variable} (let $x_2 = t \in \R$).
    \end{itemize}
    \item \textbf{Parametric Vector Form:}
    From row 1: $x_1 + 2t = 5 \implies x_1 = 5 - 2t$. \\
    From row 2: $x_3 = 1$. \\
    Expressing as particular and homogeneous vector components:
    \begin{equation*}
    \mathbf{x} = \begin{bmatrix} x_1 \\ x_2 \\ x_3 \end{bmatrix} = \begin{bmatrix} 5 - 2t \\ t \\ 1 \end{bmatrix} = \mathbf{\begin{bmatrix} 5 \\ 0 \\ 1 \end{bmatrix} + t \begin{bmatrix} -2 \\ 1 \\ 0 \end{bmatrix}}, \quad \forall t \in \R
    \end{equation*}
    The solution set is a 1-dimensional line in $\R^3$ passing through $\mathbf{x}_p = [5, 0, 1]^T$ along direction $\mathbf{v}_h = [-2, 1, 0]^T$.
\end{enumerate}

\textbf{5. Ordinary Least Squares (OLS) Normal Equations}
Given $(X^T X)\mathbf{w} = X^T \mathbf{y}$ with $X^T X = \begin{bmatrix} 4 & 2 \\ 2 & 3 \end{bmatrix}$ and $X^T \mathbf{y} = \begin{bmatrix} 10 \\ 9 \end{bmatrix}$:
\begin{enumerate}
    \item \textbf{Invertibility Check:}
    \begin{equation*}
    \det(X^T X) = (4)(3) - (2)(2) = 12 - 4 = \mathbf{8} \neq 0
    \end{equation*}
    Since $\det(X^T X) = 8 > 0$, the normal equations matrix is non-singular and strictly invertible.
    \item \textbf{Solution via Cramer's Rule:}
    \begin{align*}
    w_0 &= \frac{\det \begin{bmatrix} 10 & 2 \\ 9 & 3 \end{bmatrix}}{\det(X^T X)} = \frac{10(3) - 2(9)}{8} = \frac{30 - 18}{8} = \frac{12}{8} = \mathbf{1.5} \\
    w_1 &= \frac{\det \begin{bmatrix} 4 & 10 \\ 2 & 9 \end{bmatrix}}{\det(X^T X)} = \frac{4(9) - 10(2)}{8} = \frac{36 - 20}{8} = \frac{16}{8} = \mathbf{2.0}
    \end{align*}
    Optimal weight vector: $\mathbf{w}^* = \mathbf{\begin{bmatrix} 1.5 \\ 2.0 \end{bmatrix}}$.
    \item \textbf{Confirmation via Matrix Inversion:}
    \begin{equation*}
    (X^T X)^{-1} = \frac{1}{8} \begin{bmatrix} 3 & -2 \\ -2 & 4 \end{bmatrix}
    \end{equation*}
    \begin{equation*}
    \mathbf{w}^* = (X^T X)^{-1} (X^T \mathbf{y}) = \frac{1}{8} \begin{bmatrix} 3 & -2 \\ -2 & 4 \end{bmatrix} \begin{bmatrix} 10 \\ 9 \end{bmatrix} = \frac{1}{8} \begin{bmatrix} 3(10) - 2(9) \\ -2(10) + 4(9) \end{bmatrix} = \frac{1}{8} \begin{bmatrix} 12 \\ 16 \end{bmatrix} = \mathbf{\begin{bmatrix} 1.5 \\ 2.0 \end{bmatrix}}
    \end{equation*}
\end{enumerate}

\subsection*{Diagnostic MCQ Explanations}

\textbf{6. Contradiction Rows and System Inconsistency}
\begin{itemize}
    \item \textbf{Correct Answer: (b)}
    \item \textbf{Why (b) is correct:} Row 3 of the augmented matrix translates to the algebraic equation $0x_1 + 0x_2 + 0x_3 = 5$, which simplifies to $0 = 5$. Since zero cannot equal five, no set of real values for $(x_1, x_2, x_3)$ can ever satisfy this constraint. Under the Rouché-Capelli Theorem, $\rank(A) = 2$ while $\rank([A \mid \mathbf{b}]) = 3$. Because $\rank(A) < \rank([A \mid \mathbf{b}])$, the system is strictly inconsistent and has \textbf{no solutions}.
    \item \textbf{Why (a) is false:} Free variables only yield infinite solutions if the system is consistent (no contradiction row).
    \item \textbf{Why (c) is false:} Zero rows in REF are already correctly positioned at the bottom.
    \item \textbf{Why (d) is false:} Cramer's Rule only applies to square matrices with $\det(A) \neq 0$; here the system is non-square ($4 \times 3$).
\end{itemize}

\textbf{7. Complexity of Gaussian Elimination vs. Cramer's Rule}
\begin{itemize}
    \item \textbf{Correct Answer: (b)}
    \item \textbf{Why (b) is correct:} Cramer's Rule requires calculating $n + 1$ separate determinants of size $n \times n$. Computing a determinant via standard cofactor expansion scales factorially as $\mathcal{O}((n+1)!)$, making it impossible to solve even moderate systems (e.g. $n = 50$). In contrast, Gaussian Elimination requires only $\frac{2}{3}n^3$ arithmetic operations ($\mathcal{O}(n^3)$), which runs in milliseconds on modern hardware.
    \item \textbf{Why (a) is false:} Cramer's Rule works on negative and positive entries alike.
    \item \textbf{Why (c) is false:} Gaussian Elimination works on all matrices, but OLS solves systems where $X^T X$ is invertible.
    \item \textbf{Why (d) is false:} Cramer's Rule does not compute or alter eigenvalues.
\end{itemize}

\section*{Week 3: Vector Spaces, Subspaces \& Linear Independence}

\subsection*{Detailed Solutions}

\textbf{1. Subspace Test and Geometric Verification}
\begin{enumerate}
    \item \textbf{Subset $W_1 = \left\{ [x, y, z]^T \in \R^3 \mid 2x - y + 3z = 0 \right\}$:}
    \begin{itemize}
        \item \textbf{Zero Vector:} For $\mathbf{0} = [0, 0, 0]^T$, $2(0) - (0) + 3(0) = 0$. Thus $\mathbf{0} \in W_1$.
        \item \textbf{Closure under Addition:} Let $\mathbf{u} = [x_1, y_1, z_1]^T, \mathbf{v} = [x_2, y_2, z_2]^T \in W_1$, so $2x_1 - y_1 + 3z_1 = 0$ and $2x_2 - y_2 + 3z_2 = 0$.
        For $\mathbf{u} + \mathbf{v} = [x_1 + x_2, y_1 + y_2, z_1 + z_2]^T$:
        \begin{equation*}
        2(x_1 + x_2) - (y_1 + y_2) + 3(z_1 + z_2) = (2x_1 - y_1 + 3z_1) + (2x_2 - y_2 + 3z_2) = 0 + 0 = 0
        \end{equation*}
        Thus $\mathbf{u} + \mathbf{v} \in W_1$.
        \item \textbf{Closure under Scalar Multiplication:} For any scalar $c \in \R$:
        \begin{equation*}
        2(cx_1) - (cy_1) + 3(cz_1) = c(2x_1 - y_1 + 3z_1) = c(0) = 0 \implies c\mathbf{u} \in W_1
        \end{equation*}
    \end{itemize}
    $W_1$ satisfies all 3 criteria and is a \textbf{valid subspace of $\R^3$} (a plane passing through the origin).

\textbf{2. Linear Independence Testing in $\R^3$}
Given vectors $\mathbf{v}_1 = \begin{bmatrix} 1 \\ 2 \\ 3 \end{bmatrix}, \mathbf{v}_2 = \begin{bmatrix} 0 \\ 1 \\ 4 \end{bmatrix}, \mathbf{v}_3 = \begin{bmatrix} 5 \\ 6 \\ 0 \end{bmatrix}$:

\end{enumerate}

\textbf{1. Basis Verification \& Coordinate Representation}
\begin{enumerate}
    \item \textbf{Basis Verification in $\mathbb{R}^2$:}
    A set of 2 vectors forms a basis for $\mathbb{R}^2$ if and only if they are linearly independent. Form the matrix $M$ with the vectors as columns:
    \begin{equation*}
    M = [\mathbf{b}_1 \; \mathbf{b}_2] = \begin{bmatrix} 1 & 1 \\ 1 & -1 \end{bmatrix}

    \begin{equation*}
    \det(M) = (1)(-1) - (1)(1) = -1 - 1 = -2 \neq 0
    \end{equation*}

    \end{equation*}

\textbf{2. Finding a Basis and Dimension for Subspace $W$}
Given subspace $W = \left\{ [x, y, z]^T \in \mathbb{R}^3 \mid x + 2y - z = 0 \right\}$:
\begin{enumerate}

\begin{enumerate}

    \begin{itemize}

        \item \textbf{Normal Equations Invertibility:} The Gram matrix $X^T X \in \mathbb{R}^{5 \times 5}$ has rank equal to $\rank(X) = 3$. Because $\rank(X^T X) = 3 < 5$, $X^T X$ is \textbf{singular (non-invertible)}. \textbf{2. Testing Linearity of Mappings}
\begin{enumerate}
    \item \textbf{(a)} $T\left(\begin{bmatrix} x \\ y \end{bmatrix}\right) = \begin{bmatrix} 2x - y \\ x + 3y \end{bmatrix}$:

    \begin{equation*}

    \end{equation*}
    Since every matrix-vector product preserves vector addition ($A(\mathbf{u}+\mathbf{v}) = A\mathbf{u}+A\mathbf{v}$) and scalar multiplication ($A(c\mathbf{u}) = cA\mathbf{u}$), $T$ is a \textbf{valid linear transformation}.

    \item \textbf{(b)} $T\left(\begin{bmatrix} x \\ y \end{bmatrix}\right) = \begin{bmatrix} x^2 \\ y \end{bmatrix}$:
    Test scalar multiplication with $\mathbf{v} = \begin{bmatrix} 1 \\ 1 \end{bmatrix}$ and scalar $c = 2$:
    \begin{equation*}
    T(c\mathbf{v}) = T\left(\begin{bmatrix} 2 \\ 2 \end{bmatrix}\right) = \begin{bmatrix} 2^2 \\ 2 \end{bmatrix} = \begin{bmatrix} 4 \\ 2 \end{bmatrix}
    \end{equation*}
    However, $c\, T(\mathbf{v}) = 2 T\left(\begin{bmatrix} 1 \\ 1 \end{bmatrix}\right) = 2 \begin{bmatrix} 1^2 \\ 1 \end{bmatrix} = 2 \begin{bmatrix} 1 \\ 1 \end{bmatrix} = \begin{bmatrix} 2 \\ 2 \end{bmatrix}$.
    Since $\begin{bmatrix} 4 \\ 2 \end{bmatrix} \neq \begin{bmatrix} 2 \\ 2 \end{bmatrix}$, $T(c\mathbf{v}) \neq c\, T(\mathbf{v})$.
    Therefore, $T$ fails scalar homogeneity and is \textbf{NOT a linear transformation}.

    Test the zero vector condition:
    \begin{equation*}
    T\left(\begin{bmatrix} 0 \\ 0 \end{bmatrix}\right) = \begin{bmatrix} 0 + 1 \\ 0 - 2 \end{bmatrix} = \begin{bmatrix} 1 \\ -2 \end{bmatrix} \neq \begin{bmatrix} 0 \\ 0 \end{bmatrix}

    Because $T(\mathbf{0}) \neq \mathbf{0}$, $T$ fails the foundational necessary condition. Thus, $T$ is an affine translation, \textbf{NOT a linear transformation}.

Given $A = \begin{bmatrix} 1 & 2 \\ 0 & 3 \end{bmatrix}$ and $B = \begin{bmatrix} 3 & 0 \\ 1 & 1 \end{bmatrix}$:
\begin{enumerate}

    \end{equation*}
    \item \textbf{Traces:}
    \begin{equation*}
    \trace(A) = 1 + 3 = \mathbf{4}, \quad \trace(B) = 3 + 1 = \mathbf{4}
    \end{equation*}
    \item \textbf{Similarity Determination:}
    Because $A$ is upper triangular, its eigenvalues are its diagonal entries: $\lambda_1 = 1, \lambda_2 = 3$.
    Because $B$ is lower triangular, its eigenvalues are its diagonal entries: $\lambda_1 = 3, \lambda_2 = 1$.
    Since both $2 \times 2$ matrices have two distinct real eigenvalues $\{1, 3\}$, each is diagonalizable to the same diagonal matrix $D = \begin{bmatrix} 1 & 0 \\ 0 & 3 \end{bmatrix}$.
    Thus, there exist invertible matrices $P_1, P_2$ such that $P_1^{-1} A P_1 = D$ and $P_2^{-1} B P_2 = D$, which implies:

    B = P_2 D P_2^{-1} = P_2 (P_1^{-1} A P_1) P_2^{-1} = (P_1 P_2^{-1})^{-1} A (P_1 P_2^{-1})
    \end{equation*}
    Setting $P = P_1 P_2^{-1}$, we have $B = P^{-1} A P$. Therefore, \textbf{$A$ and $B$ are strictly similar} ($A \sim B$).
\end{enumerate}

\textbf{4. Orthogonal Matrix Properties}
Let $Q = \begin{bmatrix} \cos \theta & -\sin \theta \\ \sin \theta & \cos \theta \end{bmatrix}$:
\begin{enumerate}
    \item \textbf{Product $Q^T Q$:}
    \begin{equation*}
    Q^T Q = \begin{bmatrix} \cos\theta & \sin\theta \\ -\sin\theta & \cos\theta \end{bmatrix} \begin{bmatrix} \cos\theta & -\sin\theta \\ \sin\theta & \cos\theta \end{bmatrix} = \begin{bmatrix} \cos^2\theta + \sin^2\theta & 0 \\ 0 & \sin^2\theta + \cos^2\theta \end{bmatrix} = \begin{bmatrix} 1 & 0 \\ 0 & 1 \end{bmatrix} = I
    \end{equation*}
    Because $Q^T Q = I$, $Q$ is orthogonal and $Q^{-1} = Q^T$.
    \item \textbf{Length Preservation (Isometry):}
    \begin{equation*}
    \|Q\mathbf{x}\|^2 = (Q\mathbf{x})^T (Q\mathbf{x}) = \mathbf{x}^T (Q^T Q) \mathbf{x} = \mathbf{x}^T I \mathbf{x} = \mathbf{x}^T \mathbf{x} = \|\mathbf{x}\|^2 \implies \mathbf{\|Q\mathbf{x}\| = \|\mathbf{x}\|}
    \end{equation*}
    \item \textbf{Inner Product \& Angle Preservation:}
    \begin{equation*}
    \langle Q\mathbf{x}, Q\mathbf{y} \rangle = (Q\mathbf{x})^T (Q\mathbf{y}) = \mathbf{x}^T (Q^T Q) \mathbf{y} = \mathbf{x}^T I \mathbf{y} = \mathbf{x}^T \mathbf{y} = \langle \mathbf{x}, \mathbf{y} \rangle
    \end{equation*}
\end{enumerate}

\textbf{2. Classifying 2D Critical Points}

\textbf{5. Data Science Application: Convexity of Mean Squared Error}
For linear regression with loss $L(w) = \frac{1}{N} \sum_{i=1}^N (w x_i - y_i)^2$:
\begin{enumerate}
    \item \textbf{First Derivative:}
    \begin{equation*}
    \frac{dL}{dw} = \frac{1}{N} \sum_{i=1}^N 2(w x_i - y_i) \cdot x_i = \frac{2}{N} \sum_{i=1}^N (w x_i^2 - x_i y_i)
    \end{equation*}
    \item \textbf{Second Derivative (1D Hessian):}
    \begin{equation*}
    \frac{d^2 L}{dw^2} = \frac{2}{N} \sum_{i=1}^N x_i^2
    \end{equation*}
    \item \textbf{Convexity Verification:}
    Because $x_i \in \mathbb{R}$, $x_i^2 \ge 0$ for all $i$. Thus $\sum_{i=1}^N x_i^2 \ge 0$.
    As long as the dataset contains at least one non-zero feature value ($\sum x_i^2 > 0$), we have $\frac{d^2 L}{dw^2} > 0$ strictly for all $w \in \mathbb{R}$.
    Since the second derivative is strictly positive everywhere, $L(w)$ is \textbf{strictly convex}.
    Consequently, $L(w)$ contains no local maxima, no saddle points, and has a \textbf{single unique global minimum} given by $w^* = \frac{\sum x_i y_i}{\sum x_i^2}$.
\end{enumerate}
