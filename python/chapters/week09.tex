\chapter{Recursion, Algorithmic Search \& Divide-and-Conquer}

Fundamental computational problems in data science---from recursive decision-tree splitting in CART algorithms and hierarchical clustering to nearest-neighbor searches in high-dimensional KD-trees---rely on \index{Recursion}\textbf{Recursion} and \textbf{Divide-and-Conquer} strategies. Rather than tackling an intractable problem monolithically, divide-and-conquer breaks a dataset into disjoint subproblems, recursively solves each partition, and recombines the solutions. In Python, recursion is executed via activation records allocated on the C-level call stack. Understanding call stack mechanics, recursion limits, recurrence relations under the Master Theorem, and the logarithmic efficiency of \index{Binary Search}\textbf{Binary Search} provides the foundation for scalable algorithmic design.

\section{Mathematical \& Computational Foundations}

\subsection{Call Stack Mechanics and Activation Records}

\begin{definitionbox}{The Activation Frame Invariant}
\index{Call Stack} \index{Activation Record}
When a function is called, the CPython runtime pushes an \textbf{activation record (stack frame)} onto the call stack. Each frame encapsulates:
\begin{enumerate}
    \item Parameter references passed by the caller.
    \item Local variables declared within the current scope.
    \item The return address pointing to the caller's next bytecode instruction.
\end{enumerate}
\textbf{The CPython Recursion Constraint:} Python does \textbf{not} perform Tail-Call Optimization (TCO). Every nested call consumes physical memory on the process stack.
To prevent uncontrolled stack overflows from crashing the underlying operating system process, CPython enforces a strict limit:
\begin{lstlisting}[language=Python]
import sys
print(sys.getrecursionlimit())  # Typically 1000 frames
\end{lstlisting}
Exceeding this boundary immediately raises a \texttt{RecursionError: maximum recursion depth exceeded}.
\end{definitionbox}

\subsection{Recurrence Relations and the Master Theorem}

\begin{theorembox}{The Master Theorem for Divide-and-Conquer}
\index{Master Theorem} \index{Divide-and-Conquer}
Let an algorithm divide a problem of size $n$ into $a \ge 1$ subproblems, each of size $n/b$ (with $b > 1$), incurring recombination cost $f(n) = \Theta(n^c)$:
\begin{equation}
T(n) = a T\left(\frac{n}{b}\right) + \Theta(n^c)
\end{equation}
The asymptotic complexity is determined by comparing the work done at the leaves ($\log_b a$) with the work at the root ($c$):
\begin{enumerate}
    \item \textbf{Leaf-dominated ($c < \log_b a$):} $T(n) = \Theta(n^{\log_b a})$.
    \item \textbf{Balanced ($c = \log_b a$):} $T(n) = \Theta(n^c \log n)$.
    \item \textbf{Root-dominated ($c > \log_b a$):} $T(n) = \Theta(n^c)$.
\end{enumerate}
\textbf{Applications:}
\begin{itemize}
    \item \textbf{Binary Search:} $a = 1, b = 2, c = 0 \implies \log_2 1 = 0 = c \implies T(n) = \mathbf{\Theta(\log n)}$.
    \item \textbf{Merge Sort:} $a = 2, b = 2, c = 1 \implies \log_2 2 = 1 = c \implies T(n) = \mathbf{\Theta(n \log n)}$.
\end{itemize}
\end{theorembox}

\subsection{Binary Search Algorithm: Discrete and Continuous}

\begin{definitionbox}{Discrete vs. Continuous Binary Search}
\index{Binary Search} \index{Bisection Method}
\begin{enumerate}
    \item \textbf{Discrete Binary Search on Sorted Array $A[0 \dots n-1]$:}
    Maintains search interval $[\text{low}, \text{high}]$. In each step:
    \begin{equation}
    \text{mid} = \text{low} + \frac{\text{high} - \text{low}}{2}
    \end{equation}
    If $A[\text{mid}] == \text{target}$, return index. If $A[\text{mid}] < \text{target}$, update $\text{low} = \text{mid} + 1$; else $\text{high} = \text{mid} - 1$.
    \item \textbf{Continuous Root-Finding (Bisection Method):}
    Let $f: [a, b] \to \mathbb{R}$ be continuous with $f(a) \cdot f(b) \le 0$. The midpoint $m = \frac{a+b}{2}$ halves the interval. After $k$ iterations, the error is bounded by:
    \begin{equation}
    \epsilon_k = \frac{b - a}{2^k} \implies k = \left\lceil \log_2\left(\frac{b - a}{\text{tolerance}}\right) \right\rceil
    \end{equation}
\end{enumerate}
\end{definitionbox}

\section{Deep Intuition \& Conceptual Derivations}

\subsection{Overlapping Subproblems and Memoization}
Naive recursive Fibonacci computation exhibits recurrence $T(n) = T(n-1) + T(n-2) + \mathcal{O}(1)$, forming a binary computation tree of depth $n$ with total calls scaling as:
\begin{equation}
T(n) \ge 2^{n/2} = \Theta\left(\left(\frac{1 + \sqrt{5}}{2}\right)^n\right) \approx \Theta(1.618^n)
\end{equation}
Because the tree repeatedly recomputes identical states (overlapping subproblems), caching results via a hash table (**Memoization**) collapses the tree into a directed acyclic graph (DAG) of length $n$, reducing execution complexity from exponential $\mathcal{O}(2^n)$ to strictly linear $\mathcal{O}(n)$.

\subsection{Divide-and-Conquer: Merge Sort Mechanics}
Merge Sort recursively bisects an array into two halves of size $n/2$, sorts each recursively, and performs a two-pointer linear merge in $\mathcal{O}(n)$ time:
\begin{equation}
T(n) = 2 T(n/2) + c n \implies T(n) = c n \log_2 n = \mathcal{O}(n \log n)
\end{equation}
Unlike Quick Sort (whose worst-case degrades to $\mathcal{O}(n^2)$ on already sorted arrays without randomized pivoting), Merge Sort provides an absolute worst-case guarantee of $\mathcal{O}(n \log n)$ comparisons and is inherently stable.

\section{Explicit Gotchas \& Diagnostic Misconceptions}

\begin{trapbox}{The Lack of Tail-Call Optimization (TCO) in Python}
In languages like Scheme or Haskell, a tail-recursive function consumes $\mathcal{O}(1)$ stack space. In Python, \textbf{no tail-call elimination occurs}:
\begin{lstlisting}[language=Python]
def count_down(n):
    if n == 0:
        return 0
    return count_down(n - 1)  # Consumes full stack frame!

count_down(2000)  # Raises RecursionError!
\end{lstlisting}
Even if the recursive call is the final statement in the function, it allocates a new stack frame. Deep iterative processes in Python should always use a standard \texttt{while} loop or an explicit user-space list stack.
\end{trapbox}

\begin{trapbox}{The Mutable Default Argument Memoization Trap}
Attempting to implement a memoization cache using default arguments can leak state across tests:
\begin{lstlisting}[language=Python]
def compute(n, memo={}):  # CRITICAL BUG!
    if n in memo:
        return memo[n]
    # ...
\end{lstlisting}
The \texttt{memo} dictionary persists across all calls globally. If one call corrupts the cache or depends on external dynamic weights, all subsequent invocations receive stale, corrupted answers. Use \texttt{@functools.lru\_cache(maxsize=None)} instead.
\end{trapbox}

\begin{trapbox}{Binary Search Integer Midpoint Calculation}
In lower-level languages, calculating \texttt{mid = (low + high) // 2} overflows 32-bit signed integers if $\text{low} + \text{high} > 2^{31}-1$. While Python 3 integers have arbitrary precision and will not overflow, the form \texttt{mid = low + (high - low) // 2} remains the universal algorithmic standard across all production software.
\end{trapbox}

\begin{trapbox}{Binary Search Infinite Loop on Update Bounds}
Setting \texttt{low = mid} or \texttt{high = mid} instead of \texttt{mid + 1} / \texttt{mid - 1} causes infinite loops when \texttt{high - low == 1}:
\begin{lstlisting}[language=Python]
# BUG: If low = 0, high = 1, mid = (0 + 1)//2 = 0.
# If target > arr[0], setting low = mid sets low = 0 again! Infinite loop!
\end{lstlisting}
Always advance past the midpoint: \texttt{low = mid + 1} and \texttt{high = mid - 1}.
\end{trapbox}

\section{Python / Scientific Computing Lab}

The following production script implements pure-Python Merge Sort with recursive divide-and-conquer mechanics, and a Continuous Bisection Method Root Finder for finding mathematical roots of non-linear loss gradients.

\begin{lstlisting}[language=Python]
from typing import List, Callable

def merge_sort(arr: List[float]) -> List[float]:
    """
    Sorts an array using Divide-and-Conquer Merge Sort in O(n log n) time.
    """
    if len(arr) <= 1:
        return arr

    # Divide step: bisect array at midpoint
    mid = len(arr) // 2
    left_half = merge_sort(arr[:mid])
    right_half = merge_sort(arr[mid:])

    # Conquer/Merge step: two-pointer linear merge
    merged = []
    i = j = 0
    while i < len(left_half) and j < len(right_half):
        if left_half[i] <= right_half[j]:  # <= ensures sorting stability
            merged.append(left_half[i])
            i += 1
        else:
            merged.append(right_half[j])
            j += 1

    # Append remaining elements
    merged.extend(left_half[i:])
    merged.extend(right_half[j:])
    return merged

def bisection_root_finder(
    f: Callable[[float], float],
    a: float,
    b: float,
    tol: float = 1e-6,
    max_iters: int = 100
) -> dict:
    """
    Continuous Binary Search (Bisection Method) for finding root x* where f(x*) = 0.
    Requires f(a) and f(b) to have opposite signs: f(a)*f(b) <= 0.
    """
    fa, fb = f(a), f(b)
    if fa * fb > 0:
        raise ValueError("Root is not bracketed: f(a) and f(b) must have opposite signs.")

    low, high = a, b
    for step in range(1, max_iters + 1):
        mid = low + (high - low) / 2.0
        f_mid = f(mid)

        # Check convergence: absolute function value or interval width
        if abs(f_mid) < tol or (high - low) / 2.0 < tol:
            return {
                "converged": True,
                "root": round(mid, 6),
                "f_root": round(f_mid, 8),
                "iterations": step
            }

        # Update half-interval
        if fa * f_mid < 0:
            high = mid
            fb = f_mid
        else:
            low = mid
            fa = f_mid

    return {"converged": False, "root": round(mid, 6), "iterations": max_iters}

# Unit tests
test_data = [64.2, 25.1, 12.0, 22.8, 11.5, 90.3, 44.1]
sorted_data = merge_sort(test_data)
print("Merge Sort Output:", sorted_data)

# Find root of f(x) = x^3 - 2*x - 5 on interval [1, 3]
loss_gradient = lambda x: x**3 - 2.0*x - 5.0
root_info = bisection_root_finder(loss_gradient, a=1.0, b=3.0, tol=1e-6)
print(f"Root: x* = {root_info['root']} found in {root_info['iterations']} steps.")
# Output:
# Merge Sort Output: [11.5, 12.0, 22.8, 25.1, 44.1, 64.2, 90.3]
# Root: x* = 2.094551 found in 22 steps.
\end{lstlisting}
