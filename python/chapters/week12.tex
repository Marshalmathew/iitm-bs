\section{Mathematical \& Computational Foundations}

\subsection{Pandas Series, DataFrames, and Index Alignment}

\begin{definitionbox}{Series and DataFrame Anatomy}
\index{Series} \index{DataFrame} \index{Index Alignment}
\begin{enumerate}
    \item \textbf{Series:} A 1D labeled homogeneous array combining a NumPy data buffer with an immutable \texttt{Index} object:
    \begin{equation}
    S = \{(\text{idx}_i, v_i) \mid \text{idx}_i \in \mathcal{I}, \, v_i \in \mathcal{V}\}
    \end{equation}
    \item \textbf{DataFrame:} A 2D tabular structure composed of an ordered collection of columns, each sharing a common row \texttt{Index}. Internally, Pandas groups columns of identical \texttt{dtype} into 2D NumPy array chunks called **Block Managers** to maximize memory efficiency.
    \item \textbf{Automatic Index Alignment Invariant:} When performing arithmetic between two Series ($S_1 + S_2$), Pandas aligns data on matching index labels (set union $\mathcal{I}_1 \cup \mathcal{I}_2$), NOT by positional order. Indices present in one Series but absent in the other produce \texttt{NaN} (missing value) automatically.
\end{enumerate}
\end{definitionbox}

\subsection{Deterministic Selection: \texttt{.loc} vs. \texttt{.iloc}}

\begin{theorembox}{The Slicing Boundary Disparity}
\index{.loc} \index{.iloc}
Pandas provides two explicit indexing operators to eliminate ambiguities:
\begin{enumerate}
    \item \textbf{\texttt{.iloc[row\_pos, col\_pos]} (Integer Position):}
    Operates strictly on standard 0-indexed integer offsets $[0, N-1]$. Slicing follows standard Python half-open intervals:
    \begin{equation}
    \text{\texttt{df.iloc[0:3]}} \implies \text{Rows at positions } 0, 1, 2 \quad (\text{Stop is exclusive})
    \end{equation}
    \item \textbf{\texttt{.loc[row\_label, col\_label]} (Label Based):}
    Operates on index labels and column names. Slicing follows closed intervals:
    \begin{equation}
    \text{\texttt{df.loc['A':'C']}} \implies \text{Rows labeled } 'A', 'B', 'C' \quad (\text{\textbf{Both endpoints inclusive!}})
    \end{equation}
\end{enumerate}
\end{theorembox}

\subsection{The Split-Apply-Combine Groupby Calculus}

\begin{definitionbox}{The Formal Groupby Operator}
\index{Groupby} \index{Split-Apply-Combine}
Let a dataset $D = \{(x_i, y_i)\}_{i=1}^N$ possess a categorical key $k(x_i) \in \mathcal{K}$. The Groupby operator performs three discrete stages:
\begin{enumerate}
    \item \textbf{Split:} Partitions index set $\{1, \dots, N\}$ into disjoint subsets $I_k = \{i \mid k(x_i) = k\}$.
    \item \textbf{Apply:} Computes an aggregation or transformation function $g: \mathbb{R}^{|I_k|} \to \mathbb{R}$ independently on each partition (e.g., sample mean $\bar{y}_k = \frac{1}{|I_k|} \sum_{i \in I_k} y_i$).
    \item \textbf{Combine:} Stitches the aggregated scalar results back into a single labeled DataFrame indexed by $\mathcal{K}$.
\end{enumerate}
\end{definitionbox}

\section{Deep Intuition \& Conceptual Derivations}

\subsection{Missing Data Mechanics: The IEEE 754 NaN Invariant}
In Pandas, missing numerical entries default to \texttt{np.nan} (Not a Number), conforming to IEEE 754 floating-point standards:
\begin{equation}
\text{\texttt{np.nan == np.nan}} \implies \mathbf{False}
\end{equation}
Because \texttt{NaN} never equals itself, conditional checks like \texttt{if x == np.nan:} silently fail! We must use \texttt{pd.isna(x)} or \texttt{np.isnan(x)}.
\textbf{Imputation Strategies:}
\begin{itemize}
    \item \textbf{Mean Imputation:} $\hat{x} = \frac{1}{|V|} \sum_{i \in V} x_i$ (preserves sample mean, shrinks sample variance).
    \item \textbf{Median Imputation:} Robust to extreme outliers in skewed feature distributions.
\end{itemize}

\subsection{Statistical EDA Visualizations}
\begin{enumerate}
    \item \textbf{Box Plots and the IQR Rule:} Visualizes the five-number summary: minimum, first quartile $Q_1$, median $Q_2$, third quartile $Q_3$, and maximum. Outliers are defined as points falling outside the Tukey fences:
    \begin{equation}
    x < Q_1 - 1.5 \times \text{IQR} \quad \lor \quad x > Q_3 + 1.5 \times \text{IQR}, \quad \text{where } \text{IQR} = Q_3 - Q_1
    \end{equation}
    \item \textbf{Histograms and Density Estimation:} Partitions data into $B$ bins of width $\Delta = \frac{\max(x) - \min(x)}{B}$, estimating empirical probability density.
\end{enumerate}

\section{Explicit Gotchas \& Diagnostic Misconceptions}

\begin{trapbox}{The \texttt{SettingWithCopyWarning} Chained Indexing Bug}
Attempting to assign values using chained brackets causes silent failures or warnings:
\begin{lstlisting}[language=Python]
# DANGEROUS: Chained indexing
df[df["Salary"] > 50000]["Bonus"] = 5000  # Raises SettingWithCopyWarning!
\end{lstlisting}
The first bracket \texttt{df[df["Salary"] > 50000]} may return a temporary copy rather than a view. The subsequent assignment mutates the temporary copy, leaving the original DataFrame \texttt{df} completely unmodified!
\textbf{The Correct Atomic Indexing Solution:} Always use a single \texttt{.loc} call:
\begin{lstlisting}[language=Python]
df.loc[df["Salary"] > 50000, "Bonus"] = 5000  # Guaranteed in-place update!
\end{lstlisting}
\end{trapbox}

\begin{trapbox}{Inclusive Endpoint Slicing in \texttt{.loc} vs \texttt{.iloc}}
Programmers frequently introduce off-by-one errors when switching between \texttt{.loc} and \texttt{.iloc}:
\begin{lstlisting}[language=Python]
# df has integer index: 0, 1, 2, 3, 4
sub1 = df.iloc[0:2]  # Returns 2 rows: indices 0 and 1! (Half-open [0, 2))
sub2 = df.loc[0:2]   # Returns 3 rows: indices 0, 1, AND 2! (Closed [0, 2])
\end{lstlisting}
Remember that \texttt{.loc} treats numerical endpoints as labels, including both boundaries.
\end{trapbox}

\begin{trapbox}{Stateful \texttt{plt.} vs Object-Oriented \texttt{fig, ax}}
Relying on stateful functions (\texttt{plt.plot()}, \texttt{plt.title()}) in complex scripts causes state pollution where figures inadvertently draw on top of previously opened plots. Always instantiate explicit objects:
\begin{lstlisting}[language=Python]
fig, ax = plt.subplots(figsize=(8, 4))
ax.plot(x, y)
ax.set_title("Title")  # Method on specific axes instance!
\end{lstlisting}
\end{trapbox}

\begin{trapbox}{Inadvertent Index Corruption During Groupby}
By default, \texttt{df.groupby('Category').mean()} converts the grouping column into the DataFrame's index. To prevent downstream errors when treating the grouping key as a regular feature column, use \texttt{as\_index=False} or invoke \texttt{.reset\_index()}.
\end{trapbox}

\section{Python / Scientific Computing Lab}

The following industrial script executes an end-to-end Exploratory Data Analysis (EDA) Pipeline on customer transactions: missing value imputation, group-level feature engineering, outlier detection via the Interquartile Range, and generation of a multi-panel Matplotlib dashboard.

\begin{lstlisting}[language=Python]
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# 1. Construct raw stochastic dataset with missing records and outliers
np.random.seed(42)
n_records = 200
data = {
    "Department": np.random.choice(["Engineering", "Sales", "Marketing"], size=n_records),
    "Experience_Years": np.random.uniform(1.0, 15.0, size=n_records),
    "Salary": np.random.normal(75000, 15000, size=n_records),
    "Performance_Score": np.random.uniform(50.0, 100.0, size=n_records)
}
df = pd.DataFrame(data)

# Inject missing values and extreme outliers
df.loc[np.random.choice(n_records, 10, replace=False), "Salary"] = np.nan
df.loc[0, "Salary"] = 250000.0  # High outlier

# 2. Robust Preprocessing Pipeline
# Clean missing values using department-level median imputation
df["Salary"] = df.groupby("Department")["Salary"].transform(lambda grp: grp.fillna(grp.median()))

# Outlier Detection using IQR method
q1 = df["Salary"].quantile(0.25)
q3 = df["Salary"].quantile(0.75)
iqr = q3 - q1
lower_fence = q1 - 1.5 * iqr
upper_fence = q3 + 1.5 * iqr
outliers = df[(df["Salary"] < lower_fence) | (df["Salary"] > upper_fence)]
print(f"[EDA] Detected {len(outliers)} salary outliers.")

# Groupby Aggregation Summary
dept_summary = df.groupby("Department", as_index=False).agg(
    Avg_Salary=("Salary", "mean"),
    Median_Salary=("Salary", "median"),
    Avg_Score=("Performance_Score", "mean"),
    Headcount=("Salary", "count")
)
print("Departmental Summary:\n", dept_summary.round(2))

# 3. Multi-Panel Object-Oriented Matplotlib Visualization
fig, axes = plt.subplots(1, 2, figsize=(12, 5))

# Subplot 1: Experience vs Salary Scatter Plot with Regression Trend
ax1 = axes[0]
for dept, color in zip(["Engineering", "Sales", "Marketing"], ["blue", "green", "red"]):
    subset = df[df["Department"] == dept]
    ax1.scatter(subset["Experience_Years"], subset["Salary"], label=dept, color=color, alpha=0.6)

ax1.set_title("Salary vs. Experience by Department")
ax1.set_xlabel("Years of Experience")
ax1.set_ylabel("Salary ($)")
ax1.legend()
ax1.grid(True, linestyle="--", alpha=0.5)

# Subplot 2: Department Salary Distribution Boxplot
ax2 = axes[1]
salary_data = [df[df["Department"] == d]["Salary"] for d in ["Engineering", "Sales", "Marketing"]]
ax2.boxplot(salary_data, tick_labels=["Engineering", "Sales", "Marketing"], patch_artist=True)
ax2.set_title("Salary Distributions (Outlier Inspection)")
ax2.set_ylabel("Salary ($)")
ax2.grid(True, linestyle="--", alpha=0.5)

plt.tight_layout()
# In interactive environments: plt.show()
print("[EDA] Visual dashboard successfully compiled.")
\end{lstlisting}
