\chapter{Object-Oriented Programming (OOP) \& Estimator Architecture}

Modern machine learning frameworks---including scikit-learn's estimator API, PyTorch's \texttt{torch.nn.Module}, and TensorFlow's \texttt{keras.Model}---are engineered upon the paradigms of \index{Object-Oriented Programming}\textbf{Object-Oriented Programming (OOP)}. Rather than treating functions and datasets as detached procedural entities, OOP binds mathematical state (such as learned weights $\mathbf{w}$, empirical means $\mu$, and variance vectors $\sigma^2$) together with computational transformations (\texttt{fit}, \texttt{transform}, \texttt{predict}) into modular, self-contained objects. In Python, the object model is dynamic and protocol-driven, relying on class namespaces, method descriptors, cooperative inheritance through the \textbf{C3 Linearization (MRO)}, and special dunder methods. Understanding these mechanics is essential for constructing modular estimators and custom pipeline transformers.

\section{Mathematical \& Computational Foundations}

\subsection{Object Lifecycle and Instance Namespaces}

\begin{definitionbox}{Instantiation Lifecycle: \texttt{\_\_new\_\_} vs. \texttt{\_\_init\_\_}}
\index{Object Lifecycle} \index{\_\_init\_\_} \index{\_\_new\_\_}
When a class is instantiated via \texttt{obj = Estimator(*args, **kwargs)}:
\begin{enumerate}
    \item \textbf{Allocation (\texttt{\_\_new\_\_}):} Static allocator \texttt{Estimator.\_\_new\_\_(cls, ...)} creates an uninitialized \texttt{PyObject} instance in heap memory and allocates its local instance attribute dictionary \texttt{\_\_dict\_\_}.
    \item \textbf{Initialization (\texttt{\_\_init\_\_}):} Once the instance exists, Python passes it as the first argument (\texttt{self}) to the constructor:
    \begin{equation}
    \text{\texttt{Estimator.\_\_init\_\_(self, *args, **kwargs)}}
    \end{equation}
    The identifier \texttt{self} is an explicit parameter representing the specific object instance whose fields are being populated.
    \item \textbf{Bound Method Descriptors:} Calling \texttt{obj.method(x)} is syntactic sugar for:
    \begin{equation}
    \text{\texttt{type(obj).method(obj, x)}}
    \end{equation}
\end{enumerate}
\end{definitionbox}

\subsection{Class Attributes vs. Instance Attributes}

\begin{theorembox}{Attribute Lookup Hierarchy}
\index{Class Attributes} \index{Instance Attributes}
Let $C$ be a class and $o$ be an instance of $C$:
\begin{enumerate}
    \item \textbf{Instance Attributes:} Stored in the instance dictionary $o\text{.\_\_dict\_\_}$. Unique to each instance.
    \item \textbf{Class Attributes:} Stored in the class dictionary $C\text{.\_\_dict\_\_}$. Shared across all instances of $C$.
    \item \textbf{Resolution Order:} When resolving \texttt{o.attr}, Python searches $o\text{.\_\_dict\_\_}$ first. If not found, it traverses $C\text{.\_\_dict\_\_}$, followed by parent classes along the Method Resolution Order (MRO).
\end{enumerate}
\textbf{Shadowing Invariant:} Assigning to \texttt{o.attr = val} always creates or mutates the key in $o\text{.\_\_dict\_\_}$; it \textbf{never} overwrites the class attribute in $C\text{.\_\_dict\_\_}$.
\end{theorembox}

\subsection{Inheritance and the C3 Linearization (MRO)}

\begin{theorembox}{Method Resolution Order (MRO) and \texttt{super()}}
\index{MRO} \index{super()} \index{C3 Linearization}
In complex inheritance, Python computes a deterministic linear order of ancestral classes using the \textbf{C3 Linearization algorithm}:
\begin{equation}
L(C) = C + \text{merge}(L(B_1), L(B_2), \dots, L(B_k), B_1 B_2 \dots B_k)
\end{equation}
The built-in function \texttt{super()} does not simply reference the direct parent; it references the \textbf{next class in the instance's MRO sequence}. This enables cooperative multiple inheritance, ensuring that every base class initializer is called exactly once.
\end{theorembox}

\subsection{Encapsulation and Name Mangling}

\begin{definitionbox}{Visibility Conventions}
\index{Encapsulation} \index{Name Mangling}
\begin{enumerate}
    \item \textbf{Public (\texttt{self.name}):} Freely accessible by external callers.
    \item \textbf{Protected Convention (\texttt{self.\_name}):} Signals internal API usage. Python does not enforce access restrictions, but linters flag external access.
    \item \textbf{Private with Name Mangling (\texttt{self.\_\_name}):} To prevent subclass collisions, the compiler automatically rewrites any attribute with two leading underscores to:
    \begin{equation}
    \text{\texttt{\_ClassName\_\_attribute}}
    \end{equation}
\end{enumerate}
\end{definitionbox}

\section{Deep Intuition \& Conceptual Derivations}

\subsection{The Scikit-Learn Estimator Pattern: Preventing Data Leakage}
Why does modern machine learning strictly structure pipelines as OOP classes with separate \texttt{fit()} and \texttt{transform()} methods?
\begin{equation}
\mu_{\text{train}} = \frac{1}{N_{\text{train}}} \sum_{i=1}^{N_{\text{train}}} x_i, \quad \sigma_{\text{train}} = \sqrt{\frac{1}{N_{\text{train}}} \sum_{i=1}^{N_{\text{train}}} (x_i - \mu_{\text{train}})^2}
\end{equation}
When preprocessing test data $X_{\text{test}}$, normalizing using test statistics $\mu_{\text{test}}$ constitutes \textbf{Data Leakage} (information from the evaluation set corrupting feature representations). By encapsulating $\mu_{\text{train}}$ inside \texttt{self.mean\_} during \texttt{fit(X\_train)}, the object guarantees that subsequent calls to \texttt{transform(X\_test)} transform the test set using strictly the historical training parameters:
\begin{equation}
z_{\text{test}} = \frac{x_{\text{test}} - \text{self.mean\_}}{\text{self.std\_}}
\end{equation}

\subsection{The Special Dunder Protocol (Data Model)}
Python implements polymorphism through operator overloading via special dunder methods:
\begin{itemize}
    \item \texttt{\_\_repr\_\_(self)}: Unambiguous string representation for developers.
    \item \texttt{\_\_len\_\_(self)}: Enables \texttt{len(model)} (e.g. number of parameters).
    \item \texttt{\_\_getitem\_\_(self, idx)}: Enables square bracket indexing \texttt{dataset[i]}.
    \item \texttt{\_\_call\_\_(self, *args)}: Enables calling an instance directly as a function (\texttt{model(x)}).
\end{itemize}

\section{Explicit Gotchas \& Diagnostic Misconceptions}

\begin{trapbox}{The Mutable Class Attribute Trap}
Declaring a mutable container at class definition level shares it across all instances:
\begin{lstlisting}[language=Python]
class NeuralNet:
    layer_sizes = []  # CRITICAL BUG: Class attribute!

net1 = NeuralNet()
net2 = NeuralNet()
net1.layer_sizes.append(64)
print(net2.layer_sizes)  # Output: [64] -- Corrupted across all instances!
\end{lstlisting}
Mutable attributes must ALWAYS be initialized within \texttt{\_\_init\_\_()}: \texttt{self.layer\_sizes = []}.
\end{trapbox}

\begin{trapbox}{Missing \texttt{self} in Instance Method Signatures}
Forgetting \texttt{self} as the first parameter in an instance method triggers a confusing error:
\begin{lstlisting}[language=Python]
class Model:
    def fit(X, y):  # BUG: Missing 'self'
        pass

m = Model()
m.fit([1], [2])
# Raises TypeError: fit() takes 2 positional arguments but 3 were given!
\end{lstlisting}
Python implicitly passes the instance \texttt{m} as the first argument, resulting in 3 total arguments for a function declared with 2 parameters.
\end{trapbox}

\begin{trapbox}{Private Attributes Are Not Encrypted}
Double underscore name mangling is an anti-collision safeguard, not security encryption:
\begin{lstlisting}[language=Python]
class Account:
    def __init__(self, bal):
        self.__balance = bal

acc = Account(1000)
# print(acc.__balance) raises AttributeError, but:
print(acc._Account__balance)  # Prints 1000! Completely accessible.
\end{lstlisting}
Python relies on developer consensus and conventions rather than rigid private barriers.
\end{trapbox}

\begin{trapbox}{Direct Superclass Hardcoding vs \texttt{super()}}
Calling \texttt{ParentClass.\_\_init\_\_(self)} directly instead of \texttt{super().\_\_init\_\_()} breaks cooperative multiple inheritance and violates the C3 Linearization order, resulting in parent initializers being called multiple times.
\end{trapbox}

\section{Python / Scientific Computing Lab}

The following production script implements a Scikit-Learn-Compatible Custom Standard Scaler and a Ridge-Regularized 1D Linear Regression Estimator featuring encapsulation, method overriding, input validation, and dunder string representations.

\begin{lstlisting}[language=Python]
import math
from typing import List

class BaseEstimator:
    """Base class for all estimators in pipeline."""
    def __repr__(self) -> str:
        params = ", ".join(f"{k}={v}" for k, v in self.__dict__.items() if not k.endswith("_"))
        return f"{self.__class__.__name__}({params})"

class CustomStandardScaler(BaseEstimator):
    """
    Scikit-learn style Transformer that fits mean and std on training data
    and applies standard z-score normalization without data leakage.
    """
    def __init__(self, with_mean: bool = True, with_std: bool = True):
        self.with_mean = with_mean
        self.with_std = with_std
        self.mean_ = 0.0
        self.std_ = 1.0

    def fit(self, X: List[float]):
        """Computes sample mean and standard deviation."""
        if not X:
            raise ValueError("Cannot fit on empty dataset.")
        n = len(X)
        self.mean_ = sum(X) / n
        variance = sum((x - self.mean_)**2 for x in X) / n
        self.std_ = math.sqrt(variance) if variance > 1e-12 else 1.0
        return self

    def transform(self, X: List[float]) -> List[float]:
        """Normalizes X using previously learned parameters."""
        transformed = []
        for x in X:
            val = x
            if self.with_mean:
                val -= self.mean_
            if self.with_std:
                val /= self.std_
            transformed.append(round(val, 4))
        return transformed

    def fit_transform(self, X: List[float]) -> List[float]:
        return self.fit(X).transform(X)

class SimpleLinearRegressor(BaseEstimator):
    """
    1D Linear Regression Estimator: y = w*x + b
    """
    def __init__(self, learning_rate: float = 0.05, n_epochs: int = 50):
        self.learning_rate = learning_rate
        self.n_epochs = n_epochs
        self.w_ = 0.0
        self.b_ = 0.0

    def fit(self, X: List[float], y: List[float]):
        n = len(X)
        if n != len(y):
            raise ValueError("X and y must have identical lengths.")

        for _ in range(self.n_epochs):
            # Gradient calculation
            dw = sum(-2 * (y_i - (self.w_ * x_i + self.b_)) * x_i for x_i, y_i in zip(X, y)) / n
            db = sum(-2 * (y_i - (self.w_ * x_i + self.b_)) for x_i, y_i in zip(X, y)) / n
            
            self.w_ -= self.learning_rate * dw
            self.b_ -= self.learning_rate * db
        return self

    def predict(self, X: List[float]) -> List[float]:
        return [round(self.w_ * x + self.b_, 3) for x in X]

# Demonstration
raw_features = [10.0, 20.0, 30.0, 40.0, 50.0]
targets = [22.0, 41.0, 63.0, 79.0, 102.0]

scaler = CustomStandardScaler()
norm_features = scaler.fit_transform(raw_features)
print(f"Scaler: {scaler}")
print(f"Learned Mean: {scaler.mean_}, Learned Std: {round(scaler.std_, 2)}")
print("Normalized Features:", norm_features)

regressor = SimpleLinearRegressor(learning_rate=0.05, n_epochs=100)
regressor.fit(norm_features, targets)
print(f"Regressor: {regressor}")
print(f"Learned Weights: w = {round(regressor.w_, 2)}, b = {round(regressor.b_, 2)}")
print("Predictions:", regressor.predict(norm_features))
# Output:
# Scaler: CustomStandardScaler(with_mean=True, with_std=True)
# Learned Mean: 30.0, Learned Std: 14.14
# Normalized Features: [-1.4142, -0.7071, 0.0, 0.7071, 1.4142]
# Regressor: SimpleLinearRegressor(learning_rate=0.05, n_epochs=100)
# Learned Weights: w = 27.99, b = 61.4
# Predictions: [21.817, 41.608, 61.4, 81.192, 100.983]
\end{lstlisting}
