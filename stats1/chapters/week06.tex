\section{Random Experiments, Sample Spaces, and Events}

\begin{definitionbox}{Random Experiment and Sample Space ($\Omega$)}
\index{Random Experiment}\index{Sample Space}
\begin{itemize}
    \item \textbf{Random Experiment:} An empirical procedure or physical process whose exact outcome cannot be deterministically predicted in advance, but whose set of all possible outcomes is fully specifiable before execution.
    \item \textbf{Sample Space ($\Omega$ or $S$):} The exhaustive set of all mutually exclusive elementary outcomes:
    $$\Omega = \{\omega_1, \omega_2, \dots\}$$
    Sample spaces may be \textbf{finite discrete} (e.g. tossing two coins: $\{HH, HT, TH, TT\}$), \textbf{countably infinite} (e.g. number of attempts until first server response: $\{1, 2, 3, \dots\}$), or \textbf{uncountably continuous} (e.g. packet arrival latency: $\Omega = [0, \infty)$).
    \item \textbf{Event ($A$):} Any subset of the sample space ($A \subseteq \Omega$). The event $A$ occurs if the observed realization $\omega \in A$.
\end{itemize}
\end{definitionbox}

\section{Event Algebra and Set Operations}

Because events are formal sets of outcomes, probability logic is identical to Boolean set algebra:

\begin{definitionbox}{Set Operations on Events}
\index{Set Operations}
Let $A, B \subseteq \Omega$ be arbitrary events in sample space $\Omega$:
\begin{itemize}
    \item \textbf{Union ($A \cup B$):} The event that $A$ occurs \textbf{OR} $B$ occurs (inclusive or):
    $$A \cup B = \{\omega \in \Omega \colon \omega \in A \text{ or } \omega \in B\}$$
    \item \textbf{Intersection ($A \cap B$ or $AB$):} The event that \textbf{BOTH} $A$ and $B$ occur simultaneously:
    $$A \cap B = \{\omega \in \Omega \colon \omega \in A \text{ and } \omega \in B\}$$
    \item \textbf{Complement ($A^c$ or $\bar{A}$):} The event that $A$ does \textbf{NOT} occur:
    $$A^c = \Omega \setminus A = \{\omega \in \Omega \colon \omega \notin A\}$$
    \item \textbf{Mutually Exclusive / Disjoint Events:} Events that cannot co-occur in a single realization:
    $$A \cap B = \emptyset \implies \P(A \cap B) = 0$$
\end{itemize}
\end{definitionbox}

\begin{theorembox}{De Morgan's Laws and Duality}
\index{De Morgan's Laws}
For any collection of events $A$ and $B$:
\begin{align*}
    (A \cup B)^c &= A^c \cap B^c \quad (\text{"Neither } A \text{ nor } B \text{ occurs"}) \\
    (A \cap B)^c &= A^c \cup B^c \quad (\text{"Not both occur" } \iff \text{"At least one fails to occur"})
\end{align*}
\end{theorembox}

\section{Kolmogorov's Three Axioms of Probability}

In 1933, Soviet mathematician Andrey Kolmogorov established the rigorous axiomatic foundation of probability theory:

\begin{formulabox}{Kolmogorov's Axioms of Probability}
\index{Kolmogorov's Axioms}
A probability measure $\P$ is a real-valued function defined on the collection of events $\mathcal{F}$ of $\Omega$ satisfying:
\begin{enumerate}
    \item \textbf{Axiom 1 (Non-Negativity):} For every event $A \subseteq \Omega$:
    $$\P(A) \ge 0$$
    \item \textbf{Axiom 2 (Total Normalization):} The probability of the entire sample space is unity:
    $$\P(\Omega) = 1.00$$
    \item \textbf{Axiom 3 (Countable Additivity):} If $A_1, A_2, A_3, \dots$ is a sequence of pairwise mutually exclusive (disjoint) events ($A_i \cap A_j = \emptyset$ for all $i \neq j$):
    $$\P\left(\bigcup_{i=1}^\infty A_i\right) = \sum_{i=1}^\infty \P(A_i)$$
\end{enumerate}
\end{formulabox}

\begin{intuitionbox}{Probability as Physical Mass}
Think of probability as \textbf{unit mass ($1.0\,\text{kg}$)} distributed across the domain $\Omega$.
\begin{itemize}
    \item Axiom 1 states that no region can contain negative mass ($\text{mass} \ge 0$).
    \item Axiom 2 states that the total mass of the entire universe $\Omega$ is strictly $1.0\,\text{kg}$.
    \item Axiom 3 states that if two non-overlapping parcels of clay $A$ and $B$ are placed together on a balance scale, their combined mass is strictly the sum of their individual masses: $\text{mass}(A) + \text{mass}(B)$.
\end{itemize}
In deep learning classifiers, the \textbf{Softmax function} enforces Axioms 1 and 2 by exponentiating raw logits (ensuring $\ge 0$) and dividing by their sum (ensuring total probability $= 1.0$).
\end{intuitionbox}

\section{Derived Properties and Inclusion-Exclusion}

From Kolmogorov's three axioms, we deduce the core theorems of probability calculation:

\begin{theorembox}{Fundamental Theorems of Probability}
\index{Inclusion-Exclusion Principle}\index{Boole's Inequality}
\begin{itemize}
    \item \textbf{Empty Set Probability:} $\P(\emptyset) = 0$.
    \item \textbf{Complement Rule:} For any event $A$:
    $$\P(A^c) = 1 - \P(A)$$
    \item \textbf{Bounded Range:} $0 \le \P(A) \le 1$ for all events $A$.
    \item \textbf{Monotonicity:} If $A \subseteq B$, then $\P(A) \le \P(B)$ and $\P(B \setminus A) = \P(B) - \P(A)$.
    \item \textbf{General Addition Rule (Inclusion-Exclusion for 2 Events):}
    $$\P(A \cup B) = \P(A) + \P(B) - \P(A \cap B)$$
    \item \textbf{Inclusion-Exclusion for 3 Events:}
    $$\P(A \cup B \cup C) = \sum \P(A) - \sum \P(A \cap B) + \P(A \cap B \cap C)$$
    \item \textbf{Boole's Inequality (The Union Bound):} For \emph{any} sequence of events (disjoint or overlapping):
    $$\P\left(\bigcup_{i=1}^n A_i\right) \le \sum_{i=1}^n \P(A_i)$$
\end{itemize}
\end{theorembox}

\section{Cognitive Traps in Basic Probability}

\begin{trapbox}{Trap 1: The Non-Disjoint Addition Fallacy}
A classic error is writing $\P(A \cup B) = \P(A) + \P(B)$ when events are \textbf{not} mutually exclusive.
\begin{itemize}
    \item If event $A$ (server CPU threshold exceeded) has $\P(A) = 0.60$ and event $B$ (memory threshold exceeded) has $\P(B) = 0.50$, blindly adding them gives:
    $$\P(A) + \P(B) = 0.60 + 0.50 = 1.10$$
    violating Kolmogorov's Axiom 2 ($\P \le 1$)!
    \item The joint failure state $A \cap B$ is double-counted and must be subtracted: $\P(A \cup B) = 0.60 + 0.50 - \P(A \cap B)$.
\end{itemize}
\end{trapbox}

\begin{trapbox}{Trap 2: The "At Least One" Combinatorial Trap}
When asked to find the probability that "at least one of $n$ components succeeds" or "at least one failure occurs":
\begin{itemize}
    \item \textbf{The Exhaustive Trap:} Attempting to calculate $\P(\text{exactly 1}) + \P(\text{exactly 2}) + \dots + \P(\text{all } n)$ becomes mathematically monstrous for large $n$.
    \item \textbf{The Master Shortcut:} Always invert the problem using De Morgan's Law and the Complement Rule:
    $$\P(\text{At least one } A_i) = 1 - \P(\text{None occurs}) = 1 - \P\left(\bigcap_{i=1}^n A_i^c\right)$$
\end{itemize}
\end{trapbox}

\begin{trapbox}{Trap 3: Zero Probability Does NOT Imply Impossibility!}
In continuous sample spaces ($\Omega = \mathbb{R}$), any single exact point has probability measure \textbf{zero}:
$$\P(X = 3.14159265\dots) = 0$$
Yet, whenever a continuous experiment is run, \emph{some} exact real number with individual probability zero is realized! In continuous probability, $\P(A) = 0$ means the event is a set of Lebesgue measure zero (almost surely impossible), not mathematically impossible.
\end{trapbox}

\section{Python Implementation: Probability Simulation and Verification}

\begin{lstlisting}[caption={Verifying Kolmogorov Axioms and Inclusion-Exclusion in Python}]
import numpy as np

# Monte Carlo simulation of rolling two independent 6-sided dice
N_trials = 1_000_000
die1 = np.random.randint(1, 7, size=N_trials)
die2 = np.random.randint(1, 7, size=N_trials)

# Event A: Sum is even
A = ((die1 + die2) % 2 == 0)
# Event B: At least one die shows a 6
B = (die1 == 6) | (die2 == 6)

p_A = np.mean(A)
p_B = np.mean(B)
p_A_and_B = np.mean(A & B)
p_A_or_B = np.mean(A | B)

# Empirical verification of Inclusion-Exclusion
p_inc_exc = p_A + p_B - p_A_and_B

print(f"Empirical P(A) = {p_A:.4f}")
print(f"Empirical P(B) = {p_B:.4f}")
print(f"Empirical P(A and B) = {p_A_and_B:.4f}")
print(f"Direct P(A or B)     = {p_A_or_B:.4f}")
print(f"Inclusion-Exclusion  = {p_inc_exc:.4f} (Matches exact union within sampling noise)")
\end{lstlisting}

\subsection*{Part A: Computational and Analytical Problems}

\begin{enumerate}
    \item \textbf{Sample Spaces and Cardinalities:} Define the sample space $\Omega$ and state its cardinality $|\Omega|$ for the following experiments:
    \begin{enumerate}
        \item Flipping a fair coin 4 consecutive times.
        \item Rolling two distinguishable dice (one red, one green) and recording the ordered pair of faces.
        \item Sampling customer service tickets until the first resolved ticket is observed.
    \end{enumerate}

    \item \textbf{Set Operations on Sample Spaces:} A 6-sided die is rolled with $\Omega = \{1, 2, 3, 4, 5, 6\}$.
    Let $A = \{2, 4, 6\}$ (even), $B = \{4, 5, 6\}$ (greater than 3), and $C = \{1, 2\}$ (strictly less than 3).
    \begin{enumerate}
        \item Evaluate $A \cap B$ and describe it in words.
        \item Evaluate $A \cup C$ and state its probability $\P(A \cup C)$.
        \item Evaluate $(A \cup B)^c$ using De Morgan's Law.
        \item Are events $A$ and $C$ mutually exclusive? Explain.
    \end{enumerate}

    \item \textbf{The Complement Shortcut in Reliability:} A distributed cloud database replicates an essential table across 5 independent storage nodes. Each storage node has an independent failure probability of $p = 0.05$ per day. The database remains accessible as long as \textbf{at least one node} remains operational.
    \begin{enumerate}
        \item What is the probability that all 5 nodes fail simultaneously on a given day?
        \item Using the complement rule, compute the probability that the database remains operational.
    \end{enumerate}

    \item \textbf{Inclusion-Exclusion Principle:} In an engineering cohort of 200 data science students:
    \begin{itemize}
        \item 120 students have completed the Python prerequisite ($P$).
        \item 100 students have completed the Mathematics prerequisite ($M$).
        \item 40 students have completed both prerequisites ($P \cap M$).
    \end{itemize}
    \begin{enumerate}
        \item Find the probability that a randomly chosen student has completed Python OR Mathematics.
        \item Find the probability that a student has completed neither prerequisite.
        \item Find the probability that a student has completed Python but NOT Mathematics.
    \end{enumerate}

    \item \textbf{Kolmogorov Axioms Validation:} An untrained machine learning engineer trains a multi-class image classifier outputting predicted confidence scores for three classes:
    $$\P(\text{Class 1}) = 0.65, \quad \P(\text{Class 2}) = 0.45, \quad \P(\text{Class 3}) = -0.10$$
    Explicitly identify which of Kolmogorov's three axioms are violated by this output.
\end{enumerate}

\subsection*{Part B: Conceptual and Diagnostic Multiple Choice Questions}

\begin{enumerate}[resume]
\item \textbf{Conceptual Diagnostic: Boole's Inequality and Family-Wise Risk} \\
An algorithmic trading system operates $m = 20$ independent statistical arbitrage bots. For each bot $i$, the probability of suffering an abnormal execution slippage error on any trading day is bounded by $\P(E_i) \le 0.005$. A risk officer wishes to establish a mathematically guaranteed upper bound on the probability that \textbf{at least one} bot suffers an execution slippage error today: $\P(\bigcup_{i=1}^{20} E_i)$. Which bound is mathematically guaranteed by probability axioms without requiring independence?
\begin{enumerate}
    \item $\P\left(\bigcup_{i=1}^{20} E_i\right) \le 1 - (1 - 0.005)^{20} \approx 0.0954$
    \item $\P\left(\bigcup_{i=1}^{20} E_i\right) \le \sum_{i=1}^{20} \P(E_i) \le 20 \times 0.005 = 0.100$
    \item $\P\left(\bigcup_{i=1}^{20} E_i\right) = 0.005$, because all bots share identical risk limits.
    \item $\P\left(\bigcup_{i=1}^{20} E_i\right) \ge 0.100$, because individual risks compound multiplicatively.
\end{enumerate}

\item \textbf{Conceptual Diagnostic: Monotonicity and Probability Bounds} \\
Let $A$ and $B$ be two events in an arbitrary probability space $(\Omega, \mathcal{F}, \P)$ such that $\P(A) = 0.70$ and $\P(B) = 0.80$. What are the strictly tightest possible mathematical bounds on the joint probability of their intersection $\P(A \cap B)$?
\begin{enumerate}
    \item $0.00 \le \P(A \cap B) \le 0.56$
    \item $0.50 \le \P(A \cap B) \le 0.70$
    \item $0.56 \le \P(A \cap B) \le 1.00$
    \item $0.70 \le \P(A \cap B) \le 0.80$
\end{enumerate}
\end{enumerate}
