\chapter{Principles of Counting, Permutations \& Combinations}

Before calculating probabilities, we must master the combinatorial grammar of counting. In machine learning, calculating the exact probability of an edge case, evaluating hyperparameter search spaces, or computing the number of possible network subgraphs requires rigorous counting. If you cannot count the number of favorable outcomes and total configurations in a sample space, probabilistic modeling is impossible. In this chapter, we formalize the \textbf{Fundamental Rules of Sum and Product}, \textbf{Factorials and Stirling's Approximation}, \textbf{Permutations ($P(n, r)$)}, \textbf{Combinations ($C(n, r)$)}, \textbf{Multinomial Arrangements}, \textbf{Circular Permutations}, and the \textbf{Four Fundamental Counting Paradigms (including Stars and Bars)}.

\section{Fundamental Counting Principles}

All combinatorial enumeration derives from two foundational set-theoretic axioms:

\begin{theorembox}{The Rule of Sum and the Rule of Product}
\index{Rule of Sum}\index{Rule of Product}
Let $A$ and $B$ denote finite sets of mutually exclusive outcomes or operational tasks:
\begin{itemize}
    \item \textbf{The Rule of Sum (Disjoint Tasks / OR):} If task $A$ can be performed in $m$ distinct ways, and task $B$ can be performed in $n$ distinct ways, and tasks $A$ and $B$ cannot be performed simultaneously ($A \cap B = \emptyset$):
    $$|A \cup B| = |A| + |B| = m + n$$
    \item \textbf{The Rule of Product (Sequential Stages / AND):} If a multi-stage procedure consists of stage 1 having $n_1$ possible outcomes, and regardless of the outcome of stage 1, stage 2 has $n_2$ possible outcomes, \dots, through stage $k$ having $n_k$ possible outcomes:
    $$|A_1 \times A_2 \times \dots \times A_k| = \prod_{i=1}^k n_i = n_1 \times n_2 \times \dots \times n_k$$
\end{itemize}
\end{theorembox}

\section{Factorials and Asymptotics}

\begin{definitionbox}{Factorial ($n!$) and Stirling's Formula}
\index{Factorial}\index{Stirling's Approximation}
The number of bijective permutations of $n$ distinct elements into $n$ ordered positions is given by the \textbf{factorial}:
$$n! = \prod_{i=1}^n i = n \times (n-1) \times (n-2) \times \dots \times 2 \times 1$$
\begin{itemize}
    \item \textbf{The Zero Factorial Convention:} $0! = 1$, representing the single unique bijection of the empty set onto itself ($\emptyset \to \emptyset$).
    \item \textbf{Stirling's Asymptotic Approximation:} For large $n$ (ubiquitous in statistical physics and information theory):
    $$n! \sim \sqrt{2\pi n} \left(\frac{n}{e}\right)^n \quad \text{as } n \to \infty$$
    Taking natural logarithms: $\ln(n!) \approx n \ln n - n$, showing that combinatorial search spaces grow faster than any simple exponential $a^n$.
\end{itemize}
\end{definitionbox}

\section{The Four Fundamental Counting Paradigms}

Combinatorial problems partition into a $2 \times 2$ taxonomy governed by whether \textbf{Order Matters} (Sequence vs. Set) and whether \textbf{Replacement is Allowed}.

\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{p{3.8cm}p{4.2cm}p{4.5cm}}
\toprule
\textbf{Selection Type} & \textbf{Order Matters (Sequences)} & \textbf{Order Does NOT Matter (Subsets)} \\
\midrule
\textbf{Without Replacement} & \textbf{Permutations:} $P(n, r) = \frac{n!}{(n-r)!}$ & \textbf{Combinations:} $\binom{n}{r} = \frac{n!}{r!(n-r)!}$ \\
\textbf{With Replacement} & \textbf{Tuples / Words:} $n^r$ & \textbf{Stars and Bars (Multisets):} $\binom{n + r - 1}{r}$ \\
\bottomrule
\end{tabular}
\end{center}

\begin{definitionbox}{Permutations and Combinations Formulas}
\index{Permutations}\index{Combinations}
\begin{itemize}
    \item \textbf{Permutations without Replacement ($P(n, r)$):} Choosing and ordering $r$ distinct items from a pool of $n$:
    $$P(n, r) = n(n-1)\dots(n-r+1) = \frac{n!}{(n-r)!}$$
    \item \textbf{Combinations without Replacement ($\binom{n}{r}$):} Choosing an unordered subset of size $r$ from $n$ distinct items:
    $$\binom{n}{r} = \frac{P(n, r)}{r!} = \frac{n!}{r!(n-r)!}$$
    \item \textbf{Fundamental Algebraic Identities:}
    \begin{align*}
        \binom{n}{r} &= \binom{n}{n-r} \quad (\text{Symmetry Identity}) \\
        \binom{n}{r} &= \binom{n-1}{r-1} + \binom{n-1}{r} \quad (\text{Pascal's Identity}) \\
        \sum_{r=0}^n \binom{n}{r} &= 2^n \quad (\text{Total Subsets of an } n\text{-element Set})
    \end{align*}
\end{itemize}
\end{definitionbox}

\begin{definitionbox}{Combinations With Replacement: Stars and Bars}
\index{Stars and Bars}
The number of ways to distribute $r$ identical (indistinguishable) items into $n$ distinct (labeled) bins (or equivalently, choosing $r$ elements from $n$ types with unlimited replacement) is:
$$N = \binom{r + n - 1}{r} = \binom{r + n - 1}{n - 1}$$
\textit{Proof Idea:} Represent the $r$ identical items as $r$ stars ($\star$) and the $n$ bins separated by $n-1$ divider bars ($|$). The problem reduces to choosing $n-1$ positions for the bars out of $r + n - 1$ total symbol positions!
\end{definitionbox}

\section{Multinomial Arrangements and Circular Permutations}

\begin{theorembox}{Multinomial Arrangements (Permutations with Repeated Elements)}
\index{Multinomial Coefficient}
Suppose a collection of $n$ objects contains $k$ distinct types, with $n_1$ identical objects of type 1, $n_2$ of type 2, \dots, and $n_k$ of type $k$, such that $\sum_{i=1}^k n_i = n$.
The number of distinct, distinguishable linear sequences is:
$$\binom{n}{n_1, n_2, \dots, n_k} = \frac{n!}{n_1! \, n_2! \, \dots \, n_k!}$$
\textit{Reasoning:} Permuting identical items among themselves within their $n_i$ positions creates zero distinguishable changes, so we divide out $n_i!$ for every group.
\end{theorembox}

\begin{theorembox}{Circular Permutations}
\index{Circular Permutations}
The number of ways to arrange $n$ distinct objects around a circular table (where only relative positions matter, so rotating all individuals by any angle produces an equivalent configuration) is:
$$N_{\text{circle}} = \frac{n!}{n} = (n-1)!$$
If the circle can be flipped over (like a necklace or bracelet where clockwise and counter-clockwise arrangements are physically indistinguishable):
$$N_{\text{necklace}} = \frac{(n-1)!}{2}$$
\end{theorembox}

\section{Cognitive Traps in Combinatorics}

\begin{trapbox}{Trap 1: The "Combination Lock" Misnomer}
Physical suitcase and padlock locks are universally labeled "Combination Locks", yet if the code is 3-7-1, entering 7-1-3 fails to unlock it!
\begin{itemize}
    \item Because the sequence of numbers matters, it is mathematically a \textbf{Permutation Lock} (specifically, an ordered tuple with replacement: $10^3 = 1{,}000$ codes).
    \item A true mathematical "Combination Lock" would open for any anagram permutation of the selected digits. Always ask: \textbf{"Does swapping the position of two chosen items change the outcome?"} If yes $\implies$ Permutations. If no $\implies$ Combinations.
\end{itemize}
\end{trapbox}

\begin{trapbox}{Trap 2: The Multi-Step Selection Overcounting Fallacy}
A classic blunder occurs when attempting to enforce a condition (such as "at least one Ace") by selecting items in sequential stages.
\begin{itemize}
    \item \textbf{Fallacy:} "To form a 5-card poker hand with at least one Ace from a 52-card deck, first choose 1 Ace in $\binom{4}{1} = 4$ ways, then pick any 4 cards from the remaining 51 cards in $\binom{51}{4}$ ways: $4 \times \binom{51}{4} = 999{,}600$."
    \item \textbf{Why it fails:} This severely \textbf{overcounts}! A hand with two Aces (e.g., $A\spadesuit, A\heartsuit, K, Q, J$) is counted twice: once when $A\spadesuit$ is picked first, and once when $A\heartsuit$ is picked first.
    \item \textbf{Correct Method (Complement Rule):}
    $$\text{Hands with } \ge 1 \text{ Ace} = \binom{52}{5} - \binom{48}{5} = 2{,}598{,}960 - 1{,}712{,}304 = \mathbf{886{,}656}$$
\end{itemize}
\end{trapbox}

\begin{trapbox}{Trap 3: The Circular Seating Rotation Illusion}
When seating $n$ individuals at a round table, students frequently write $n!$.
\begin{itemize}
    \item In a circle, there is no designated "Seat \#1" unless a fixed landmark (e.g., a podium or doorway) establishes an absolute frame of reference.
    \item Shifting everyone one seat clockwise preserves all neighbor relationships. To break rotational symmetry, you must fix the first person in an arbitrary seat ($1$ way), leaving the remaining $(n-1)$ people to be arranged linearly: $(n-1)!$.
\end{itemize}
\end{trapbox}

\section{Python Implementation: Combinatorics in Standard Library}

\begin{lstlisting}[caption={Permutations, Combinations, and Multisets in Python}]
import math
from itertools import permutations, combinations, combinations_with_replacement

# 1. Permutations: P(8, 3) = 8! / 5! = 336
n, r = 8, 3
p_calc = math.perm(n, r)
print(f"P({n}, {r}) = {p_calc}")

# 2. Combinations: C(10, 4) = 10! / (4! * 6!) = 210
c_calc = math.comb(10, 4)
print(f"C(10, 4) = {c_calc} (Identical to C(10, 6) = {math.comb(10, 6)})")

# 3. Multinomial: Anagrams of "STATISTICS" (S:3, T:3, I:2, A:1, C:1)
total_letters = 10
word_counts = [3, 3, 2, 1, 1]
multinomial = math.factorial(total_letters)
for cnt in word_counts:
    multinomial //= math.factorial(cnt)
print(f"Distinct anagrams of 'STATISTICS': {multinomial}")

# 4. Stars and Bars: Distributing r=7 identical compute tasks to n=3 servers
# C(r + n - 1, r) = C(7 + 3 - 1, 7) = C(9, 7) = 36
tasks, servers = 7, 3
stars_and_bars = math.comb(tasks + servers - 1, tasks)
print(f"Allocating 7 identical tasks to 3 servers: {stars_and_bars} allocations")
\end{lstlisting}

\subsection*{Part A: Computational and Analytical Problems}

\begin{enumerate}
    \item \textbf{Cascading Rule of Product:} A cloud microservice architecture provides configurable API endpoints. A developer must select:
    \begin{itemize}
        \item Exactly 1 authentication protocol out of 3 options (OAuth2, API Key, JWT).
        \item Exactly 1 compute runtime out of 4 options (Python, Go, Node.js, Rust).
        \item Either an AWS region (out of 5 regions) OR a GCP region (out of 3 regions), but not both.
    \end{itemize}
    How many distinct deployment configurations can be established?

    \item \textbf{Permutations in Competitive Benchmarking:} An AI research team benchmarks 8 competitive foundation models.
    \begin{enumerate}
        \item How many unique top-3 rank-ordered leaderboards (Gold, Silver, Bronze) are possible assuming zero ties?
        \item State the formal combinatorial formula utilized and explain why combinations cannot be applied.
    \end{enumerate}

    \item \textbf{Multinomial Anagrams:} A natural language processing engineer tests token generation by finding all distinct permutations of the characters in the string:
    $$\textbf{P R O B A B I L I T Y}$$
    \begin{enumerate}
        \item Count the length of the string and the frequency of each repeated letter.
        \item Compute the exact number of unique, distinguishable character sequences.
    \end{enumerate}

    \item \textbf{Combinatorial Committee Selection:} A data ethics review board of 5 members is formed from a pool of 6 computer scientists and 7 legal scholars.
    \begin{enumerate}
        \item How many distinct committees can be formed with no constraints?
        \item How many distinct committees contain exactly 3 computer scientists and 2 legal scholars?
        \item How many distinct committees contain at least 4 computer scientists?
    \end{enumerate}

    \item \textbf{Circular Permutations and Landmarks:} 7 software engineers are seated at a circular conference table for a sprint retrospective.
    \begin{enumerate}
        \item In how many distinct circular seating arrangements can they sit if only relative neighbor positions matter?
        \item Suppose the conference room has a single projector screen directly in front of one seat, making that seat unique. How many distinct seating arrangements exist in this case?
    \end{enumerate}
\end{enumerate}

\subsection*{Part B: Conceptual and Diagnostic Multiple Choice Questions}

\begin{enumerate}[resume]
\item \textbf{Conceptual Diagnostic: Stars and Bars Resource Allocation} \\
A cloud orchestrator must distribute $r = 10$ identical batch processing jobs across $n = 4$ distinct server nodes such that every server node receives \textbf{at least one job} (no server remains idle). How many distinct allocation configurations are possible?
\begin{enumerate}
    \item $\binom{10 + 4 - 1}{10} = \binom{13}{10} = 286$
    \item $\binom{10 - 1}{4 - 1} = \binom{9}{3} = 84$
    \item $4^{10} = 1{,}048{,}576$
    \item $P(10, 4) = 5{,}040$
\end{enumerate}

\item \textbf{Conceptual Diagnostic: Multi-Step Selection and Overcounting} \\
A cybersecurity auditor wants to construct an audit panel of $5$ servers chosen from a cluster of $8$ Linux servers and $6$ Windows servers, subject to the condition that the panel must contain \textbf{at least one Linux server}. An apprentice analyst computes the answer as:
$$\text{Apprentice Solution} = \binom{8}{1} \times \binom{13}{4} = 8 \times 715 = 5{,}720$$
What error was committed, and what is the true number of valid panels?
\begin{enumerate}
    \item The apprentice undercounted by ignoring Windows servers; the true answer is $\binom{14}{5} = 2{,}002$.
    \item The apprentice committed the multi-step overcounting fallacy by creating an artificial ordering; the true answer via the complement rule is $\binom{14}{5} - \binom{6}{5} = 2{,}002 - 6 = 1{,}996$.
    \item The apprentice should have used permutations because audit servers are sampled sequentially; the true answer is $P(14, 5)$.
    \item The apprentice's formula is exact because choosing the mandatory Linux server first satisfies the condition without distortion.
\end{enumerate}
\end{enumerate}
