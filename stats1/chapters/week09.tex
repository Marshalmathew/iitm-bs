\chapter{Expectation, Variance \& LOTUS of Random Variables}

Once a discrete random variable and its probability mass function (PMF) are defined, we require theoretical summary metrics to quantify its central location and dispersion across infinitely many hypothetical repetitions. Just as sample data is described by the sample mean $\bar{x}$ and sample variance $s^2$, the probabilistic behavior of a random variable is governed by its \textbf{Expected Value ($\E[X]$)} and \textbf{Variance ($\Var(X)$)}. In this chapter, we master \textbf{Mathematical Expectation}, the \textbf{Law of the Unconscious Statistician (LOTUS)}, the \textbf{Unconditional Linearity of Expectation}, the \textbf{Quadratic Scaling of Variance}, \textbf{Chebyshev's Inequality}, and the critical cognitive traps: \textbf{Jensen's non-linear expectation gap}, \textbf{the negative variance fallacy}, and \textbf{why variances of differences add}.

\vspace{0.5cm}
\begin{tcolorbox}[colback=primarylight, colframe=primarydark, title={\textbf{What You'll Learn in this Chapter}}]
\begin{itemize}
    \item Mathematical formalization of Expected Value as probability-weighted center of mass: $\E[X] = \sum x p(x)$.
    \item The Law of the Unconscious Statistician (LOTUS) for functions of random variables: $\E[g(X)] = \sum g(x) p(x)$.
    \item Universal Linearity of Expectation: $\E[aX + bY + c] = a\E[X] + b\E[Y] + c$ (holds without requiring independence).
    \item Theoretical Variance $\Var(X) = \E[(X - \mu)^2]$ and its computational shortcut $\Var(X) = \E[X^2] - (\E[X])^2$.
    \item Properties of Variance and Standard Deviation: $\Var(aX + b) = a^2\Var(X)$ and $\SD(aX + b) = |a|\SD(X)$.
    \item Variance of sums and differences: $\Var(X \pm Y) = \Var(X) + \Var(Y) \pm 2\Cov(X, Y)$.
    \item Cognitive traps: $\E[g(X)] \neq g(\E[X])$, negative variance errors, and subtracting variances.
    \item Two-tier exercises with diagnostic multiple choice questions on affine transformations and independent differences.
\end{itemize}
\end{tcolorbox}

\section{Mathematical Expectation (The Expected Value)}

\begin{definitionbox}{Expected Value (Mean) of a Discrete Random Variable}
\index{Expected Value}\index{Mathematical Expectation}
Let $X$ be a discrete random variable with support $\mathcal{R}_X$ and PMF $p_X(x)$. The \textbf{Expected Value} of $X$, denoted $\E[X]$ or $\mu_X$, is defined by:
$$\E[X] = \sum_{x \in \mathcal{R}_X} x \cdot p_X(x)$$
provided that the sum is absolutely convergent: $\sum_{x} |x| p_X(x) < \infty$.
\begin{itemize}
    \item \textbf{Center of Mass Interpretation:} $\E[X]$ represents the physical balance point (fulcrum) of the PMF histogram on the real line.
    \item \textbf{Non-Support Realization:} $\E[X]$ does \textbf{not} have to belong to the support $\mathcal{R}_X$! (e.g. For a fair 6-sided die, $\mathcal{R}_X = \{1, 2, 3, 4, 5, 6\}$, yet $\E[X] = \frac{1+2+3+4+5+6}{6} = 3.5$, a value the die can never show).
\end{itemize}
\end{definitionbox}

\begin{theorembox}{The Law of the Unconscious Statistician (LOTUS)}
\index{LOTUS}
Let $X$ be a discrete random variable and $g \colon \mathbb{R} \to \mathbb{R}$ be any real-valued function. To compute the expected value of the transformed random variable $Y = g(X)$, one does \textbf{not} need to derive the PMF of $Y$; one simply evaluates:
$$\E[g(X)] = \sum_{x \in \mathcal{R}_X} g(x) \cdot p_X(x)$$
\end{theorembox}

\begin{theorembox}{Universal Linearity of Expectation}
\index{Linearity of Expectation}
Expectation is a strictly linear functional. For any real constants $a, b, c \in \mathbb{R}$ and any random variables $X$ and $Y$ (whether independent or arbitrarily dependent):
$$\E[aX + bY + c] = a\E[X] + b\E[Y] + c$$
\textit{Crucial Insight:} Linearity holds universally across all joint dependency structures. Even if $X$ and $Y$ are fiercely correlated, $\E[X + Y] = \E[X] + \E[Y]$ holds without exception!
\end{theorembox}

\section{Theoretical Variance and Standard Deviation}

\begin{definitionbox}{Variance and Standard Deviation}
\index{Variance}\index{Standard Deviation}
The \textbf{Variance} of a random variable $X$, denoted $\Var(X)$ or $\sigma_X^2$, measures the expected squared distance of $X$ from its mean $\mu_X = \E[X]$:
$$\Var(X) = \E[(X - \E[X])^2] = \sum_{x \in \mathcal{R}_X} (x - \mu_X)^2 \cdot p_X(x)$$
\begin{itemize}
    \item \textbf{Computational Shortcut Formula:}
    $$\Var(X) = \E[X^2] - (\E[X])^2$$
    where $\E[X^2] = \sum_{x} x^2 p_X(x)$ is the second raw moment via LOTUS.
    \item \textbf{Standard Deviation ($\sigma_X$):} The positive square root of variance, restoring dispersion to original units:
    $$\sigma_X = \SD(X) = \sqrt{\Var(X)}$$
\end{itemize}
\end{definitionbox}

\begin{theorembox}{Properties of Variance (Non-Linearity)}
\index{Properties of Variance}
Unlike Expectation, Variance is \textbf{not linear}:
\begin{enumerate}
    \item \textbf{Variance of a Constant:} $\Var(c) = 0$ for any constant $c \in \mathbb{R}$.
    \item \textbf{Shift Invariance:} $\Var(X + c) = \Var(X)$. Shifting a distribution horizontally alters its center but leaves its spread identical.
    \item \textbf{Quadratic Scaling:} For any constant $a \in \mathbb{R}$:
    $$\Var(aX) = a^2 \Var(X) \implies \SD(aX) = |a|\SD(X)$$
    Combining shift and scale: $\Var(aX + b) = a^2\Var(X)$ and $\SD(aX + b) = |a|\SD(X)$.
    \item \textbf{Variance of Sums and Differences:}
    $$\Var(X + Y) = \Var(X) + \Var(Y) + 2\Cov(X, Y)$$
    $$\Var(X - Y) = \Var(X) + \Var(Y) - 2\Cov(X, Y)$$
    If $X$ and $Y$ are independent ($\Cov(X, Y) = 0$):
    $$\Var(X + Y) = \Var(X) + \Var(Y) \quad \text{and} \quad \Var(X - Y) = \Var(X) + \Var(Y)$$
\end{enumerate}
\end{theorembox}

\section{Cognitive Traps in Expectation and Variance}

\begin{trapbox}{Trap 1: The Non-Linear Expectation Fallacy ($\E[g(X)] \neq g(\E[X])$)}
A catastrophic blunder is pushing the expectation operator inside non-linear functions:
\begin{itemize}
    \item \textbf{Powers:} $\E[X^2] \neq (\E[X])^2$ (In fact, $\E[X^2] - (\E[X])^2 = \Var(X) > 0$ for any non-constant random variable!).
    \item \textbf{Inverses (Jensen's Inequality):} If $X$ is positive and non-degenerate, $\E[1/X] > \frac{1}{\E[X]}$.
    \item \textbf{Square Roots / Logs:} $\E[\sqrt{X}] < \sqrt{\E[X]}$ and $\E[\ln X] < \ln(\E[X])$ for strictly concave functions.
    \item \textbf{Rule:} $\E[g(X)] = g(\E[X])$ holds \textbf{strictly and only if} $g(x) = ax + b$ is affine!
\end{itemize}
\end{trapbox}

\begin{trapbox}{Trap 2: The Negative Variance Hallucination}
Variance is the expectation of a squared quantity $(X - \mu)^2$.
\begin{itemize}
    \item Therefore, variance is \textbf{strictly non-negative}: $\Var(X) \ge 0$ always.
    \item Under negative scaling:
    $$\Var(-3X + 5) = (-3)^2 \Var(X) = \mathbf{+9\Var(X)}$$
    Writing $-9\Var(X)$ or $-3\Var(X)$ is a severe mathematical violation!
\end{itemize}
\end{trapbox}

\begin{trapbox}{Trap 3: Variances of Independent Differences DO NOT Subtract!}
When subtracting two independent random variables $X$ and $Y$:
\begin{itemize}
    \item \textbf{The Fallacy:} Writing $\Var(X - Y) = \Var(X) - \Var(Y)$.
    \item \textbf{The Physical Reality:} Subtracting two noisy signals introduces \textbf{more} uncertainty, not less!
    $$\Var(X - Y) = \Var(X + (-1)Y) = \Var(X) + (-1)^2\Var(Y) = \mathbf{\Var(X) + \Var(Y)}$$
    Dispersion always compounds positively when combining independent random sources!
\end{itemize}
\end{trapbox}

\section{Python Implementation: LOTUS and Moment Calculations}

\begin{lstlisting}[caption={Numerical Verification of LOTUS and Linearity in Python}]
import numpy as np

# Discrete random variable support and PMF
support = np.array([-2, -1, 0, 1, 3])
pmf = np.array([0.10, 0.20, 0.40, 0.20, 0.10])

# 1. Expectation E[X]
mean_x = np.sum(support * pmf)

# 2. Second raw moment E[X^2] via LOTUS
e_x2 = np.sum((support**2) * pmf)

# 3. Variance Var(X) = E[X^2] - (E[X])^2
var_x = e_x2 - (mean_x**2)
sd_x = np.sqrt(var_x)

print(f"E[X]     = {mean_x:.4f}")
print(f"E[X^2]   = {e_x2:.4f}")
print(f"Var(X)   = {var_x:.4f}")
print(f"SD(X)    = {sd_x:.4f}")

# 4. Verifying affine scaling: Y = -3X + 5
y_vals = -3 * support + 5
mean_y = np.sum(y_vals * pmf)
var_y = np.sum((y_vals - mean_y)**2 * pmf)

print(f"\nTransformed Y = -3X + 5:")
print(f"Empirical E[Y] = {mean_y:.4f} (Matches -3*E[X]+5 = {-3*mean_x+5:.4f})")
print(f"Empirical Var(Y) = {var_y:.4f} (Matches (-3)^2 * Var(X) = {9 * var_x:.4f})")
\end{lstlisting}

\subsection*{Part A: Computational and Analytical Problems}

\begin{enumerate}
    \item \textbf{Expectation in Financial Decision Making:} An angel investment fund reviews a seed-stage AI startup. The venture has:
    \begin{itemize}
        \item A $2\%$ probability of generating an IPO payout of \$10,000,000.
        \item An $8\%$ probability of an acquisition payout of \$1,000,000.
        \item A $90\%$ probability of total company failure (payout of \$0).
    \end{itemize}
    \begin{enumerate}
        \item Define the PMF of the gross payout $X$.
        \item Calculate the Expected Gross Payout $\E[X]$.
        \item If the fund must invest \$250,000 to acquire this equity stake, compute the expected net profit. Is the investment statistically rational?
    \end{enumerate}

    \item \textbf{Variance and Moment Calculation via LOTUS:} A discrete random variable $Z$ has PMF:
    $$p_Z(-2) = 0.15, \quad p_Z(0) = 0.50, \quad p_Z(2) = 0.35$$
    \begin{enumerate}
        \item Calculate $\E[Z]$.
        \item Using LOTUS, compute the second moment $\E[Z^2]$.
        \item Calculate $\Var(Z)$ using the shortcut formula $\E[Z^2] - (\E[Z])^2$.
        \item Calculate the Standard Deviation $\SD(Z)$.
    \end{enumerate}

    \item \textbf{Affine Scaling of Expectation and Variance:} Ambient temperature sensor readings $C$ in Celsius have mean $\E[C] = 20^\circ\text{C}$ and standard deviation $\SD(C) = 5^\circ\text{C}$. The sensor values are converted to Fahrenheit via $F = 1.8 C + 32$.
    \begin{enumerate}
        \item Compute $\E[F]$.
        \item Compute $\Var(F)$.
        \item Compute $\SD(F)$.
    \end{enumerate}

    \item \textbf{Variance of Independent Sums and Differences:} Let $X_1$ and $X_2$ be two independent random variables measuring component weights (in grams), with:
    $$\E[X_1] = 120, \quad \Var(X_1) = 16, \qquad \E[X_2] = 80, \quad \Var(X_2) = 9$$
    \begin{enumerate}
        \item Compute $\E[X_1 + X_2]$ and $\Var(X_1 + X_2)$.
        \item Compute $\E[X_1 - X_2]$ and $\Var(X_1 - X_2)$.
        \item State $\SD(X_1 - X_2)$.
    \end{enumerate}

    \item \textbf{Jensen's Inequality Demonstration:} Let $X$ take values $2$ and $8$ with equal probability: $\P(X = 2) = 0.5, \P(X = 8) = 0.5$.
    \begin{enumerate}
        \item Compute $\E[X]$ and evaluate $g(\E[X])$ for $g(x) = x^2$.
        \item Compute $\E[X^2]$ directly via LOTUS.
        \item Show that $\E[X^2] > (\E[X])^2$ and verify that the difference equals $\Var(X)$.
    \end{enumerate}
\end{enumerate}

\subsection*{Part B: Conceptual and Diagnostic Multiple Choice Questions}

\begin{enumerate}[resume]
\item \textbf{Conceptual Diagnostic: Transformations of Variance and Standard Deviation} \\
Let $X$ be a random variable with mean $\E[X] = 10$ and standard deviation $\SD(X) = 3$. An engineer defines the linear transformation $W = -5X + 12$. What are the variance $\Var(W)$ and standard deviation $\SD(W)$ of the transformed variable?
\begin{enumerate}
    \item $\Var(W) = -45, \quad \SD(W) = -15$
    \item $\Var(W) = 225, \quad \SD(W) = 15$
    \item $\Var(W) = 225, \quad \SD(W) = -15$
    \item $\Var(W) = 45, \quad \SD(W) = 15$
\end{enumerate}

\item \textbf{Conceptual Diagnostic: Uncertainty in Difference of Independent Processes} \\
A quality assurance lab measures the thickness of a coating by taking the difference between total coated thickness $T \sim (\mu_T = 5.0\,\text{mm}, \sigma_T^2 = 0.04)$ and baseline substrate thickness $S \sim (\mu_S = 4.2\,\text{mm}, \sigma_S^2 = 0.01)$, where $T$ and $S$ are measured independently. An intern reports that the variance of the coating thickness $C = T - S$ is $\sigma_C^2 = 0.04 - 0.01 = 0.03\,\text{mm}^2$. Which of the following correctly identifies the error in the intern's calculation?
\begin{enumerate}
    \item The intern should have multiplied the variances because thicknesses are on a ratio scale.
    \item The calculation is correct because subtraction of physical variables subtracts their respective variances.
    \item The intern committed the variance subtraction fallacy; because $T$ and $S$ are independent, their variances add: $\Var(T - S) = \Var(T) + (-1)^2\Var(S) = 0.04 + 0.01 = 0.05\,\text{mm}^2$.
    \item The intern should have reported the standard deviation directly as $0.2 - 0.1 = 0.1\,\text{mm}$.
\end{enumerate}
\end{enumerate}
