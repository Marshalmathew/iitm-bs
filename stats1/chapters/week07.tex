\section{Conditional Probability: Sample Space Restriction}

When an observer learns that event $B$ has occurred, all outcomes outside $B$ become impossible ($\Omega \setminus B = \emptyset$). The event $B$ becomes the \textbf{new, restricted sample space}.

\begin{definitionbox}{Conditional Probability}
\index{Conditional Probability}
Let $A$ and $B$ be events in sample space $\Omega$ with $\P(B) > 0$. The \textbf{conditional probability of $A$ given $B$} is:
$$\P(A \mid B) = \frac{\P(A \cap B)}{\P(B)}$$
\begin{itemize}
    \item \textbf{Measure Consistency:} For any fixed conditioning event $B$ with $\P(B) > 0$, the function $\P(\cdot \mid B)$ is a legitimate probability measure that satisfies all three Kolmogorov axioms: $\P(A \mid B) \ge 0$, $\P(\Omega \mid B) = 1$, and $\P(A_1 \cup A_2 \mid B) = \P(A_1 \mid B) + \P(A_2 \mid B)$ for disjoint $A_1, A_2$.
\end{itemize}
\end{definitionbox}

\section{The Multiplication Rule and the Chain Rule}

Rearranging the definition of conditional probability yields the method for computing joint intersections:

\begin{theorembox}{The General Multiplication Rule and Chain Rule}
\index{Multiplication Rule}\index{Chain Rule}
\begin{itemize}
    \item \textbf{Two Events:} For any events $A$ and $B$:
    $$\P(A \cap B) = \P(B) \cdot \P(A \mid B) = \P(A) \cdot \P(B \mid A)$$
    \item \textbf{Kolmogorov Chain Rule (Multiple Events):} For any finite sequence of events $A_1, A_2, \dots, A_n$:
    $$\P(A_1 \cap A_2 \cap \dots \cap A_n) = \P(A_1) \cdot \P(A_2 \mid A_1) \cdot \P(A_3 \mid A_1 \cap A_2) \dots \P(A_n \mid \bigcap_{i=1}^{n-1} A_i)$$
\end{itemize}
\textbf{Data Science Impact:} Modern autoregressive Large Language Models (LLMs like GPT) compute the joint probability of generating a text sequence $w_1, w_2, \dots, w_T$ via this exact chain rule: $\P(w_1, \dots, w_T) = \prod_{t=1}^T \P(w_t \mid w_1, \dots, w_{t-1})$.
\end{theorembox}

\section{Statistical Independence}

\begin{definitionbox}{Statistical Independence}
\index{Statistical Independence}
Two events $A$ and $B$ are statistically independent if and only if knowledge of $B$ imparts zero information regarding $A$:
$$\P(A \mid B) = \P(A) \iff \P(B \mid A) = \P(B)$$
Multiplying by $\P(B)$ yields the formal product criterion:
$$\P(A \cap B) = \P(A) \cdot \P(B)$$
\begin{itemize}
    \item \textbf{Pairwise vs. Mutual Independence:} For three events $A, B, C$, pairwise independence ($\P(AB) = \P(A)\P(B)$, $\P(BC) = \P(B)\P(C)$, $\P(AC) = \P(A)\P(C)$) does \textbf{not} imply mutual independence! Mutual independence requires in addition that:
    $$\P(A \cap B \cap C) = \P(A) \cdot \P(B) \cdot \P(C)$$
\end{itemize}
\end{definitionbox}

\section{The Law of Total Probability and Bayes' Theorem}

\begin{definitionbox}{Partitions of a Sample Space}
\index{Partition}
A collection of events $\{B_1, B_2, \dots, B_k\}$ forms a \textbf{partition} of sample space $\Omega$ if:
\begin{enumerate}
    \item They are pairwise mutually exclusive: $B_i \cap B_j = \emptyset$ for all $i \neq j$.
    \item They are collectively exhaustive: $\bigcup_{i=1}^k B_i = \Omega$.
    \item Each component has non-zero probability: $\P(B_i) > 0$ for all $i$.
\end{enumerate}
\end{definitionbox}

\begin{theorembox}{The Law of Total Probability (LTP)}
\index{Law of Total Probability}
Let $\{B_1, \dots, B_k\}$ be a partition of $\Omega$. For any event $A \subseteq \Omega$:
$$\P(A) = \sum_{i=1}^k \P(A \cap B_i) = \sum_{i=1}^k \P(B_i) \cdot \P(A \mid B_i)$$
\end{theorembox}

\begin{theorembox}{Bayes' Theorem (Inverting Conditional Probabilities)}
\index{Bayes' Theorem}
Given a partition $\{B_1, \dots, B_k\}$ and an observed piece of evidence $A$ with $\P(A) > 0$:
$$\P(B_i \mid A) = \frac{\P(B_i \cap A)}{\P(A)} = \frac{\P(B_i) \cdot \P(A \mid B_i)}{\sum_{j=1}^k \P(B_j) \cdot \P(A \mid B_j)}$$
\begin{itemize}
    \item \textbf{Prior Probability $\P(B_i)$:} Baseline belief in hypothesis $B_i$ before observing evidence.
    \item \textbf{Likelihood $\P(A \mid B_i)$:} The probability that evidence $A$ would be generated if hypothesis $B_i$ were true.
    \item \textbf{Marginal Evidence $\P(A)$:} Total probability of evidence across all competing hypotheses.
    \item \textbf{Posterior Probability $\P(B_i \mid A)$:} Updated belief in hypothesis $B_i$ in light of evidence $A$.
\end{itemize}
\end{theorembox}

\section{Cognitive Traps in Conditional Probability}

\begin{trapbox}{Trap 1: The Disjoint vs. Independent Catastrophe!}
\index{Disjoint vs. Independent}
Students constantly confuse "Mutually Exclusive (Disjoint)" with "Independent". They are diametrically opposite concepts!
\begin{itemize}
    \item \textbf{Mutually Exclusive ($A \cap B = \emptyset$):} The events \textbf{cannot happen together}. If $A$ happens, $B$ is guaranteed not to happen ($\P(B \mid A) = 0$).
    \item \textbf{Independent ($\P(A \cap B) = \P(A)\P(B)$):} Knowing that $A$ occurred provides \textbf{zero information} about $B$ ($\P(B \mid A) = \P(B)$).
    \item \textbf{The Fatal Contradiction:} If two non-trivial events ($\P(A) > 0, \P(B) > 0$) are mutually exclusive, then:
    $$\P(A \cap B) = 0 \neq \P(A) \cdot \P(B) > 0$$
    Therefore, two non-trivial events that are disjoint can \textbf{NEVER be independent}! Disjointness represents extreme, maximal statistical dependence!
\end{itemize}
\end{trapbox}

\begin{trapbox}{Trap 2: The Transposition Fallacy (The Prosecutor's Fallacy)}
Never equate $\P(A \mid B)$ with $\P(B \mid A)$!
\begin{itemize}
    \item The probability that an individual is a professional basketball player given that they are over 7 feet tall ($\P(\text{NBA} \mid >7\text{ft})$) is moderately high ($\approx 17\%$).
    \item The probability that an individual is over 7 feet tall given that they are an NBA player ($\P(>7\text{ft} \mid \text{NBA})$) is tiny ($\approx 1\%$).
    \item In forensic science and courtrooms, confusing $\P(\text{Match} \mid \text{Innocent})$ with $\P(\text{Innocent} \mid \text{Match})$ has led to wrongful convictions (the Prosecutor's Fallacy).
\end{itemize}
\end{trapbox}

\begin{trapbox}{Trap 3: The Base Rate Fallacy in Rare Event Detection}
\index{Base Rate Fallacy}
Suppose a medical condition has prevalence $\P(D) = 0.001$ ($0.1\%$). A diagnostic test has $99\%$ sensitivity ($\P(+ \mid D) = 0.99$) and $99\%$ specificity ($\P(- \mid H) = 0.99 \implies \P(+ \mid H) = 0.01$).
A patient tests positive. What is the probability they actually have the disease?
\begin{itemize}
    \item \textbf{Naive Intuition:} "The test is 99\% accurate, so the patient has a 99\% chance of being sick!" (Completely false!).
    \item \textbf{Bayesian Reality:} Out of $100{,}000$ people, $100$ are sick (yielding $\approx 99$ true positives) and $99{,}900$ are healthy (yielding $\approx 999$ false positives).
    $$\P(D \mid +) = \frac{0.001 \times 0.99}{(0.001 \times 0.99) + (0.999 \times 0.01)} = \frac{0.00099}{0.00099 + 0.00999} \approx \mathbf{9.0\%}$$
    Because the base rate ($0.1\%$) is so low, false positives from the massive healthy cohort dwarf true positives!
\end{itemize}
\end{trapbox}

\section{Comparative Synthesis: Disjoint vs. Independent Events}

\begin{tabular}{p{3.2cm}p{4.2cm}p{4.2cm}}
\toprule
\textbf{Dimension} & \textbf{Mutually Exclusive (Disjoint)} & \textbf{Statistically Independent} \\
\midrule
\textbf{Mathematical Condition} & $A \cap B = \emptyset \implies \P(A \cap B) = 0$ & $\P(A \cap B) = \P(A) \cdot \P(B)$ \\
\textbf{Conditional Probability} & $\P(A \mid B) = 0$ (Complete suppression) & $\P(A \mid B) = \P(A)$ (Zero impact) \\
\textbf{Physical Meaning} & Events cannot co-occur in the same experiment & Occurrence of one gives zero information about the other \\
\textbf{Venn Diagram Representation} & Circles are completely separated (zero overlap) & Circles overlap with area exactly $\P(A)\P(B)$ \\
\textbf{Coexistence Rule} & If $\P(A)>0, \P(B)>0$, they \textbf{cannot} be independent & If $\P(A)>0, \P(B)>0$, they \textbf{cannot} be disjoint \\
\bottomrule
\end{tabular}
\end{center}

\section{Python Implementation: Bayesian Updating and Diagnostic Testing}

\begin{lstlisting}[caption={Bayes' Theorem and Base Rate Sensitivity in Python}]
import numpy as np

def compute_posterior(prior, sensitivity, false_positive_rate):
    """
    Computes P(State | Positive Test) using Bayes' Theorem
    """
    p_pos = (prior * sensitivity) + ((1.0 - prior) * false_positive_rate)
    posterior = (prior * sensitivity) / p_pos
    return posterior, p_pos

# Base rate sensitivity analysis
priors = [0.0001, 0.001, 0.01, 0.05, 0.20]  # varying base rate prevalence
sens = 0.99                                  # 99% true positive rate
fpr = 0.01                                   # 1% false positive rate (99% specificity)

print(f"{'Base Rate (Prior)':<18} | {'P(D | +) Posterior':<20} | {'Total P(+)':<12}")
print("-" * 56)
for p in priors:
    post, p_test = compute_posterior(p, sens, fpr)
    print(f"{p:<18.4f} | {post:<20.4%} | {p_test:<12.5f}")
\end{lstlisting}

\subsection*{Part A: Computational and Analytical Problems}

\begin{enumerate}
    \item \textbf{Conditional Probability in Poker Deck Selection:} A standard 52-card deck is thoroughly shuffled.
    \begin{enumerate}
        \item What is the unconditional probability $\P(K)$ that a randomly selected card is a King?
        \item Given that an observer reveals the selected card is a Red Card ($R$), find $\P(K \mid R)$.
        \item Given that an observer reveals the selected card is a Face Card ($F$), find $\P(K \mid F)$.
        \item Are the events $K$ (King) and $R$ (Red) statistically independent? Prove your answer.
    \end{enumerate}

    \item \textbf{Cascading Chain Rule Without Replacement:} A server cache contains 10 memory blocks: 6 valid cached blocks and 4 corrupted blocks. An operating system routine reads 3 memory blocks consecutively \textbf{without replacement}. Compute the exact probability that all 3 selected blocks are corrupted.

    \item \textbf{Testing Mathematical Independence:} Let $A$ and $B$ be two events in an experiment with $\P(A) = 0.50$, $\P(B) = 0.60$, and $\P(A \cup B) = 0.80$.
    \begin{enumerate}
        \item Compute $\P(A \cap B)$.
        \item Compute $\P(A \mid B)$ and $\P(B \mid A)$.
        \item State whether $A$ and $B$ are statistically independent, and justify your answer.
    \end{enumerate}

    \item \textbf{Bayesian Medical Diagnosis and False Positives:} A rare autoimmune disease has a population base rate of $\P(D) = 0.002$ ($0.2\%$). A diagnostic blood screening has a sensitivity of $98\%$ ($\P(+ \mid D) = 0.98$) and a false positive rate of $2\%$ ($\P(+ \mid H) = 0.02$).
    \begin{enumerate}
        \item Compute the overall probability $\P(+)$ that a randomly chosen citizen tests positive.
        \item If a citizen tests positive, calculate the posterior probability $\P(D \mid +)$ that they actually possess the disease.
    \end{enumerate}

    \item \textbf{Multinomial Bayes Classifier in Spam Detection:} An automated spam filter classifies incoming messages into $\text{Spam}$ ($\P(S) = 0.15$) and $\text{Ham}$ ($\P(H) = 0.85$). The word "Cryptocurrency" appears in $70\%$ of spam emails ($\P(W \mid S) = 0.70$) and in $5\%$ of legitimate ham emails ($\P(W \mid H) = 0.05$). An incoming email arrives containing the word "Cryptocurrency". Compute the posterior probability $\P(S \mid W)$ that the email is spam.
\end{enumerate}

\subsection*{Part B: Conceptual and Diagnostic Multiple Choice Questions}

\begin{enumerate}[resume]
\item \textbf{Conceptual Diagnostic: The Disjoint vs. Independent Fallacy} \\
Let $E_1$ and $E_2$ be two events in a random experiment such that $\P(E_1) = 0.35$ and $\P(E_2) = 0.40$. An engineering student asserts: \textit{"Because $E_1$ and $E_2$ are mutually exclusive ($E_1 \cap E_2 = \emptyset$), they must also be statistically independent, because neither event influences the other."} Which of the following statements provides the \textbf{most mathematically rigorous critique} of the student's assertion?
\begin{enumerate}
    \item The student is correct because mutually exclusive events share no outcomes, which is the definition of independence.
    \item The student is wrong; because $\P(E_1 \cap E_2) = 0$ while $\P(E_1)\P(E_2) = 0.35 \times 0.40 = 0.14 \neq 0$, the events are fundamentally dependent. Mutually exclusive non-trivial events can never be independent.
    \item The student is wrong because independence can only be evaluated for continuous distributions, not discrete events.
    \item The student is correct provided that the sample space $\Omega$ is finite.
\end{enumerate}

\item \textbf{Conceptual Diagnostic: Transposition and the Prosecutor's Fallacy} \\
In an automated fraud classification pipeline, an algorithm flags transactions with a feature anomaly flag $A$. Forensic audit data shows that \textbf{$99\%$ of fraudulent transactions exhibit anomaly $A$} ($\P(A \mid \text{Fraud}) = 0.99$). A junior analyst reviews an incoming transaction exhibiting anomaly $A$ and declares: \textit{"Because $\P(A \mid \text{Fraud}) = 0.99$, this flagged transaction has a $99\%$ probability of being fraudulent."} What foundational statistical error has the analyst committed?
\begin{enumerate}
    \item Violating Kolmogorov's Third Axiom by assuming fraud is countably additive.
    \item Transposition of conditionals ($\P(\text{Fraud} \mid A) \neq \P(A \mid \text{Fraud})$); the analyst ignored the base rate of fraud and the false alarm rate of anomaly $A$ among legitimate transactions.
    \item Conflating an Ordinal scale with an Interval scale during feature extraction.
    \item Confusing sample covariance with the Pearson correlation coefficient.
\end{enumerate}
\end{enumerate}
