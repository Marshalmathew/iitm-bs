\chapter{The Poisson Distribution \& Poisson Processes}

While the Binomial distribution models counts of successes across a fixed integer number of trials $n$, countless empirical data science scenarios operate where $n$ is uncountably massive while the occurrence probability $p$ of any single event is microscopically small. How many customer requests will an AWS server receive in the next sixty seconds? How many network packet collisions occur per megabyte? How many typos appear in a 500-page manuscript? The \index{Poisson Distribution}\textbf{Poisson Distribution} mathematically models the frequency of rare, independent events occurring within a continuous exposure interval of time, space, or volume. In this chapter, we master the \textbf{Poisson Limit of the Binomial Distribution}, the \textbf{Poisson Probability Mass Function}, the \textbf{Equidispersion Property ($\E[X] = \Var(X) = \lambda$)}, the \textbf{Superposition of Independent Poisson Processes}, and the cognitive traps: \textbf{Overdispersion in Machine Learning}, and \textbf{Interval Mismatch}.

\vspace{0.5cm}
\begin{tcolorbox}[colback=primarylight, colframe=primarydark, title={\textbf{What You'll Learn in this Chapter}}]
\begin{itemize}
    \item Mathematical derivation of the Poisson distribution as the asymptotic limit of the Binomial ($n \to \infty, p \to 0$ with $\lambda = np$ fixed).
    \item The Poisson PMF: $p_X(k) = \frac{e^{-\lambda}\lambda^k}{k!}$ on countably infinite support $\mathcal{R}_X = \{0, 1, 2, \dots\}$.
    \item The Equidispersion Theorem: why Expected Value and Variance are identically $\lambda$.
    \item Scaling arrival rate parameters across variable exposure windows: $X_t \sim \text{Poisson}(\lambda t)$.
    \item Superposition: the sum of independent Poisson random variables is strictly Poisson ($\lambda_{\text{sum}} = \sum \lambda_i$).
    \item Cognitive traps: overdispersion failure in applied counts ($\Var > \E$), and invalid Poisson approximation when $p$ is large.
    \item Two-tier exercises with diagnostic multiple choice questions on overdispersion and multi-stream Poisson queues.
\end{itemize}

\section{The Poisson Limit of the Binomial Distribution}

In 1837, French mathematician Siméon Denis Poisson derived this distribution by investigating how the Binomial distribution behaves when the number of trials grows without bound while the event remains rare.

\begin{theorembox}{The Law of Rare Events (Poisson Limit Theorem)}
\index{Poisson Limit Theorem}
Let $X_n \sim \text{Binomial}(n, p_n)$. Suppose $n \to \infty$ and $p_n \to 0$ in such a manner that the expected number of successes remains constant:
$$\lim_{n \to \infty} n p_n = \lambda \quad (\text{where } \lambda > 0)$$
Then for any fixed integer $k \in \{0, 1, 2, \dots\}$:
$$\lim_{n \to \infty} \P(X_n = k) = \frac{e^{-\lambda}\lambda^k}{k!}$$
\end{theorembox}

\begin{intuitionbox}{Derivation of the Poisson Formula}
Substitute $p = \frac{\lambda}{n}$ into the Binomial PMF:
$$\P(X_n = k) = \binom{n}{k} \left(\frac{\lambda}{n}\right)^k \left(1 - \frac{\lambda}{n}\right)^{n - k} = \frac{n(n-1)\dots(n-k+1)}{k!} \cdot \frac{\lambda^k}{n^k} \cdot \left(1 - \frac{\lambda}{n}\right)^n \left(1 - \frac{\lambda}{n}\right)^{-k}$$
Now evaluate limits as $n \to \infty$ with $k$ fixed:
\begin{enumerate}
    \item $\lim_{n \to \infty} \frac{n(n-1)\dots(n-k+1)}{n^k} = 1$.
    \item By the definition of Euler's constant: $\lim_{n \to \infty} \left(1 - \frac{\lambda}{n}\right)^n = e^{-\lambda}$.
    \item $\lim_{n \to \infty} \left(1 - \frac{\lambda}{n}\right)^{-k} = 1^{-k} = 1$.
\end{enumerate}
Multiplying these limits together yields exactly: $\mathbf{\frac{\lambda^k e^{-\lambda}}{k!}}$!
\end{intuitionbox}

\begin{theorembox}{Rule of Thumb for Poisson Approximation}
\index{Poisson Approximation}
A Binomial distribution $\text{Binomial}(n, p)$ can be accurately approximated by $\text{Poisson}(\lambda = np)$ whenever:
$$n \ge 20 \quad \text{and} \quad p \le 0.05 \quad (\text{or } n \ge 100 \text{ and } np \le 10)$$
Computing $\binom{100{,}000}{3}(0.00002)^3 (0.99998)^{99997}$ causes severe floating-point underflow in software, whereas computing $\frac{2^3 e^{-2}}{3!}$ is exact, instantaneous, and immune to numerical instability.
\end{theorembox}

\section{The Poisson Probability Mass Function and Moments}

\begin{definitionbox}{The Poisson Distribution}
\index{Poisson Distribution}
A discrete random variable $X$ follows a \textbf{Poisson Distribution} with rate parameter $\lambda > 0$, denoted $X \sim \text{Poisson}(\lambda)$, if its support is countably infinite $\mathcal{R}_X = \{0, 1, 2, \dots\}$ and its PMF is:
$$p_X(k) = \P(X = k) = \frac{e^{-\lambda} \lambda^k}{k!} \quad \text{for } k = 0, 1, 2, \dots$$
\begin{itemize}
    \item \textbf{Total Probability Verification:}
    $$\sum_{k=0}^\infty p_X(k) = \sum_{k=0}^\infty \frac{e^{-\lambda}\lambda^k}{k!} = e^{-\lambda} \sum_{k=0}^\infty \frac{\lambda^k}{k!} = e^{-\lambda} \cdot e^\lambda = e^0 = \mathbf{1.00}$$
\end{itemize}
\end{definitionbox}

\begin{theorembox}{The Equidispersion Theorem: $\E[X] = \Var(X) = \lambda$}
\index{Equidispersion}
For $X \sim \text{Poisson}(\lambda)$:
\begin{itemize}
    \item \textbf{Expected Value:} $\E[X] = \mathbf{\lambda}$
    \item \textbf{Variance:} $\Var(X) = \mathbf{\lambda}$
    \item \textbf{Standard Deviation:} $\sigma_X = \sqrt{\lambda}$
\end{itemize}
\textit{Why does $\Var(X) = \lambda$?} In the Binomial limit, $\Var(X) = np(1-p)$. As $p \to 0$, $(1 - p) \to 1$, leaving $\Var(X) \to np = \lambda$!
\end{theorembox}

\section{Properties of Poisson Processes}

\begin{theorembox}{Linear Scaling Across Time Windows}
\index{Poisson Scaling}
If events arrive according to a homogeneous Poisson process with mean rate $\lambda$ per unit time, then the number of arrivals $X_t$ in an interval of duration $t > 0$ follows:
$$X_t \sim \text{Poisson}(\lambda \cdot t)$$
\end{theorembox}

\begin{theorembox}{Superposition (Sum of Independent Poisson Variables)}
\index{Superposition of Poisson}
If $X_1 \sim \text{Poisson}(\lambda_1)$ and $X_2 \sim \text{Poisson}(\lambda_2)$ are \textbf{independent} random variables, their sum is strictly Poisson:
$$S = X_1 + X_2 \sim \text{Poisson}(\lambda_1 + \lambda_2)$$
\textit{Data Science Relevance:} If web traffic arrives from desktop users at rate $\lambda_1 = 40\,\text{req/sec}$ and mobile users at rate $\lambda_2 = 60\,\text{req/sec}$ independently, the total load on the load balancer is $X_{\text{total}} \sim \text{Poisson}(100\,\text{req/sec})$.
\end{theorembox}

\section{Cognitive Traps in Poisson Modeling}

\begin{trapbox}{Trap 1: The Equidispersion Trap and Overdispersion in Real Data}
\index{Overdispersion}
The Poisson distribution strictly enforces that the variance \textbf{must equal the mean} ($\Var(X) = \E[X] = \lambda$).
\begin{itemize}
    \item In real-world data science (e.g. daily website visits, e-commerce purchases, medical doctor visits), empirical data almost always displays \textbf{Overdispersion}:
    $$\Var(X) \gg \E[X]$$
    \item If you fit a Poisson regression model to overdispersed count data, the model drastically underestimates the true standard errors, falsely declaring uninformative features to be "highly statistically significant" (producing severe false positive discoveries).
    \item \textbf{Remedy:} When overdispersion is present, data scientists switch to the \textbf{Negative Binomial Distribution}, which includes a dispersion parameter $\alpha$ allowing $\Var(X) = \mu + \alpha \mu^2$.
\end{itemize}
\end{trapbox}

\begin{trapbox}{Trap 2: The Exposure Window Mismatch Error}
A frequent exam blunder is failing to adjust $\lambda$ when the problem shifts time scales.
\begin{itemize}
    \item If a server crashes at a rate of $\lambda = 2$ crashes per \emph{week}, what is the probability of 0 crashes in a \emph{month} (4 weeks)?
    \item Writing $\P(X = 0) = e^{-2} \approx 0.135$ is a catastrophic unit error!
    \item The correct rate for a 4-week window is $\lambda_{\text{month}} = 2 \times 4 = 8$. The true probability is $\P(X = 0) = e^{-8} \approx \mathbf{0.000335}$. Always scale $\lambda$ to match the observation interval before evaluating the PMF!
\end{itemize}
\end{trapbox}

\section{Python Implementation: Poisson Modeling and Superposition}

\begin{lstlisting}[caption={Poisson PMF, Cumulative Distribution, and Superposition in SciPy}]
import numpy as np
from scipy.stats import poisson

# Base rate: Lambda = 4 requests per second
lmbda = 4.0

# 1. Exact probability of receiving exactly 4 requests in 1 second
p_exact_4 = poisson.pmf(k=4, mu=lmbda)
print(f"P(X = 4 in 1 sec)  = {p_exact_4:.4f}")

# 2. Probability of at least 1 request in 1 second: 1 - P(0)
p_at_least_1 = 1.0 - poisson.pmf(k=0, mu=lmbda)
print(f"P(X >= 1 in 1 sec) = {p_at_least_1:.4f}")

# 3. Scaling interval: 10-second window -> lambda_10 = 4 * 10 = 40
lmbda_10 = lmbda * 10.0
p_overload = 1.0 - poisson.cdf(k=50, mu=lmbda_10)
print(f"P(X > 50 in 10 sec) = {p_overload:.4f}")

# 4. Superposition of independent streams: Stream A (lambda=3) + Stream B (lambda=5)
lmbda_total = 3.0 + 5.0  # lambda_total = 8
p_sum_8 = poisson.pmf(k=8, mu=lmbda_total)
print(f"Combined P(Sum = 8) = {p_sum_8:.4f}")
\end{lstlisting}

\subsection*{Part A: Computational and Analytical Problems}

\begin{enumerate}
    \item \textbf{Basic Poisson Probabilities:} A data center encounters disk drive failures at an average rate of $\lambda = 2.5$ drive failures per month. Assuming disk failures follow a Poisson process:
    \begin{enumerate}
        \item What is the probability that exactly 2 drives fail in a given month?
        \item What is the probability that exactly 0 drives fail in a given month?
        \item What is the probability that at least 1 drive fails in a given month?
    \end{enumerate}

    \item \textbf{Scaling Arrival Rates Across Time Windows:} A cloud microservice receives traffic at an average rate of $\lambda = 12$ incoming requests per minute.
    \begin{enumerate}
        \item Find the expected number of requests and the variance in a 30-second window.
        \item Compute the probability of receiving exactly 5 requests in that 30-second window.
        \item Find the probability of receiving 0 requests in a 15-second window.
    \end{enumerate}

    \item \textbf{Poisson Approximation to Binomial Accuracy:} An optical telecommunications cable experiences single-bit transmission errors with probability $p = 0.0005$ per byte. An engineer transmits a packet of $n = 4{,}000$ bytes.
    \begin{enumerate}
        \item What is the parameter $\lambda = np$ for the Poisson approximation?
        \item Compute the approximate probability that the packet contains exactly 2 bit errors.
        \item Compute the approximate probability that the packet transmits completely error-free ($k = 0$).
    \end{enumerate}

    \item \textbf{Superposition of Independent Queues:} An e-commerce API gateway handles two independent transaction channels: Credit Card checkouts ($\lambda_1 = 3\,\text{req/sec}$) and Digital Wallet checkouts ($\lambda_2 = 5\,\text{req/sec}$).
    \begin{enumerate}
        \item State the distribution of the combined transaction stream $S = X_1 + X_2$.
        \item Compute the probability that exactly 8 transactions arrive in a 1-second window.
        \item Compute the expected number of transactions in a 10-second window.
    \end{enumerate}

    \item \textbf{Moment Calculations for Poisson Variables:} Let $Y \sim \text{Poisson}(\lambda = 5)$.
    \begin{enumerate}
        \item State $\E[Y]$ and $\Var(Y)$.
        \item Using the computational variance formula $\Var(Y) = \E[Y^2] - (\E[Y])^2$, compute the second raw moment $\E[Y^2]$.
        \item Evaluate $\E[Y(Y - 1)]$ using LOTUS.
    \end{enumerate}
\end{enumerate}

\subsection*{Part B: Conceptual and Diagnostic Multiple Choice Questions}

\begin{enumerate}[resume]
\item \textbf{Conceptual Diagnostic: Equidispersion and Overdispersion Breakdown} \\
A healthcare data science team fits a generalized linear model to predict the count of patient hospital admissions per year. The dataset yields an empirical sample mean of $\bar{x} = 2.10$ admissions and a sample variance of $s^2 = 14.70$. An analyst suggests fitting a standard Poisson regression model. What fundamental modeling flaw exists in this approach?
\begin{enumerate}
    \item The sample mean $\bar{x} = 2.10$ is too small for a Poisson distribution.
    \item The data displays severe \textbf{overdispersion} ($s^2 = 14.70 \gg 2.10 = \bar{x}$), violently violating the Poisson equidispersion requirement ($\Var(X) = \E[X]$); fitting a Poisson model will drastically underestimate standard errors and produce spurious statistical significance.
    \item The data is underdispersed, meaning a Binomial distribution must be used instead.
    \item Poisson models require continuous target variables, whereas patient counts are discrete integers.
\end{enumerate}

\item \textbf{Conceptual Diagnostic: Poisson Approximation Boundary Conditions} \\
A machine learning engineer evaluates whether to approximate a Binomial distribution $X \sim \text{Binomial}(n, p)$ using a Poisson distribution. Under which of the following parameter regimes is the Poisson approximation \textbf{most mathematically accurate and justified}?
\begin{enumerate}
    \item $n = 10, \quad p = 0.50$
    \item $n = 100{,}000, \quad p = 0.00003$
    \item $n = 50, \quad p = 0.85$
    \item $n = 10{,}000, \quad p = 0.60$
\end{enumerate}
\end{enumerate}
