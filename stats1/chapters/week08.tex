\section{Introduction to Random Variables}

\begin{definitionbox}{Random Variable}
\index{Random Variable}
Let $(\Omega, \mathcal{F}, \P)$ be a probability space. A \textbf{Random Variable ($X$)} is a deterministic function that maps every elementary outcome $\omega \in \Omega$ to a real number on the real line:
$$X \colon \Omega \to \mathbb{R}$$
For any subset $B \subseteq \mathbb{R}$, we write $\P(X \in B)$ as shorthand for the probability of the preimage event:
$$\P(\{\omega \in \Omega \colon X(\omega) \in B\})$$
\begin{itemize}
    \item \textbf{Support ($\mathcal{R}_X$):} The set of all possible numerical values that $X$ can assume with non-zero probability:
    $$\mathcal{R}_X = \{x \in \mathbb{R} \colon \P(X = x) > 0\}$$
    \item \textbf{Discrete Random Variable:} The support $\mathcal{R}_X$ is a finite or countably infinite set (e.g., $\{0, 1, 2, \dots\}$).
    \item \textbf{Continuous Random Variable:} The support $\mathcal{R}_X$ forms an uncountably infinite continuum or interval on $\mathbb{R}$.
\end{itemize}
\end{definitionbox}

\begin{intuitionbox}{A Random Variable is a Function, Not a Variable!}
Despite its name, a random variable is \textbf{neither random nor a variable}.
\begin{itemize}
    \item The mapping $X(\omega)$ is a $100\%$ \textbf{deterministic mathematical function}.
    \item The only "randomness" resides in which elementary outcome $\omega \in \Omega$ the universe realizes!
    \item For example, in tossing two coins ($\Omega = \{HH, HT, TH, TT\}$), the random variable "number of heads" is the fixed rule:
    $$X(HH) = 2, \quad X(HT) = 1, \quad X(TH) = 1, \quad X(TT) = 0$$
\end{itemize}
\end{intuitionbox}

\section{The Probability Mass Function (PMF)}

For a discrete random variable, probability is assigned to isolated numerical values in its support.

\begin{definitionbox}{Probability Mass Function (PMF)}
\index{Probability Mass Function}
The \textbf{Probability Mass Function} of a discrete random variable $X$, denoted $p_X(x)$ or $f(x)$, is:
$$p_X(x) = \P(X = x)$$
\begin{itemize}
    \item \textbf{Kolmogorov Validity Conditions:} For $p_X(x)$ to be a valid mathematical PMF:
    \begin{enumerate}
        \item \textbf{Non-Negativity:} $p_X(x) \ge 0$ for all $x \in \mathbb{R}$ (with $p_X(x) = 0$ for all $x \notin \mathcal{R}_X$).
        \item \textbf{Total Probability Normalization:}
        $$\sum_{x \in \mathcal{R}_X} p_X(x) = 1.00$$
    \end{enumerate}
\end{itemize}
\end{definitionbox}

\section{The Cumulative Distribution Function (CDF)}

The CDF aggregates probability from $-\infty$ up to a given real threshold $x$.

\begin{definitionbox}{Cumulative Distribution Function (CDF)}
\index{Cumulative Distribution Function}
The \textbf{Cumulative Distribution Function} of a random variable $X$, denoted $F_X(x)$, is defined for all $x \in \mathbb{R}$ by:
$$F_X(x) = \P(X \le x) = \sum_{t \le x, \, t \in \mathcal{R}_X} p_X(t)$$
\end{definitionbox}

\begin{theorembox}{The Four Axiomatic Properties of a CDF}
\index{Properties of CDF}
Any mathematically valid CDF $F(x)$ satisfies:
\begin{enumerate}
    \item \textbf{Bounded Range:} $0 \le F(x) \le 1$ for all $x \in \mathbb{R}$.
    \item \textbf{Monotonically Non-Decreasing:} If $x_1 < x_2$, then $F(x_1) \le F(x_2)$.
    \item \textbf{Asymptotic Extremes:}
    $$\lim_{x \to -\infty} F(x) = 0 \quad \text{and} \quad \lim_{x \to +\infty} F(x) = 1$$
    \item \textbf{Right-Continuity:} The function is continuous from the right at every point $x$:
    $$\lim_{h \to 0^+} F(x + h) = F(x)$$
\end{enumerate}
\end{theorembox}

\begin{theorembox}{Jump Discontinuities and Point Probabilities}
\index{Jump Discontinuity}
For a discrete random variable, $F_X(x)$ is a \textbf{step function}.
\begin{itemize}
    \item The CDF is flat (horizontal) between support points.
    \item At every support value $x_k \in \mathcal{R}_X$, $F_X(x)$ exhibits a vertical jump discontinuity.
    \item The \textbf{height of the jump} equals the exact PMF probability mass at that point:
    $$\P(X = x) = F_X(x) - F_X(x^-) = F_X(x) - \lim_{t \to x^-} F_X(t)$$
\end{itemize}
\end{theorembox}

\section{Cognitive Traps in Discrete CDF Analysis}

\begin{trapbox}{Trap 1: Strict vs. Weak Inequalities in Discrete CDFs}
In continuous probability, $\P(X \le x) = \P(X < x)$ because individual points carry zero probability. \textbf{In discrete probability, strict vs. weak inequality is completely different!}
Let $X$ take integer values:
\begin{itemize}
    \item $\P(X \le b) = F(b)$.
    \item $\P(X < b) = F(b - 1) = F(b) - \P(X = b)$.
    \item $\P(a < X \le b) = F(b) - F(a)$.
    \item $\P(a \le X \le b) = F(b) - F(a - 1) = F(b) - F(a) + \P(X = a)$.
    \item $\P(X > a) = 1 - \P(X \le a) = 1 - F(a)$.
    \item $\P(X \ge a) = 1 - \P(X < a) = 1 - F(a - 1)$.
\end{itemize}
Failing to shift the boundary by $1$ in discrete integer intervals is the single most common exam calculation error!
\end{trapbox}

\begin{trapbox}{Trap 2: The Non-Integer CDF Evaluation Fallacy}
Students often believe that if $X$ takes integer values $\{1, 2, 3\}$, then $F(2.7)$ is either undefined or zero.
\begin{itemize}
    \item By definition, $F(2.7) = \P(X \le 2.7)$.
    \item The only integers less than or equal to $2.7$ are $1$ and $2$.
    \item Therefore:
    $$F(2.7) = \P(X = 1) + \P(X = 2) = F(2)$$
    The CDF is fully defined for \textbf{all real numbers $x \in \mathbb{R}$}, remaining constant across intervals between support points.
\end{itemize}
\end{trapbox}

\section{Comparative Synthesis: PMF vs. CDF}

\begin{tabular}{p{3.2cm}p{4.2cm}p{4.2cm}}
\toprule
\textbf{Feature} & \textbf{Probability Mass Function (PMF)} & \textbf{Cumulative Distribution (CDF)} \\
\midrule
\textbf{Notation} & $p_X(x)$ or $\P(X = x)$ & $F_X(x)$ or $\P(X \le x)$ \\
\textbf{Operational Meaning} & Point probability mass at exact coordinate $x$ & Accumulated probability mass from $-\infty$ to $x$ \\
\textbf{Domain} & Defined on support $\mathcal{R}_X$ (zero elsewhere) & Defined on the entire real line $\mathbb{R}$ \\
\textbf{Range} & $p_X(x) \in [0, 1]$, summing to $1.00$ & $F_X(x) \in [0, 1]$, increasing from $0$ to $1$ \\
\textbf{Graphical Form} & Spike / stick plot at discrete values & Step function with right-continuous jumps \\
\textbf{Conversion Formula} & $p_X(x) = F_X(x) - F_X(x^-)$ & $F_X(x) = \sum_{t \le x} p_X(t)$ \\
\bottomrule
\end{tabular}

\section{Python Implementation: PMF and Step CDF in SciPy}

\begin{lstlisting}[caption={PMF, Step CDF, and Jump Discontinuities in Python}]
import numpy as np
import scipy.stats as stats
import matplotlib.pyplot as plt

# Custom discrete random variable support and PMF
support = np.array([1, 2, 4, 7])
pmf_vals = np.array([0.15, 0.35, 0.30, 0.20])

# Validating Kolmogorov normalization
assert np.isclose(np.sum(pmf_vals), 1.0), "PMF must sum to 1.0!"

# Custom CDF step function evaluation
def discrete_cdf(x_query, supp, pmf):
    return np.sum(pmf[supp <= x_query])

# Querying CDF across integer and non-integer values
print(f"F(0.5) = {discrete_cdf(0.5, support, pmf_vals):.2f} (Before support)")
print(f"F(2.0) = {discrete_cdf(2.0, support, pmf_vals):.2f} (Includes x=1, 2)")
print(f"F(3.5) = {discrete_cdf(3.5, support, pmf_vals):.2f} (Flat step between 2 and 4)")
print(f"F(10.0)= {discrete_cdf(10.0, support, pmf_vals):.2f} (Full mass accumulated)")

# Recovering point mass from jump discontinuity at x=4: F(4) - F(4-)
f_4 = discrete_cdf(4.0, support, pmf_vals)
f_4_minus = discrete_cdf(3.999, support, pmf_vals)
recovered_p4 = f_4 - f_4_minus
print(f"Jump at x=4: {f_4:.2f} - {f_4_minus:.2f} = {recovered_p4:.2f} (Matches PMF {pmf_vals[2]})")
\end{lstlisting}

\subsection*{Part A: Computational and Analytical Problems}

\begin{enumerate}
    \item \textbf{PMF Normalization Parameter:} A discrete random variable $X$ has support $\mathcal{R}_X = \{1, 2, 3, 4, 5\}$ and PMF:
    $$p_X(x) = c \cdot (6 - x) \quad \text{for } x \in \{1, 2, 3, 4, 5\}$$
    \begin{enumerate}
        \item Find the normalization constant $c$.
        \item Calculate $\P(X \ge 3)$.
        \item Calculate $\P(X \text{ is even})$.
    \end{enumerate}

    \item \textbf{Random Variable Construction: Maximum of Two Dice:} Two fair 6-sided dice are rolled independently. Let $Y = \max(D_1, D_2)$.
    \begin{enumerate}
        \item State the support $\mathcal{R}_Y$.
        \item Derive the PMF $p_Y(y)$ for all $y \in \mathcal{R}_Y$.
        \item Compute $\P(Y \le 4)$.
    \end{enumerate}

    \item \textbf{Step CDF Construction:} A server fault counter $K$ has the following PMF:
    $$p_K(0) = 0.40, \quad p_K(1) = 0.35, \quad p_K(2) = 0.15, \quad p_K(3) = 0.10$$
    \begin{enumerate}
        \item Formulate the complete piecewise algebraic definition of the CDF $F_K(k)$ for all $k \in \mathbb{R}$.
        \item Evaluate $F_K(-1)$, $F_K(1.8)$, and $F_K(5)$.
    \end{enumerate}

    \item \textbf{Interval Probabilities via CDF:} Using the CDF from Exercise 3:
    \begin{enumerate}
        \item Compute $\P(1 < K \le 3)$ using the CDF.
        \item Compute $\P(1 \le K \le 3)$ using the CDF and verify against the PMF.
        \item Compute $\P(K > 1)$ using the complement rule on the CDF.
    \end{enumerate}

    \item \textbf{Recovering PMF from Jump Discontinuities:} The CDF of a discrete random variable $W$ is given by:
    $$F_W(w) = \begin{cases}
    0.00, & w < -1 \\
    0.25, & -1 \le w < 1 \\
    0.70, & 1 \le w < 3 \\
    1.00, & w \ge 3
    \end{cases}$$
    Identify the support $\mathcal{R}_W$ and calculate the PMF probability mass $p_W(w)$ at each support point.
\end{enumerate}

\subsection*{Part B: Conceptual and Diagnostic Multiple Choice Questions}

\begin{enumerate}[resume]
\item \textbf{Conceptual Diagnostic: Discrete Interval Boundary Shifting} \\
Let $N$ be a discrete random variable taking integer values $\{0, 1, 2, \dots, 10\}$ with known Cumulative Distribution Function $F_N(n) = \P(N \le n)$. A network reliability analyst needs to calculate the probability that the number of network drops satisfies $3 \le N \le 7$. Which algebraic formula correctly computes this probability using the CDF?
\begin{enumerate}
    \item $F_N(7) - F_N(3)$
    \item $F_N(7) - F_N(2)$
    \item $F_N(8) - F_N(3)$
    \item $1 - F_N(3)$
\end{enumerate}

\item \textbf{Conceptual Diagnostic: Mathematical Validity of CDFs} \\
Consider a proposed candidate function $G(x)$ for the Cumulative Distribution Function of a random variable $X$:
$$G(x) = \begin{cases}
0.0, & x < 0 \\
0.3 + 0.2x, & 0 \le x < 2 \\
0.6, & 2 \le x < 4 \\
1.0, & x \ge 4
\end{cases}$$
Why does $G(x)$ \textbf{fail} to be a valid mathematical CDF?
\begin{enumerate}
    \item It violates non-decreasing monotonicity because $G(2) < \lim_{x \to 2^-} G(x)$.
    \item It violates the asymptotic upper limit because $G(4) = 1.0$ is reached too quickly.
    \item It violates non-negativity because $G(x) = 0$ for $x < 0$.
    \item It violates right-continuity at $x = 0$ because $G(0) = 0.3 \neq \lim_{x \to 0^+} G(x)$.
\end{enumerate}
\end{enumerate}
