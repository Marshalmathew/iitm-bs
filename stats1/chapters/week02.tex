\chapter{Describing Categorical Data}

In data science pipelines and exploratory data analysis (EDA), categorical variables frequently represent targets (e.g., churn vs. retention, fraud vs. legitimate) or structural groupings (e.g., geographic regions, experimental treatment arms). Unlike quantitative variables, categorical features cannot be summarized using algebraic means or standard deviations. Instead, we summarize categorical distributions through \textbf{Frequency and Relative Frequency Distributions}, \textbf{Contingency Tables}, \textbf{Mode and Ordinal Medians}, and visual architectures including \textbf{Bar, Pareto, and Mosaic plots}. We also analyze the classic cognitive trap: \textbf{Simpson's Paradox}.

\section{Frequency Distribution Tables}

Let a categorical variable $X$ take values in $k$ mutually exclusive categories $\{c_1, c_2, \dots, c_k\}$ across a sample of size $n$.

\begin{definitionbox}{Frequency and Relative Frequency}
\index{Frequency}\index{Relative Frequency}
\begin{itemize}
    \item \textbf{Absolute Frequency ($f_i$):} The observed count of instances falling into category $c_i$:
    $$f_i = \sum_{j=1}^n \mathbb{I}(x_j = c_i) \quad \text{such that} \quad \sum_{i=1}^k f_i = n$$
    \item \textbf{Relative Frequency ($rf_i$):} The empirical proportion of observations in category $c_i$:
    $$rf_i = \frac{f_i}{n} \quad \text{such that} \quad \sum_{i=1}^k rf_i = 1.00 \quad (100\%)$$
    \item \textbf{Cumulative Relative Frequency ($crf_m$):} For \emph{Ordinal} categories ordered $c_1 < c_2 < \dots < c_k$:
    $$crf_m = \sum_{i=1}^m rf_i = \frac{1}{n}\sum_{i=1}^m f_i$$
\end{itemize}
\end{definitionbox}

\begin{intuitionbox}{Class Imbalance in Machine Learning}
Checking relative frequencies is the first diagnostic step before training a classifier. If a credit card fraud dataset contains $f_{\text{legitimate}} = 999{,}000$ ($99.9\%$) and $f_{\text{fraud}} = 1{,}000$ ($0.1\%$), a naive model that blindly predicts "Legitimate" for all transactions achieves a deceptive $99.9\%$ accuracy while catching zero fraudulent transactions. In imbalanced settings, accuracy is meaningless; precision, recall, and PR-AUC must be computed instead.
\end{intuitionbox}

\section{Two-Way Contingency Tables}

When analyzing the relationship between two categorical variables ($X$ with $R$ rows and $Y$ with $C$ columns), observations are organized in a \index{Contingency Table}\textbf{Contingency Table} (cross-tabulation).

\begin{definitionbox}{Joint, Marginal, and Conditional Relative Frequencies}
\index{Marginal Distribution}\index{Conditional Distribution}
Let $n_{ij}$ denote the frequency count in cell $(i, j)$ and $n = \sum_{i=1}^R \sum_{j=1}^C n_{ij}$.
\begin{itemize}
    \item \textbf{Joint Relative Frequency:} The proportion of the total sample in cell $(i, j)$:
    $$p_{ij} = \frac{n_{ij}}{n}$$
    \item \textbf{Marginal Row and Column Proportions:} The overall proportion for a single category:
    $$p_{i\cdot} = \frac{\sum_{j=1}^C n_{ij}}{n} = \sum_{j=1}^C p_{ij}, \qquad p_{\cdot j} = \frac{\sum_{i=1}^R n_{ij}}{n} = \sum_{i=1}^R p_{ij}$$
    \item \textbf{Conditional Relative Frequency (Given Row $i$):} The distribution of $Y$ restricted to row $i$:
    $$p(j \mid i) = \frac{n_{ij}}{\sum_{j=1}^C n_{ij}} = \frac{p_{ij}}{p_{i\cdot}}$$
    Two categorical variables are \textbf{independent} if and only if $p(j \mid i) = p_{\cdot j}$ for all $i$ and $j$.
\end{itemize}
\end{definitionbox}

\section{Measures of Central Tendency for Categorical Data}

\begin{definitionbox}{Mode and Ordinal Median}
\index{Mode}\index{Median}
\begin{itemize}
    \item \textbf{Mode:} The category with the highest frequency count ($f_{\text{mode}} = \max_i f_i$).
    \begin{itemize}
        \item The mode is the \emph{only} legitimate central tendency measure for \textbf{Nominal} variables.
        \item A dataset may be \emph{unimodal} (single peak), \emph{bimodal} (two categories tied or near-tied for highest count), or \emph{multimodal}.
    \end{itemize}
    \item \textbf{Median (Strictly for Ordinal Data):} The category containing the $50^{\text{th}}$ percentile ($crf \ge 0.50$) when categories are arranged in their strict natural order.
    $$\text{Median} = c_m \quad \text{where} \quad crf_{m-1} < 0.50 \le crf_m$$
    \textbf{Critical Rule:} The median is completely undefined for Nominal data because changing the arbitrary display order would change the median!
\end{itemize}
\end{definitionbox}

\section{Cognitive Traps in Categorical Analysis}

\begin{trapbox}{Trap 1: The Truncated Y-Axis Optical Illusion}
In bar charts, the vertical axis \textbf{must strictly begin at zero}.
\begin{itemize}
    \item If Product A has a satisfaction score of $92\%$ and Product B has $96\%$, a bar chart starting at $90\%$ will show Product B's bar as three times taller than Product A's bar ($6\%$ visible vs. $2\%$ visible).
    \item This creates a fraudulent visual impression of a $300\%$ advantage when the actual difference is only $4.3\%$.
\end{itemize}
\end{trapbox}

\begin{trapbox}{Trap 2: Simpson's Paradox (Reversal in Aggregated Data)}
\index{Simpson's Paradox}
\textbf{Simpson's Paradox} occurs when a trend, association, or advantage observed within multiple distinct subgroups \textbf{disappears or completely reverses} when the subgroups are pooled together into an aggregated total.
\begin{itemize}
    \item \textbf{Mathematical Cause:} Arises when an unobserved or ignored \emph{confounding variable} (a lurking variable) is strongly associated with both the grouping variable and the outcome variable, creating disproportionate subgroup weights.
    \item \textbf{Data Science Rule:} Never base strategic or clinical decisions purely on aggregated marginal rates without stratifying by key confounding variables!
\end{itemize}
\end{trapbox}

\section{Python Implementation: Contingency Tables and Simpson's Paradox}

\begin{lstlisting}[caption={Contingency Tables, Normalization, and Simpson's Paradox in Python}]
import pandas as pd
import numpy as np

# Demonstrating Simpson's Paradox: Treatment A vs B across Mild and Severe cases
data = {
    'Severity': ['Mild']*100 + ['Mild']*400 + ['Severe']*400 + ['Severe']*100,
    'Treatment': ['A']*100 + ['B']*400 + ['A']*400 + ['B']*100,
    'Recovered': [1]*90 + [0]*10 + [1]*340 + [0]*60 +  # Mild: A=90%, B=85%
                 [1]*240 + [0]*160 + [1]*50 + [0]*50   # Severe: A=60%, B=50%
}
df = pd.DataFrame(data)

# 1. Stratified Recovery Rates by Subgroup
stratified = df.groupby(['Severity', 'Treatment'])['Recovered'].mean().unstack()
print("--- Stratified Recovery Rates (Treatment A beats B in BOTH cohorts) ---")
print(stratified)

# 2. Aggregated Recovery Rates (Simpson's Paradox Reversal!)
aggregated = df.groupby('Treatment')['Recovered'].mean()
print("\n--- Aggregated Recovery Rates (B falsely appears superior!) ---")
print(aggregated)
\end{lstlisting}

\subsection*{Part A: Computational and Analytical Problems}

\begin{enumerate}
    \item \textbf{Frequency and Relative Frequency Calculations:} In a market survey of $n = 500$ consumers regarding their primary streaming platform:
    \begin{itemize}
        \item Netflix: 215 subscribers
        \item Amazon Prime: 145 subscribers
        \item Disney+: 90 subscribers
        \item Other platforms: Remaining respondents
    \end{itemize}
    Construct a complete frequency and relative frequency distribution table.

    \item \textbf{Pareto Analysis in Quality Engineering:} A cloud hosting company analyzes $n = 1{,}000$ server downtime incidents across five root causes: Network Outage (480), Database Deadlock (290), Software Bug (130), Hardware Fault (70), Human Operator Error (30).
    \begin{enumerate}
        \item Compute the percentage and cumulative percentage for each category.
        \item Determine which causes constitute the "vital few" responsible for at least $75\%$ of all incidents.
    \end{enumerate}

    \item \textbf{Contingency Table Analysis:} An e-commerce store tracks 400 customer visits categorized by Device Type and Purchase Outcome:
    \begin{center}
    \begin{tabular}{l|cc|c}
    & Purchased & Abandoned & Total \\
    \hline
    Mobile & 60 & 180 & 240 \\
    Desktop & 80 & 80 & 160 \\
    \hline
    Total & 140 & 260 & 400 \\
    \end{tabular}
    \end{center}
    \begin{enumerate}
        \item Compute the overall marginal conversion rate $P(\text{Purchased})$.
        \item Compute the conditional conversion rates $P(\text{Purchased} \mid \text{Mobile})$ and $P(\text{Purchased} \mid \text{Desktop})$.
        \item Are Device Type and Purchase Outcome independent?
    \end{enumerate}

    \item \textbf{Ordinal Median Determination:} A software usability test records feedback from $n = 80$ users on a 5-point Likert scale:
    \begin{center}
    \begin{tabular}{ccccc}
    Strongly Disagree (1) & Disagree (2) & Neutral (3) & Agree (4) & Strongly Agree (5) \\
    \hline
    6 & 14 & 25 & 27 & 8 \\
    \end{tabular}
    \end{center}
    Determine the Mode and the Median category. Show cumulative counts.

    \item \textbf{Graphing Ethics Analysis:} An executive dashboard displays a bar chart comparing quarterly sales: Q1 (\$10.2M) and Q2 (\$10.8M). The bar chart's vertical axis begins at \$10.0M rather than \$0.0M. Calculate the ratio of the physical heights of the two bars on the screen and compare it to the true percentage increase in sales.
\end{enumerate}

\subsection*{Part B: Conceptual and Diagnostic Multiple Choice Questions}

\begin{enumerate}[resume]
\item \textbf{Conceptual Diagnostic: Mechanism of Simpson's Paradox} \\
A medical center evaluates two surgical procedures, Procedure A and Procedure B. In clinical trials, Procedure A achieves a higher success rate than Procedure B on mild cases ($95\%$ vs $90\%$) and also on severe cases ($70\%$ vs $60\%$). However, when all patient records are aggregated across the entire hospital, Procedure B exhibits an overall success rate of $84\%$, while Procedure A exhibits an overall success rate of only $75\%$. What mathematical condition is responsible for this paradoxical reversal?
\begin{enumerate}
    \item A calculation error in summing percentages, because percentages cannot be added directly.
    \item Confounding caused by an unequal distribution of patient severity: Procedure A was assigned to a vastly higher proportion of severe cases with low baseline survival, while Procedure B was assigned predominantly to easy, mild cases.
    \item Violation of Stevens' scale of measurement, because medical outcomes are continuous ratio variables.
    \item Class imbalance in the total number of patients, which invalidates the law of total probability.
\end{enumerate}

\item \textbf{Conceptual Diagnostic: Central Tendency for Ordinal Categorical Distributions} \\
A human resources department assesses employee engagement using an ordinal scale: Unsatisfied ($U$), Neutral ($N$), Satisfied ($S$), and Highly Satisfied ($H$). Across 101 employees, the sorted responses are: 20 $U$, 25 $N$, 40 $S$, and 16 $H$. The HR director assigns numerical values $U=1, N=2, S=3, H=4$, calculates the arithmetic mean $\bar{x} = 2.51$, and reports: \textit{"Our average employee is slightly above Neutral."} Which of the following evaluations of the director's methodology is \textbf{most scientifically accurate}?
\begin{enumerate}
    \item The methodology is completely valid because any discrete ordinal scale can be treated as an interval scale once mapped to integers.
    \item The methodology is flawed because ordinal categories do not guarantee equidistant spacing; the mathematically rigorous measure of location is the median category, which is \textbf{Satisfied ($S$)}.
    \item The methodology is flawed because the mode is \textbf{Neutral ($N$)}, and the mode is the only permitted measure for any categorical data.
    \item The methodology is valid, but the median should be reported as $2.5$ rather than a category label.
\end{enumerate}
\end{enumerate}
