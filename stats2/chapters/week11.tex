\chapter{Hypothesis Testing: Size, Power, Z-Test \& p-Values}

Hypothesis testing provides the formal decision-theoretic machinery for separating genuine scientific discovery from random statistical noise. In quantitative data science and modern A/B testing, hypothesis testing governs product launch decisions, algorithmic uplift validation, and regulatory clinical trials. In this chapter, we master \textbf{Null vs. Alternative Hypotheses}, the decision matrix of \textbf{Type I ($\alpha$, Size)} and \textbf{Type II ($\beta$, Power $1-\beta$)} errors, the \index{One-Sample Z-Test}\textbf{One-Sample Z-Test}, the mathematical derivation of \textbf{$p$-values}, the proof that $p \sim \text{Uniform}(0, 1)$ under $H_0$, and the classic cognitive fallacies that mislead unwary analysts.

\section{Null and Alternative Hypotheses, Errors, and Power}

\begin{definitionbox}{Hypotheses and Decision Matrix}
\index{Hypotheses and Decision Matrix}
Statistical hypothesis testing tests competing claims regarding population parameter $\theta$:
\begin{itemize}
    \item \textbf{Null Hypothesis ($H_0$):} Baseline status-quo assumption of no effect, no difference, or no uplift (e.g. $H_0 \colon \mu = \mu_0$). Assumed true unless empirical evidence overwhelmingly contradicts it.
    \item \textbf{Alternative Hypothesis ($H_1$ or $H_a$):} The research claim of discovery or directional shift (e.g. $H_1 \colon \mu > \mu_0$ or $H_1 \colon \mu \neq \mu_0$).
\end{itemize}
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{c|cc}
\toprule
\textbf{Reality} & \textbf{Retain $H_0$ (Fail to Reject)} & \textbf{Reject $H_0$ (Declare Significance)} \\
\midrule
\textbf{$H_0$ is True} & Correct Decision (Specificity $1 - \alpha$) & \textbf{Type I Error ($\alpha$, False Positive)} \\
\textbf{$H_0$ is False} & \textbf{Type II Error ($\beta$, False Negative)} & Correct Decision (\textbf{Statistical Power $1 - \beta$}) \\
\bottomrule
\end{tabular}
\end{center}
\end{definitionbox}

\begin{formulabox}{Significance Level ($\alpha$), Size, and Statistical Power ($1 - \beta$)}
\begin{itemize}
    \item \textbf{Significance Level / Size ($\alpha$):} The maximum permissible probability of committing a Type I error:
    $$\alpha = \P(\text{Reject } H_0 \mid H_0 \text{ is true})$$
    Typically fixed upfront by the experimenter (conventionally $\alpha = 0.05$ or $0.01$).
    \item \textbf{Type II Error Probability ($\beta$):} The probability of failing to detect a real effect:
    $$\beta = \P(\text{Fail to Reject } H_0 \mid H_1 \text{ is true})$$
    \item \textbf{Statistical Power ($1 - \beta$):} The probability of successfully rejecting a false null:
    $$\text{Power} = 1 - \beta = \P(\text{Reject } H_0 \mid H_1 \text{ is true})$$
    Power increases with larger sample size $n$, larger true effect size $|\mu - \mu_0|$, and higher permitted $\alpha$.
\end{itemize}
\end{formulabox}

\section{The One-Sample Z-Test}

When the population variance $\sigma^2$ is known (or $n \ge 30$ via the CLT), the sampling distribution of $\bar{X}_n$ under $H_0 \colon \mu = \mu_0$ is Gaussian $\mathcal{N}\left(\mu_0, \frac{\sigma^2}{n}\right)$.

\begin{formulabox}{Z-Test Statistic and Decision Rules}
Standardize the sample mean into the test statistic:
$$Z_{\text{calc}} = \frac{\bar{X}_n - \mu_0}{\sigma / \sqrt{n}}$$
Decision rules at significance level $\alpha$:
\begin{itemize}
    \item \textbf{Two-Tailed ($H_1 \colon \mu \neq \mu_0$):} Reject $H_0$ if $|Z_{\text{calc}}| \ge z_{\alpha/2}$.
    \item \textbf{Right-Tailed ($H_1 \colon \mu > \mu_0$):} Reject $H_0$ if $Z_{\text{calc}} \ge z_{\alpha}$.
    \item \textbf{Left-Tailed ($H_1 \colon \mu < \mu_0$):} Reject $H_0$ if $Z_{\text{calc}} \le -z_{\alpha}$.
\end{itemize}
Standard critical values: For $\alpha = 0.05$, $z_{0.025} = 1.960$ (two-tailed) and $z_{0.05} = 1.645$ (one-tailed).
\end{formulabox}

\section{The $p$-Value and its Mathematical Distribution}

\begin{definitionbox}{Mathematical Definition of the $p$-Value}
The \textbf{$p$-value} is the probability, calculated under the assumption that the null hypothesis $H_0$ is strictly true, of observing a test statistic at least as extreme as the value $z_{\text{calc}}$ actually computed from the sample:
\begin{itemize}
    \item \textbf{Right-Tailed Test:} $p\text{-value} = \P(Z \ge z_{\text{calc}} \mid H_0) = 1 - \Phi(z_{\text{calc}})$.
    \item \textbf{Left-Tailed Test:} $p\text{-value} = \P(Z \le z_{\text{calc}} \mid H_0) = \Phi(z_{\text{calc}})$.
    \item \textbf{Two-Tailed Test:} $p\text{-value} = \P(|Z| \ge |z_{\text{calc}}| \mid H_0) = 2 \left[ 1 - \Phi(|z_{\text{calc}}|) \right]$.
\end{itemize}
\textbf{Universal Decision Rule:} Reject $H_0$ if and only if $p\text{-value} \le \alpha$.
\end{definitionbox}

\begin{theorembox}{The Distribution of the $p$-Value Under $H_0$ is $\text{Uniform}(0, 1)$}
\index{Distribution of p-value under H0}
Let $T$ be a continuous test statistic with continuous strictly increasing cumulative distribution function $F_0(t)$ under $H_0$. 
Then, \textbf{when $H_0$ is true, the $p$-value random variable $P$ follows a continuous $\text{Uniform}(0, 1)$ distribution}:
$$P \mid H_0 \sim \text{Uniform}(0, 1)$$
\textbf{Proof:}
For a right-tailed test, $P = 1 - F_0(T)$. By probability integral transform, for any $u \in [0, 1]$:
$$\P(P \le u \mid H_0) = \P(1 - F_0(T) \le u) = \P(F_0(T) \ge 1 - u) = 1 - \P(F_0(T) \le 1 - u)$$
Because $F_0(T) \sim \text{Uniform}(0, 1)$, $\P(F_0(T) \le 1 - u) = 1 - u$:
$$\P(P \le u \mid H_0) = 1 - (1 - u) = u$$
Because the CDF of $P$ is $F_P(u) = u$ on $[0, 1]$, $P \sim \text{Uniform}(0, 1)$. 
This guarantees that $\P(P \le \alpha \mid H_0) = \alpha$, validating the exact nominal Type I error rate.
\end{theorembox}

\begin{lstlisting}[language=Python, caption=Data Science: Verifying that p-values are Uniform under H0]
import numpy as np
import scipy.stats as stats

# Simulating 50,000 independent experiments where H0 is TRUE (true mu = 0.0)
np.random.seed(42)
n_simulations = 50_000
samples = np.random.normal(loc=0.0, scale=1.0, size=(n_simulations, 30))

# Compute two-tailed one-sample t-test p-value against popmean = 0.0
_, p_values = stats.ttest_1samp(samples, popmean=0.0, axis=1)

# Under H0, exactly 5% of p-values fall below alpha = 0.05!
false_positive_rate = np.mean(p_values <= 0.05)
print(f"Empirical Type I Error Rate: {false_positive_rate:.4f} (Nominal: 0.0500)")

# Check Kolmogorov-Smirnov test against Uniform(0, 1)
ks_stat, ks_pval = stats.kstest(p_values, 'uniform')
print(f"KS test p-value for Uniformity: {ks_pval:.4f} (Uniform confirmed!)")
\end{lstlisting}

\section{Exam Gotchas \& Cognitive Traps}

\begin{trapbox}{The Three Universal $p$-Value Fallacies}
\textbf{Trap 1: The Transposition Fallacy (Inverse Probability):}
Believing that $p = 0.03$ means "there is a 3\% probability that $H_0$ is true."
\begin{itemize}
    \item \textbf{Correction:} The $p$-value is $\P(\text{Data} \mid H_0)$, \textbf{not} $\P(H_0 \mid \text{Data})$! Conflating them is the Base-Rate Fallacy. In Frequentist statistics, hypotheses do not have probabilities.
\end{itemize}

\textbf{Trap 2: The Alternative Probability Fallacy ($1 - p$):}
Believing that $p = 0.03$ means "there is a $97\%$ probability that the alternative hypothesis is true."
\begin{itemize}
    \item \textbf{Correction:} $1 - p$ simply measures the probability of seeing less extreme data under $H_0$. It has zero mathematical connection to the truth probability of $H_1$.
\end{itemize}

\textbf{Trap 3: The "Acceptance" Fallacy (Absence of Evidence):}
Claiming that $p = 0.28 > 0.05$ "proves that the null hypothesis is true" or that "treatment has zero effect."
\begin{itemize}
    \item \textbf{Correction:} Absence of evidence is not evidence of absence! A high $p$-value frequently occurs when a real, massive effect is present but the experiment had insufficient sample size (low power). We strictly state: \textbf{"Fail to reject $H_0$"}, never "Accept $H_0$".
\end{itemize}
\end{trapbox}

\begin{trapbox}{Statistical Significance ($p < 0.05$) vs. Practical Effect Size}
\textbf{The Trap:} Confusing a tiny $p$-value with a large, meaningful practical impact.

\textbf{The Reality:} As sample size $n \to \infty$, the standard error $\sigma/\sqrt{n} \to 0$. Consequently:
$$Z_{\text{calc}} = \frac{\bar{X}_n - \mu_0}{\sigma / \sqrt{n}} \xrightarrow{n \to \infty} \pm \infty \implies p\text{-value} \xrightarrow{n \to \infty} 0$$
For huge datasets ($n = 10,000,000$), a completely meaningless uplift (e.g. conversion improving from $10.00\%$ to $10.001\%$) will produce $p < 0.0000001$. Conversely, in clinical studies with $n = 15$, a dramatic 50\% reduction in mortality might yield $p = 0.09$ due to large sampling variance. \textbf{Always report Effect Size (Cohen's $d$) alongside $p$-values!}
\end{trapbox}

\subsection*{Part B: Conceptual Diagnostic Traps (Exam Gotchas)}

\begin{enumerate}[resume]
\item \textbf{Diagnostic Trap (The Distribution of $p$ Under a True Null):}
A medical researcher tests 10,000 completely inert sugar pills against a placebo at $\alpha = 0.05$. In reality, $H_0$ is strictly true for every single pill. What distribution will the resulting 10,000 $p$-values follow, and approximately how many false discoveries will be made?
\begin{enumerate}
    \item The $p$-values will cluster near 1.0, producing zero false discoveries.
    \item The $p$-values will follow a Standard Normal distribution centered at 0.
    \item The $p$-values will follow a continuous $\text{Uniform}(0, 1)$ distribution, resulting in approximately $10,000 \times 0.05 = 500$ false positive discoveries.
    \item The $p$-values will follow an Exponential distribution whose rate depends on sample size $n$.
\end{enumerate}

\item \textbf{Diagnostic Trap (Statistical Significance vs. Practical Meaning):}
An analyst evaluates an algorithm on a massive dataset of $n = 20,000,000$ user clicks. The test statistic yields $p = 0.00002$, and the measured increase in click-through rate is from $5.000\%$ to $5.002\%$. How should a senior data scientist evaluate this result?
\begin{enumerate}
    \item The result is statistically significant and proves that the algorithm has an enormous commercial impact.
    \item Because $p < 0.05$, the probability that the uplift is zero is exactly $0.002\%$.
    \item The result is statistically significant purely due to immense sample size $n$, but the practical effect size ($0.002\%$) is likely negligible and economically irrelevant.
    \item The test is invalid because $p$-values cannot be calculated when $n > 100,000$.
\end{enumerate}
\end{enumerate}
