\chapter{Functions of Continuous RVs, Expectations \& Mixed Distributions}

When we apply a mathematical mapping $Y = g(X)$ to a continuous feature $X$---such as logarithmic feature scaling, squaring signals, or transforming uniform pseudo-random numbers into arbitrary target distributions---the probability density is stretched, compressed, and warped by the derivative of the transformation. In this chapter, we master the \textbf{CDF Transformation Method}, the \textbf{Change-of-Variables (Jacobian) Formula}, \textbf{Continuous LOTUS Expectations}, and \textbf{Joint Discrete-Continuous (Mixed) Distributions} that form the mathematical bedrock of generative mixture models.

\section{Transformations of Continuous Random Variables}

To find the probability density function $f_Y(y)$ of a transformed variable $Y = g(X)$ given the original PDF $f_X(x)$, there are two primary methods:

\begin{formulabox}{Method 1: The Universal CDF Method}
The CDF method works for \textbf{any} transformation (monotonic or non-monotonic):
\begin{enumerate}
    \item Express the cumulative probability of $Y$ in terms of events on $X$:
    $$F_Y(y) = \P(Y \le y) = \P(g(X) \le y) = \int_{\{x \colon g(x) \le y\}} f_X(x) \, dx$$
    \item Differentiate $F_Y(y)$ with respect to $y$ by the Fundamental Theorem of Calculus:
    $$f_Y(y) = \frac{d}{dy} F_Y(y)$$
\end{enumerate}
\end{formulabox}

\begin{theorembox}{Method 2: Change of Variables Formula (Strictly Monotonic Functions)}
If $g(x)$ is strictly monotonic (strictly increasing or strictly decreasing) and differentiable over the support of $X$, then the mapping is one-to-one with unique inverse $x = g^{-1}(y)$. The density of $Y$ is:
$$f_Y(y) = f_X(g^{-1}(y)) \cdot \left| \frac{d}{dy} g^{-1}(y) \right| = \frac{f_X(x)}{\left| g'(x) \right|} \Bigg|_{x = g^{-1}(y)}$$
The absolute value term $\left| \frac{dx}{dy} \right|$ is the 1D \index{Jacobian}\textbf{Jacobian}, ensuring that probability density remains strictly non-negative even when the transformation function is decreasing.
\end{theorembox}

\begin{examplebox}{Linear Transformation $Y = aX + b$ ($a \neq 0$)}
Let $X$ have density $f_X(x)$. Here, $y = ax + b \implies x = \frac{y - b}{a}$, and $\frac{dx}{dy} = \frac{1}{a}$.
$$f_Y(y) = f_X\left(\frac{y - b}{a}\right) \cdot \left| \frac{1}{a} \right| = \frac{1}{|a|} f_X\left(\frac{y - b}{a}\right)$$
In particular, if $X \sim \mathcal{N}(\mu, \sigma^2)$, then $Y = aX + b \sim \mathcal{N}(a\mu + b, a^2 \sigma^2)$.
\end{examplebox}

\section{Expectation and Variance of Continuous RVs}

\begin{definitionbox}{Expectation $\E[g(X)]$ via Continuous LOTUS}
For a continuous random variable $X$ with PDF $f_X(x)$, the expected value of any real-valued function $g(X)$ is computed directly without first deriving the distribution of $g(X)$:
$$\E[g(X)] = \int_{-\infty}^{\infty} g(x) f_X(x) \, dx$$
provided the integral converges absolutely: $\int_{-\infty}^{\infty} |g(x)| f_X(x) \, dx < \infty$.
\begin{itemize}
    \item \textbf{Mean (First Moment):} $\mu = \E[X] = \int_{-\infty}^{\infty} x f_X(x) \, dx$.
    \item \textbf{Second Moment:} $\E[X^2] = \int_{-\infty}^{\infty} x^2 f_X(x) \, dx$.
    \item \textbf{Variance:} $\sigma^2 = \Var(X) = \E[(X - \mu)^2] = \E[X^2] - (\E[X])^2$.
\end{itemize}
\end{definitionbox}

\section{Joint Distributions of Discrete and Continuous RVs}

\begin{intuitionbox}{Gaussian Mixtures as Latent Cluster Generators}
In customer segmentation or speech recognition, data points do not originate from a single Gaussian bell curve. Instead, each observation first selects an underlying latent cluster $N \in \{1, 2, \dots, K\}$ with categorical probability $\pi_k$. Conditional on that cluster, the continuous feature is drawn from a cluster-specific normal distribution $\mathcal{N}(\mu_k, \sigma_k^2)$. The overall observed data density is a weighted blend (mixture) of these individual bells!
\end{intuitionbox}

\begin{lstlisting}[language=Python, caption=Data Science: Simulating and Visualizing a Gaussian Mixture Model]
import numpy as np

# Simulating a two-component Gaussian Mixture Model (GMM)
n_samples = 10_000
weights = [0.70, 0.30] # 70% Cluster 0 (Normal transactions), 30% Cluster 1 (Outliers)

# 1. Sample discrete component labels
cluster_labels = np.random.choice([0, 1], size=n_samples, p=weights)

# 2. Sample continuous amounts conditionally
amounts = np.where(
    cluster_labels == 0,
    np.random.normal(loc=50.0, scale=10.0, size=n_samples),   # Cluster 0: N(50, 100)
    np.random.normal(loc=250.0, scale=40.0, size=n_samples)   # Cluster 1: N(250, 1600)
)

print(f"Overall Mixed Mean: ${amounts.mean():.2f}")
print(f"Overall Mixed StdDev: ${amounts.std():.2f}")
\end{lstlisting}

\begin{definitionbox}{Joint Discrete-Continuous Modeling}
Let $N$ be a discrete random variable taking values in $\{1, \dots, K\}$ with PMF $p_N(n)$, and let $X$ be a continuous random variable.
\begin{itemize}
    \item \textbf{Conditional PDF of $X$ given $N=n$:} $f_{X|N}(x \mid n)$.
    \item \textbf{Marginal PDF of $X$ (Continuous Law of Total Probability):}
    $$f_X(x) = \sum_{n} f_{X|N}(x \mid n) \cdot p_N(n)$$
    \item \textbf{Posterior Probability of Discrete Class given Continuous Evidence (Bayes' Rule):}
    $$\P(N = n \mid X = x) = \frac{f_{X|N}(x \mid n) \cdot p_N(n)}{f_X(x)} = \frac{f_{X|N}(x \mid n) \cdot p_N(n)}{\sum_{k} f_{X|N}(x \mid k) \cdot p_N(k)}$$
\end{itemize}
\end{definitionbox}

\section{Exam Gotchas \& Cognitive Traps}

\begin{trapbox}{Forgetting the Absolute Value of the Derivative}
\textbf{The Trap:} Writing $f_Y(y) = f_X(g^{-1}(y)) \frac{dx}{dy}$ when the mapping $g(x)$ is decreasing.

\textbf{The Reality:} If $g(x)$ is decreasing, $\frac{dx}{dy}$ is strictly negative. Dropping the absolute value produces a \textbf{negative probability density}, which is a catastrophic exam blunder!
For instance, consider $X \sim \text{Uniform}(0, 1)$ and $Y = -X$. Here $x = -y$, so $\frac{dx}{dy} = -1$.
$$f_Y(y) = f_X(-y) \cdot |-1| = 1 \cdot 1 = 1 \quad \text{for } y \in [-1, 0]$$
The density is positive 1, not $-1$!
\end{trapbox}

\begin{trapbox}{Non-Monotonic Transformations Require Branch Summation}
\textbf{The Trap:} Directly using $f_Y(y) = f_X(\sqrt{y})\frac{1}{2\sqrt{y}}$ for $Y = X^2$ when the support of $X$ contains both positive and negative values.

\textbf{The Reality:} The function $y = x^2$ is not one-to-one; both $x = +\sqrt{y}$ and $x = -\sqrt{y}$ map to the same $y$. One \textbf{must sum over all inverse branches}:
$$F_Y(y) = \P(X^2 \le y) = \P(-\sqrt{y} \le X \le \sqrt{y}) = F_X(\sqrt{y}) - F_X(-\sqrt{y})$$
Differentiating with respect to $y$ gives:
$$f_Y(y) = \frac{1}{2\sqrt{y}} \left[ f_X(\sqrt{y}) + f_X(-\sqrt{y}) \right] \quad \text{for } y > 0$$
Only when $X$ is strictly non-negative does the second branch vanish.
\end{trapbox}

\begin{trapbox}{Mixture Distribution vs. Sum of Random Variables}
\textbf{The Trap:} Confusing a Gaussian Mixture Model $X \sim 0.5 \mathcal{N}(0, 1) + 0.5 \mathcal{N}(10, 1)$ with the sum of two independent Gaussians $S = 0.5 X_1 + 0.5 X_2$.

\textbf{The Reality:}
\begin{itemize}
    \item \textbf{A Mixture} is an \textbf{OR} choice: you pick a coin, flip it, and draw from either the first bell OR the second bell. The resulting distribution is \textbf{bimodal} with two separate humps at 0 and 10!
    \item \textbf{A Sum} is an \textbf{AND} convolution: you sample from both curves and add their values. The sum of two Gaussians is \textbf{always unimodal Gaussian} centered at the average mean: $S \sim \mathcal{N}(5, 0.5)$.
\end{itemize}
\end{trapbox}

\subsection*{Part A: Analytical \& Computational Problems}

\begin{enumerate}
\item \textbf{CDF Transformation Method:} Let $X \sim \text{Uniform}(0, 1)$ with $f_X(x) = 1$ for $x \in (0, 1)$. Find the PDF $f_Y(y)$ of $Y = -\ln(X)$ for $y > 0$. Identify the distribution of $Y$.

\item \textbf{Linear Transformation of Normal RV:} Let $X \sim \mathcal{N}(0, 1)$ be a standard normal random variable. Define $Y = 3X + 5$. Write the PDF of $Y$ and state its mean and variance.

\item \textbf{Continuous LOTUS Expectation:} Let $X$ have PDF $f_X(x) = 2x$ for $0 \le x \le 1$. Compute $\E[X^2]$.

\item \textbf{Variance of Uniform RV:} Compute $\Var(X)$ for $X \sim \text{Uniform}(0, 4)$ using the integral definition $\E[X^2] - (\E[X])^2$.

\item \textbf{Data Science Application (Gaussian Mixture Model Marginal PDF):} A binary classifier models customer transaction amount $X$ conditional on fraud status $N \in \{0, 1\}$.
\begin{itemize}
    \item Prior fraud probability: $p_N(1) = 0.05$, $p_N(0) = 0.95$.
    \item Non-fraud distribution: $X \mid N=0 \sim \mathcal{N}(\mu_0 = 50, \sigma_0^2 = 100)$.
    \item Fraud distribution: $X \mid N=1 \sim \mathcal{N}(\mu_1 = 300, \sigma_1^2 = 400)$.
\end{itemize}
    Write the overall marginal PDF expression $f_X(x)$ for transaction amounts across all customers.
\end{enumerate}

\subsection*{Part B: Conceptual Diagnostic Traps (Exam Gotchas)}

\begin{enumerate}[resume]
\item \textbf{Diagnostic Trap (Non-Monotonic Transformation $Y = X^2$):}
Let $X \sim \mathcal{N}(0, 1)$ be a standard normal random variable with symmetric density $f_X(x) = \frac{1}{\sqrt{2\pi}} e^{-x^2/2}$. Define $Y = X^2$. For $y > 0$, which of the following expressions correctly gives the PDF $f_Y(y)$?
\begin{enumerate}
    \item $f_Y(y) = \frac{1}{\sqrt{2\pi}} e^{-y/2} \cdot \frac{1}{2\sqrt{y}}$
    \item $f_Y(y) = \frac{1}{\sqrt{2\pi y}} e^{-y/2}$
    \item $f_Y(y) = \frac{1}{2\sqrt{y}} \left[ f_X(\sqrt{y}) - f_X(-\sqrt{y}) \right] = 0$
    \item $f_Y(y) = 2 \cdot f_X(y) = \frac{2}{\sqrt{2\pi}} e^{-y^2/2}$
\end{enumerate}

\item \textbf{Diagnostic Trap (Gaussian Mixture vs. Sum of Independent Gaussians):}
Consider two independent normal variables $X_1 \sim \mathcal{N}(-10, 1)$ and $X_2 \sim \mathcal{N}(10, 1)$. Model A defines $S = 0.5 X_1 + 0.5 X_2$. Model B defines a mixture $M$ where with probability $0.5$ an observation is drawn from $\mathcal{N}(-10, 1)$ and with probability $0.5$ it is drawn from $\mathcal{N}(10, 1)$. Which of the following statements is strictly correct?
\begin{enumerate}
    \item Model A and Model B have identical probability density functions.
    \item Model A has a bimodal distribution with peaks at $-10$ and $10$, whereas Model B is unimodal.
    \item Model A has a unimodal Gaussian distribution centered at $0$, whereas Model B has a bimodal distribution with two distinct peaks at $-10$ and $10$.
    \item The variance of Model A is strictly greater than the variance of Model B.
\end{enumerate}
\end{enumerate}
