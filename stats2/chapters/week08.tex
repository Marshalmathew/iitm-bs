\section{Moment Generating Functions (MGF)}

The \index{Moment Generating Function}\textbf{Moment Generating Function} $M_X(t)$ provides an operational transform domain (analogous to the Laplace and Fourier transforms in engineering) that uniquely characterizes a probability distribution.

\begin{definitionbox}{Moment Generating Function $M_X(t)$}
For a random variable $X$, its \textbf{Moment Generating Function (MGF)} is defined for real $t$ by:
$$M_X(t) = \E[e^{tX}] = \begin{cases} \sum_x e^{tx} p_X(x) & \text{(Discrete)} \\ \int_{-\infty}^{\infty} e^{tx} f_X(x) \, dx & \text{(Continuous)} \end{cases}$$
The MGF is mathematically valid \textbf{if and only if} the expectation exists and is finite in an open interval $(-h, h)$ containing $t = 0$ for some $h > 0$.
\end{definitionbox}

\begin{theorembox}{Properties of Moment Generating Functions}
\begin{enumerate}
    \item \textbf{Moment Extraction:} Expanding $e^{tX} = \sum_{k=0}^\infty \frac{(tX)^k}{k!}$ shows that the $k$-th derivative evaluated at $t = 0$ yields the $k$-th raw moment:
    $$\left. \frac{d^k M_X(t)}{dt^k} \right|_{t=0} = \E[X^k] \implies M_X'(0) = \E[X], \quad M_X''(0) = \E[X^2]$$
    \item \textbf{Linear Transformation Scaling:} $M_{aX + b}(t) = \E[e^{t(aX + b)}] = e^{bt} M_X(at)$.
    \item \textbf{Factorization for Independent Sums:} If $X_1, \dots, X_n$ are mutually independent, then:
    $$M_{\sum X_i}(t) = \E\left[ \exp\left(t \sum X_i\right) \right] = \prod_{i=1}^n \E[e^{tX_i}] = \prod_{i=1}^n M_{X_i}(t)$$
    \item \textbf{Uniqueness Theorem:} If two random variables share the identical MGF on an open interval $(-h, h)$ around zero, they possess the \textbf{exact same probability distribution}.
\end{enumerate}
\end{theorembox}

\begin{formulabox}{Canonical MGF Table}
\begin{itemize}
    \item \textbf{Bernoulli($p$):} $M(t) = (1 - p) + p e^t$.
    \item \textbf{Binomial($n, p$):} $M(t) = \left[(1 - p) + p e^t\right]^n$.
    \item \textbf{Poisson($\lambda$):} $M(t) = \exp(\lambda(e^t - 1))$.
    \item \textbf{Exponential($\lambda$):} $M(t) = \frac{\lambda}{\lambda - t}$ for $t < \lambda$.
    \item \textbf{Gaussian $\mathcal{N}(\mu, \sigma^2)$:} $M(t) = \exp\left(\mu t + \frac{1}{2}\sigma^2 t^2\right)$.
\end{itemize}
\end{formulabox}

\section{The Central Limit Theorem (CLT)}

\begin{intuitionbox}{The Universality of the Bell Curve}
Take \textbf{any} distribution whatsoever---an exponential decay, a discrete uniform roll, or a multi-modal mixture. If you draw samples of size $n$, compute their average $\bar{X}_n$, and repeat this process thousands of times, the histogram of those averages will \textbf{inevitably form a symmetrical Gaussian bell curve}. The idiosyncratic skewness and spikes of the original distribution are smoothed away under summation!
\end{intuitionbox}

\begin{lstlisting}[language=Python, caption=Data Science: Demonstrating the Central Limit Theorem]
import numpy as np
import scipy.stats as stats

# Drawing from a heavily skewed Exponential distribution (lambda = 0.5, mean = 2.0)
np.random.seed(42)
n_experiments = 10_000
sample_size = 64

# Generate 10,000 samples of size 64
samples = np.random.exponential(scale=2.0, size=(n_experiments, sample_size))

# Compute the sample mean of each experiment
sample_means = np.mean(samples, axis=1)

# Theoretical SE = sigma / sqrt(n) = 2.0 / sqrt(64) = 0.25
print(f"Empirical Mean of Means: {sample_means.mean():.4f} (True: 2.0000)")
print(f"Empirical Std Error:     {sample_means.std():.4f}  (Theoretical: 0.2500)")

# Check Gaussian normality via skewness of sample means (should be ~0)
print(f"Sample Means Skewness:   {stats.skew(sample_means):.4f} (Nearly perfect 0!)")
\end{lstlisting}

\begin{theorembox}{The Central Limit Theorem (Lindeberg-Lévy CLT)}
Let $X_1, X_2, \dots, X_n$ be an i.i.d. sequence of random variables with finite mean $\mu$ and finite variance $\sigma^2 > 0$.
As $n \to \infty$, the standardized sample mean $Z_n$ \index{Convergence in Distribution}\textbf{converges in distribution} to the Standard Normal Distribution $\mathcal{N}(0, 1)$:
$$Z_n = \frac{\bar{X}_n - \mu}{\sigma / \sqrt{n}} = \frac{\sum_{i=1}^n X_i - n\mu}{\sigma \sqrt{n}} \xrightarrow{d} \mathcal{N}(0, 1)$$
That is, for every real $z \in \mathbb{R}$:
$$\lim_{n \to \infty} \P\left( \frac{\bar{X}_n - \mu}{\sigma / \sqrt{n}} \le z \right) = \Phi(z) = \int_{-\infty}^z \frac{1}{\sqrt{2\pi}} e^{-u^2/2} \, du$$
\end{theorembox}

\section{Exam Gotchas \& Cognitive Traps}

\begin{trapbox}{The Raw Data Does NOT Become Normal!}
\textbf{The Trap:} Stating that "by the Central Limit Theorem, if we collect $n = 100,000$ customer purchase amounts, the purchase amounts become normally distributed."

\textbf{The Reality:} This is the single most common conceptual trap in statistical inference!
\begin{itemize}
    \item The distribution of the \textbf{raw data} $X_i$ is determined by physics/human behavior (e.g. power-law or exponential). Collecting more data simply reveals the true non-normal shape with sharper resolution.
    \item The Central Limit Theorem governs \textbf{only the sample mean $\bar{X}_n$} (or sample sum $S_n$). It is the \emph{estimator}, not the underlying physical data, whose sampling distribution converges to a Gaussian bell curve.
\end{itemize}
\end{trapbox}

\begin{trapbox}{The Continuity Correction in Discrete CLT Approximations}
\textbf{The Trap:} Approximating a discrete integer sum (such as Binomial $S \sim \text{Binomial}(n, p)$) by evaluating $Z$ directly at integer $k$: $\P(S \ge 10) \approx \P\left(Z \ge \frac{10 - np}{\sqrt{np(1-p)}}\right)$.

\textbf{The Reality:} A discrete probability mass at integer $k$ occupies the continuous interval $[k - 0.5, k + 0.5]$.
Omitting the \index{Continuity Correction}\textbf{continuity correction} creates systematic approximation error:
\begin{itemize}
    \item $\P(S \ge k) \implies$ Include bar $k \implies$ Integrate from $k - 0.5$: $\P(S \ge k) \approx 1 - \Phi\left(\frac{k - 0.5 - \mu}{\sigma}\right)$.
    \item $\P(S \le k) \implies$ Include bar $k \implies$ Integrate up to $k + 0.5$: $\P(S \le k) \approx \Phi\left(\frac{k + 0.5 - \mu}{\sigma}\right)$.
    \item $\P(S = k) \implies$ Area over single bar: $\P\left(\frac{k - 0.5 - \mu}{\sigma} \le Z \le \frac{k + 0.5 - \mu}{\sigma}\right)$.
\end{itemize}
\end{trapbox}

\subsection*{Part A: Analytical \& Computational Problems}

\begin{enumerate}
\item \textbf{MGF Derivative Moment Extraction:} The MGF of a random variable $X$ is $M_X(t) = \frac{1}{1 - 2t}$ for $t < 0.5$.
    \begin{enumerate}
        \item Compute $\E[X] = M_X'(0)$.
        \item Compute $\E[X^2] = M_X''(0)$ and variance $\Var(X)$.
    \end{enumerate}

\item \textbf{MGF of Independent Sums:} Let $X \sim \text{Poisson}(\lambda_1 = 3)$ and $Y \sim \text{Poisson}(\lambda_2 = 5)$ be independent. Use MGF product $M_{X+Y}(t) = M_X(t) M_Y(t)$ to prove that $X + Y \sim \text{Poisson}(8)$.

\item \textbf{CLT Probability Approximation for Sample Means:} A battery's lifetime in hours has mean $\mu = 40$ and standard deviation $\sigma = 8$. A company tests an i.i.d. sample of $n = 64$ batteries. Given $\Phi(1.5) = 0.9332$, use the CLT to approximate the probability that the sample mean lifetime $\bar{X}_{64}$ exceeds $41.5$ hours.

\item \textbf{CLT Approximation for Binomial Counts:} A coin with probability of heads $p = 0.5$ is tossed $n = 100$ times. Use the Normal approximation (CLT) to find $\P(45 \le S \le 55)$ where $S = \sum_{i=1}^{100} X_i$. Given $\Phi(1.1) = 0.8643$.

\item \textbf{Data Science Application (A/B Test Conversion Sample Size):} An e-commerce platform compares conversion rates ($p = 0.10$, $\sigma = \sqrt{p(1-p)} = 0.30$). Using the CLT, how many visitor sessions $n$ are required so that the standard error of the sample conversion rate $\text{SE}(\bar{X}_n) = \frac{\sigma}{\sqrt{n}}$ is at most $0.005$?
\end{enumerate}

\subsection*{Part B: Conceptual Diagnostic Traps (Exam Gotchas)}

\begin{enumerate}[resume]
\item \textbf{Diagnostic Trap (The Target of the Central Limit Theorem):}
A data science intern analyzes a web server's response times, which are heavily skewed following an Exponential distribution with rate $\lambda = 0.2$. The intern collects $1,000,000$ response times and claims: "Because the sample size is one million, the histogram of these one million response times will follow a Normal bell curve due to the Central Limit Theorem." Is the intern's claim correct?
\begin{enumerate}
    \item Yes, any dataset with $n > 30$ is guaranteed to be normally distributed by the CLT.
    \item No, the raw data will remain exponentially distributed; the CLT states only that the \emph{sample mean} of repeated batches would follow a normal distribution.
    \item Yes, provided the observations are standardized by subtracting the mean and dividing by the standard deviation.
    \item No, the CLT only applies to discrete Bernoulli trials.
\end{enumerate}

\item \textbf{Diagnostic Trap (Continuity Correction Mechanics):}
Let $S \sim \text{Binomial}(n = 100, p = 0.5)$ be the number of successes in 100 trials ($\mu = 50, \sigma = 5$). To approximate $\P(S \ge 60)$ using the continuous standard normal CDF $\Phi(z)$, which continuity-corrected $Z$-score should be evaluated?
\begin{enumerate}
    \item $Z = \frac{60 - 50}{5} = 2.0$
    \item $Z = \frac{60.5 - 50}{5} = 2.1$
    \item $Z = \frac{59.5 - 50}{5} = 1.9$
    \item $Z = \frac{60 - 50}{50} = 0.2$
\end{enumerate}
\end{enumerate}
