\section{Joint Probability Mass Function (Joint PMF)}

When two discrete random variables $X$ and $Y$ are defined on the same sample space, their combined behavior is specified by their \index{Joint PMF}\textbf{Joint PMF}.

\begin{definitionbox}{Joint PMF $p_{X,Y}(x, y)$}
The \textbf{Joint Probability Mass Function} of discrete random variables $X$ and $Y$ is defined as:
$$p_{X,Y}(x, y) = \P(X = x \text{ and } Y = y) = \P(X = x, Y = y)$$
A valid joint PMF must satisfy two fundamental conditions:
\begin{enumerate}
    \item Non-negativity: $p_{X,Y}(x, y) \ge 0$ for all $(x, y)$.
    \item Normalization: $\sum_{x} \sum_{y} p_{X,Y}(x, y) = 1$.
\end{enumerate}
\end{definitionbox}

\begin{examplebox}{Two Dice Roll Joint PMF}
Let $X$ be the number shown on die 1 and $Y$ be the number on die 2. Since the dice are independent and fair:
$$p_{X,Y}(x, y) = \frac{1}{36} \quad \text{for } x \in \{1,\dots,6\}, y \in \{1,\dots,6\}$$
The sum of all 36 probabilities is $36 \times \frac{1}{36} = 1$.
\end{examplebox}

\section{Marginal PMFs}

\begin{intuitionbox}{Marginalization is "Squashing" Data}
Imagine a 3D bar chart where $X$ is on the floor pointing left, $Y$ is on the floor pointing forward, and the bars grow upwards representing probability. "Marginalizing out Y" is like shining a flashlight from the front so that all the bars squash against the back wall. You lose the specific $Y$ information, but you perfectly preserve the total probability of each $X$ slice!
\end{intuitionbox}

\begin{lstlisting}[language=Python, caption=Data Science: Marginal distributions via pandas crosstab]
import pandas as pd

# Creating a joint frequency table
df = pd.DataFrame({"Device": ["Mobile", "Mobile", "Desktop", "Desktop"],
                   "Click":  [0, 1, 0, 1]})

# 'margins=True' computes the Marginal Distributions!
joint_pmf = pd.crosstab(df['Device'], df['Click'], margins=True, normalize=True)
print(joint_pmf)
\end{lstlisting}

Given the joint PMF $p_{X,Y}(x, y)$, we can recover the individual distribution of $X$ alone (or $Y$ alone) by summing over all possible values of the other variable. This process is called \index{Marginalization}\textbf{Marginalization}.

\begin{formulabox}{Marginal PMF Formulas}
\begin{itemize}
    \item \textbf{Marginal PMF of $X$:}
\end{itemize}
  $$p_X(x) = \P(X = x) = \sum_{y} p_{X,Y}(x, y)$$
\begin{itemize}
    \item \textbf{Marginal PMF of $Y$:}
\end{itemize}
  $$p_Y(y) = \P(Y = y) = \sum_{x} p_{X,Y}(x, y)$$
\end{formulabox}

\begin{examplebox}{Step-by-Step Marginalization}
Consider a joint PMF table for $X$ (rows: 0, 1) and $Y$ (cols: 1, 2):
$p_{X,Y}(0,1) = 0.1, \quad p_{X,Y}(0,2) = 0.3$
$p_{X,Y}(1,1) = 0.4, \quad p_{X,Y}(1,2) = 0.2$
To find the marginal probability $p_X(0) = \P(X=0)$, we sum across the row for $X=0$:
$$p_X(0) = p_{X,Y}(0,1) + p_{X,Y}(0,2) = 0.1 + 0.3 = 0.4$$
Similarly, $p_X(1) = 0.4 + 0.2 = 0.6$. Note that $0.4 + 0.6 = 1$.
\end{examplebox}

\textbf{Data Science Relevance:} In tabular data (e.g. customer churn vs age bracket), computing column totals or row totals in a contingency table corresponds directly to finding marginal PMFs!

\section{Conditional PMFs}

How does knowing the value of $Y$ update our belief about the distribution of $X$? This is captured by the \index{Conditional PMF}\textbf{Conditional PMF}.

\begin{definitionbox}{Conditional PMF $p_{X|Y}(x \mid y)$}
For any $y$ such that $p_Y(y) > 0$, the \index{Conditional PMF}\textbf{Conditional PMF} of $X$ given $Y = y$ is:
$$p_{X|Y}(x \mid y) = \P(X = x \mid Y = y) = \frac{p_{X,Y}(x, y)}{p_Y(y)}$$
For a fixed $y$, $\sum_{x} p_{X|Y}(x \mid y) = 1$, making $p_{X|Y}(\cdot \mid y)$ a valid probability distribution over $x$.
\end{definitionbox}

\begin{warningbox}{Conditioning vs Marginalizing}
A very common mistake is confusing $p_{X|Y}(x \mid y)$ with $p_{X,Y}(x, y)$. The joint probability $p_{X,Y}(x, y)$ is the probability that \textbf{both} $X=x$ and $Y=y$ happen out of the entire sample space. The conditional probability $p_{X|Y}(x \mid y)$ is the probability that $X=x$ happens, \textbf{restricting the universe only to cases where $Y=y$ has already happened}. The denominator $p_Y(y)$ is the scaling factor that forces the new restricted universe to sum to 1!
\end{warningbox}

\begin{tabular}{c|cc}
$p_{X,Y}(x, y)$ & $Y = 1$ & $Y = 2$ \\
\hline
$X = 0$ & $0.1$ & $0.3$ \\
$X = 1$ & $0.2$ & $k$ \\
\end{tabular}
\end{center}
Find the value of constant $k$ so that $p_{X,Y}$ is a valid joint PMF.

\item \textbf{Marginal Distribution Computation:} Using the completed table from Exercise 1:
    \begin{enumerate}
        \item Find the marginal PMF $p_X(x)$ for $x \in \{0, 1\}$.
        \item Find the marginal PMF $p_Y(y)$ for $y \in \{1, 2\}$.
    \end{enumerate}

\item \textbf{Conditional Probability Calculation:} Calculate $p_{X|Y}(1 \mid 2)$, the conditional probability that $X = 1$ given $Y = 2$.

\item \textbf{Triangular Domain Joint PMF:} Let $X$ and $Y$ have joint PMF $p_{X,Y}(x, y) = c(x + y)$ for $x \in \{1, 2\}$ and $y \in \{1, 2\}$.
    \begin{enumerate}
        \item Find the normalization constant $c$.
        \item Find $p_X(1)$.
    \end{enumerate}

\item \textbf{Data Science Application (A/B Testing Click-Through Distribution):} In an online ad campaign, $X \in \{0, 1\}$ represents device type ($0 = \text{Mobile}, 1 = \text{Desktop}$) and $Y \in \{0, 1\}$ represents click outcome ($0 = \text{No Click}, 1 = \text{Click}$). Given joint probabilities:
$$p_{X,Y}(0,0) = 0.50, \quad p_{X,Y}(0,1) = 0.10, \quad p_{X,Y}(1,0) = 0.32, \quad p_{X,Y}(1,1) = 0.08$$
Compare the Click-Through Rate (CTR) for Mobile users $\P(Y=1 \mid X=0)$ versus Desktop users $\P(Y=1 \mid X=1)$.
\end{enumerate}
