\chapter{Continuous Random Variables, CDF, PDF \& Standard Distributions}

Unlike discrete random variables whose supports are countable collections of isolated points, \index{Continuous Random Variables}\textbf{Continuous Random Variables} model quantities taking uncountably infinite real values over intervals---such as sensor latencies, stock price returns, and physical measurements. In this chapter, we master the fundamental mathematical calculus of continuous distributions: the \textbf{Cumulative Distribution Function (CDF)}, the \textbf{Probability Density Function (PDF)}, and the core triumvirate of standard continuous models: \index{Uniform Distribution}\textbf{Uniform}, \index{Exponential Distribution}\textbf{Exponential}, and \index{Gaussian (Normal) Distribution}\textbf{Gaussian (Normal)}.

\section{Cumulative Distribution Function (CDF) and Probability Density Function (PDF)}

\begin{definitionbox}{Cumulative Distribution Function $F_X(x)$}
The \textbf{Cumulative Distribution Function (CDF)} of any random variable $X$ (discrete, continuous, or mixed) is defined for all $x \in \mathbb{R}$ by:
$$F_X(x) = \P(X \le x)$$
Every valid CDF satisfies three axiomatic properties:
\begin{enumerate}
    \item \textbf{Monotonicity:} $x_1 < x_2 \implies F_X(x_1) \le F_X(x_2)$ (non-decreasing).
    \item \textbf{Asymptotic Limits:} $\lim_{x \to -\infty} F_X(x) = 0$ and $\lim_{x \to \infty} F_X(x) = 1$.
    \item \textbf{Right-Continuity:} $\lim_{h \to 0^+} F_X(x + h) = F_X(x)$ for all $x \in \mathbb{R}$.
\end{enumerate}
\end{definitionbox}

\begin{definitionbox}{Probability Density Function $f_X(x)$}
A random variable $X$ is \index{Continuous}\textbf{Continuous} if its CDF can be expressed as the definite integral of a non-negative function $f_X(x) \ge 0$:
$$F_X(x) = \int_{-\infty}^{x} f_X(t) \, dt \implies f_X(x) = \frac{d}{dx} F_X(x) \quad \text{(wherever } F_X \text{ is differentiable)}$$
\textbf{The Fundamental Integral Law:} For continuous $X$, the probability of landing in any interval $[a, b]$ is the definite area under the density curve:
$$\P(a \le X \le b) = \int_{a}^{b} f_X(x) \, dx = F_X(b) - F_X(a)$$
\textbf{Total Area Axiom:} $\int_{-\infty}^{\infty} f_X(x) \, dx = 1$.
\end{definitionbox}

\section{Standard Continuous Distributions}

\begin{intuitionbox}{The Standard Normal $Z$-Score as a Universal Ruler}
Why do we standardize via $Z = \frac{X - \mu}{\sigma}$? Because the Gaussian integral $\int e^{-x^2} dx$ possesses no elementary algebraic anti-derivative. Standardizing transforms every arbitrary Gaussian bell curve into the universal reference template $\mathcal{N}(0, 1)$ with mean $0$ and unit variance $1$. The $Z$-score simply counts how many standard deviations an observation lies away from the population center.
\end{intuitionbox}

\begin{lstlisting}[language=Python, caption=Data Science: Continuous Distributions in SciPy Stats]
from scipy.stats import norm, expon, uniform

# 1. Standard Normal Z: P(Z <= 1.96)
print(f"P(Z <= 1.96): {norm.cdf(1.96):.4f}") # 0.9750

# 2. General Normal: X ~ N(mu=50, sigma=10), P(X >= 65)
p_val = 1 - norm.cdf(65, loc=50, scale=10)
print(f"P(X >= 65): {p_val:.4f}") # 0.0668

# 3. Exponential: WARNING - SciPy uses 'scale = 1/lambda'!
# For lambda = 2 (mean = 0.5), scale = 1/2 = 0.5
p_expon = 1 - expon.cdf(1.0, scale=0.5)
print(f"Exponential P(T > 1.0) with lambda=2: {p_expon:.4f}") # e^(-2) = 0.1353
\end{lstlisting}

\begin{formulabox}{1. Uniform Distribution $\text{Uniform}(a, b)$}
\begin{itemize}
    \item \textbf{PDF:} $f(x) = \frac{1}{b - a}$ for $x \in [a, b]$ (zero outside).
    \item \textbf{CDF:} $F(x) = \frac{x - a}{b - a}$ for $x \in [a, b]$.
    \item \textbf{Mean \& Variance:} $\E[X] = \frac{a + b}{2}, \quad \Var(X) = \frac{(b - a)^2}{12}$.
\end{itemize}
\end{formulabox}

\begin{formulabox}{2. Exponential Distribution $\text{Exponential}(\lambda)$}
\begin{itemize}
    \item \textbf{PDF:} $f(x) = \lambda e^{-\lambda x}$ for $x \ge 0$ ($\lambda > 0$ is the arrival rate).
    \item \textbf{CDF:} $F(x) = 1 - e^{-\lambda x}$ for $x \ge 0$.
    \item \textbf{Mean \& Variance:} $\E[X] = \frac{1}{\lambda}, \quad \Var(X) = \frac{1}{\lambda^2}$.
    \item \textbf{Memoryless Property:} $\P(X > s + t \mid X > s) = \P(X > t)$ for all $s, t \ge 0$.
\end{itemize}
\end{formulabox}

\begin{formulabox}{3. Normal (Gaussian) Distribution $\mathcal{N}(\mu, \sigma^2)$}
\begin{itemize}
    \item \textbf{PDF:} $f(x) = \frac{1}{\sigma \sqrt{2\pi}} \exp\left( -\frac{(x - \mu)^2}{2\sigma^2} \right)$ for $x \in \mathbb{R}$.
    \item \textbf{Symmetry:} Bell-shaped around $x = \mu$; points of inflection at $x = \mu \pm \sigma$.
    \item \textbf{Empirical 68--95--99.7 Rule:}
    $$\P(\mu - \sigma \le X \le \mu + \sigma) \approx 0.6827, \quad \P(\mu - 2\sigma \le X \le \mu + 2\sigma) \approx 0.9545$$
    \item \textbf{Standardization:} $Z = \frac{X - \mu}{\sigma} \sim \mathcal{N}(0, 1)$ with CDF $\Phi(z) = \P(Z \le z)$.
\end{itemize}
\end{formulabox}

\section{Exam Gotchas \& Cognitive Traps}

\begin{trapbox}{PDF Density is NOT a Probability ($f(x)$ Can Exceed 1!)}
\textbf{The Trap:} Believing that because probabilities must lie in $[0, 1]$, the density value $f_X(x)$ must also be $\le 1$.

\textbf{The Reality:} Probability density $f(x)$ represents probability \emph{per unit length} (concentration), not absolute probability!
Consider $X \sim \text{Uniform}(0, 0.05)$:
$$f_X(x) = \frac{1}{0.05 - 0} = 20 \quad \text{for } x \in [0, 0.05]$$
The density is $20$, which is far greater than 1! Absolute probability is only recovered upon integrating over a finite interval width: $\P(0 \le X \le 0.01) = \int_0^{0.01} 20 \, dx = 0.20 \le 1$.
\end{trapbox}

\begin{trapbox}{Point Probabilities in Continuous RVs are Strictly Zero}
\textbf{The Trap:} Trying to calculate $\P(X = 3)$ for a continuous random variable and writing $f(3)$.

\textbf{The Reality:} For any continuous random variable $X$ and any single real number $c$:
$$\P(X = c) = \int_c^c f(x) \, dx = 0$$
Consequently, \textbf{endpoint inclusion does not alter continuous probabilities}:
$$\P(a \le X \le b) = \P(a < X \le b) = \P(a \le X < b) = \P(a < X < b)$$
(Notice that this is strictly false for discrete variables, where $\P(X \le 3) = \P(X < 3) + \P(X = 3)$).
\end{trapbox}

\begin{trapbox}{Exponential Parameterization: Rate $\lambda$ vs. Scale $\beta$}
\textbf{The Trap:} Confusing rate parameterization $f(x) = \lambda e^{-\lambda x}$ with scale parameterization $f(x) = \frac{1}{\beta} e^{-x/\beta}$.

\textbf{The Reality:}
\begin{itemize}
    \item In textbook / exam notation: $\lambda$ denotes the arrival \textbf{rate}. Here, $\E[X] = \frac{1}{\lambda}$. If a server handles $\lambda = 5$ queries/sec, average waiting time is $\frac{1}{5} = 0.2$ seconds.
    \item In Python SciPy (`scipy.stats.expon`): the parameter `scale` represents $\beta = \frac{1}{\lambda}$, which is the \textbf{mean duration}! Setting `expon(scale=5)` generates a process with rate $0.2$, not $5$!
\end{itemize}
\end{trapbox}

\section{Practice Exercises}

\subsection*{Part A: Analytical \& Computational Problems}

\begin{enumerate}
\item \textbf{PDF Normalization and Probability:} A continuous random variable $X$ has PDF:
    $$f_X(x) = \begin{cases} c x^2 & \text{for } 0 \le x \le 2 \\ 0 & \text{otherwise} \end{cases}$$
    \begin{enumerate}
        \item Find the constant $c$.
        \item Compute $\P(1 \le X \le 2)$.
    \end{enumerate}

\item \textbf{CDF to PDF Conversion:} The CDF of a continuous random variable $X$ is:
    $$F_X(x) = \begin{cases} 0 & \text{if } x < 0 \\ 1 - e^{-3x} & \text{if } x \ge 0 \end{cases}$$
    Identify the distribution of $X$, write its PDF $f_X(x)$, and calculate $\E[X]$.

\item \textbf{Exponential Memoryless Property:} Let $T \sim \text{Exponential}(\lambda = 0.5)$ represent customer service call duration in minutes. Compute $\P(T > 4 \mid T > 2)$.

\item \textbf{Standard Normal Z-Score Calculation:} Let $X \sim \mathcal{N}(\mu = 50, \sigma^2 = 100)$. Given $\Phi(1.5) = 0.9332$, compute $\P(X \ge 65)$.

\item \textbf{Data Science Application (Feature Normalization Range):} A model feature $X$ follows a $\text{Uniform}(10, 50)$ distribution.
    \begin{enumerate}
        \item Find the mean $\mu_X$ and standard deviation $\sigma_X$.
        \item Compute the probability that a random feature observation falls within 1 standard deviation of the mean ($\P(\mu_X - \sigma_X \le X \le \mu_X + \sigma_X)$).
    \end{enumerate}
\end{enumerate}

\subsection*{Part B: Conceptual Diagnostic Traps (Exam Gotchas)}

\begin{enumerate}[resume]
\item \textbf{Diagnostic Trap (Magnitude of Probability Density):}
Let $X \sim \text{Uniform}(0, 0.25)$. Consider the probability density function $f_X(x)$ evaluated at $x = 0.1$. Which of the following statements is strictly correct?
\begin{enumerate}
    \item $f_X(0.1) = 4$, which means the probability of observing exactly $X = 0.1$ is $400\%$.
    \item $f_X(0.1) = 4$, but the point probability is strictly $\P(X = 0.1) = 0$.
    \item $f_X(0.1)$ cannot exceed 1.0 because probabilities must belong to $[0, 1]$; therefore the distribution is invalid.
    \item $\P(X \le 0.1) = 4 \times 0.25 = 1.0$.
\end{enumerate}

\item \textbf{Diagnostic Trap (Endpoint Boundary Sensitivity):}
Let $X$ be a continuous random variable with continuous PDF $f_X(x)$ and CDF $F_X(x)$. How does the quantity $\P(a < X < b)$ compare with $\P(a \le X \le b)$?
\begin{enumerate}
    \item $\P(a < X < b) < \P(a \le X \le b)$ because two outcomes are excluded.
    \item $\P(a < X < b) = \P(a \le X \le b) - f_X(a) - f_X(b)$.
    \item $\P(a < X < b) = \P(a \le X \le b)$ identically, because the probability mass assigned to any single real coordinate is zero.
    \item It depends on whether the distribution is symmetric around the interval midpoint $\frac{a+b}{2}$.
\end{enumerate}
\end{enumerate}
