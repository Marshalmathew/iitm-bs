\section{i.i.d. Samples and Sample Statistics}

\begin{definitionbox}{i.i.d. Random Variables}
\index{i.i.d. Random Variables}
Random variables $X_1, X_2, \dots, X_n$ are \textbf{Independent and Identically Distributed (i.i.d.)} if:
\begin{enumerate}
    \item \textbf{Identically Distributed:} Every individual $X_i$ shares the exact same marginal probability distribution (identical CDF $F(x)$, identical mean $\mu = \E[X_i]$, and identical variance $\sigma^2 = \Var(X_i)$).
    \item \textbf{Mutually Independent:} The joint distribution factorizes completely into the product of marginals:
    $$f_{X_1, \dots, X_n}(x_1, \dots, x_n) = \prod_{i=1}^n f(x_i)$$
\end{enumerate}
\end{definitionbox}

\begin{formulabox}{Sample Mean $\bar{X}_n$ Properties}
The \index{Sample Mean}\textbf{Sample Mean} $\bar{X}_n = \frac{1}{n} \sum_{i=1}^n X_i$ is itself a random variable (a sample statistic) with fundamental properties:
\begin{itemize}
    \item \textbf{Unbiasedness:} $\E[\bar{X}_n] = \E\left[ \frac{1}{n} \sum_{i=1}^n X_i \right] = \frac{1}{n} \sum_{i=1}^n \E[X_i] = \frac{1}{n} (n\mu) = \mu$.
    \item \textbf{Variance Decay:} Because $X_i$ are independent ($\Cov(X_i, X_j) = 0$ for $i \neq j$):
    $$\Var(\bar{X}_n) = \Var\left( \frac{1}{n} \sum_{i=1}^n X_i \right) = \frac{1}{n^2} \sum_{i=1}^n \Var(X_i) = \frac{1}{n^2} (n \sigma^2) = \frac{\sigma^2}{n}$$
    \item \textbf{Standard Error of the Mean:} $\text{SE}(\bar{X}_n) = \sqrt{\Var(\bar{X}_n)} = \frac{\sigma}{\sqrt{n}}$.
\end{itemize}
\end{formulabox}

\section{Empirical Distribution Function}

When the theoretical population distribution $F(x)$ is completely unknown, we reconstruct it directly from an observed sample $x_1, \dots, x_n$.

\begin{definitionbox}{Empirical Cumulative Distribution Function (ECDF)}
The \index{Empirical CDF}\textbf{Empirical CDF} $\hat{F}_n(x)$ represents the empirical proportion of observations less than or equal to threshold $x$:
$$\hat{F}_n(x) = \frac{1}{n} \sum_{i=1}^n \mathbb{I}(x_i \le x)$$
where $\mathbb{I}(\cdot)$ is the binary indicator function.
\begin{itemize}
    \item For any fixed $x$, the count $\sum_{i=1}^n \mathbb{I}(X_i \le x) \sim \text{Binomial}(n, p = F(x))$.
    \item Unbiased: $\E[\hat{F}_n(x)] = F(x)$.
    \item Variance: $\Var(\hat{F}_n(x)) = \frac{F(x)(1 - F(x))}{n} \xrightarrow{n \to \infty} 0$.
    \item By the \index{Glivenko-Cantelli Theorem}\textbf{Glivenko-Cantelli Theorem}, $\sup_{x} |\hat{F}_n(x) - F(x)| \xrightarrow{a.s.} 0$, guaranteeing uniform convergence of the empirical curve to the true population curve.
\end{itemize}
\end{definitionbox}

\section{The Weak Law of Large Numbers (WLLN)}

\begin{intuitionbox}{Why the Casino "House Edge" is an Inviolable Law}
When a single gambler plays European roulette, the outcome is volatile: the gambler might win 35 times their bet. However, the casino hosts millions of independent bets daily. As $n \to \infty$, the sample mean return locks with mathematical certainty onto the true population expectation: a guaranteed house profit of $\frac{1}{37} \approx 2.70\%$ per spin. The WLLN guarantees that high sample volumes extinguish risk.
\end{intuitionbox}

\begin{lstlisting}[language=Python, caption=Data Science: Simulating the Law of Large Numbers in Python]
import numpy as np

# Simulating Law of Large Numbers for a fair die (true mean = 3.5)
np.random.seed(42)
n_trials = 50_000
rolls = np.random.randint(1, 7, size=n_trials)

# Compute cumulative running sample means
running_means = np.cumsum(rolls) / np.arange(1, n_trials + 1)

print(f"Sample mean at n=10:     {running_means[9]:.4f}")
print(f"Sample mean at n=1,000:  {running_means[999]:.4f}")
print(f"Sample mean at n=50,000: {running_means[-1]:.4f} (Locked onto 3.5000!)")
\end{lstlisting}

\begin{theorembox}{Weak Law of Large Numbers (WLLN)}
Let $X_1, X_2, \dots, X_n$ be i.i.d. random variables with finite population mean $\mu$ and finite variance $\sigma^2$. For any arbitrarily small margin of error $\epsilon > 0$:
$$\lim_{n \to \infty} \P(|\bar{X}_n - \mu| \ge \epsilon) = 0 \iff \lim_{n \to \infty} \P(|\bar{X}_n - \mu| < \epsilon) = 1$$
In statistical notation, the sample mean \index{Convergence in Probability}\textbf{converges in probability} to the population mean: $\bar{X}_n \xrightarrow{P} \mu$.

\textbf{Direct Proof via Chebyshev's Inequality:}
Applying Chebyshev's inequality to the random variable $\bar{X}_n$:
$$\P(|\bar{X}_n - \mu| \ge \epsilon) \le \frac{\Var(\bar{X}_n)}{\epsilon^2} = \frac{\sigma^2}{n \epsilon^2}$$
For any fixed $\epsilon > 0$ and finite $\sigma^2$, taking the limit as $n \to \infty$:
$$\lim_{n \to \infty} \frac{\sigma^2}{n \epsilon^2} = 0 \implies \lim_{n \to \infty} \P(|\bar{X}_n - \mu| \ge \epsilon) = 0$$
\end{theorembox}

\section{Exam Gotchas \& Cognitive Traps}

\begin{trapbox}{The Gambler's Fallacy: Dilution vs. Compensation}
\textbf{The Trap:} Believing that because a fair coin landed Heads 10 times in a row, the next flips are "due" or more likely to be Tails to "balance the average."

\textbf{The Reality:} The coin has no memory; each flip remains independent with $\P(\text{Heads}) = 0.5$. The Law of Large Numbers operates through \textbf{dilution}, never \textbf{compensation}!
\begin{itemize}
    \item Suppose you start with 10 excess Heads in the first 10 flips: $\bar{X}_{10} = \frac{10}{10} = 1.0$.
    \item In the next 10,000 flips, you expect roughly 5,000 Heads and 5,000 Tails.
    \item The total tally is now 5,010 Heads out of 10,010 flips:
    $$\bar{X}_{10,010} = \frac{10 + 5000}{10 + 10000} = \frac{5010}{10010} \approx 0.5005$$
\end{itemize}
The original 10 excess Heads were never "cancelled" by extra Tails; they were drowned out by the huge denominator $n$!
\end{trapbox}

\begin{trapbox}{The Cauchy Distribution Trap: When LLN Fails}
\textbf{The Trap:} Assuming that the sample mean $\bar{X}_n$ of any i.i.d. sample always converges to a constant as $n \to \infty$.

\textbf{The Reality:} The Law of Large Numbers strictly requires that the population has a \textbf{finite first moment} $\E[|X|] < \infty$.
For the \index{Cauchy Distribution}\textbf{Standard Cauchy Distribution} ($f(x) = \frac{1}{\pi(1+x^2)}$), the mean does not exist (the integral $\int_{-\infty}^\infty |x| f(x) dx = \infty$).
Remarkably, for an i.i.d. Cauchy sample, the sample mean $\bar{X}_n$ has the \textbf{exact same Cauchy distribution as a single observation $X_1$} for all $n$! Increasing $n$ does not reduce variance or cause convergence.
\end{trapbox}

\begin{trapbox}{Population Standard Deviation $\sigma$ vs. Standard Error $\text{SE}(\bar{X}_n)$}
\textbf{The Trap:} Confusing $\sigma$ with $\frac{\sigma}{\sqrt{n}}$ in exam problem statements.

\textbf{The Reality:}
\begin{itemize}
    \item Population Standard Deviation $\sigma$: A fixed characteristic of the underlying physical system (does not change as you collect more data).
    \item Standard Error $\text{SE}(\bar{X}_n) = \frac{\sigma}{\sqrt{n}}$: The dispersion of the \emph{estimator} $\bar{X}_n$ across repeated experiments. Shrinks to 0 at rate $1/\sqrt{n}$. To cut estimator uncertainty in half, you must quadruple the sample size $n$!
\end{itemize}
\end{trapbox}

\subsection*{Part A: Analytical \& Computational Problems}

\begin{enumerate}
\item \textbf{Sample Mean Standard Error:} An i.i.d. sample of size $n = 100$ is drawn from a population with mean $\mu = 80$ and variance $\sigma^2 = 400$.
    \begin{enumerate}
        \item Compute $\E[\bar{X}_{100}]$.
        \item Compute the standard error $\text{SE}(\bar{X}_{100}) = \sqrt{\Var(\bar{X}_{100})}$.
    \end{enumerate}

\item \textbf{Chebyshev Sample Size Determination for WLLN:} How large a sample size $n$ is required to ensure that the sample mean $\bar{X}_n$ lies within $\epsilon = 0.5$ units of the true mean $\mu$ with probability at least $95\%$ ($\P(|\bar{X}_n - \mu| < 0.5) \ge 0.95$), if population variance $\sigma^2 = 4$?

\item \textbf{Empirical CDF Evaluation:} Given a sample of 5 observations: $x = [2.1, 4.5, 1.8, 6.0, 4.5]$.
    \begin{enumerate}
        \item Calculate the value of the Empirical CDF $\hat{F}_5(4.0)$.
        \item Calculate $\hat{F}_5(4.5)$.
    \end{enumerate}

\item \textbf{Sum of i.i.d. Uniform RVs:} Let $X_1, X_2, \dots, X_{12}$ be i.i.d. $\text{Uniform}(0, 1)$ random variables. Let $S = \sum_{i=1}^{12} X_i$. Find $\E[S]$ and $\Var(S)$.

\item \textbf{Data Science Application (Monte Carlo Pi Estimation Bounds):} In a Monte Carlo algorithm, $n$ independent random points are dropped in a unit square. Let $I_i = 1$ if point $i$ falls inside a unit circle ($\P(I_i = 1) = p = \pi/4 \approx 0.7854$) and $0$ otherwise.
    Use Chebyshev's inequality to find the sample size $n$ needed so that the Monte Carlo estimate $\hat{p}_n = \frac{1}{n} \sum_{i=1}^n I_i$ is within $0.01$ of $p$ with at least $90\%$ probability.
\end{enumerate}

\subsection*{Part B: Conceptual Diagnostic Traps (Exam Gotchas)}

\begin{enumerate}[resume]
\item \textbf{Diagnostic Trap (The Mechanism of the Law of Large Numbers):}
A roulette player observes that the ball has landed on Red 8 consecutive times. The player argues: "By the Law of Large Numbers, the average proportion of Reds must converge to 0.5; therefore Black has a significantly higher probability of occurring on the 9th spin." What is the fundamental flaw in the player's reasoning?
\begin{enumerate}
    \item The Law of Large Numbers only applies to coin flips, not casino roulette wheels.
    \item The WLLN functions through dilution via an increasing sample size $n$, not through historical compensation of independent future trials.
    \item The player is correct; the probability of Black increases to $1 - (0.5)^8$.
    \item The WLLN only holds when sample size $n$ exceeds 100,000.
\end{enumerate}

\item \textbf{Diagnostic Trap (Boundary Conditions of the WLLN):}
Let $X_1, X_2, \dots, X_n$ be an i.i.d. sample drawn from a standard Cauchy distribution ($f(x) = \frac{1}{\pi(1+x^2)}$). As $n \to \infty$, what happens to the sample mean $\bar{X}_n$?
\begin{enumerate}
    \item $\bar{X}_n$ converges in probability to 0 because the distribution is symmetric around 0.
    \item $\bar{X}_n$ converges to $\pi$ by the Central Limit Theorem.
    \item $\bar{X}_n$ does not converge to any constant; its distribution remains identical to a single Cauchy random variable for all $n$.
    \item $\Var(\bar{X}_n)$ decays at rate $1/n$.
\end{enumerate}
\end{enumerate}
