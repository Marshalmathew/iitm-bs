\section{Mathematical \& Theoretical Foundations}

The curriculum of Mathematics for Data Science 1 forms the foundational language of modern artificial intelligence. Rather than isolated topics, the five structural units constitute a single interconnected pipeline:

\subsection{The Five Pillars of Foundation Mathematics}

\begin{enumerate}
    \item \textbf{Pillar I: Sets, Relations, and Mappings (Weeks 1 \& 5)}
    \begin{itemize}
        \item Sets define data domains ($\mathcal{X} \subseteq \R^d$) and categorical label spaces ($\mathcal{Y} = \{0, 1\}$).
        \item Relations model relational databases (SQL joins as Cartesian intersections) and clustering partitions via equivalence relations:
        \begin{equation}
        [x] = \{y \in X \mid x \sim y\}, \quad X / \sim = \bigcup [x], \quad [x] \cap [y] = \emptyset \text{ for } x \not\sim y
        \end{equation}
        \item Invertible bijections enable lossless coordinate transformations, feature normalization, and logit-sigmoid link functions:
        \begin{equation}
        \sigma(z) = \frac{1}{1 + e^{-z}}, \quad \sigma^{-1}(p) = \operatorname{logit}(p) = \ln\left(\frac{p}{1 - p}\right)
        \end{equation}
    \end{itemize}

    \item \textbf{Pillar II: Analytic Geometry \& Linear Models (Week 2)}
    \begin{itemize}
        \item Linear equations $w_1 x_1 + w_2 x_2 + b = 0$ define hyperplanes separating classification classes.
        \item The perpendicular geometric margin from sample point $(x_0, y_0)$ to decision boundary $Ax + By + C = 0$ is:
        \begin{equation}
        d = \frac{|Ax_0 + By_0 + C|}{\sqrt{A^2 + B^2}}
        \end{equation}
        This metric forms the core objective maximized in Support Vector Machines (SVM).
    \end{itemize}

    \item \textbf{Pillar III: Non-linear Approximations \& Information Scaling (Weeks 3, 4, 6)}
    \begin{itemize}
        \item Quadratic loss functions $L(w) = a(w - w^*)^2 + c$ yield closed-form OLS solutions and convex regularizations ($w^* = \frac{w_{\text{OLS}}}{1 + \lambda}$).
        \item Higher-degree polynomials provide non-linear curve fitting via Taylor/Weierstrass approximations, but require degree control to prevent Runge's phenomenon and high-variance overfitting.
        \item Logarithms compress massive exponential spreads, eliminate floating-point probability underflow via log-likelihood $\ell(\theta) = \sum \ln p_i$, and measure uncertainty via Shannon entropy:
        \begin{equation}
        H(X) = -\sum_{i=1}^k p_i \log_2(p_i)
        \end{equation}
    \end{itemize}

    \item \textbf{Pillar IV: Infinitesimal Calculus \& Optimization (Weeks 7, 8, 9)}
    \begin{itemize}
        \item Derivatives measure instantaneous rates of change. The Chain Rule powers deep learning backpropagation:
        \begin{equation}
        \frac{d}{dx}[f(g(x))] = f'(g(x)) \cdot g'(x)
        \end{equation}
        \item Optimization is guided by stationary points ($f'(w^*) = 0$) and convexity conditions ($f''(w^*) > 0$).
        \item Gradient descent updates $w_{t+1} = w_t - \eta \nabla L(w_t)$ converge if step size satisfies the Lipschitz stability bound $0 < \eta < \frac{2}{L}$.
        \item Integration computes cumulative probabilities $\int_a^b p(x) \dx = 1$ and risk expectations $\E[L] = \int L(x) p(x) \dx$.
    \end{itemize}

    \item \textbf{Pillar V: Discrete Relational Structures \& Graph Optimization (Weeks 10 \& 11)}
    \begin{itemize}
        \item Graphs $G = (V, E)$ formalize network topologies, knowledge graphs, and relational constraints.
        \item The Handshaking Lemma bounds connectivity: $\sum \deg(v) = 2|E|$.
        \item Directed Acyclic Graphs (DAGs) govern task pipelines and autograd execution engines via Topological Sorting.
        \item Shortest path algorithms (Dijkstra: $\mathcal{O}(|E| \log |V|)$, Bellman-Ford: $\mathcal{O}(|V| \cdot |E|)$) and Minimum Spanning Trees (Kruskal, Prim) optimize physical network topologies and manifold learning distances.
    \end{itemize}
\end{enumerate}

\section{Deep Intuition \& Conceptual Derivations}

\subsection{The Universal Convergence of Gradient Descent}
Consider the minimization of a strictly convex, $L$-Lipschitz smooth quadratic loss function:
\begin{equation}
J(w) = \frac{1}{2} L (w - w^*)^2, \quad \text{where } L > 0
\end{equation}
The gradient is $J'(w) = L(w - w^*)$. Gradient descent with learning rate $\eta > 0$ applies the update rule:
\begin{equation}
w_{t+1} = w_t - \eta J'(w_t) = w_t - \eta L (w_t - w^*)
\end{equation}
Subtracting the optimal parameter $w^*$ from both sides:
\begin{equation}
w_{t+1} - w^* = (w_t - w^*) - \eta L (w_t - w^*) = (1 - \eta L)(w_t - w^*)
\end{equation}
Applying this recurrence relation recursively across $t$ steps yields:
\begin{equation}
w_t - w^* = (1 - \eta L)^t (w_0 - w^*)
\end{equation}
For the parameter error to converge to zero as $t \to \infty$, the contraction factor must strictly satisfy:
\begin{equation}
|1 - \eta L| < 1 \iff -1 < 1 - \eta L < 1 \iff 0 < \eta < \frac{2}{L}
\end{equation}
\begin{itemize}
    \item If $\eta = \frac{1}{L}$: $|1 - 1| = 0$, convergence occurs in exactly \textbf{1 step} (the Newton step).
    \item If $0 < \eta < \frac{1}{L}$: Monotonic geometric convergence without oscillations.
    \item If $\frac{1}{L} < \eta < \frac{2}{L}$: Damped oscillatory convergence around $w^*$.
    \item If $\eta \ge \frac{2}{L}$: Immediate numerical explosion and divergence to $\pm \infty$.
\end{itemize}

\section{Explicit Gotchas \& Diagnostic Misconceptions}

\begin{trapbox}{Trap 1: Stationarity Does Not Imply Optimality}
Setting $f'(w) = 0$ identifies critical points, which can be local minima, local maxima, or inflection points. For $f(w) = w^3$, $f'(0) = 0$, but $w=0$ is an inflection saddle point, not an extremum. Rigorous optimization always checks the Second Derivative Test ($f''(w) > 0$ for minima) or establishes global convexity.
\end{trapbox}

\begin{trapbox}{Trap 2: Ignoring Domain Restrictions in Inverse \& Log Functions}
Equations involving logarithms $\ln(g(x))$ require $g(x) > 0$. Solving $\ln(x - 2) + \ln(x - 3) = \ln(2)$ algebraically gives $x^2 - 5x + 6 = 2 \implies x^2 - 5x + 4 = 0 \implies x = 4 \text{ or } x = 1$. The candidate root $x = 1$ is completely extraneous because $\ln(1 - 2) = \ln(-1)$ is undefined in real numbers!
\end{trapbox}

\begin{trapbox}{Trap 3: Linear Parallel Line Distance Normalization Oversight}
The distance between parallel lines $Ax + By + C_1 = 0$ and $Ax + By + C_2 = 0$ is $d = \frac{|C_1 - C_2|}{\sqrt{A^2 + B^2}}$. If the equations are given as $2x + 4y + 5 = 0$ and $x + 2y - 3 = 0$, you cannot compute $|5 - (-3)| = 8$. You must first scale the second equation to matching coefficients: $2x + 4y - 6 = 0 \implies d = \frac{|5 - (-6)|}{\sqrt{2^2 + 4^2}} = \frac{11}{\sqrt{20}}$.
\end{trapbox}

\begin{trapbox}{Trap 4: Shortest Path Blindness on Graphs with Negative Weights}
Executing Dijkstra's algorithm on a graph with negative edges yields incorrect distances because Dijkstra permanently closes nodes upon extraction from the min-heap. When negative edges exist, the Bellman-Ford algorithm must be used.
\end{trapbox}

\section{Python Scientific Computing Lab}

The following laboratory executes an integrated end-to-end demonstration combining feature normalization, gradient descent optimization of a convex loss function, and computational graph DAG topological execution.

\begin{lstlisting}[language=Python, caption=Integrated Mathematics 1 Capstone Pipeline]
import numpy as np
from collections import deque

# 1. Feature Transformation and Min-Max Normalization (Week 1 & 5)
raw_features = np.array([12.0, 45.0, 78.0, 110.0, 150.0])
x_min, x_max = raw_features.min(), raw_features.max()
norm_features = (raw_features - x_min) / (x_max - x_min)
print("Normalized Features [0, 1]:", np.round(norm_features, 3))

# 2. Convex Loss Optimization via Gradient Descent (Week 7 & 8)
# Loss function: L(w) = 0.5 * L_smooth * (w - w_true)^2
L_smooth = 4.0
w_true = 3.5
w = 0.0 # Initial guess
eta = 0.2 # Learning rate satisfying 0 < eta < 2 / L_smooth (0 < 0.2 < 0.5)

print("\n--- Gradient Descent Optimization ---")
for step in range(1, 11):
    grad = L_smooth * (w - w_true)
    w = w - eta * grad
    loss = 0.5 * L_smooth * (w - w_true)**2
    if step in [1, 2, 5, 10]:
        print(f"Step {step:2d} | Weight: {w:.4f} | Loss: {loss:.6e}")

# 3. Continuous Probability Verification via Integration (Week 9)
from scipy.integrate import quad
pdf = lambda x: (3.0 / 64.0) * x**2 # Defined on [0, 4]
total_p, _ = quad(pdf, 0, 4)
mean_val, _ = quad(lambda x: x * pdf(x), 0, 4)
print(f"\nPDF Integral [0, 4]: {total_p:.4f} (Must equal 1.0)")
print(f"Expected Value E[X]: {mean_val:.4f} (Expected: 3.0)")

# 4. Computational Dependency DAG Topological Sort (Week 10)
# Nodes: X (Input), W (Weights), Z (Linear: X*W), A (Activation: Sigmoid), L (Loss)
dependencies = {
    'X': ['Z'],
    'W': ['Z'],
    'Z': ['A'],
    'A': ['L'],
    'L': []
}

in_deg = {u: 0 for u in dependencies}
for u in dependencies:
    for v in dependencies[u]:
        in_deg[v] += 1

q = deque([u for u in in_deg if in_deg[u] == 0])
forward_pass = []
while q:
    u = q.popleft()
    forward_pass.append(u)
    for v in dependencies[u]:
        in_deg[v] -= 1
        if in_deg[v] == 0:
            q.append(v)

print("\nAutograd Execution Order:", " -> ".join(forward_pass))
\end{lstlisting}

    \begin{enumerate}

\subsection*{Part B: Diagnostic Misconception MCQs}

\begin{enumerate}
    \setcounter{enumi}{5}
    \item A machine learning engineer runs Gradient Descent on a strictly convex loss function $L(w)$ whose second derivative satisfies $L''(w) \le 8$ everywhere. The engineer initializes at $w_0 = 10.0$ and sets the constant learning rate $\eta = 0.30$. What will happen to the parameter iterates $w_t$ as $t \to \infty$?
    \begin{enumerate}
        \item The iterates will converge monotonically to the global minimum $w^*$.
        \item The iterates will oscillate with decreasing amplitude, eventually converging to $w^*$.
        \item The iterates will diverge to $\pm \infty$ because the learning rate exceeds the critical stability threshold $\eta_{\text{max}} = \frac{2}{L_{\text{smooth}}} = \frac{2}{8} = 0.25$.
        \item The iterates will become trapped in a local minimum because the learning rate is too small.
    \end{enumerate}

    \item A student is asked whether adding an edge of weight 0 between two existing vertices in a connected weighted graph can ever increase the total weight of the graph's Minimum Spanning Tree (MST). Which response is rigorously correct?
    \begin{enumerate}
        \item Yes, because adding an edge increases the number of vertices by 1.
        \item No; by the Cut Property, adding any edge can only either decrease or maintain the total weight of the MST, as the new edge can replace a heavier edge in a cycle or be ignored.
        \item Yes, if the edge forms a cycle with negative edges.
        \item It depends on whether Prim's or Kruskal's algorithm is used.
    \end{enumerate}
\end{enumerate}
