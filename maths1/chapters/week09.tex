\section{Mathematical \& Theoretical Foundations}

Differential calculus decomposes continuous processes into instantaneous rates of change. Conversely, \textbf{integral calculus} synthesizes infinitely many infinitesimal quantities into macroscopic totals. In machine learning, continuous optimization, Bayesian inference, and probabilistic modelling rely fundamentally on integration to compute normalization constants, marginal distributions, and risk expectations.

\subsection{Riemann Sums and the Definite Integral}

Let $f: [a, b] \to \R$ be a bounded function on a compact interval $[a, b]$. 

\begin{definitionbox}{Partition and Riemann Sum}
A \textbf{partition} $P$ of $[a, b]$ is a finite ordered sequence of points:
\begin{equation}
P = \{x_0, x_1, x_2, \dots, x_n\}, \quad \text{where } a = x_0 < x_1 < x_2 < \dots < x_n = b
\end{equation}
The width of the $i$-th subinterval $[x_{i-1}, x_i]$ is $\Delta x_i = x_i - x_{i-1}$. The \textbf{mesh norm} of the partition is:
\begin{equation}
\|P\| = \max_{1 \le i \le n} \Delta x_i
\end{equation}
For a uniform partition into $n$ equal subintervals, $\Delta x = \frac{b - a}{n}$.

For any selection of sample evaluation points $x_i^* \in [x_{i-1}, x_i]$, the \textbf{Riemann Sum} $R(f, P)$ is:
\begin{equation}
R(f, P) = \sum_{i=1}^n f(x_i^*) \Delta x_i
\end{equation}
Depending on the choice of $x_i^*$:
\begin{itemize}
    \item \textbf{Left Riemann Sum:} $x_i^* = x_{i-1}$
    \item \textbf{Right Riemann Sum:} $x_i^* = x_i$
    \item \textbf{Midpoint Riemann Sum:} $x_i^* = \frac{x_{i-1} + x_i}{2}$
\end{itemize}
\end{definitionbox}

\begin{definitionbox}{The Definite Integral}
A function $f$ is \textbf{Riemann integrable} on $[a, b]$ if there exists a number $I \in \R$ such that for every $\varepsilon > 0$, there exists $\delta > 0$ such that for every partition $P$ with $\|P\| < \delta$ and every choice of sample points $x_i^*$:
\begin{equation}
\left| \sum_{i=1}^n f(x_i^*) \Delta x_i - I \right| < \varepsilon
\end{equation}
We write this limit as the \textbf{definite integral}:
\begin{equation}
\int_a^b f(x) \dx = \lim_{\|P\| \to 0} \sum_{i=1}^n f(x_i^*) \Delta x_i
\end{equation}
\textbf{Integrability Guarantee:} If $f$ is continuous on $[a, b]$, or bounded with a finite number of jump discontinuities, $f$ is guaranteed to be Riemann integrable on $[a, b]$.
\end{definitionbox}

\subsection{The Fundamental Theorem of Calculus}

The Fundamental Theorem of Calculus establishes that differentiation and integration are inverse operations.

\begin{theorembox}{Fundamental Theorem of Calculus (FTC)}
Let $f: [a, b] \to \R$ be continuous on $[a, b]$.
\begin{itemize}
    \item \textbf{FTC Part 1 (Accumulation Derivative):} Define the area accumulation function $g: [a, b] \to \R$ by:
    \begin{equation}
    g(x) = \int_a^x f(t) \, dt
    \end{equation}
    Then $g$ is continuous on $[a, b]$, differentiable on $(a, b)$, and:
    \begin{equation}
    g'(x) = \frac{d}{dx} \left[ \int_a^x f(t) \, dt \right] = f(x)
    \end{equation}
    \item \textbf{FTC Part 2 (Evaluation Theorem):} If $F: [a, b] \to \R$ is any antiderivative of $f$ on $[a, b]$ (that is, $F'(x) = f(x)$ for all $x \in [a, b]$), then:
    \begin{equation}
    \int_a^b f(x) \dx = F(b) - F(a) = \Big[ F(x) \Big]_a^b
    \end{equation}
\end{itemize}
\end{theorembox}

\subsection{Indefinite Integrals and Standard Rules}

An \textbf{indefinite integral} denotes the entire family of antiderivatives of $f(x)$, parameterized by an arbitrary integration constant $C \in \R$:
\begin{equation}
\int f(x) \dx = F(x) + C \iff \frac{d}{dx}[F(x) + C] = f(x)
\end{equation}

\begin{formulabox}{Essential Antiderivative Rules}
\begin{itemize}
    \item \textbf{Power Rule:} $\int x^n \dx = \frac{x^{n+1}}{n+1} + C \quad (\text{for all } n \neq -1)$
    \item \textbf{Logarithmic Rule:} $\int \frac{1}{x} \dx = \ln|x| + C \quad (\text{for } x \neq 0)$
    \item \textbf{Exponential Rule:} $\int e^{kx} \dx = \frac{1}{k}e^{kx} + C \quad (k \neq 0)$
    \item \textbf{Linearity:} $\int [\alpha f(x) + \beta g(x)] \dx = \alpha \int f(x) \dx + \beta \int g(x) \dx$
\end{itemize}
\end{formulabox}

\subsection{Integration by Substitution (The Chain Rule in Reverse)}

When an integrand contains a composite function multiplied by the derivative of the inner function, substitution transforms the integral into an elementary form.

\begin{theorembox}{The Substitution Rule}
Let $u = g(x)$ be a differentiable function whose range is an interval $I$, and let $f$ be continuous on $I$. Then:
\begin{equation}
\int f(g(x)) g'(x) \dx = \int f(u) \, du
\end{equation}
For definite integrals, transforming the variable requires transforming the limits of integration:
\begin{equation}
\int_a^b f(g(x)) g'(x) \dx = \int_{g(a)}^{g(b)} f(u) \, du
\end{equation}
\end{theorembox}

\subsection{Area Between Curves}

If $f(x)$ and $g(x)$ are continuous on $[a, b]$ with $f(x) \ge g(x)$ for all $x \in [a, b]$, the geometric area $A$ enclosed between the curves and the vertical boundaries $x = a, x = b$ is:
\begin{equation}
A = \int_a^b [f(x) - g(x)] \dx
\end{equation}
If the curves cross at intermediate points $c \in (a, b)$, the total geometric area is computed by partitioning the integral:
\begin{equation}
A = \int_a^b |f(x) - g(x)| \dx = \int_a^c [f(x) - g(x)] \dx + \int_c^b [g(x) - f(x)] \dx
\end{equation}

\section{Deep Intuition \& Conceptual Derivations}

\subsection{Proof of the Fundamental Theorem of Calculus (Part 1)}
To understand why the derivative of an accumulation function recovers the original rate, consider the Newton difference quotient of $g(x) = \int_a^x f(t) \, dt$:
\begin{equation}
g'(x) = \lim_{h \to 0} \frac{g(x+h) - g(x)}{h} = \lim_{h \to 0} \frac{1}{h} \left[ \int_a^{x+h} f(t) \, dt - \int_a^x f(t) \, dt \right] = \lim_{h \to 0} \frac{1}{h} \int_x^{x+h} f(t) \, dt
\end{equation}
By the \textbf{Mean Value Theorem for Integrals}, because $f$ is continuous on $[x, x+h]$, there exists a point $c_h \in [x, x+h]$ such that:
\begin{equation}
\int_x^{x+h} f(t) \, dt = f(c_h) \cdot h \implies \frac{1}{h} \int_x^{x+h} f(t) \, dt = f(c_h)
\end{equation}
As $h \to 0$, the interval $[x, x+h]$ shrinks to the single point $x$, forcing $c_h \to x$. By the continuity of $f$:
\begin{equation}
\lim_{h \to 0} f(c_h) = f(x) \implies g'(x) = f(x) \quad \blacksquare
\end{equation}

\subsection{Derivation of the Evaluation Theorem (FTC Part 2)}
Let $F(x)$ be any antiderivative of $f(x)$, meaning $F'(x) = f(x)$. From FTC Part 1, we know that $g(x) = \int_a^x f(t) \, dt$ also satisfies $g'(x) = f(x)$. 
Because two functions with identical derivatives differ only by a constant:
\begin{equation}
F(x) = g(x) + C = \int_a^x f(t) \, dt + C
\end{equation}
Evaluating at $x = a$:
\begin{equation}
F(a) = \int_a^a f(t) \, dt + C = 0 + C = C
\end{equation}
Now evaluating at $x = b$:
\begin{equation}
F(b) = \int_a^b f(t) \, dt + C = \int_a^b f(t) \, dt + F(a)
\end{equation}
Rearranging terms yields FTC Part 2:
\begin{equation}
\int_a^b f(t) \, dt = F(b) - F(a) \quad \blacksquare
\end{equation}

\section{Explicit Gotchas \& Diagnostic Misconceptions}

\begin{trapbox}{Trap 1: Omitting the $+ C$ in Indefinite Integrals}
An indefinite integral represents an entire equivalence class of vertically shifted functions. Omitting $+C$ causes boundary-value problems and differential equation solutions to fail. For example, solving $\frac{dy}{dx} = 2x$ with initial condition $y(0) = 5$ yields $y = x^2 + C \implies y = x^2 + 5$. Dropping $+C$ yields the wrong function $y = x^2$.
\end{trapbox}

\begin{trapbox}{Trap 2: Conflating Net Signed Area with Total Geometric Area}
The definite integral $\int_a^b f(x) \dx$ computes \textit{net signed area}, where regions below the $x$-axis contribute negative quantities. For example, $\int_0^{2\pi} \sin(x) \dx = [-\cos(x)]_0^{2\pi} = -1 - (-1) = 0$. If an engineer or data scientist seeks the \textit{total physical/geometric area}, they must compute:
\begin{equation*}
\int_a^b |f(x)| \dx = \int_0^\pi \sin(x) \dx + \int_\pi^{2\pi} (-\sin(x)) \dx = 2 + 2 = 4
\end{equation*}
\end{trapbox}

\begin{trapbox}{Trap 3: Forgetting to Transform Limits During $u$-Substitution}
When evaluating $\int_a^b f(g(x))g'(x) \dx$ via substitution $u = g(x)$, the bounds $a$ and $b$ apply to $x$, not $u$. Writing $\int_a^b f(u) \, du$ is mathematically false. You must either update bounds to $[g(a), g(b)]$ immediately:
\begin{equation*}
\int_0^1 2x e^{x^2} \dx = \int_{0^2}^{1^2} e^u \, du = \int_0^1 e^u \, du = e^1 - e^0 = e - 1
\end{equation*}
or solve the indefinite integral first, substitute back $x$, and evaluate at the original limits.
\end{trapbox}

\begin{trapbox}{Trap 4: Applying FTC Across Discontinuities (The Cauchy Principal Failure)}
FTC Part 2 requires $f$ to be continuous on the entire closed interval $[a, b]$. Blindly integrating across an essential pole produces non-physical, contradictory results. For example:
\begin{equation*}
\int_{-1}^1 \frac{1}{x^2} \dx \ne \left[ -\frac{1}{x} \right]_{-1}^1 = -1 - (1) = -2 \quad \text{\textbf{(WRONG!)}}
\end{equation*}
The integrand $\frac{1}{x^2} > 0$ everywhere on its domain, yet blind evaluation yields a negative area! The integral is an improper integral that diverges to $+\infty$ at $x = 0$.
\end{trapbox}

\section{Python Scientific Computing Lab}

The following laboratory illustrates numerical quadrature (`scipy.integrate.quad`), Riemann sum approximations, and symbolic integration (`sympy`).

\begin{lstlisting}[language=Python, caption=Numerical Quadrature and Riemann Approximations]
import numpy as np
from scipy.integrate import quad
import sympy as sp

# 1. Exact Symbolic Integration via SymPy
x = sp.Symbol('x')
f_sym = 3 * x**2 + 2 * x + 1

# Indefinite integral: returns x**3 + x**2 + x (constant omitted by convention)
indef_res = sp.integrate(f_sym, x)
print(f"Indefinite Integral: {indef_res} + C")

# Definite integral from 0 to 2: (8 + 4 + 2) - 0 = 14
def_res = sp.integrate(f_sym, (x, 0, 2))
print(f"SymPy Definite Integral [0, 2]: {def_res}")

# 2. High-Precision Numerical Quadrature via SciPy
f_num = lambda t: 3 * t**2 + 2 * t + 1
integral_val, abs_error = quad(f_num, 0, 2)
print(f"SciPy quad result: {integral_val:.6f} (Error est: {abs_error:.2e})")

# 3. Riemann Sum Convergence
def riemann_sum(func, a, b, n, method='midpoint'):
    dx = (b - a) / n
    if method == 'left':
        x_pts = np.linspace(a, b - dx, n)
    elif method == 'right':
        x_pts = np.linspace(a + dx, b, n)
    elif method == 'midpoint':
        x_pts = np.linspace(a + dx/2, b - dx/2, n)
    return np.sum(func(x_pts) * dx)

for n in [5, 20, 100, 1000]:
    r_mid = riemann_sum(f_num, 0, 2, n, method='midpoint')
    print(f"n = {n:4d} | Midpoint Riemann Sum: {r_mid:.6f} | Error: {abs(r_mid - 14):.6e}")

# 4. Continuous Probability Density Verification
# PDF: p(x) = (3/8)*x^2 on [0, 2]
p = lambda x: (3.0 / 8.0) * x**2
total_prob, _ = quad(p, 0, 2)
expected_x, _ = quad(lambda x: x * p(x), 0, 2)
print(f"Total Probability Area: {total_prob:.6f}") # Must equal 1.0
print(f"Expected Value E[X]: {expected_x:.6f}")    # Integral of (3/8)x^3 is (3/32)(16) = 1.5
\end{lstlisting}
