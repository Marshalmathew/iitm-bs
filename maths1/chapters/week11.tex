\section{Mathematical \& Theoretical Foundations}

In weighted networks, structural topology is enriched by numerical weights $w: E \to \R$ representing distances, financial costs, latencies, or dissimilarities. Two fundamental optimization problems dominate graph algorithms:
\begin{enumerate}
    \item \textbf{Shortest Path Problem:} Finding the path connecting two vertices that minimizes the total sum of edge weights.
    \item \textbf{Minimum Spanning Tree Problem:} Finding a cycle-free backbone that connects all vertices while minimizing total network construction cost.
\end{enumerate}

\subsection{Transitive Closure and Warshall's Algorithm}

Given a directed graph $G = (V, E)$, the \textbf{transitive closure} is a graph $G^* = (V, E^*)$ where $(i, j) \in E^*$ if and only if there exists a directed path of any non-zero length from $i$ to $j$ in $G$.

\begin{theorembox}{Warshall's Dynamic Programming Algorithm}
Let vertices be indexed $V = \{1, 2, \dots, n\}$. Define $R^{(k)} \in \{0, 1\}^{n \times n}$ where $R^{(k)}_{ij} = 1$ if there exists a directed path from $i$ to $j$ using only intermediate vertices from the subset $\{1, 2, \dots, k\}$.
\begin{itemize}
    \item \textbf{Base Case ($k = 0$):} $R^{(0)}_{ij} = A_{ij}$ (direct adjacency).
    \item \textbf{Recurrence Relation ($k = 1, \dots, n$):}
    \begin{equation}
    R^{(k)}_{ij} = R^{(k-1)}_{ij} \lor \left( R^{(k-1)}_{ik} \land R^{(k-1)}_{kj} \right)
    \end{equation}
    A path exists from $i$ to $j$ through $\{1, \dots, k\}$ if either:
    \begin{enumerate}
        \item There was already a path using only $\{1, \dots, k-1\}$, OR
        \item There is a path from $i \to k$ AND a path from $k \to j$ using intermediate nodes from $\{1, \dots, k-1\}$.
    \end{enumerate}
    \item \textbf{Complexity:} Time $\Theta(|V|^3)$, Space $\Theta(|V|^2)$.
\end{itemize}
\end{theorembox}

\subsection{Single-Source Shortest Paths: Dijkstra's Algorithm}

Let $G = (V, E, w)$ be a directed or undirected graph with edge weights $w(u, v) \ge 0$.

\begin{definitionbox}{Edge Relaxation and Dijkstra's Mechanism}
For a source vertex $s \in V$, maintain an array of tentative distances $\text{dist}[v]$ initialized to $\text{dist}[s] = 0$ and $\text{dist}[v] = \infty$ for all $v \neq s$.
\begin{itemize}
    \item \textbf{Relaxation Step:} For directed edge $(u, v)$ with weight $w(u, v)$:
    \begin{equation}
    \text{if } \text{dist}[u] + w(u, v) < \text{dist}[v] \implies \text{dist}[v] = \text{dist}[u] + w(u, v), \quad \text{parent}[v] = u
    \end{equation}
    \item \textbf{Greedy Loop:} Maintain a priority queue of unvisited vertices keyed on $\text{dist}[\cdot]$. In each iteration:
    \begin{enumerate}
        \item Extract vertex $u$ with the minimum tentative distance: $u = \operatorname{argmin}_{v \notin S} \text{dist}[v]$.
        \item Mark $u$ as finalized ($u \in S$).
        \item Relax all outgoing edges $(u, v) \in E$ where $v \notin S$.
    \end{enumerate}
    \item \textbf{Precondition:} \textbf{All edge weights must be non-negative} ($w(e) \ge 0$).
    \item \textbf{Complexity:} $\mathcal{O}((|V| + |E|) \log |V|)$ using a binary min-heap.
\end{itemize}
\end{definitionbox}

\subsection{Shortest Paths with Negative Weights: Bellman-Ford}

When edge weights may be negative (e.g. financial returns or energy gains), Dijkstra's greedy finalization fails.

\begin{theorembox}{The Bellman-Ford Algorithm}
\begin{enumerate}
    \item Initialize $\text{dist}[s] = 0$ and all other distances to $\infty$.
    \item \textbf{Relaxation Passes:} Repeat $|V| - 1$ times:
    \begin{itemize}
        \item For every directed edge $(u, v) \in E$:
        \begin{equation}
        \text{if } \text{dist}[u] + w(u, v) < \text{dist}[v] \implies \text{dist}[v] = \text{dist}[u] + w(u, v)
        \end{equation}
    \end{itemize}
    \item \textbf{Negative Cycle Detection ($|V|$-th Pass):} Iterate over all edges $(u, v) \in E$ one additional time. If any edge can still be relaxed ($\text{dist}[u] + w(u, v) < \text{dist}[v]$), the graph contains a \textbf{negative-weight cycle} reachable from $s$, meaning shortest paths are undefined ($-\infty$).
    \item \textbf{Complexity:} $\mathcal{O}(|V| \cdot |E|)$.
\end{enumerate}
\end{theorembox}

\subsection{Minimum Spanning Trees (MST)}

A \textbf{spanning tree} of a connected, undirected graph $G = (V, E)$ is an acyclic connected subgraph $T = (V, E_T)$ that spans all vertices in $V$. Any spanning tree on $n$ vertices contains exactly $n - 1$ edges.
A \textbf{Minimum Spanning Tree (MST)} is a spanning tree that minimizes total edge weight:
\begin{equation}
w(T) = \sum_{e \in E_T} w(e)
\end{equation}

\begin{theorembox}{The Cut Property of MSTs}
Let $G = (V, E)$ be a connected undirected graph with real edge weights. Let $S \subset V$ be any non-empty proper subset of vertices, defining a cut $(S, V \setminus S)$. 
If edge $e = \{u, v\}$ is the \textbf{strictly minimum-weight edge} crossing the cut (i.e., $u \in S$ and $v \in V \setminus S$), then $e$ must belong to every Minimum Spanning Tree of $G$.
\end{theorembox}

\begin{definitionbox}{Kruskal's Algorithm (Edge-Centric Greedy)}
\begin{enumerate}
    \item Sort all edges in non-decreasing order of weight: $w(e_1) \le w(e_2) \le \dots \le w(e_m)$.
    \item Initialize an empty forest $T = \emptyset$.
    \item Initialize a Disjoint Set Union (DSU / Union-Find) data structure where each vertex $v \in V$ forms its own component.
    \item For each edge $e = \{u, v\}$ in the sorted sequence:
    \begin{itemize}
        \item If $\text{Find}(u) \neq \text{Find}(v)$ (i.e. $u$ and $v$ belong to different components):
        \item Add $e$ to $T$, and perform $\text{Union}(u, v)$.
    \end{itemize}
    \item Terminate when $T$ contains $|V| - 1$ edges.
    \item \textbf{Complexity:} $\mathcal{O}(|E| \log |E|) = \mathcal{O}(|E| \log |V|)$.
\end{enumerate}
\end{definitionbox}

\begin{definitionbox}{Prim's Algorithm (Vertex-Centric Greedy)}
\begin{enumerate}
    \item Select an arbitrary starting vertex $s \in V$. Initialize visited set $S = \{s\}$.
    \item Maintain a min-priority queue of all crossing edges between $S$ and $V \setminus S$.
    \item While $|S| < |V|$:
    \begin{itemize}
        \item Extract the minimum weight crossing edge $e = \{u, v\}$ where $u \in S, v \notin S$.
        \item Add $e$ to the MST, add $v$ to $S$.
        \item Insert all incident edges of $v$ connecting to $V \setminus S$ into the priority queue.
    \end{itemize}
    \item \textbf{Complexity:} $\mathcal{O}(|E| \log |V|)$ using a binary min-heap.
\end{enumerate}
\end{definitionbox}

\section{Deep Intuition \& Conceptual Derivations}

\subsection{Proof of Correctness for Dijkstra's Algorithm}
We prove by mathematical induction that whenever a vertex $u$ is added to the finalized set $S$, $\text{dist}[u]$ is the exact shortest path distance from source $s$, denoted $\delta(s, u)$.

\textbf{Base Case:} When $S = \{s\}$, $\text{dist}[s] = 0 = \delta(s, s)$ because all edge weights are non-negative ($w(e) \ge 0$), meaning no path can have negative length.

\textbf{Inductive Step:} Suppose that for all vertices currently in $S$, $\text{dist}[v] = \delta(s, v)$. Let $u \in V \setminus S$ be the next vertex chosen by the algorithm (i.e., $u = \operatorname{argmin}_{v \notin S} \text{dist}[v]$).
Suppose, for contradiction, that $\text{dist}[u] > \delta(s, u)$. Then there must exist an active true shortest path $P$ from $s$ to $u$ whose total weight is strictly less than $\text{dist}[u]$:
\begin{equation}
w(P) = \delta(s, u) < \text{dist}[u]
\end{equation}
Since $s \in S$ and $u \notin S$, the path $P$ must at some point cross the boundary from $S$ to $V \setminus S$. Let $(x, y)$ be the first edge along $P$ such that $x \in S$ and $y \notin S$.
Because $x \in S$, by our inductive hypothesis $\text{dist}[x] = \delta(s, x)$. When $x$ was finalized, edge $(x, y)$ was relaxed, ensuring:
\begin{equation}
\text{dist}[y] \le \text{dist}[x] + w(x, y) = \delta(s, x) + w(x, y) = \delta(s, y)
\end{equation}
Because edge weights are non-negative ($w(e) \ge 0$), the subpath from $s$ to $y$ has length less than or equal to the total path $P$ from $s$ to $u$:
\begin{equation}
\delta(s, y) \le \delta(s, u)
\end{equation}
Combining these inequalities:
\begin{equation}
\text{dist}[y] \le \delta(s, y) \le \delta(s, u) < \text{dist}[u]
\end{equation}
However, the algorithm selected vertex $u \in V \setminus S$ precisely because $\text{dist}[u]$ was minimal among all unvisited vertices, which requires $\text{dist}[u] \le \text{dist}[y]$. This contradiction proves that $\text{dist}[u] = \delta(s, u) \quad \blacksquare$.

\subsection{Proof of the Cut Property for Minimum Spanning Trees}
Let $T$ be an arbitrary MST of $G = (V, E)$, and let $e = \{u, v\}$ be the strictly lightest edge crossing cut $(S, V \setminus S)$.
Assume, for contradiction, that $e \notin T$.
Since $T$ is a spanning tree, adding edge $e$ to $T$ must create a unique simple cycle $C$. Because $u \in S$ and $v \in V \setminus S$, traversing this cycle $C$ requires crossing back across the cut $(S, V \setminus S)$ via at least one other edge $e' = \{x, y\}$ with $x \in S, y \in V \setminus S$.
Now consider the new subgraph $T' = (T \setminus \{e'\}) \cup \{e\}$ obtained by removing $e'$ and inserting $e$.
$T'$ is connected, touches all vertices, and has $|V| - 1$ edges, so $T'$ is a valid spanning tree. The total weight of $T'$ is:
\begin{equation}
w(T') = w(T) - w(e') + w(e)
\end{equation}
Since $e$ is the strictly minimum-weight crossing edge, $w(e) < w(e')$, which implies:
\begin{equation}
w(T') < w(T)
\end{equation}
This contradicts the premise that $T$ was a Minimum Spanning Tree! Hence, $e$ must belong to $T \quad \blacksquare$.

\section{Explicit Gotchas \& Diagnostic Misconceptions}

\begin{trapbox}{Trap 1: Applying Dijkstra to Graphs with Negative Edge Weights}
Dijkstra's greedy paradigm permanently "freezes" the distance to a vertex the moment it is popped from the priority queue. If a later edge has a negative weight, a path through that edge could provide a shorter route to a finalized vertex, but Dijkstra will never revisit or update finalized nodes! Always use Bellman-Ford or Floyd-Warshall when negative weights are possible.
\end{trapbox}

\begin{trapbox}{Trap 2: Adding a Large Positive Constant $C$ to Make All Edges Positive}
Students often suggest: "If edges have negative weights, just add a constant $C = |\min w| + 1$ to every edge and run Dijkstra!" 
This fails completely. If Path 1 has 1 edge of weight 5, and Path 2 has 3 edges of weights $(1, 1, 1)$, total weights are 5 vs 3 (Path 2 is shorter). Adding $C = 10$ makes the edges:
Path 1: $5 + 10 = 15$.
Path 2: $(1+10) + (1+10) + (1+10) = 33$.
Now Path 1 appears shorter! Adding a constant penalizes paths with more edges, altering the fundamental shortest path geometry.
\end{trapbox}

\begin{trapbox}{Trap 3: Conflating Shortest Paths with Minimum Spanning Trees}
An edge in a Minimum Spanning Tree does \textit{not} guarantee the shortest path between its endpoints, and the shortest path tree from source $s$ is generally \textit{not} an MST. An MST minimizes total infrastructure cost $\sum_{e \in T} w(e)$, whereas a shortest path tree minimizes point-to-point distances $\text{dist}(s, v)$ from a single origin.
\end{trapbox}

\begin{trapbox}{Trap 4: Assuming Minimum Spanning Trees are Always Unique}
An MST is guaranteed to be unique \textbf{if and only if} all edge weights in the graph are distinct. If multiple edges share identical weights, there may exist dozens or thousands of distinct minimum spanning trees with identical total sum $w(T)$.
\end{trapbox}

\section{Python Scientific Computing Lab}

The following laboratory implements Dijkstra's shortest path algorithm using a binary min-heap (`heapq`) and Kruskal's MST algorithm with Disjoint Set Union (Union-Find).

\begin{lstlisting}[language=Python, caption=Dijkstra Shortest Paths and Kruskal MST Implementation]
import heapq

# 1. Dijkstra's Algorithm with Min-Heap
def dijkstra(graph, start):
    # graph: dict of {node: [(neighbor, weight)]}
    distances = {node: float('inf') for node in graph}
    distances[start] = 0
    pq = [(0, start)] # (distance, node)
    visited = set()
    
    while pq:
        curr_dist, u = heapq.heappop(pq)
        if u in visited:
            continue
        visited.add(u)
        
        for v, weight in graph[u]:
            if v not in visited and curr_dist + weight < distances[v]:
                distances[v] = curr_dist + weight
                heapq.heappush(pq, (distances[v], v))
                
    return distances

weighted_graph = {
    'A': [('B', 4), ('C', 2)],
    'B': [('A', 4), ('C', 1), ('D', 5)],
    'C': [('A', 2), ('B', 1), ('D', 8), ('E', 10)],
    'D': [('B', 5), ('C', 8), ('E', 2)],
    'E': [('C', 10), ('D', 2)]
}

dists_from_A = dijkstra(weighted_graph, 'A')
print("Dijkstra Shortest Distances from 'A':", dists_from_A)

# 2. Kruskal's Minimum Spanning Tree with Union-Find (DSU)
class DisjointSet:
    def __init__(self, vertices):
        self.parent = {v: v for v in vertices}
        self.rank = {v: 0 for v in vertices}
        
    def find(self, item):
        if self.parent[item] != item:
            self.parent[item] = self.find(self.parent[item]) # Path compression
        return self.parent[item]
        
    def union(self, root1, root2):
        if self.rank[root1] < self.rank[root2]:
            self.parent[root1] = root2
        elif self.rank[root1] > self.rank[root2]:
            self.parent[root2] = root1
        else:
            self.parent[root2] = root1
            self.rank[root1] += 1

def kruskal_mst(vertices, edges):
    # edges: list of (u, v, weight)
    edges.sort(key=lambda x: x[2])
    dsu = DisjointSet(vertices)
    mst = []
    total_cost = 0
    
    for u, v, weight in edges:
        root_u = dsu.find(u)
        root_v = dsu.find(v)
        if root_u != root_v:
            mst.append((u, v, weight))
            total_cost += weight
            dsu.union(root_u, root_v)
            if len(mst) == len(vertices) - 1:
                break
                
    return mst, total_cost

nodes = ['A', 'B', 'C', 'D', 'E']
edge_list = [
    ('A', 'B', 4), ('A', 'C', 2),
    ('B', 'C', 1), ('B', 'D', 5),
    ('C', 'D', 8), ('C', 'E', 10),
    ('D', 'E', 2)
]

mst_edges, mst_weight = kruskal_mst(nodes, edge_list)
print("\nKruskal MST Selected Edges:", mst_edges)
print(f"Total Minimum Spanning Tree Weight: {mst_weight}")
\end{lstlisting}
