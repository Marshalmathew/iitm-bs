\textbf{1. Set Algebra, Cardinality, and Power Sets}
Given $A = \{x \in \Z \mid -2 \le x \le 2\}$ and $B = \{x \in \N \mid x^2 \le 9\}$ with $\N = \{1, 2, 3, \dots\}$:
\begin{enumerate}
    \item \textbf{Element Listings:}
    $$A = \{-2, -1, 0, 1, 2\}, \qquad B = \{1, 2, 3\}$$
    $$A \cap B = \{1, 2\}, \qquad A \cup B = \{-2, -1, 0, 1, 2, 3\}$$
    \item \textbf{Set Differences:}
    $$A \setminus B = \{-2, -1, 0, 1, 2\} \setminus \{1, 2, 3\} = \mathbf{\{-2, -1, 0\}}$$
    $$B \setminus A = \{1, 2, 3\} \setminus \{-2, -1, 0, 1, 2\} = \mathbf{\{3\}}$$
    \item \textbf{Symmetric Difference $A \Delta B$:}
    $$A \Delta B = (A \setminus B) \cup (B \setminus A) = \{-2, -1, 0\} \cup \{3\} = \mathbf{\{-2, -1, 0, 3\}}$$
    (Equivalently, $(A \cup B) \setminus (A \cap B) = \{-2, -1, 0, 1, 2, 3\} \setminus \{1, 2\} = \{-2, -1, 0, 3\}$).
    \item \textbf{Power Set Cardinality $|\mathcal{P}(A \cap B)|$:}
    Since $A \cap B = \{1, 2\}$, its cardinality is $|A \cap B| = 2$.
    $$|\mathcal{P}(A \cap B)| = 2^{|A \cap B|} = 2^2 = \mathbf{4}$$
    The explicit subsets are $\mathcal{P}(A \cap B) = \{\emptyset, \{1\}, \{2\}, \{1, 2\}\}$.
\end{enumerate}

\textbf{2. Equivalence Relations and Quotient Partitions}
Relation on $\Z$: $a R b \iff 5 \mid (2a + 3b)$.
\begin{enumerate}
    \item \textbf{Reflexivity:} For any $a \in \Z$:
    $$2a + 3a = 5a = 5(a)$$
    Since $a \in \Z$, $5 \mid 5a$. Thus $a R a$ holds for all $a \in \Z$, so $R$ is reflexive.
    \item \textbf{Symmetry:} Assume $a R b$. Then $2a + 3b = 5k$ for some integer $k$. Consider $2b + 3a$:
    $$(2b + 3a) + (2a + 3b) = 5a + 5b = 5(a + b)$$
    $$2b + 3a = 5(a + b) - (2a + 3b) = 5(a + b) - 5k = 5(a + b - k)$$
    Since $a + b - k \in \Z$, $5 \mid (2b + 3a)$, which implies $b R a$. Thus, $R$ is symmetric.
    \item \textbf{Transitivity:} Assume $a R b$ and $b R c$. Then $2a + 3b = 5k$ and $2b + 3c = 5m$ for integers $k, m$.
    Adding both equations:
    $$(2a + 3b) + (2b + 3c) = 2a + 5b + 3c = 5(k + m)$$
    $$2a + 3c = 5(k + m) - 5b = 5(k + m - b)$$
    Since $k + m - b \in \Z$, $5 \mid (2a + 3c)$, which implies $a R c$. Thus, $R$ is transitive.
    
    Since $R$ is reflexive, symmetric, and transitive, $R$ is an \textbf{equivalence relation}.
    \item \textbf{Partition into Equivalence Classes:}
    Notice $2a + 3b = 2(a - b) + 5b$. Hence $5 \mid (2a + 3b) \iff 5 \mid 2(a - b) \iff 5 \mid (a - b) \iff a \equiv b \pmod 5$.
    Thus $R$ partitions $\Z$ into the 5 standard residue classes modulo 5:
    \begin{align*}
        [0] &= \{\dots, -10, -5, 0, 5, 10, \dots\} = \{5k \mid k \in \Z\} \\
        [1] &= \{\dots, -9, -4, 1, 6, 11, \dots\} = \{5k + 1 \mid k \in \Z\} \\
        [2] &= \{\dots, -8, -3, 2, 7, 12, \dots\} = \{5k + 2 \mid k \in \Z\} \\
        [3] &= \{\dots, -7, -2, 3, 8, 13, \dots\} = \{5k + 3 \mid k \in \Z\} \\
        [4] &= \{\dots, -6, -1, 4, 9, 14, \dots\} = \{5k + 4 \mid k \in \Z\}
    \end{align*}
\end{enumerate}

\textbf{3. Domain and Codomain Impacts on Injectivity and Surjectivity}
Consider $f(x) = \frac{x}{x^2 + 1}$.
\begin{enumerate}
    \item \textbf{Injectivity on $\func{f}{\R}{\R}$:}
    Notice that $f(2) = \frac{2}{2^2 + 1} = \frac{2}{5}$, and $f(1/2) = \frac{1/2}{(1/2)^2 + 1} = \frac{1/2}{5/4} = \frac{2}{5}$.
    Since $2 \neq \frac{1}{2}$ but $f(2) = f(1/2)$, $f$ is \textbf{not injective} on $\R$. (In fact, $f(x) = f(1/x)$ for all $x \neq 0$).
    \item \textbf{Surjectivity on $\func{f}{\R}{\R}$:}
    Set $y = 1$ (which lies in codomain $\R$).
    $$\frac{x}{x^2 + 1} = 1 \implies x^2 - x + 1 = 0$$
    The discriminant is $\Delta = (-1)^2 - 4(1)(1) = 1 - 4 = -3 < 0$. There are no real solutions $x \in \R$.
    Because $y = 1$ has no pre-image, $\text{Range}(f) \neq \R$ (the true range is $[-\frac{1}{2}, \frac{1}{2}]$). Therefore, $f$ is \textbf{not surjective} on $\R$.
    \item \textbf{Bijectivity on $\func{f}{[1, \infty)}{(0, \frac{1}{2}]}$:}
    Differentiating: $f'(x) = \frac{(x^2 + 1)(1) - x(2x)}{(x^2 + 1)^2} = \frac{1 - x^2}{(x^2 + 1)^2}$.
    For all $x > 1$, $x^2 > 1 \implies f'(x) < 0$. The function is strictly monotonically decreasing on $[1, \infty)$, guaranteeing that distinct inputs produce distinct outputs, so $f$ is \textbf{strictly injective}.
    Evaluating boundaries: $f(1) = \frac{1}{1 + 1} = \frac{1}{2}$, and $\lim_{x \to \infty} f(x) = 0$. Since $f$ is continuous, by the Intermediate Value Theorem its range is precisely $(0, \frac{1}{2}]$, matching the declared codomain. Thus, $f$ is \textbf{surjective}.
    Being both injective and surjective, $f$ is a \textbf{bijection}.
\end{enumerate}

\textbf{4. Feature Scaling as a Bijective Isomorphism}
Given $f(x) = \frac{x - x_{\min}}{x_{\max} - x_{\min}}$ for $x \in [x_{\min}, x_{\max}]$ with $x_{\max} > x_{\min}$:
\begin{enumerate}
    \item \textbf{Injectivity:} Assume $f(x_1) = f(x_2)$:
    $$\frac{x_1 - x_{\min}}{x_{\max} - x_{\min}} = \frac{x_2 - x_{\min}}{x_{\max} - x_{\min}}$$
    Since $x_{\max} - x_{\min} > 0$, multiplying both sides gives $x_1 - x_{\min} = x_2 - x_{\min} \implies x_1 = x_2$. Hence, $f$ is injective.
    \item \textbf{Surjectivity:} Let $y \in [0, 1]$. We solve for $x$:
    $$y = \frac{x - x_{\min}}{x_{\max} - x_{\min}} \implies x = x_{\min} + y(x_{\max} - x_{\min})$$
    Since $0 \le y \le 1$, $0 \le y(x_{\max} - x_{\min}) \le x_{\max} - x_{\min}$. Adding $x_{\min}$ gives $x_{\min} \le x \le x_{\max}$.
    Thus, every $y \in [0, 1]$ has a pre-image in $[x_{\min}, x_{\max}]$, proving surjectivity.
    \item \textbf{Inverse Function and Identity Composition:}
    $$f^{-1}(y) = x_{\min} + y(x_{\max} - x_{\min})$$
    Composing:
    $$(f^{-1} \circ f)(x) = f^{-1}(f(x)) = x_{\min} + \left( \frac{x - x_{\min}}{x_{\max} - x_{\min}} \right)(x_{\max} - x_{\min}) = x_{\min} + (x - x_{\min}) = \mathbf{x}$$
    This proves that Min-Max scaling is a lossless, perfectly invertible transformation.
\end{enumerate}

\textbf{5. Rational Function Composition and Inversion}
Given $\func{f}{\R \setminus \{1\}}{\R \setminus \{2\}}$ with $f(x) = \frac{2x + 1}{x - 1}$:
\begin{enumerate}
    \item \textbf{Bijectivity:}
    \begin{itemize}
        \item \textit{Injective:} $f(x_1) = f(x_2) \implies \frac{2x_1 + 1}{x_1 - 1} = \frac{2x_2 + 1}{x_2 - 1} \implies (2x_1 + 1)(x_2 - 1) = (2x_2 + 1)(x_1 - 1) \implies 2x_1 x_2 - 2x_1 + x_2 - 1 = 2x_1 x_2 - 2x_2 + x_1 - 1 \implies -3x_1 = -3x_2 \implies x_1 = x_2$.
        \item \textit{Surjective:} Let $y \in \R \setminus \{2\}$. Set $y = \frac{2x + 1}{x - 1} \implies y(x - 1) = 2x + 1 \implies x(y - 2) = y + 1 \implies x = \frac{y + 1}{y - 2}$. Since $y \neq 2$, this $x$ is well-defined. If $x = 1$, then $\frac{y + 1}{y - 2} = 1 \implies y + 1 = y - 2 \implies 1 = -2$ (contradiction). Hence $x \in \R \setminus \{1\}$.
    \end{itemize}
    Thus $f$ is a bijection.
    \item \textbf{Explicit Inverse:}
    $$f^{-1}(y) = \mathbf{\frac{y + 1}{y - 2}}, \qquad y \neq 2$$
    \item \textbf{Composition $(f \circ f)(x)$ and Natural Domain:}
    $$(f \circ f)(x) = f(f(x)) = \frac{2\left(\frac{2x+1}{x-1}\right) + 1}{\left(\frac{2x+1}{x-1}\right) - 1} = \frac{2(2x+1) + (x-1)}{(2x+1) - (x-1)} = \frac{4x + 2 + x - 1}{2x + 1 - x + 1} = \mathbf{\frac{5x + 1}{x + 2}}$$
    The domain requires $x \neq 1$ (for inner $f(x)$) and $f(x) \neq 1 \implies \frac{2x+1}{x-1} \neq 1 \implies 2x+1 \neq x-1 \implies x \neq -2$.
    Thus, the natural domain of $(f \circ f)$ is $\mathbf{\R \setminus \{-2, 1\}}$.
\end{enumerate}

\textbf{6. Conceptual Diagnostic: Element Membership vs. Subset Inclusion}
\begin{itemize}
    \item \textbf{Correct Answer: (d)}
    \item \textbf{Why (d) is strictly false:} $2 \notin S$. The elements of $S = \{\emptyset, 1, \{1, 2\}\}$ are precisely $\emptyset$, $1$, and the set $\{1, 2\}$. The number $2$ is an element of $\{1, 2\}$, but set membership is \textbf{not transitive}: $x \in A$ and $A \in S$ does \textbf{not} imply $x \in S$. Thus, $2 \in S$ is mathematically false.
    \item \textbf{Why (a) is true:} $\emptyset \in S$ is true because $\emptyset$ is explicitly listed as the first element of $S$. $\emptyset \subseteq S$ is true because the empty set is vacuously a subset of every set.
    \item \textbf{Why (b) is true:} $\{1\} \subseteq S$ is true because its only element, $1$, is in $S$. $\{1, 2\} \in S$ is true because the set $\{1, 2\}$ is listed as an element of $S$.
    \item \textbf{Why (c) is true:} $\{\{1, 2\}\} \subseteq S$ is true because its sole element is $\{1, 2\}$, which belongs to $S$. However, $\{1, 2\} \subseteq S$ is false because $2 \notin S$.

\textbf{7. Conceptual Diagnostic: Invertibility and Bijective Constraints}
\begin{itemize}
    \item \textbf{Correct Answer: (c)}
    \item \textbf{Why (c) is correct:} By the Fundamental Inversion Theorem of mappings, a function $\func{h}{A}{B}$ possesses a well-defined two-sided inverse function $\func{h^{-1}}{B}{A}$ if and only if $h$ is a \textbf{bijection} (simultaneously injective and surjective). For finite sets, bijectivity immediately implies $|A| = |B|$. Injectivity ensures that each $b \in B$ has at most one pre-image (preventing $h^{-1}$ from being one-to-many), and surjectivity ensures that each $b \in B$ has at least one pre-image (preventing $h^{-1}$ from being undefined on elements of $B$).
    \item \textbf{Why (a) is false:} Injectivity alone only guarantees a left inverse; if $|A| < |B|$, there exist elements in $B$ without pre-images, so $h^{-1}$ cannot be a valid total function on $B$.
    \item \textbf{Why (b) is false:} Surjectivity alone only guarantees a right inverse; if $|A| > |B|$, multiple elements in $A$ map to the same $b$, making $h^{-1}$ multi-valued (not a function).
    \item \textbf{Why (d) is false:} An equivalence relation is defined on a set to itself ($A \times A$), not between arbitrary domain and codomain sets $A$ and $B$, and equivalence relations are generally many-to-many.
\end{itemize}

\textbf{1. Distances, Section Formulas, and Centroids}
Given points $A(-2, 3)$, $B(4, -1)$, and $C(6, 5)$:
\begin{enumerate}
    \item \textbf{Collinearity and Triangle Verification:}
    Compute the slopes between pairs of points:
    $$m_{AB} = \frac{-1 - 3}{4 - (-2)} = \frac{-4}{6} = -\frac{2}{3}$$
    $$m_{BC} = \frac{5 - (-1)}{6 - 4} = \frac{6}{2} = 3$$
    Since $m_{AB} \neq m_{BC}$ ($-\frac{2}{3} \neq 3$), the three points are non-collinear and form a non-degenerate triangle $\triangle ABC$.
    \item \textbf{Perimeter via Euclidean Distance:}
    $$d(A, B) = \sqrt{(4 - (-2))^2 + (-1 - 3)^2} = \sqrt{6^2 + (-4)^2} = \sqrt{52} = 2\sqrt{13} \approx 7.2111$$
    $$d(B, C) = \sqrt{(6 - 4)^2 + (5 - (-1))^2} = \sqrt{2^2 + 6^2} = \sqrt{40} = 2\sqrt{10} \approx 6.3246$$
    $$d(C, A) = \sqrt{(6 - (-2))^2 + (5 - 3)^2} = \sqrt{8^2 + 2^2} = \sqrt{68} = 2\sqrt{17} \approx 8.2462$$
    $$\text{Perimeter} = \sqrt{52} + \sqrt{40} + \sqrt{68} \approx 7.2111 + 6.3246 + 8.2462 = \mathbf{21.7819}$$
    \item \textbf{Geometric Centroid $G$:}
    $$x_G = \frac{x_A + x_B + x_C}{3} = \frac{-2 + 4 + 6}{3} = \mathbf{\frac{8}{3} \approx 2.6667}$$
    $$y_G = \frac{y_A + y_B + y_C}{3} = \frac{3 + (-1) + 5}{3} = \mathbf{\frac{7}{3} \approx 2.3333}$$
    $$G = \left( \frac{8}{3}, \frac{7}{3} \right)$$
    \item \textbf{Internal Section Point $P$ ($2 : 1$ Ratio):}
    Using the section formula with $m = 2, n = 1$:
    $$x_P = \frac{2(4) + 1(-2)}{2 + 1} = \frac{8 - 2}{3} = \mathbf{2.0}$$
    $$y_P = \frac{2(-1) + 1(3)}{2 + 1} = \frac{-2 + 3}{3} = \mathbf{\frac{1}{3} \approx 0.3333}$$
    $$P = \left( 2, \frac{1}{3} \right)$$
\end{enumerate}

\textbf{2. Line Equations, Intercepts, and Orthogonal Projections}
Given $L_1: 3x - 4y + 12 = 0$:
\begin{enumerate}
    \item \textbf{Intercepts and Slope:}
    \begin{itemize}
        \item $x$-intercept ($y=0$): $3x + 12 = 0 \implies x = -4 \implies \mathbf{a = -4}$.
        \item $y$-intercept ($x=0$): $-4y + 12 = 0 \implies y = 3 \implies \mathbf{b = 3}$.
        \item Slope: $4y = 3x + 12 \implies y = \frac{3}{4}x + 3 \implies \mathbf{m_1 = \frac{3}{4} = 0.75}$.
    \end{itemize}
    \item \textbf{Perpendicular Line $L_2$ Through $(5, -2)$:}
    The perpendicular slope is $m_2 = -\frac{1}{m_1} = -\frac{4}{3}$.
    $$y - (-2) = -\frac{4}{3}(x - 5) \implies 3(y + 2) = -4(x - 5) \implies 3y + 6 = -4x + 20$$
    $$\mathbf{4x + 3y - 14 = 0}$$
    \item \textbf{Intersection Point and Orthogonal Projection:}
    Solving the linear system:
    \begin{align*}
        3x - 4y &= -12 \quad \times 3 \implies 9x - 12y = -36 \\
        4x + 3y &= 14 \quad \times 4 \implies 16x + 12y = 56
    \end{align*}
    Adding the equations gives $25x = 20 \implies \mathbf{x = \frac{4}{5} = 0.80}$.
    Substituting $x = 0.80$ into $4x + 3y = 14$:
    $$4(0.80) + 3y = 14 \implies 3.2 + 3y = 14 \implies 3y = 10.8 \implies \mathbf{y = \frac{18}{5} = 3.60}$$
    Intersection point is $\mathbf{(0.80, 3.60)}$.
    
    Verifying via projection formula:
    $$t = \frac{Ax_0 + By_0 + C}{A^2 + B^2} = \frac{3(5) - 4(-2) + 12}{3^2 + (-4)^2} = \frac{15 + 8 + 12}{25} = \frac{35}{25} = 1.40$$
    $$x_{\text{proj}} = 5 - (1.40)(3) = 5 - 4.2 = 0.80$$
    $$y_{\text{proj}} = -2 - (1.40)(-4) = -2 + 5.6 = 3.60$$
    The intersection matches the orthogonal projection point identically!
\end{enumerate}

\textbf{3. Support Vector Machine Decision Margins}
Given hyperplanes $H_1: 2x - 3y + 6 = 0$ and $H_2: 4x - 6y - 14 = 0$:
\begin{enumerate}
    \item \textbf{Normalized Form of $H_2$:}
    Dividing the entire equation of $H_2$ by $2$:
    $$\mathbf{2x - 3y - 7 = 0}$$
    Now $A = 2, B = -3, C_1 = 6, C_2 = -7$.
    \item \textbf{Perpendicular Margin Separation $d$:}
    $$d = \frac{|C_1 - C_2|}{\sqrt{A^2 + B^2}} = \frac{|6 - (-7)|}{\sqrt{2^2 + (-3)^2}} = \frac{13}{\sqrt{4 + 9}} = \frac{13}{\sqrt{13}} = \mathbf{\sqrt{13} \approx 3.6056}$$
    \item \textbf{Midway Decision Boundary $H_0$:}
    The central decision line has the same slope and an intercept constant equal to the arithmetic mean of $C_1$ and $C_2$:
    $$C_0 = \frac{C_1 + C_2}{2} = \frac{6 + (-7)}{2} = -\frac{1}{2} = -0.5$$
    $$2x - 3y - 0.5 = 0 \implies \mathbf{4x - 6y - 1 = 0}$$
\end{enumerate}

\textbf{5. Ordinary Least Squares (OLS) Closed-Form Derivation}
Data: $\{(1, 2), (2, 3), (3, 5), (4, 4), (5, 6)\}$, $N = 5$.
\begin{enumerate}
    \item \textbf{Sample Means:}
    $$\bar{x} = \frac{1 + 2 + 3 + 4 + 5}{5} = \frac{15}{5} = \mathbf{3.0}$$
    $$\bar{y} = \frac{2 + 3 + 5 + 4 + 6}{5} = \frac{20}{5} = \mathbf{4.0}$$
    \item \textbf{Optimal Slope $m$ and Intercept $c$:}
    Compute deviance sums:
    $$\sum_{i=1}^5 (x_i - \bar{x})^2 = (-2)^2 + (-1)^2 + 0^2 + 1^2 + 2^2 = 4 + 1 + 0 + 1 + 4 = 10$$
    $$\sum_{i=1}^5 (x_i - \bar{x})(y_i - \bar{y}) = (-2)(-2) + (-1)(-1) + (0)(1) + (1)(0) + (2)(2) = 4 + 1 + 0 + 0 + 4 = 9$$
    $$m = \frac{\sum (x_i - \bar{x})(y_i - \bar{y})}{\sum (x_i - \bar{x})^2} = \frac{9}{10} = \mathbf{0.90}$$
    $$c = \bar{y} - m\bar{x} = 4.0 - (0.90)(3.0) = 4.0 - 2.70 = \mathbf{1.30}$$
    Fitted regression line: $\mathbf{\hat{y} = 0.90x + 1.30}$.
    \item \textbf{Target Prediction for $x = 6$:}
    $$\hat{y}(6) = 0.90(6) + 1.30 = 5.40 + 1.30 = \mathbf{6.70}$$
    \item \textbf{Sum of Squared Errors (SSE):}
    \begin{align*}
        e_1 &= 2 - (0.9(1) + 1.3) = 2 - 2.2 = -0.2 \implies e_1^2 = 0.04 \\
        e_2 &= 3 - (0.9(2) + 1.3) = 3 - 3.1 = -0.1 \implies e_2^2 = 0.01 \\
        e_3 &= 5 - (0.9(3) + 1.3) = 5 - 4.0 = +1.0 \implies e_3^2 = 1.00 \\
        e_4 &= 4 - (0.9(4) + 1.3) = 4 - 4.9 = -0.9 \implies e_4^2 = 0.81 \\
        e_5 &= 6 - (0.9(5) + 1.3) = 6 - 5.8 = +0.2 \implies e_5^2 = 0.04
    \end{align*}
    $$\text{SSE} = \sum_{i=1}^5 e_i^2 = 0.04 + 0.01 + 1.00 + 0.81 + 0.04 = \mathbf{1.90}$$
\end{enumerate}

\textbf{7. Conceptual Diagnostic: OLS Vertical Errors vs. Orthogonal PCA Projection}
\begin{itemize}
    \item \textbf{Correct Answer: (c)}
    \item \textbf{Why (c) is correct:} Ordinary Least Squares regression solves the asymmetric loss minimization problem $\min_{m, c} \sum_{i=1}^N (y_i - (mx_i + c))^2$, which measures strictly \textbf{vertical deviations} parallel to the $y$-axis. This formulation mathematically assumes that feature $x$ is measured without noise, attributing all residual variance to $y$. In contrast, Principal Component Analysis (orthogonal regression / Total Least Squares) minimizes the perpendicular Euclidean distance $\frac{|Ax_i + By_i + C|}{\sqrt{A^2 + B^2}}$, treating both $x$ and $y$ symmetrically as noisy random variables. Consequently, OLS attenuates the estimated slope relative to PCA whenever $x$ has noise.
    \item \textbf{Why (a) is false:} Both OLS and PCA work perfectly on centered or uncentered data; mean-centering is simply standard algebraic convenience.
    \item \textbf{Why (b) is false:} Both OLS simple regression and first-component PCA define strictly linear straight lines in 2D space.
    \item \textbf{Why (d) is false:} PCA maximizes variance along the principal projection axis, not minimizes it.
\end{itemize}

\textbf{1. Vieta's Relations and Polynomial Transformation}
Given quadratic equation $2x^2 - 7x + 3 = 0$ with roots $r_1, r_2$:
\begin{enumerate}
    \item \textbf{Vieta's Formulas ($a = 2, b = -7, c = 3$):}
    $$r_1 + r_2 = -\frac{b}{a} = -\frac{-7}{2} = \mathbf{\frac{7}{2} = 3.50}$$

    \begin{equation*}

    \end{equation*}
    $(A^2)_{ii}$ represents the number of walks of length 2 from vertex $i$ back to itself. In a simple graph, any walk of length 2 from $i$ must travel to an adjacent neighbor $k$ and immediately return to $i$. Therefore, $(A^2)_{ii}$ equals the \textbf{degree of vertex $i$}.
    \item \textbf{Off-Diagonal Entry $(A^k)_{ij}$:}
    The $(i, j)$-th entry of $A^k$ is the exact number of distinct \textbf{walks of length $k$} from vertex $i$ to vertex $j$.
\end{enumerate}

\textbf{4. DFS and Cycle Detection}
In the three-color scheme:
\begin{itemize}
    \item \textbf{White:} Vertex has not yet been discovered.
    \item \textbf{Gray:} Vertex has been discovered, and its subtree is actively being explored on the recursion stack.
    \item \textbf{Black:} Vertex and all its descendants have been completely processed.
\end{itemize}
When traversing a directed edge $(u, v)$:
\begin{itemize}
    \item If $v$ is White: Tree edge (standard recursive descent).
    \item If $v$ is Gray: $v$ is an active ancestor of $u$ on the call stack. The directed edge $(u, v)$ points back to an active ancestor, forming a \textbf{back-edge}.
    \item \textbf{Theorem:} A directed graph contains a directed cycle if and only if DFS encounters at least one \textbf{back-edge}.
\end{itemize}
