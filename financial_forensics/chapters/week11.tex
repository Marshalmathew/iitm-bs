\section{Unstructured Text in Financial Investigations}

While quantitative ledgers capture numerical balances, over $80\%$ of enterprise intelligence resides in unstructured text: regulatory filings, auditor notes, internal chat/email archives, and wire transfer payment narrations.

\begin{definitionbox}{Deceptive Linguistic Markers}
\index{Linguistic Deception}
Psycholinguistic research shows that corporate executives writing deceptive reports exhibit consistent linguistic fingerprints:
\begin{itemize}
    \item \textbf{First-Person Pronoun Displacement:} Reduction in first-person singular pronouns (``I'', ``my'') and increased reliance on passive voice or third-person entities (``the company'', ``management'') to psychologically detach from fraudulent statements.
    \item \textbf{Superficial Cognitive Complexity:} Overuse of high-syllable legal jargon and convoluted phrasing designed to obfuscate financial reality.
    \item \textbf{Extreme Positive Sentiment Overcompensation:} Unwarranted usage of exuberant superlatives (``tremendous'', ``exceptional'') to mask deteriorating fundamentals.
\end{itemize}
\end{definitionbox}

\section{Entity Resolution: The Disambiguation Puzzle}

In Anti-Money Laundering (AML) and vendor procurement investigations, adversaries intentionally disguise their identity using minor variations in spelling, acronyms, or fictitious holding structures. \index{entity resolution}\textbf{Entity Resolution} (also termed record linkage or de-duplication) is the task of recognizing whether distinct database records represent the exact same real-world entity.

\end{videocard}

\begin{theorembox}{String Similarity Metrics for Entity Matching}
\index{Levenshtein Distance}\index{Jaro-Winkler}
\begin{enumerate}
    \item \textbf{Levenshtein Edit Distance:} The minimum number of single-character edits (insertions, deletions, or substitutions) required to transform string $s_1$ into $s_2$:
    \begin{equation}
    \text{Sim}_{\text{Lev}}(s_1, s_2) = 1 - \frac{\text{Lev}(s_1, s_2)}{\max(|s_1|, |s_2|)}
    \end{equation}
    \item \textbf{Jaro-Winkler Similarity:} Specifically optimized for person and business names, giving higher weight to common initial prefixes:
    \begin{equation}
    d_j = \frac{1}{3}\left(\frac{m}{|s_1|} + \frac{m}{|s_2|} + \frac{m - t}{m}\right)
    \end{equation}
    \begin{equation}
    d_w = d_j + \ell \cdot p \cdot (1 - d_j)
    \end{equation}
    where $m$ is the count of matching characters, $t$ is the number of transpositions, $\ell$ is the length of common prefix (up to 4 chars), and $p = 0.1$ is the constant scaling factor.
\end{enumerate}
\end{theorembox}

\begin{intuitionbox}{Connecting the Shell Company Dots}
Consider payments issued to:
\begin{itemize}
    \item ``Acme Global Solutions Ltd.''
    \item ``ACME GLOBAL SOLNS PVT LTD''
    \item ``Acme Globel Solutions''
\end{itemize}
A naive exact-match query treats these as three unrelated businesses. Entity resolution identifies that all three strings have a Jaro-Winkler similarity $>0.92$, share the same bank routing code, and share a common director address, instantly exposing a centralized shell company funnel!
\end{intuitionbox}

\section{Data Science Relevance: Automated Forensic Entity Resolution}

\begin{lstlisting}[caption={Automated Entity Resolution and Vendor Matching in Python}]
import difflib
import pandas as pd
import numpy as np

def jaro_winkler_similarity(s1: str, s2: str) -> float:
    """Computes string sequence matcher ratio as a robust proxy."""
    return difflib.SequenceMatcher(None, s1.lower().strip(), s2.lower().strip()).ratio()

def resolve_vendor_entities(vendor_list: list, match_threshold: float = 0.85) -> pd.DataFrame:
    """
    Groups distinct vendor text strings into unified canonical entity clusters.
    """
    n = len(vendor_list)
    clusters = {}
    cluster_id = 0
    assigned = [False] * n
    
    for i in range(n):
        if assigned[i]:
            continue
        # Create new entity cluster
        cluster_id += 1
        canonical_name = vendor_list[i]
        clusters[vendor_list[i]] = (cluster_id, canonical_name)
        assigned[i] = True
        
        for j in range(i + 1, n):
            if assigned[j]:
                continue
            sim = jaro_winkler_similarity(vendor_list[i], vendor_list[j])
            if sim >= match_threshold:
                clusters[vendor_list[j]] = (cluster_id, canonical_name)
                assigned[j] = True

    res = pd.DataFrame([
        {'Raw_Vendor_Name': k, 'Cluster_ID': v[0], 'Canonical_Entity': v[1]}
        for k, v in clusters.items()
    ])
    return res

# Messy procurement ledger entries
raw_vendors = [
    "Reliance Industries Ltd",
    "Reliance Inds Ltd",
    "Tata Consultancy Services",
    "TCS Ltd",
    "Reliance Ind.",
    "Tata Consultancy Serv.",
    "Shell Oil Corp"
]

matched_entities = resolve_vendor_entities(raw_vendors, match_threshold=0.72)
print("Entity Resolution Disambiguation Results:\n", matched_entities)
\end{lstlisting}
